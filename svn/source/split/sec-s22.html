稀疏检出：只拉你要的那部分

◈图 22-1　--depth 的四种取值Depth
bash稀疏检出实战
# ---- 检出时只拉一层 ----
svn checkout ^/trunk wc --depth immediates

# ---- 事后按需加深某个子目录 ----
cd wc
svn update --set-depth infinity src/      # src 完整拉下来
svn update --set-depth files docs/        # docs 只拉文件

# ---- 把整个工作副本变回完整 ----
svn update --set-depth infinity .

# ---- 反过来：缩减已检出的目录（会删除本地文件！）----
svn update --set-depth empty art/         # 清空 art 目录内容，保留目录
svn update --set-depth exclude art/       # 整个目录从工作副本移除（更强）

# ---- 查看当前目录的深度设置 ----
svn info wc
# 输出里有 Depth: immediates

# ---- 切换分支时保持稀疏 ----
svn switch ^/branches/feature-login --depth immediates

# ---- 在稀疏工作副本里提交（只提交拉下来的部分）----
cd wc/src
svn commit -m "只修改了 src 下的文件"

changelist：给改动分组

bashchangelist 用法
# ---- 创建变更列表并归类文件 ----
svn changelist doc-fix docs.md README.md
svn changelist code-fix src/main.cs src/utils.cs
svn changelist test-fix tests/*.cs

# ---- 查看分组后的状态 ----
svn status
# 输出会按 changelist 分组显示：
# --- Changelist 'doc-fix':
# M       docs.md
# --- Changelist 'code-fix':
# M       src/main.cs

# ---- 只提交某个列表 ----
svn commit --changelist doc-fix -m "文档修正：补充安装说明"

# ---- 只 diff 某个列表 ----
svn diff --changelist code-fix

# ---- 从列表中移除 ----
svn changelist --remove src/main.cs
svn changelist --remove --recursive .          # 全部移除

# ---- 典型用途：把「顺手改的东西」和「本次任务」分开提交 ----
changelist 是「小步提交」的助推器

很多人攒着一大堆改动一起提交，导致日志写不清、review 困难、回退时牵连太广。 changelist 让你能把「这次任务真正相关的改动」单独挑出来提交， 剩下的（比如顺手改的错别字、格式化的空行）放到下一个提交里。

其他实用技巧合集

svn move：保留历史的重命名
bashmove vs 手工 mv
# ✓ 用 svn move（保留历史关联）
svn move old.cs new.cs
# 执行后 status 会这样显示（实测）：
#   A  +    new.cs          ← 第4列的 + 表示「带复制的历史」
#        > moved from old.cs   ← 附加说明行（不在固定列位）
#   D       old.cs
#        > moved to new.cs

svn commit -m "重命名 old.cs 为 new.cs"
# log -v 里会显示带来源的复制记录：
#    D /trunk/old.cs
#    A /trunk/new.cs (from /trunk/old.cs:21)

# ✗ 手工 mv（历史断开）
mv old.cs new.cs
svn delete old.cs
svn add new.cs
# 历史里看成「删了一个 + 加了一个」，没有 from/to 的来源标注

# 查历史时：SVN 默认【就会】跟随复制历史，无需额外选项
svn log new.cs
# 想「只看这个文件自己的历史、不追溯来源」才需要反转默认行为：
svn log --stop-on-copy new.cs

# 目录也能移动
svn move src/ legacy-src/
svn move ^/trunk/lib ^/branches/lib-experimental
svn blame：逐行追责
bashblame 用法
# 每一行前面标出：版本号 作者
svn blame README.md
#      8     kelang  版本最终：包含登录功能

# 带日期
svn blame -v README.md

# 换成 XML 便于脚本处理
svn blame --xml src/main.cs

# 只看某个版本区间的责任
svn blame -r 10:20 src/main.cs

# 忽略空白差异（格式化改动造成的假变更）
svn blame -x -w src/main.cs

# 追查某一行是谁引入的
svn blame src/main.cs | grep "问题代码"
svn cat / export：不落盘查看与导出
bash用法
# 直接看某个版本的某个文件（不检出）
svn cat -r 8 README.md
svn cat -r 8 file:///repo/trunk/README.md

# 导出干净副本（不含 .svn，可直接打包）
svn export ^/trunk release-build/
svn export -r 520 ^/trunk release-r520/
svn export ^/tags/v1.0.0 v1.0.0/
svn export --force ^/trunk ./     # 覆盖已有目录

# export vs checkout 的选择
# export  → 构建产物、给别人一份、不打算提交
# checkout → 要提交、要更新、要 diff

# 用 export 避免 .svn 污染构建产物
svn export ^/tags/v2.0 ./dist && cd dist && make
svn info --show-item：脚本友好输出
bash等价于只取一个字段
# 只输出一个值，无需 grep/awk 解析
svn info --show-item revision
# 20

svn info --show-item last-changed-revision
svn info --show-item url
svn info --show-item relative-url
# ^/trunk

svn info --show-item repos-root-url
svn info --show-item wc-root
svn info --show-item kind
svn info --show-item revision --no-newline    # 不带换行，方便拼字符串

# 脚本示例：拿到当前版本号做版本标记
REV=$(svn info --show-item last-changed-revision --no-newline)
echo "BUILD_VERSION=1.0.0-r${REV}" > version.env
bash批量操作与自动化脚本
# ---- 批量给新加入的文件设置 svn:eol-style ----
svn status | grep '^A' | awk '{print $2}' | \
  xargs -I{} svn propset svn:eol-style native {}
svn commit -m "统一新增文件的换行符属性"

# ---- 找出所有未加入版本控制的文件并批量添加 ----
svn status | grep '^?' | awk '{print $2}' | xargs svn add

# ---- 批量解决冲突 ----
svn status | grep '^C' | awk '{print $2}' | \
  xargs -I{} svn resolve --accept working {}

# ---- 提交前自动检查：有没有残留冲突标记 ----
if grep -rn '^<<<<<<<' --include="*.cs" --include="*.js" . ; then
  echo "发现未解决的冲突标记，请先处理！"
  exit 1
fi

# ---- 把工作副本恢复到干净状态（危险操作，谨慎）----
svn revert -R .
svn cleanup
svn status --no-ignore | grep '^?' | awk '{print $2}' | xargs rm -rf

# ---- 找出最近一周的提交，生成变更报告 ----
svn log -r "{$(date -d '7 days ago' +%Y-%m-%d)}:HEAD" -v > weekly-changes.txt

# ---- 统计每个人的提交次数 ----
svn log -q -r 1:HEAD | grep '^r' | awk -F'|' '{print $2}' | \
  tr -d ' ' | sort | uniq -c | sort -rn

# ---- 对比两个分支的差异清单 ----
svn diff ^/trunk ^/branches/feature --summarize

# ---- 找出某个文件的所有修改者 ----
svn log -q src/main.cs | grep '^r' | awk -F'|' '{print $2}' | sort -u
