    一、装一个 Lua 解释器

    学 Lua 不需要 Unity。装一个独立的解释器，就能练习所有语法。

    平台方式说明
    Windows官网下载 lua-5.4.x_Win64_bin.zip解压即用
    Windows（推荐）用 VS 编译源码能得到最新的 5.4.7
    macOSbrew install luaHomebrew 安装
    Linuxsudo apt install lua5.4包管理器
    免安装在线 Lua 运行网站仅用于快速试验
    自学建议：用 VS 编译源码 为什么推荐自己编译：
    · 官网的 Windows 二进制包往往版本较旧，且包含的是 32 位版本
    · 自己编译能得到最新版 + 64 位 + DLL（DLL 可以给 C# 用）

    步骤（VS 2022）：
    ① 下载源码：https://www.lua.org/ftp/lua-5.4.7.tar.gz
    ② 解压后进入 src/ 目录
    ③ 用 vcvars64.bat 初始化环境
    ④ cl /O2 /LD /DLUA_BUILD_AS_DLL *.c /Fe:lua54.dll（编译 DLL）
    ⑤ cl /O2 *.c /Fe:lua.exe（编译解释器）
    ⑥ 运行 lua.exe -v 验证

    整个流程不到 10 分钟，而且你会得到最新版本。

    二、编辑器与调试

    工具用途推荐度
    VS Code + Lua 插件语法高亮、跳转、提示强烈推荐
    Lua Debug (VS Code)断点调试推荐
    ZeroBrane Studio轻量 Lua IDE可选
    Notepad++快速看文件不够用
    调试能力很重要 游戏里的 Lua 代码通常在 Unity 编辑器里调试（通过 xLua 的调试支持）。
    但纯 Lua 逻辑（配置解析、算法、数据转换）可以在独立环境里调试——速度快得多，而且能断点。

    建议：把可独立运行的部分写成独立脚本，先在 VS Code 里调通，再集成到 Unity。
    这能大幅提升调试效率——因为 Unity 里调 Lua 一次要等很久。

    三、第一个脚本

    创建 hello.lua，写入下面的内容，然后运行 lua hello.lua。

    -- ============================================================
--  第一段 Lua 脚本
-- ============================================================
-- 运行方式：lua hello.lua
--
-- 注意几个与 C# 不同的地方：
--   1. 没有类、没有命名空间、没有 main 函数
--   2. 没有分号（可以有，但不必要）
--   3. 没有花括号，用 then/do/end 划分块
--   4. 注释是 --（单行）和 --[[ ]]（多行）

print("Hello, Lua!")
print("版本:", _VERSION)

-- ============================================================
--  变量：默认是全局的（这是最需要注意的点）
-- ============================================================
name = "张三"          -- 全局变量（不推荐）
local age = 25         -- 局部变量（推荐）

print("姓名:", name, "年龄:", age)

-- 不存在的变量是 nil（不是错误）
print("不存在的变量:", undefinedVar)     -- 输出 nil

-- ============================================================
--  八种类型
-- ============================================================
print()
print("=== 八种类型 ===")
local samples = {
  ["nil"] = nil,
  ["boolean"] = true,
  ["number"] = 42,
  ["string"] = "text",
  ["table"] = { 1, 2, 3 },
  ["function"] = print,
  ["thread"] = coroutine.create(function() end),
}
for name, value in pairs(samples) do
  print(string.format("  %-10s -> %s", name, type(value)))
end
-- userdata 需要 C 库，这里不演示

-- ============================================================
--  表（Lua 唯一的数据结构）
-- ============================================================
print()
print("=== 表 ===")

-- 数组（索引从 1 开始！）
local arr = { "a", "b", "c" }
print("  数组长度:", #arr)
print("  arr[1]:", arr[1])
print("  arr[0]:", arr[0])        -- nil（不是错误）

-- 字典
local dict = { name = "李四", level = 30 }
print("  字典:", dict.name, dict.level)
print("  用方括号访问:", dict["name"])

-- ============================================================
--  控制流
-- ============================================================
print()
print("=== 控制流 ===")

local score = 85

if score >= 90 then
  print("  优秀")
elseif score >= 60 then
  print("  及格")
else
  print("  不及格")
end

-- 数值 for
for i = 1, 5 do
  io.write(i, " ")
end
print()

-- 泛型 for
for i, v in ipairs(arr) do
  io.write(i, "=", v, " ")
end
print()

-- while
local count = 0
while count < 3 do
  count = count + 1
end
print("  while 循环次数:", count)

-- ============================================================
--  函数
-- ============================================================
print()
print("=== 函数 ===")

local function add(a, b)
  return a + b
end

print("  add(3, 4) =", add(3, 4))

-- 多返回值（Lua 特色）
local function multiReturn()
  return 1, 2, 3
end

local x, y, z = multiReturn()
print("  多返回值:", x, y, z)

-- 函数是值
local fn = add
print("  函数可以赋给变量:", fn(10, 20))

-- ============================================================
--  常用标准库
-- ============================================================
print()
print("=== 常用标准库 ===")

-- string
print("  string.upper:", string.upper("hello"))
print("  string.format:", string.format("%s 有 %d 个", "玩家", 5))
print("  string.rep:", string.rep("ab", 3))
print("  字符串方法语法:", ("hello"):upper())     -- 用冒号调用

-- table
local t = { 3, 1, 2 }
table.sort(t)
print("  table.sort 后:", table.concat(t, ", "))
table.insert(t, 4)
print("  table.insert 后:", table.concat(t, ", "))
print("  table.remove 返回:", table.remove(t))

-- math
print("  math.floor(3.7):", math.floor(3.7))
print("  math.max(1,5,3):", math.max(1, 5, 3))
print("  math.random:", math.random(1, 100))

-- os
print("  os.time():", os.time())
print("  os.clock():", math.floor(os.clock() * 1000) / 1000)

-- ============================================================
--  错误处理
-- ============================================================
print()
print("=== 错误处理 ===")

-- pcall：捕获错误
local ok, err = pcall(function()
  error("这是一个错误")
end)
print("  pcall 结果:", ok, err)

-- 正常执行时
local ok2, result = pcall(function()
  return "成功"
end)
print("  正常执行:", ok2, result)

-- ============================================================
--  下一步
-- ============================================================
print()
print("=== 下一步 ===")
print("  你已经跑通了第一段 Lua 代码。")
print("  接下来去 03 页学基础语法 —— 重点注意三件事：")
print("    1. 索引从 1 开始（不是 0）")
print("    2. 只有 nil 和 false 是假值（0 是真值！）")
print("    3. 不加 local 就是全局变量")
