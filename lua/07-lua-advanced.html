    一、协程：Lua 的并发原语 ★

    协程不是线程，而是可以暂停与恢复的函数。它的价值在于——用同步的写法表达异步的逻辑。

    -- ============================================================
--  Lua 高级特性：协程、错误处理、GC、环境
-- ============================================================

print("=== 1. 协程（coroutine）：Lua 的并发原语 ===")
-- 协程不是线程，而是【可以暂停与恢复的函数】。
-- 它解决了"状态机要写一堆变量来记住进度"的问题。

local co = coroutine.create(function(a, b)
  print("  协程开始, 参数:", a, b)

  local c = coroutine.yield(a + b)      -- 暂停，交出控制权
  print("  协程恢复, 收到:", c)

  local d = coroutine.yield(c * 2)
  print("  再次恢复, 收到:", d)

  return "完成"
end)

print("  状态:", coroutine.status(co))                   -- suspended
print("  第一次 resume:", coroutine.resume(co, 10, 20))  -- true, 30
print("  状态:", coroutine.status(co))                   -- suspended
print("  第二次 resume:", coroutine.resume(co, 5))       -- true, 10
print("  第三次 resume:", coroutine.resume(co, 99))      -- true, "完成"
print("  状态:", coroutine.status(co))                   -- dead

-- 对已结束的协程 resume
local ok, err = coroutine.resume(co)
print("  对已结束的协程 resume:", ok, err)

-- status 的第四个取值 normal：A 里 resume B 时，B 看 A 就是 "normal"
local A, B
A = coroutine.create(function() coroutine.resume(B) end)
B = coroutine.create(function()
  print("  A resume B 时，B 看 A 的状态:", coroutine.status(A))   -- normal
end)
coroutine.resume(A)

print()
print("=== 2. 用协程写迭代器（最实用的场景）===")
-- 协程配合 coroutine.wrap 可以写非常优雅的遍历器

-- 用协程生成斐波那契数列（无限序列）
local function fibGenerator()
  return coroutine.wrap(function()
    local a, b = 0, 1
    while true do
      coroutine.yield(a)
      a, b = b, a + b
    end
  end)
end

local fib = fibGenerator()
local result = {}
for i = 1, 10 do
  result[#result + 1] = fib()
end
print("  斐波那契前 10 项:", table.concat(result, ", "))

-- 用协程遍历树（比递归+回调更直观）
local tree = {
  value = 1,
  children = {
    { value = 2, children = { { value = 4 }, { value = 5 } } },
    { value = 3, children = { { value = 6 } } },
  }
}

local function traverse(node)
  return coroutine.wrap(function()
    local function walk(n)
      coroutine.yield(n.value)
      if n.children then
        for _, child in ipairs(n.children) do
          walk(child)
        end
      end
    end
    walk(node)
  end)
end

local values = {}
for v in traverse(tree) do
  values[#values + 1] = v
end
print("  树遍历结果:", table.concat(values, ", "))

print("=== 3. 错误处理：pcall / xpcall / error ===")

-- pcall：protected call，捕获异常
local ok2, err2 = pcall(function()
  error("出错了")
end)
print("  pcall 捕获:", ok2, err2)

-- 【重要】error 的第二个参数决定错误位置信息
-- level 0：不添加位置信息
-- level 1（默认）：报错位置是 error 被调用的地方
-- level 2：报错位置是【调用 error 所在函数】的地方
local function badFunc()
  error("内部错误", 2)   -- 报错会指向调用 badFunc 的地方
end
local ok3, err3 = pcall(badFunc)
print("  level 2 的错误信息:", err3)

-- xpcall：带消息处理器（可以获取堆栈）
local ok4, err4 = xpcall(
  function() error("带堆栈的错误") end,
  function(e)
    return "处理过的: " .. tostring(e)
  end
)
print("  xpcall:", ok4, err4)

-- 【5.2+ 新特性】xpcall 支持额外参数
local ok5, err5 = xpcall(
  function(a, b) error("参数是 " .. a .. "," .. b) end,
  function(e) return e end,
  "X", "Y"
)
print("  xpcall 传参:", ok5, err5)

-- 【错误处理的最佳实践】返回 nil + 错误信息 而不是抛异常
local function divideSafely(a, b)
  if b == 0 then
    return nil, "除数不能为 0"
  end
  return a / b
end
local r, e = divideSafely(10, 0)
print("  安全除法:", r, e)

-- 【重要】弱表：对象不被引用时自动回收
local weakCache = setmetatable({}, { __mode = "v" })

-- 强引用存在时，弱表条目保留
local strongRef = { name = "被引用" }
weakCache.a = strongRef
collectgarbage("collect")
print("  强引用还在:", weakCache.a ~= nil)

-- 去掉强引用后，弱表条目消失
strongRef = nil
collectgarbage("collect")
print("  强引用移除后:", weakCache.a)

-- 【实战应用】弱表做对象缓存，不会阻止对象被回收
local function createWeakCache()
  return setmetatable({}, { __mode = "k" })
end

-- _ENV 是 Lua 5.2+ 的机制，取代了旧版的 setfenv/getfenv
local function sandboxed()
  -- 通过 _ENV 限制这个函数能访问的全局变量
  local env = {
    print = print,
    result = nil,
  }

  local f = load([[
    print("  沙箱内可以调用 print")
    result = 42
  ]], "sandbox", "t", env)

  f()
  return env.result
end
print("  沙箱返回值:", sandboxed())

-- __gc：对象被回收前调用（需要在元表里【提前】设置）
local gcLog = {}
do
  local mt = {
    __gc = function(obj)
      gcLog[#gcLog + 1] = obj.name
    end
  }
  -- 【注意】必须在 setmetatable 时元表就已经有 __gc
  local a = setmetatable({ name = "objA" }, mt)

end
collectgarbage("collect")
print("  __gc 触发:", table.concat(gcLog, ", "))

print("=== 9. 实用工具函数 ===")

-- 深拷贝（处理循环引用）
local function deepCopy(obj, seen)
  if type(obj) ~= "table" then return obj end
  seen = seen or {}
  if seen[obj] then return seen[obj] end

  local copy = {}
  seen[obj] = copy
  for k, v in pairs(obj) do
    copy[deepCopy(k, seen)] = deepCopy(v, seen)
  end
  return setmetatable(copy, getmetatable(obj))
end

local original = { a = 1, b = { c = 2 } }
original.self = original                   -- 循环引用
local copied = deepCopy(original)
print("  深拷贝:", copied.a, copied.b.c, "循环引用:", copied.self == copied)

-- 表序列化（用于调试输出）
local function serialize(t, indent)
  indent = indent or ""
  if type(t) ~= "table" then
    if type(t) == "string" then return string.format("%q", t) end
    return tostring(t)
  end

  local parts = {}
  for k, v in pairs(t) do
    parts[#parts + 1] = indent .. "  [" .. serialize(k) .. "] = " .. serialize(v, indent .. "  ")
  end
  return "{\n" .. table.concat(parts, ",\n") .. "\n" .. indent .. "}"
end

print("  序列化:", serialize({ x = 1, name = "test" }))

  local args = { ... }

  end
end

    函数作用返回值
    coroutine.create创建协程协程对象
    coroutine.resume启动/恢复true/false, 返回值
    coroutine.yield暂停并交出值—
    coroutine.status查询状态4 种取值：running（自己）/ suspended（没跑或让出）/ normal（A 里 resume B 时，B 看 A）/ dead
    coroutine.wrap包装成函数直接调用，出错会抛出
    resume 不会抛异常，wrap 会 coroutine.resume(co) 返回 false, errmsg——错误被捕获，不会中断调用方。
    coroutine.wrap(f) 返回一个函数，调用它时如果协程内部出错，错误会直接抛出。

    怎么选：
    · 需要处理错误 → 用 resume（能拿到错误信息）
    · 写迭代器、确信不会出错 → 用 wrap（代码更简洁）
    协程最实用的场景：写迭代器 看本页代码里的两个例子：
    ① 无限序列（斐波那契）——协程可以暂停在任意位置，天然适合生成无限序列。
    ② 树的递归遍历——用递归写逻辑，用协程把递归结果变成线性序列。

    对比一下：如果不用协程，这两个都要自己维护状态变量。协程把"状态"变成了"代码位置"。

    二、错误处理

    Lua 的错误处理有两种风格，混用会导致灾难。

    风格写法适用
    抛异常error() + pcall真正的意外
    返回 nil+msgreturn nil, "错误"预期内的失败
    error 的第二个参数（level） 默认 level = 1：错误信息里包含调用 error 的那一行。
    level = 2：错误信息指向调用当前函数的那一行——写工具函数时应该用这个。
    level = 0：不添加位置信息，错误信息就是纯文本。

    为什么 level = 2 更合适：如果你写的 validateNumber(n) 内部 error("参数非法")，用默认 level 会报错在 validateNumber 内部——但真正的问题是调用方传了错参数。
    xpcall 能拿到堆栈，pcall 不能 pcall(f) 只给你错误消息，看不到调用栈。
    xpcall(f, handler) 让你在 handler 里用 debug.traceback 获取完整堆栈。

    实战建议：框架的顶层调用应该用 xpcall + debug.traceback——出问题时能看到完整调用链，而不是一行孤立的错误信息。
    Lua 5.2+ 的 xpcall 还支持给函数传参，避免用闭包包装。

    最佳实践：预期内的失败（文件不存在、参数非法）用 return nil, msg；真正的意外（数组越界、类型错误）才用 error。

    三、垃圾回收与弱表 ★

    弱表：不阻止对象被回收的容器 __mode = "v"：值是弱引用——值不被其他地方引用时，这个条目自动消失。
    __mode = "k"：键是弱引用——键对象被回收时，整个条目消失。
    __mode = "kv"：两者都是弱引用。

    最实用的场景：给对象挂"附加数据"而不阻止它被回收。
    例如：local extraData = setmetatable({}, {__mode = "k"})
    extraData[someEntity] = {...}
    当 someEntity 被回收时，对应的附加数据自动清理——不需要手动管理。

    这正是"缓存"的一种实现（见算法站页 27）。
    collectgarbage 不要频繁调用 collectgarbage("collect") 会触发一次完整的 GC 循环，是同步阻塞的。
    在游戏的帧循环里调用它，会造成明显卡顿。

    什么时候可以调：
    · 加载界面（玩家不在意这点延迟）
    · 场景切换时
    · 内存告警时的兜底

    什么时候不要调：Update 里、战斗过程中、任何帧率敏感的位置。

    四、环境与沙箱

    _ENV 与沙箱 Lua 5.2+ 用 _ENV 取代了旧版的 setfenv。
    每个函数都有一个 _ENV 上值，所有全局变量访问实际是在访问 _ENV 的字段。

    沙箱的用途：
    · 让配置脚本只能访问白名单内的 API
    · 热更脚本的安全加载（防止恶意代码）
    · 玩家自定义脚本（UGC）

    做法：load(code, name, "t", customEnv) —— 第四个参数指定环境表。
    这样脚本里的 print、os.execute 等都会去 customEnv 里找，找不到就是 nil。
