local ResourceManager = {}
ResourceManager.__index = ResourceManager

function ResourceManager.new()
  return setmetatable({
    cache = {},           -- path -> { handle, refCount, data }
    loadingQueue = {},    -- 异步加载队列
    totalLoads = 0,       -- 统计：真实加载次数
    cacheHits = 0,        -- 统计：缓存命中次数
  }, ResourceManager)
end

-- 模拟真实加载（实际项目中这里调 xLua 的 C# 接口）
local function realLoad(path)
  return {
    path = path,
    data = "资源数据:" .. path,
    size = #path * 100,
  }
end

function ResourceManager:load(path)
  local entry = self.cache[path]

  if entry then
    -- 缓存命中
    entry.refCount = entry.refCount + 1
    self.cacheHits = self.cacheHits + 1
    return entry.data
  end

  -- 真实加载
  self.totalLoads = self.totalLoads + 1
  local data = realLoad(path)

  self.cache[path] = {
    data = data,
    refCount = 1,
  }

  return data
end

function ResourceManager:release(path)
  local entry = self.cache[path]
  if not entry then
    error("释放了未加载的资源: " .. tostring(path))
  end

  entry.refCount = entry.refCount - 1

  if entry.refCount <= 0 then
    -- 引用计数归零，真正卸载
    self.cache[path] = nil
    return true          -- 表示真正卸载了
  end

  return false           -- 还有引用，不卸载
end

function ResourceManager:stats()
  local cachedCount = 0
  for _ in pairs(self.cache) do cachedCount = cachedCount + 1 end

  return {
    cached = cachedCount,
    loads = self.totalLoads,
    hits = self.cacheHits,
    hitRate = (self.totalLoads + self.cacheHits) > 0
              and self.cacheHits / (self.totalLoads + self.cacheHits) or 0,
  }
end

local TimerManager = {}
TimerManager.__index = TimerManager

function TimerManager.new()
  return setmetatable({
    heap = {},            -- 最小堆，按 fireTime 排序
    nextId = 1,
    currentTime = 0,
  }, TimerManager)
end

-- 堆操作
function TimerManager:_siftUp(i)
  local heap = self.heap
  while i > 1 do
    local parent = math.floor(i / 2)
    if heap[parent].fireTime <= heap[i].fireTime then break end
    heap[parent], heap[i] = heap[i], heap[parent]
    i = parent
  end
end

function TimerManager:_siftDown(i)
  local heap = self.heap
  local n = #heap
  while true do
    local smallest = i
    local l, r = i * 2, i * 2 + 1
    if l <= n and heap[l].fireTime < heap[smallest].fireTime then smallest = l end
    if r <= n and heap[r].fireTime < heap[smallest].fireTime then smallest = r end
    if smallest == i then break end
    heap[i], heap[smallest] = heap[smallest], heap[i]
    i = smallest
  end
end

-- 添加定时器
function TimerManager:schedule(delay, callback, repeatCount, owner)
  local timer = {
    id = self.nextId,
    fireTime = self.currentTime + delay,
    interval = delay,
    callback = callback,
    repeatCount = repeatCount or 1,      -- -1 表示无限
    firedCount = 0,
    owner = owner,
  }
  self.nextId = self.nextId + 1

  self.heap[#self.heap + 1] = timer
  self:_siftUp(#self.heap)

  return timer.id
end

-- 取消定时器（标记删除，不立即从堆移除）
function TimerManager:cancel(id)
  -- 【设计选择】不立即从堆移除，而是标记
  -- 因为从堆中间删除元素是 O(n)
  for i = 1, #self.heap do
    if self.heap[i].id == id then
      self.heap[i].cancelled = true
      return true
    end
  end
  return false
end

-- 每帧更新
function TimerManager:update(deltaTime)
  self.currentTime = self.currentTime + deltaTime

  local fired = {}

  -- 取出所有已到期的定时器
  while #self.heap > 0 do
    local top = self.heap[1]
    if top.fireTime > self.currentTime then break end

    -- 弹出堆顶
    self.heap[1] = self.heap[#self.heap]
    self.heap[#self.heap] = nil
    if #self.heap > 0 then self:_siftDown(1) end

    -- 跳过已取消的
    if not top.cancelled then
      fired[#fired + 1] = top

      -- 重复定时器：重新入堆
      if top.repeatCount == -1 or top.firedCount < top.repeatCount - 1 then
        top.firedCount = top.firedCount + 1
        top.fireTime = top.fireTime + top.interval
        self.heap[#self.heap + 1] = top
        self:_siftUp(#self.heap)
      end
    end
  end

  -- 【重要】先收集再执行 —— 避免在回调里修改堆导致问题
  for _, timer in ipairs(fired) do
    timer.callback(timer.id, timer.owner)
  end

  return #fired
end

function TimerManager:pendingCount()
  local n = 0
  for i = 1, #self.heap do
    if not self.heap[i].cancelled then n = n + 1 end
  end
  return n
end

local ConfigManager = {}
ConfigManager.__index = ConfigManager

function ConfigManager.new()
  local self = setmetatable({
    tables = {},          -- name -> { byId, byName, all }
  }, ConfigManager)

  -- 用元表实现只读（见页 04）
  return setmetatable(self, {
    __index = ConfigManager,
    __newindex = function(t, k, v)
      if k == "tables" then
        rawset(t, k, v)
      else
        error("配置管理器是只读的，不能修改字段: " .. tostring(k), 2)
      end
    end,
  })
end

-- 加载配置并建立索引
function ConfigManager:loadTable(name, data, idField, nameField)
  local tableData = {
    all = data,
    byId = {},
    byName = {},
    byType = {},
  }

  -- 【关键】一次性建立所有常用索引
  for _, item in ipairs(data) do
    if idField and item[idField] then
      tableData.byId[item[idField]] = item
    end
    if nameField and item[nameField] then
      tableData.byName[item[nameField]] = item
    end
    if item.type then
      tableData.byType[item.type] = tableData.byType[item.type] or {}
      -- 【易错点】这里必须取到【类型对应的那个数组】，再追加
      -- 写错成 table[...] 会静默失败（往全局 table 库上塞元素）
      local bucket = tableData.byType[item.type]
      bucket[#bucket + 1] = item
    end
  end

  self.tables[name] = tableData
  return tableData
end

function ConfigManager:getById(tableName, id)
  local t = self.tables[tableName]
  return t and t.byId[id]
end

function ConfigManager:getByName(tableName, name)
  local t = self.tables[tableName]
  return t and t.byName[name]
end

function ConfigManager:getByType(tableName, itemType)
  local t = self.tables[tableName]
  return t and t.byType[itemType] or {}
end

function ConfigManager:getAll(tableName)
  local t = self.tables[tableName]
  return t and t.all or {}
end

    二、定时器模块

    定时器的两种实现 方案一：每帧遍历（简单） · 维护一个列表，每帧检查每个定时器是否到期 · O(n) 每帧——定时器少时够用 方案二：最小堆（高效） · 用堆按"触发时间"排序（见算法站页 07） · 每帧只需检查堆顶是否到期 · O(log n) 插入，O(1) 检查 选择依据： · 定时器 < 100 个 → 方案一（更简单） · 定时器 > 1000 个（比如每个 Buff 一个）→ 方案二 实战中，方案二更常用——因为服务器端的定时任务数量往往很大。
    定时器最容易出的三个 bug ① 定时器泄漏：对象销毁了，但定时器还在，回调时访问已销毁对象。 解决：把定时器绑定到对象，对象销毁时批量清理。 ② 同一帧触发多次：如果帧间隔大于定时器周期，可能一帧触发多次。 解决：明确设计——是"每次到期触发一次"还是"补足所有遗漏的触发"。 ③ 在回调里删自己：遍历中修改列表 → 索引错位。 解决：用快照遍历或标记删除（和页 13 的事件系统一样）。

    三、配置表模块

    配置表的三个设计要点 ① 索引优于遍历 加载配置时建立常用索引： · byId[id] = config · byName[name] = config · byType[type] = { config1, config2 } 把 O(n) 的遍历变成 O(1) 的查找。 ② 只读保护 配置加载后应该只读——防止业务代码意外修改。 用元表的 __newindex 实现（见页 04）。 ③ 支持热重载 开发期能重新加载配置而不重启游戏。这能大幅提升调试效率。 实现：重新解析文件 + 重建索引 + 通知订阅者刷新。
