local function fileEntry(path, content)
  -- 真实项目里用 MD5；这里用长度 + 简单哈希模拟
  local hash = 0
  for i = 1, #content do
    hash = (hash * 31 + content:byte(i)) % 2147483647
  end
  return {
    path = path,
    size = #content,
    hash = hash,
  }
end

local function diff(localIdx, serverIdx)
  local toDownload = {}   -- 新增或更新
  local toDelete = {}     -- 服务器已删
  local unchanged = {}

  -- 找出需要下载的
  for path, serverFile in pairs(serverIdx) do
    local localFile = localIdx[path]

    if not localFile then
      toDownload[#toDownload + 1] = { path = path, reason = "新增" }
    elseif localFile.hash ~= serverFile.hash then
      toDownload[#toDownload + 1] = { path = path, reason = "更新" }
    else
      unchanged[#unchanged + 1] = path
    end
  end

  -- 找出需要删除的
  for path in pairs(localIdx) do
    if not serverIdx[path] then
      toDelete[#toDelete + 1] = path
    end
  end

  -- 排序，让输出稳定
  table.sort(toDownload, function(a, b) return a.path < b.path end)
  table.sort(toDelete)

  return toDownload, toDelete, unchanged
end

local FileSystem = {
  dirs = {},          -- 模拟文件系统：dirName -> { files }
}

function FileSystem.mkdir(name)
  FileSystem.dirs[name] = FileSystem.dirs[name] or {}
  return FileSystem.dirs[name]
end

function FileSystem.write(dir, path, content)
  FileSystem.mkdir(dir)
  FileSystem.dirs[dir][path] = content
end

function FileSystem.exists(dir, path)
  return FileSystem.dirs[dir] and FileSystem.dirs[dir][path] ~= nil
end

-- 热更管理器
local HotUpdateManager = {}
HotUpdateManager.__index = HotUpdateManager

function HotUpdateManager.new()
  return setmetatable({
    currentVersion = "v1",       -- 当前生效的版本
    stagingVersion = nil,        -- 正在下载的版本
    activeDir = "scripts_v1",    -- 当前使用的目录
    stagingDir = nil,            -- 临时目录
    completedFiles = {},         -- 已完成的文件
    totalFiles = 0,
    versionHistory = {},    -- 历史版本列表（用于回滚）
    log = {},
  }, HotUpdateManager)
end

function HotUpdateManager:_log(msg)
  self.log[#self.log + 1] = msg
  print("    " .. msg)
end

-- 开始下载
function HotUpdateManager:beginDownload(version, fileCount)
  self.stagingVersion = version
  self.stagingDir = "scripts_" .. version .. "_staging"
  self.totalFiles = fileCount
  self.completedFiles = {}

  FileSystem.mkdir(self.stagingDir)   -- 创建临时目录
  self:_log("开始下载 " .. version .. " 到 " .. self.stagingDir .. "（临时目录）")
end

-- 下载单个文件
function HotUpdateManager:downloadFile(path, content)
  FileSystem.write(self.stagingDir, path, content)
  self.completedFiles[#self.completedFiles + 1] = path
  self:_log(string.format("  下载 %s (%d/%d)", path, #self.completedFiles, self.totalFiles))
end

-- 校验（模拟 MD5 校验）
function HotUpdateManager:verify()
  if #self.completedFiles < self.totalFiles then
    self:_log(string.format("  校验失败：只完成 %d/%d 个文件",
      #self.completedFiles, self.totalFiles))
    return false
  end
  self:_log("校验通过：所有文件完整")
  return true
end

-- 【关键】原子切换
function HotUpdateManager:commit()
  if not self:verify() then
    self:_log("【拒绝切换】校验未通过，保持在旧版本")
    return false
  end

  -- 保留旧版本目录（用于回滚）
  local oldDir = self.activeDir
  local oldVersion = self.currentVersion

  -- 【关键】把当前版本压入历史列表
  self.versionHistory[#self.versionHistory + 1] = {
    dir = self.activeDir,
    version = oldVersion,
  }

  -- 把 staging 目录重命名为正式目录（模拟原子重命名）
  local finalDir = self.stagingDir:gsub("_staging$", "")
  FileSystem.dirs[finalDir] = FileSystem.dirs[self.stagingDir]
  FileSystem.dirs[self.stagingDir] = nil

  -- 切换
  self.activeDir = finalDir
  self.currentVersion = self.stagingVersion
  self.stagingDir = nil
  self.stagingVersion = nil

  self:_log("【切换成功】")
  self:_log("  当前版本: " .. self.currentVersion)
  self:_log("  使用目录: " .. self.activeDir .. "（由 staging 重命名而来）")
  self:_log("  旧版本保留在: " .. oldDir .. "（可回滚）")

  return true
end

-- 回滚
-- 【设计】用历史版本列表，而不是扫描目录名（staging 与正式目录命名不同）
function HotUpdateManager:rollback()
  if #self.versionHistory == 0 then
    self:_log("【无法回滚】没有历史版本")
    return false
  end

  local prev = table.remove(self.versionHistory)

  self:_log("【回滚】")
  self:_log("  从 " .. self.activeDir .. " 切回 " .. prev.dir)
  self.activeDir = prev.dir
  self.currentVersion = prev.version
  self:_log("  当前版本: " .. self.currentVersion)

  return true
end

local function checkVersionCompat(clientVer, serverVer, protocolVer, minProtocol)
  -- 规则 1：协议版本必须匹配
  if protocolVer < minProtocol then
    return false, "协议版本过旧，必须更新客户端"
  end

  -- 规则 2：脚本版本可以不同，但差距不能太大
  local cMajor = tonumber(clientVer:match("v(%d+)"))
  local sMajor = tonumber(serverVer:match("v(%d+)"))

  if sMajor - cMajor > 2 then
    return false, "客户端版本过旧，请更新"
  end

  return true, "版本兼容"
end

    二、四个上线后才会遇到的问题 ★★★

    问题一：热更后玩家卡在加载界面 原因：下载了一半的脚本，校验失败，但客户端已经用新脚本启动了。
    典型场景：玩家弱网下载到 80% 退出，下次启动读到不完整的文件。

    解决方案：
    ① 下载到临时目录，全部完成后再切换（原子操作）
    ② 每个文件都要 MD5 校验
    ③ 保留完整的旧版本，新版本有问题就切回去

    这三个措施缺一个都可能出事故。
    问题二：部分玩家没更新，联机出问题 原因：热更不是强制的，有的玩家更新了，有的没有。
    后果：两人联机时协议不一致 → 数据错乱或崩溃。

    解决方案：
    ① 登录时强制校验版本——版本不一致就要求更新
    ② 协议版本独立——协议变了必须是强更
    ③ 服务器同时支持新旧协议（过渡期）

    这一条在页 24（API 速查）会再次出现——它属于"上线检查清单"的必备项。
    问题三：热更把游戏更坏了，怎么回滚 这是最要命的场景——推了一个有 bug 的脚本，所有玩家立刻受影响。

    必须有回滚能力：
    ① 保留上一版本的完整脚本（不要覆盖）
    ② 服务器切换"当前版本号"即可回滚（秒级）
    ③ 客户端下次启动自动拉回旧版本

    更保险的做法：
    · 灰度发布——先推 1% 玩家，观察 30 分钟
    · 错误率监控——异常自动触发回滚
    · 一键停更——紧急情况下暂停所有热更

    我在 C# 站页 16 说过一句话：没有回滚方案的热更，等于把线上项目放在没有刹车的车上。这里再重复一次——因为它值得重复。
    问题四：状态残留导致疑难 bug 场景：热更后重载了某模块，但它持有的旧状态还在。
    表现：
    · 玩家数据错乱（用了旧缓存）
    · 界面显示旧数据
    · 诡异且难以复现

    解决方案（回顾页 15）：
    ① 模块提供 reset()，热更后统一调用
    ② 热更时做状态迁移——保存旧状态，注入新模块
    ③ 把状态与逻辑分离——模块只管逻辑，数据放在专门的 Data 层

    方案 ③ 是最彻底的——它让模块变成"无状态的"，热更时不需要担心状态问题。这是成熟框架的普遍做法。

    三、上线检查清单

    检查项为什么不做的后果
    文件校验防止下载损坏热更后无法启动
    原子切换防止半包可用卡加载界面
    版本强制校验防止协议不一致联机数据错乱
    完整回滚出问题能恢复线上事故无法止损
    灰度发布小范围验证全量玩家受影响
    错误上报及时发现问题靠玩家反馈
    停更开关紧急暂停持续推送错误版本
    把这些变成工具 不要指望"每次都记得检查"——人会忘。
    做法：把这些检查写成自动化脚本：
    · 打包后自动校验所有文件
    · 上传前自动生成清单
    · 发布后自动监控错误率

    这回到 C# 站页 19 讲的原则：同一个错误犯过两次，就该写工具防止第三次。
