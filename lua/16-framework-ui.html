local function class(name, base)
  local c = {}
  c.__name = name
  c.__base = base
  c.__index = c
  if base then
    setmetatable(c, { __index = base })
    c.super = base
  end
  function c.new(...)
    local obj = setmetatable({}, c)
    if obj.ctor then obj:ctor(...) end
    return obj
  end
  return c
end

local ViewBase = class("ViewBase")

function ViewBase:ctor(name, viewType)
  self.name = name
  self.viewType = viewType or ViewType.FULL
  self.isOpen = false
  self.isCreated = false
  self.createdTime = nil
  self.openCount = 0        -- 统计打开次数
  self.subscriptions = {}   -- 记录本界面订阅的事件
  self.dirty = false        -- 脏标记（数据变了但没刷新）
end

function ViewBase:onClose()
  -- 【关键】取消所有订阅，防止"界面关闭后还在刷新"
  for _, sub in ipairs(self.subscriptions) do
    sub.eventManager:off(sub.eventName, sub.handler)
  end
  self.subscriptions = {}
end

-- 便捷方法：订阅事件（自动记录，便于关闭时取消）
function ViewBase:subscribe(eventManager, eventName, handler)
  eventManager:on(eventName, handler, self)
  self.subscriptions[#self.subscriptions + 1] = {
    eventManager = eventManager,
    eventName = eventName,
    handler = handler,
  }
end

-- 标记数据变化（惰性刷新）
function ViewBase:markDirty()
  if not self.isOpen then return end   -- 没打开就不需要刷新
  self.dirty = true
end

--  3. 事件系统（简化版，来自页 13）
-- ============================================================
local EventManager = {}
EventManager.__index = EventManager

function EventManager.new()
  return setmetatable({ handlers = {} }, EventManager)
end

function EventManager:on(eventName, handler, target)
  self.handlers[eventName] = self.handlers[eventName] or {}
  local list = self.handlers[eventName]
  list[#list + 1] = { fn = handler, target = target }
  return handler
end

function EventManager:off(eventName, handler)
  local list = self.handlers[eventName]
  if not list then return end
  for i = #list, 1, -1 do
    if list[i].fn == handler then table.remove(list, i) end
  end
end

function EventManager:emit(eventName, ...)
  local list = self.handlers[eventName]
  if not list then return 0 end

  -- 快照遍历
  local snapshot = {}
  for i = 1, #list do snapshot[i] = list[i] end

  local count = 0
  for _, item in ipairs(snapshot) do
    if item.target then
      item.fn(item.target, ...)
    else
      item.fn(...)
    end
    count = count + 1
  end
  return count
end

function EventManager:listenerCount(eventName)
  return self.handlers[eventName] and #self.handlers[eventName] or 0
end

--  4. UI 管理器（界面栈）
-- ============================================================
local UIManager = {}
UIManager.__index = UIManager

function UIManager.new(eventManager)
  return setmetatable({
    eventManager = eventManager,
    stack = {},              -- 界面栈
    views = {},              -- 所有已创建的界面（name -> view）
    popups = {},             -- 单独的弹窗列表（不属于栈）
    log = {},                -- 操作日志（便于观察）
    refreshCount = 0,
  }, UIManager)
end

function UIManager:_log(msg)
  self.log[#self.log + 1] = msg
  print("    " .. msg)
end

-- 注册界面（只创建一次实例）
function UIManager:register(view)
  self.views[view.name] = view
  return view
end

-- 打开界面
function UIManager:open(name, ...)
  local view = self.views[name]
  if not view then
    error("界面未注册: " .. tostring(name))
  end

  -- 首次打开：执行 onCreate
  if not view.isCreated then
    view.isCreated = true
    view.createdTime = os.clock()
    view:onCreate()
    self:_log("[OnCreate] " .. name)
  end

  -- 已经是打开的，不重复打开
  if view.isOpen then
    self:_log("界面已打开，忽略: " .. name)
    return view
  end

  -- 【关键】全屏界面：隐藏栈顶的界面
  if view.viewType == ViewType.FULL then
    -- 【重要】如果这个界面已经躺在栈里（只是被上层盖住了），
    -- 说明玩家是在"回退"到它，而不是"新开"它 —— 必须先把上面几层关掉，
    -- 否则同一个界面会被重复压栈，返回键的层级就全乱了。
    for i = #self.stack, 1, -1 do
      if self.stack[i] == view then
        while #self.stack > i do
          local above = table.remove(self.stack)
          if above.isOpen then
            above.isOpen = false
            above:onClose()
            self:_log("[退回] 关闭上层 " .. above.name)
          end
        end
        break
      end
    end

    local top = self.stack[#self.stack]
    if top and top ~= view and top.isOpen then
      top.isOpen = false
      top:onClose()
      self:_log("[隐藏] " .. top.name .. "（被 " .. name .. " 覆盖）")
    end
    -- 已经在栈里就不再 push
    if self.stack[#self.stack] ~= view then
      self.stack[#self.stack + 1] = view
    end
  elseif view.viewType == ViewType.POPUP then
    -- 弹窗不入栈，单独管理
    self.popups[#self.popups + 1] = view
  end

  view.isOpen = true
  view.openCount = view.openCount + 1
  view:onOpen()
  self:_log("[OnOpen] " .. name)

  if view.dirty then
    view:onRefresh()
    self.refreshCount = self.refreshCount + 1
    self:_log("[OnRefresh] " .. name)
  end

  return view
end

-- 关闭（返回上一级）
function UIManager:close(name)
  local view = self.views[name]
  if not view or not view.isOpen then return false end

  view.isOpen = false
  view:onClose()
  self:_log("[OnClose] " .. name)

  -- 从栈里移除
  if view.viewType == ViewType.FULL then
    for i = #self.stack, 1, -1 do
      if self.stack[i] == view then
        table.remove(self.stack, i)
        break
      end
    end
  else
    for i = #self.popups, 1, -1 do
      if self.popups[i] == view then
        table.remove(self.popups, i)
        break
      end
    end
  end

  -- 【关键】恢复被覆盖的全屏界面
  local top = self.stack[#self.stack]
  if top and not top.isOpen then
    top.isOpen = true
    top:onOpen()
    self:_log("[恢复] " .. top.name)
  end

  return true
end

-- 返回键处理
function UIManager:onBackPressed()
  -- 优先关闭弹窗（弹窗在最上层）
  if #self.popups > 0 then
    local topPopup = self.popups[#self.popups]
    self:_log("返回键 -> 关闭弹窗 " .. topPopup.name)
    self:close(topPopup.name)
    return true
  end

  -- 再关全屏界面
  if #self.stack > 1 then    -- 保留最底层的界面
    local top = self.stack[#self.stack]
    self:_log("返回键 -> 关闭界面 " .. top.name)
    self:close(top.name)
    return true
  end

  self:_log("返回键 -> 已在最底层，忽略（或弹退出确认）")
  return false
end

-- 每帧刷新（惰性刷新）
function UIManager:update()
  for _, view in pairs(self.views) do
    if view.isOpen and view.dirty then
      view:onRefresh()
      self.refreshCount = self.refreshCount + 1
      self:_log("[OnRefresh] " .. view.name .. "（数据变化）")
    end
  end
end

function UIManager:destroyAll()
  for name, view in pairs(self.views) do
    view:onDestroy()
    self:_log("[OnDestroy] " .. name)
  end
  self.views = {}
  self.stack = {}
  self.popups = {}
end

    二、界面栈：为什么一定是栈 ★★★

    栈天然解决了"返回上一级" 场景：主界面 → 背包 → 装备详情 → 使用确认
    用栈的做法：push(主界面) → push(背包) → …
    按返回键 pop()——自动回到上一层。

    不用栈的问题：
    · 你需要自己记住"打开详情前是什么界面"
    · 处理任意层级的嵌套
    · 返回顺序可能出错

    栈的三种操作：
    · Push——打开新界面（旧界面可能隐藏但保留）
    · Pop——返回上一级
    · Replace——替换当前界面（不增加层级）
    栈最容易写错的一处：重复压栈 场景：主界面 → 背包，然后从背包里点「回主城」——又 open 了一次主界面。

    天真的实现（只看 view.isOpen）：主界面此刻是关闭状态（被背包盖住时已经 OnClose 过）， 于是它被再压一次栈，栈变成 [主城, 行囊, 主城]。
    后果：栈深度虚增、返回键的层级全乱、同一个界面的 OnCreate 只会跑一次但 OnOpen 会被跑两轮 （订阅如果没清干净就会重复注册）。

    正确做法：open 之前先在栈里找一遍——
    · 找到了 → 把它在栈里的上层全部弹出并关闭，然后恢复它，不要再 push
    · 没找到 → 才走正常的压栈流程

    本页代码样本的第 8 段就是这段逻辑的实测演示（修复后栈里不会出现重复项）。
    三种界面类型要区分 ① 全屏界面（如主界面、背包）
    · 打开时隐藏下层
    · 返回时恢复下层

    ② 弹窗（如确认框、提示）
    · 打开时下层仍然可见（半透明遮罩）
    · 可以同时存在多个

    ③ 飘字/Toast（如获得道具提示）
    · 不入栈，自动消失
    · 可以有多个同时显示

    混淆这三种类型会导致：返回键行为混乱、界面叠在一起、飘字挡住按钮。这是 UI 框架设计时最容易出错的地方。

    三、数据绑定：两种模式

    模式做法优缺点
    拉模式（Poll）每帧或定时检查数据变化简单但浪费
    推模式（Push）数据变化时通知 UI高效，主流
    混合事件通知 + 惰性刷新最优
    推模式的经典问题：界面已关闭但还在刷新 问题：界面关闭了，但事件订阅没取消。
    数据变化时回调仍然执行——访问已销毁的 UI 对象 → 报错。

    这是 Lua UI 框架最常见的 bug。
    解决方法：
    ① OnClose 时取消所有订阅（必须成对）
    ② 在回调里检查界面是否还活着
    ③ 用弱引用表管理订阅（见页 07）

    页 13 的事件系统必须提供"按所有者批量取消订阅"的接口——这是框架设计要提前考虑的。
