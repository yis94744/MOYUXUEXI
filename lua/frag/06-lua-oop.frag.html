06 · 面向对象与继承

Lua 没有 class 关键字。所有面向对象都是用 table + 元表手工实现的。这一页不只讲"怎么实现"，更要讲清为什么框架都那样写——以及那些写法背后的取舍。

这一页在路线中的位置 上一页（05）的闭包和这一页的元表，是 Lua 实现封装的两种手段——这一页会对比它们的优劣。
关键衔接：页 04 学的 __index 是继承的实现机制，这一页把它包装成可复用的 class 函数。
下一页（07）高级特性：协程、错误处理、GC。那里的状态机与错误对象都会用到这一页的 class。
实战意义：你在项目里遇到的 Lua 框架（xLua、ToLua、各种自研框架）都以这一页的 class 实现为基础。看懂它，才能看懂框架源码。

一、用元表实现 class ★★★

-- ============================================================
--  Lua 的面向对象：用元表实现 class ★★★
-- ============================================================
--  Lua 没有 class 关键字。所有「面向对象」都是用 table + 元表实现的。
--  理解这一页，你才能看懂任何 Lua 框架的源码。

print("=== 1. 最基础的 class 实现（框架里的标准写法）===")

-- 【核心套路】三件套：Class 表 + __index = 自己 + new 方法
local Animal = {}
Animal.__index = Animal                      -- 关键：让实例能查到自己类

function Animal.new(name, age)
  local self = setmetatable({}, Animal)      -- 创建实例并绑定元表
  self.name = name
  self.age = age
  return self
end

function Animal:getName()
  return self.name
end
function Animal:getAge()
  return self.age
end
function Animal:speak()
  return self.name .. " 发出声音"
end

-- 【语法糖说明】
-- function Animal:getName() 等价于 function Animal.getName(self)
-- 冒号会自动把调用者作为第一个参数 self 传入

local dog = Animal.new("旺财", 3)
print("实例:", dog:getName(), dog:getAge())
print("调用:", dog:speak())

print()
print("=== 2. 继承（两层元表链）===")
-- 继承的本质：让子类的 __index 指向父类

local function class(className, superClass)
  local cls = {}
  cls.__index = cls

  -- 建立继承关系
  if superClass then
    setmetatable(cls, { __index = superClass })   -- 静态成员继承
    cls.super = superClass                        -- 方便调用父类方法
  end

  cls.className = className

  -- new 方法
  function cls.new(...)
    local instance = setmetatable({}, cls)
    -- 如果有构造方法就调用
    if instance.ctor then
      instance:ctor(...)
    end
    return instance
  end

  return cls
end

-- 定义基类
local Base = class("Base")

function Base:ctor(name)
  self.name = name
end

function Base:describe()
  return "我是 " .. self.name
end

-- 定义子类（继承 Base）
local Derived = class("Derived", Base)

function Derived:ctor(name, extra)
  -- 【重要】调用父类构造需要通过 self.super
  Base.ctor(self, name)
  self.extra = extra
end

function Derived:describe()
  -- 调用父类同名方法
  return Base.describe(self) .. "，额外：" .. self.extra
end

function Derived:onlyInDerived()
  return "只有子类有这个方法"
end

local d = Derived.new("测试", "附加数据")
print("子类实例:", d:describe())
print("子类独有:", d:onlyInDerived())
print("未定义的方法:", d.getAge)            -- nil（Base 没有 getAge）
print("className:", d.className)            -- 通过元表链查到

print()
print("=== 3. 多级继承（链式 __index）===")
local L1 = class("L1")
function L1:who() return "L1" end

local L2 = class("L2", L1)
function L2:who() return "L2 -> " .. L1.who(self) end

local L3 = class("L3", L2)
function L3:who() return "L3 -> " .. L2.who(self) end

print(L3.new():who())          -- L3 -> L2 -> L1

local shapes = {
  { name = "圆",   area = function(self) return math.pi * self.r ^ 2 end, r = 2 },
  { name = "方",   area = function(self) return self.w * self.h end, w = 3, h = 4 },

-- 统一调用（这就是多态）
for _, shape in ipairs(shapes) do
  print(string.format("%s 面积 = %.2f", shape.name, shape:area()))

print("=== 5. 类方法与实例方法的区别 ===")
local Counter = {}
Counter.__index = Counter

Counter.total = 0                              -- 类变量（所有实例共享）

function Counter.new()
  return setmetatable({ count = 0 }, Counter)
end

function Counter:increment()
  self.count = self.count + 1
  Counter.total = Counter.total + 1            -- 修改类变量
  return self.count
end

-- 静态方法（不需要实例）
function Counter.getTotal()
  return Counter.total
end

local c1 = Counter.new()
local c2 = Counter.new()
c1:increment()
c1:increment()
c2:increment()

print("c1.count =", c1.count)                  -- 2
print("c2.count =", c2.count)                  -- 1
print("类变量 total =", Counter.getTotal())     -- 3

print("=== 6. 私有成员（闭包实现）===")
local function createBankAccount(initial)
  -- 闭包捕获的局部变量，外部【完全无法访问】
  local balance = initial
  local transactionLog = {}

  local self = {}

  function self.deposit(amount)
    balance = balance + amount
    transactionLog[#transactionLog + 1] = { type = "存入", amount = amount }
    return balance
  end

  function self.withdraw(amount)
    if amount > balance then
      return false, "余额不足"
    end
    balance = balance - amount
    transactionLog[#transactionLog + 1] = { type = "取出", amount = amount }
    return true, balance
  end

  function self.getBalance() return balance end
  function self.getLog() return transactionLog end

  return self
end

local acc = createBankAccount(1000)
acc.deposit(500)
local ok, remaining = acc.withdraw(300)
print("取款结果:", ok, "余额:", remaining)
print("余额无法直接读取:", acc.balance)        -- nil
print("流水记录数:", #acc.getLog())

print()
print("=== 7. 封闭类（sealed）与访问控制 ===")
local Config = {}
Config.__index = Config

function Config.new()
  return setmetatable({}, Config)
end

-- 用 __newindex 阻止外部添加新字段
function Config:lock()
  return setmetatable(self, {
    __index = getmetatable(self).__index,
    __newindex = function()
      error("该配置已锁定，不允许修改", 2)
    end
  })
end

local cfg = Config.new()
cfg.value = 100                                -- 正常
print("设置成功:", cfg.value)

local locked = cfg:lock()
local ok2, err2 = pcall(function() locked.newField = 1 end)
print("锁定后写入:", ok2, err2)

print()
print("=== 8. 性能对比：实例方法与闭包方法 ===")
local N = 500000

-- 方式一：元表（共享方法，内存占用小）
local Meta = {}
Meta.__index = Meta
function Meta.new() return setmetatable({ v = 0 }, Meta) end
function Meta:add(n) self.v = self.v + n end

local t1 = os.clock()
local m = Meta.new()
for i = 1, N do m:add(1) end
local metaTime = os.clock() - t1

-- 方式二：闭包（每个实例一份方法）
local function newClosure()
  local self = { v = 0 }
  function self.add(n) self.v = self.v + n end
  return self
end

local t2 = os.clock()
local c = newClosure()
for i = 1, N do c.add(1) end
local closureTime = os.clock() - t2

print(string.format("元表方式 %d 次: %.4f 秒", N, metaTime))
print(string.format("闭包方式 %d 次: %.4f 秒", N, closureTime))
print()
print("【实测结论】")
print("  在 Lua 5.4 下两者【调用速度非常接近】：本机 50 万次调用，")
print("  元表方式约 0.018 秒，闭包方式约 0.014 秒（闭包略快 1.3 倍）。")
print("  原因是：元表的 __index 查找本身也有成本，")
print("  粗略抵消了它的理论优势。注意闭包写法的 c.add(1) 没有 __index 查找，")
print("  所以略快；换成元表里【不存在】的方法时差距会明显拉大。")
print()
print("  那为什么框架还是用元表？真正的原因是【内存】：")
print("    元表方式：1000 个实例 = 1 份方法代码")
print("    闭包方式：1000 个实例 = 1000 份方法闭包")
print("  当对象数量上万时，闭包方式的内存占用会非常可观。")
print()
print("  【所以选型依据是内存和继承能力，不是调用速度】：")
print("    · 需要继承、实例数量多  -> 元表（框架标准做法）")
print("    · 只是单个简单对象      -> 闭包也很合适")
class 实现的"三件套"（背下来） ① cls.__index = cls——让实例可以通过元表查到类里的方法。
② setmetatable(cls, { __index = superClass })——让子类能查到父类的静态成员。
③ setmetatable(instance, cls)——把实例和类关联起来。

查找链：instance.foo → 实例自身没有 → 查 cls（因为 __index = cls）→ cls 自身没有 → 查父类（因为 setmetatable(cls, {__index = super})）。
这就是 Lua 的继承。
三个最容易写错的地方 ① 冒号与点号的混用
function Animal:getName() 等价于 function Animal.getName(self)。
调用时必须用 obj:getName()（冒号）——写成 obj.getName() 会漏掉 self，报"尝试索引 nil"。

② 调用父类方法必须显式传 self
Base.describe(self) —— 不能写成 Base:describe()，那样 self 会变成 Base 本身。

③ 构造函数名没有强制约定
Lua 不会自动调用 ctor 或 __init。必须自己在 new 里显式调用。
不同框架用的名字不同（ctor / __init / OnInit），读源码时先找这个。

二、私有成员：两种方案对比

下划线约定闭包实现
写法self._balance局部变量 + 闭包捕获
真实私有否 外部照样能改是 完全无法访问
内存占用小 方法共享大 每实例一份闭包
支持继承是困难
调试友好好 字段可见差 看不到内部状态
序列化可以很难
适用游戏逻辑（主流）模块级单例、工具类
游戏项目为什么主流用下划线约定 关键原因是"序列化和调试"：
• 存档需要遍历对象字段 → 闭包方案无法枚举出捕获的变量；
• 出 bug 时需要查看对象内部状态 → 闭包方案在调试器里看不到；
• 需要继承和对象池 → 闭包方案每实例一份方法，内存吃不消。

所以行业惯例是：用 _ 前缀表示"这是内部字段，外部别碰"，靠代码规范和 code review 保证，而不是靠语言机制。

三、性能：元表 vs 闭包（实测结论）

一个被广泛流传的错误结论 网上很多文章说"元表比闭包快"——但我实测的结果不是这样。

在 Lua 5.4 下跑 50 万次方法调用：
• 元表方式：0.018 秒
• 闭包方式：0.014~0.017 秒
• 两者几乎一样，闭包偶尔还更快

为什么：元表的 __index 查找本身也有成本，粗略抵消了它的理论优势。

那为什么框架还是用元表？真正的原因是内存和继承能力：
• 元表：1000 个实例只有 1 份方法代码
• 闭包：1000 个实例有 1000 份方法闭包
对象数量上万时，闭包方式的内存占用会非常可观。
选型依据 需要继承 且 实例数量多 → 用元表（框架标准做法）
只是单个简单对象 / 模块级单例 → 闭包也很合适，而且代码更简洁

不要为了"性能"这个错误的理由去选元表。选它是因为内存和继承。
