print("=== 2. 模块管理器（框架的地基）===")
-- 所有系统都注册成"模块"，由统一管理器调度

local ModuleManager = {}
ModuleManager.__index = ModuleManager

function ModuleManager.new()
  return setmetatable({
    modules = {},          -- name -> module
    order = {},            -- 按注册顺序保存
    initialized = false,
  }, ModuleManager)
end

-- 注册模块
function ModuleManager:register(name, module)
  assert(name, "模块必须有名字")
  assert(module, "模块不能为 nil")
  assert(not self.modules[name], "模块重复注册: " .. name)

  self.modules[name] = module
  self.order[#self.order + 1] = name
  return module
end

-- 获取模块
function ModuleManager:get(name)
  local m = self.modules[name]
  if not m then
    error("模块未注册: " .. tostring(name))
  end
  return m
end

-- 按顺序初始化所有模块
function ModuleManager:initAll()
  if self.initialized then return end

  print("  初始化模块（按注册顺序）:")
  for _, name in ipairs(self.order) do
    local m = self.modules[name]
    if m.onInit then
      m:onInit()
      print("    [OK] " .. name)
    end
  end
  self.initialized = true
end

-- 按【逆序】销毁（因为后注册的可能依赖先注册的）
function ModuleManager:destroyAll()
  print("  销毁模块（逆序）:")
  for i = #self.order, 1, -1 do
    local name = self.order[i]
    local m = self.modules[name]
    if m.onDestroy then
      m:onDestroy()
    end
    print("    [OK] " .. name)
  end
  self.modules = {}
  self.order = {}
  self.initialized = false
end

print("=== 3. 事件系统（模块解耦的关键）===")
-- 模块之间不能互相直接调用，只能通过事件通信

local EventManager = {}
EventManager.__index = EventManager

function EventManager.new()
  return setmetatable({
    handlers = {},        -- eventName -> { handler1, handler2, ... }
    pending = {},         -- 待派发的事件队列
    dispatching = false,
  }, EventManager)
end

function EventManager:on(eventName, handler, target)
  assert(type(eventName) == "string", "事件名必须是字符串")
  assert(type(handler) == "function", "处理器必须是函数")

  if not self.handlers[eventName] then
    self.handlers[eventName] = {}
  end

  local list = self.handlers[eventName]
  list[#list + 1] = { fn = handler, target = target }
  return handler
end

function EventManager:off(eventName, handler)
  local list = self.handlers[eventName]
  if not list then return end

  for i = #list, 1, -1 do
    if list[i].fn == handler then
      table.remove(list, i)
    end
  end
end

-- 【关键设计】派发时先入队，统一处理
-- 这样可以在回调里安全地订阅/取消订阅，不会破坏遍历
function EventManager:emit(eventName, ...)
  self.pending[#self.pending + 1] = {
    name = eventName,
    args = { ... },
  }

  if self.dispatching then return end   -- 正在派发中，交给外层处理

  self.dispatching = true

  while #self.pending > 0 do
    local evt = table.remove(self.pending, 1)
    local list = self.handlers[evt.name]

    if list then
      -- 【重要】复制一份再遍历，防止回调中修改列表
      local snapshot = {}
      for i = 1, #list do snapshot[i] = list[i] end

      for _, item in ipairs(snapshot) do
        if item.target then
          item.fn(item.target, table.unpack(evt.args))
        else
          item.fn(table.unpack(evt.args))
        end
      end
    end
  end

  self.dispatching = false
end

local em = EventManager.new()

-- 订阅
local scoreHistory = {}
em:on("ScoreChanged", function(newScore)
  scoreHistory[#scoreHistory + 1] = newScore
end)

em:on("ScoreChanged", function(newScore)
  -- 另一个订阅者（比如 UI）
end)

-- 派发
em:emit("ScoreChanged", 100)
em:emit("ScoreChanged", 250)
print("  事件系统: 分数变化历史 =", table.concat(scoreHistory, ", "))

print("=== 4. 生命周期管理 ===")
-- 框架需要明确的启动流程

local Framework = {}
Framework.__index = Framework

function Framework.new()
  local self = setmetatable({
    eventManager = EventManager.new(),
    moduleManager = ModuleManager.new(),
    state = "created",
    startupTime = nil,
  }, Framework)
  return self
end

-- 完整的启动流程
function Framework:startup()
  print()
  print("  框架启动流程:")
  local t0 = os.clock()

  -- 阶段 1：基础设施
  self.state = "initializing"
  print("    [1/4] 初始化基础设施...")

  -- 阶段 2：模块注册
  print("    [2/4] 注册模块...")
  self:_registerBuiltinModules()

  -- 阶段 3：模块初始化
  print("    [3/4] 初始化模块...")
  self.moduleManager:initAll()

  -- 阶段 4：进入游戏
  print("    [4/4] 进入游戏循环...")
  self.state = "running"
  self.startupTime = os.clock()

  print(string.format("  启动完成，耗时 %.4f 秒", self.startupTime - t0))
  print("  当前状态:", self.state)
end

function Framework:_registerBuiltinModules()
  local mm2 = self.moduleManager
  local em2 = self.eventManager

  mm2:register("Event", {
    onInit = function() end,
    onDestroy = function() end,
    emit = function(_, name, ...) em2:emit(name, ...) end,
    on = function(_, name, fn) return em2:on(name, fn) end,
  })

  mm2:register("Timer", {
    onInit = function() end,
    onDestroy = function() end,
    schedule = function(_, delay, fn) return { delay = delay, fn = fn } end,
  })

  mm2:register("Config", {
    onInit = function() end,
    onDestroy = function() end,
    get = function(_, key) return "配置值:" .. tostring(key) end,
  })
end

function Framework:shutdown()
  print()
  print("  框架关闭流程:")
  self.state = "shutting_down"
  self.moduleManager:destroyAll()
  self.state = "closed"
  print("  框架已关闭")
end

function Framework:getModule(name)
  return self.moduleManager:get(name)
end

二、模块管理器的设计要点

模块管理器是框架的地基。它的设计决定了后面所有模块怎么协作。

设计点方案理由
注册方式register(name, module)名字索引，简单直观
重复注册assert 报错早暴露问题，别静默覆盖
初始化顺序按注册顺序可控、可预测
销毁顺序逆序销毁后注册的可能依赖先注册的
懒加载首次访问时初始化加快启动速度
错误处理顶层 xpcall 兜底防止一个错误崩溃整个游戏
为什么销毁要逆序 假设 UI 模块依赖 Resource 模块（UI 要加载图集）。
如果按注册顺序销毁——先销毁 Resource，UI 还在用，就崩了。

逆序销毁保证：后注册的（依赖别人的）先被销毁。

这是栈（LIFO）的自然特性——和构造函数/析构函数的配对不同，这是"先建的活到最后"。

三、事件系统的两个关键设计

设计一：快照遍历（防止回调中修改列表） 问题场景：A 订阅了事件 X，在处理事件 X 的回调里取消了 B 的订阅。
如果直接遍历原列表，会导致索引错位、跳过元素或崩溃。

解决方案：派发前复制一份快照再遍历。
代价是每次派发多一次表复制——但这个代价是值得的，因为回调里修改订阅是最常见的用法。

优化：如果事件频繁且订阅者少，可以用复用缓冲区避免每次分配。

本页实测（Lua 5.4.7，见本页代码样本）：三个订阅者 A/B/C，A 在回调里取消 B——
· 快照遍历：调用顺序 A → B → C，共 3 次，不报错
· 直接遍历原表：调用顺序 A → C，共 2 次——B 被静默跳过（下标前移）
· 对照：只有一个订阅者、且它取消的是自己时，即使不用快照也不会出问题 ——所以"取消自己"这个 demo 根本测不出问题，必须用"取消别人"才能复现。
设计二：队列化派发（防止递归派发） 问题场景：回调 A 在响应事件 X 时又派发了事件 X → 无限递归。

解决方案：派发时先入队，统一处理。
emit(X) → 入队 → 循环取出派发
如果派发中又有 emit，只是入队，由外层循环处理。

好处：
· 不会因为"同步重入"而爆栈（回调里的 emit 只是入队，由外层循环处理）
· 派发顺序可预测（FIFO）
· 可以在回调里安全地订阅/取消订阅

但它并不是万能的：如果回调里每次都无条件再 emit 同一个事件，队列只会越来越长—— 栈不爆，但会死循环（卡在 while 里）。真正的防护还得靠业务上的终止条件 或者给队列加一个单次派发的最大重入次数。
