04 · table 与元表 ★★★

这一页是整个 Lua 语言最重要的内容。Lua 没有数组、字典、类、模块——所有这些都用 table 实现。而元表（metatable）让 table 能重载运算符、定义缺失行为、实现继承。不懂元表，你就看不懂任何 Lua 框架的源码。

这一页在路线中的位置 上一页（03）讲的是语法基础。这一页讲 Lua 的核心数据结构——它是后面所有内容的基石。
为什么这一页标三个星：接下来你会遇到的面向对象（06）、框架设计（13-17）、xLua 的类型映射（10），全部建立在元表之上。
下一页（05）函数与闭包——Lua 的另一个"魔法"来源。
验证方式：本页所有代码在 Lua 5.4.7 解释器里实际运行过，输出结果是真实的。

一、table 基础 ★

--  table：Lua 唯一的数据结构
-- ============================================================
--  Lua 没有数组、字典、类、模块的概念，
--  所有这些都是用 table 实现的。理解 table 就理解了 Lua 的一半。

print("=== 1. 两种初始化方式 ===")
-- 数组式（隐式索引 1,2,3...）

print("数组:", arr[1], arr[2], arr[3])

-- 字典式（显式键）

-- 【高频错误点】这是 Lua 与几乎所有其他语言最大的差异
local t = { "a", "b", "c" }
print("t[0] =", t[0])          -- nil（不存在！）
print("t[1] =", t[1])          -- "a"
print("#t   =", #t)            -- 3

print("=== 3. 长度运算符 # 的真相 ===")
local seq = { 1, 2, 3, 4, 5 }
print("连续数组 #:", #seq)               -- 5

-- 【重要】# 只对「序列」可靠。有空洞的表长度是未定义的
local hole = { 1, 2, nil, 4 }
print("有空洞的表 #:", #hole)           -- 结果不确定（可能是 4，也可能是 2）
print("空洞表用 select 数:", select("#", table.unpack(hole, 1, 4)))

-- 安全做法：自己数
local function countSafe(tbl)
  local n = 0
  for _ in pairs(tbl) do n = n + 1 end
  return n
end
print("安全计数:", countSafe(hole))

local data = { 10, 20, 30, name = "test", [100] = "hundred" }

print("=== 6. 表是引用类型 ===")
local a = { 1, 2, 3 }
local b = a               -- 不是复制！b 和 a 指向同一个表

local nums = { 5, 2, 8, 1, 9, 3 }

table.sort(nums)

-- 自定义排序（降序）

-- 拼接

-- 解包（5.2+ 用 table.unpack）
print("unpack:", table.unpack({ "a", "b", "c" }))

-- 打包（5.2+）
local packed = table.pack(1, 2, 3, nil, 5)
print("pack 后 n =", packed.n)

print("=== 8. 性能：预分配大小 ===")
-- Lua 的 table 会自动扩容，但扩容有成本。
-- 已知大小时用 table.new（LuaJIT 扩展）或在循环外创建。
local bigTable = {}
local start = os.clock()
for i = 1, 500000 do
  bigTable[i] = i
end
print(string.format("插入 50 万次耗时: %.3f 秒", os.clock() - start))
print("验证:", bigTable[500000])

-- 【最佳实践】复用表而不是反复创建
-- 差的写法：每次调用都建新表
-- 好的写法：传一个「输出表」进来，或使用对象池
反直觉之三：索引从 1 开始 Lua 的数组索引从 1 开始，不是 0。这意味着：
local t = {"a","b","c"} —— t[1] 是 "a"，t[0] 是 nil。

为什么容易出错：
• 从 C# / C / Java 转过来的人会习惯性写 for i = 0, #t - 1，结果少遍历一个还报错；
• 与外部系统交互（C# 数组、网络协议）时，索引偏移是最常见的 bug 来源。

正确写法：for i = 1, #t do 或 for i, v in ipairs(t) do。
长度运算符 # 的真相（重要） #t 只对"序列"可靠——也就是从 1 开始、连续、没有空洞的表。

含 nil 的表，#t 的结果是"未定义的"——不同 Lua 版本、不同内部实现可能给出不同结果。
在 Lua 5.4 实测中：{1,2,nil,4} 的 #t 返回 4，{1,2,3,nil} 返回 3——看起来"对"，但这是实现细节，不是语言保证。

要准确计数，必须用 select("#", ...) 或自己遍历 pairs。
这一条在代码 review 时经常被忽略，但在处理网络数据、可变参数时会导致真实 bug。

二、元表：Lua 的全部魔法 ★★★

元表让一个普通 table 可以"重载"运算符、定义缺失行为、实现继承。Lua 的面向对象就是用元表实现的。

-- ============================================================
--  元表（metatable）：Lua 最强大的机制 ★★★
-- ============================================================
--  元表让一个普通的 table 能够"重载"运算符、定义缺失行为、
--  实现面向对象。Lua 的 OOP 就是用元表实现的。
--  理解元表 = 理解 Lua 的全部魔法。

print("=== 1. setmetatable / getmetatable ===")
local t = {}
local mt = {}
setmetatable(t, mt)
print("getmetatable(t) == mt :", getmetatable(t) == mt)

-- 保护元表：加了 __metatable 字段后就取不到了
setmetatable(t, { __metatable = "locked" })
print("被保护的元表:", getmetatable(t))          -- "locked"

print()
print("=== 2. __index：访问不存在的键时触发 ===")
-- 这是【最重要的元方法】，OOP 继承就靠它

-- 形式一：__index 是表 —— 去那个表里找
local base = { greeting = "hello" }
local derived = setmetatable({}, { __index = base })
print("derived.greeting =", derived.greeting)     -- hello（从 base 查到的）
print("rawget 绕过 __index:", rawget(derived, "greeting"))  -- nil

-- 形式二：__index 是函数 —— 自定义查找逻辑
local dynamic = setmetatable({}, {
  __index = function(tbl, key)
    return "动态生成:" .. tostring(key)
  end
})
print(dynamic.anything)              -- "动态生成:anything"
print(dynamic.foo)                   -- "动态生成:foo"

-- 缓存优化：查到后写回表，下次就不走 __index 了
local cached = setmetatable({}, {
  __index = function(tbl, key)
    local v = "计算出的" .. tostring(key)
    rawset(tbl, key, v)              -- 写回，后续直接命中
    return v
  end
})
print("首次（走 __index 计算）:", cached.expensive)
print("rawget 直接取（能看到已写回）:", rawget(cached, "expensive"))
print("二次访问（命中缓存，不走 __index）:", cached.expensive)

print()
print("=== 3. __newindex：写入新键时触发 ===")
local log = {}
local readonly = setmetatable({}, {
  __newindex = function(tbl, key, value)
    log[#log + 1] = string.format("尝试写入 %s = %s", tostring(key), tostring(value))
    rawset(tbl, key, value)          -- 想真正写入必须用 rawset
  end
})
readonly.name = "测试"
readonly.value = 123
print("写入日志:")
for _, line in ipairs(log) do print("  " .. line) end

-- 只读表（禁止写入）
local locked = setmetatable({ fixed = 1 }, {
  __newindex = function()
    error("该表是只读的", 2)          -- 层级 2 让报错指向调用方
  end
})
print("读取只读表:", locked.fixed)
local ok, err = pcall(function() locked.newKey = 1 end)
print("写入只读表:", ok, err)

print()
print("=== 4. 运算符重载 ===")
local Vector = {}
Vector.__index = Vector

function Vector.new(x, y)
  return setmetatable({ x = x, y = y }, Vector)
end

-- 算术运算符
Vector.__add = function(a, b) return Vector.new(a.x + b.x, a.y + b.y) end
Vector.__sub = function(a, b) return Vector.new(a.x - b.x, a.y - b.y) end
Vector.__mul = function(a, b)
  if type(b) == "number" then return Vector.new(a.x * b, a.y * b) end
  return a.x * b.x + a.y * b.y            -- 点积
end
Vector.__unm = function(a) return Vector.new(-a.x, -a.y) end   -- 一元负号

-- 比较运算符
Vector.__eq = function(a, b) return a.x == b.x and a.y == b.y end
Vector.__lt = function(a, b) return (a.x * a.x + a.y * a.y) < (b.x * b.x + b.y * b.y) end
-- 【重要】Lua 5.4 起，__le【不会】再从 __lt 自动推导——必须显式定义 __le，
-- 否则 a <= b 会报 "attempt to compare two table values"（5.3 及更早会自动推导）。
Vector.__le = function(a, b) return (a.x * a.x + a.y * a.y) <= (b.x * b.x + b.y * b.y) end

-- 长度与字符串
Vector.__len = function(a) return 2 end
Vector.__tostring = function(a)
  return string.format("Vec(%g, %g)", a.x, a.y)
end

local v1 = Vector.new(1, 2)
local v2 = Vector.new(3, 4)

print("加法 v1+v2 =", tostring(v1 + v2))
print("减法 v2-v1 =", tostring(v2 - v1))
print("数乘 v1*3  =", tostring(v1 * 3))
print("点积 v1*v2 =", v1 * v2)
print("取负 -v1   =", tostring(-v1))
print("相等 v1==v1:", v1 == Vector.new(1, 2))
print("小于 v1<v2 :", v1 < v2)
print("小于等于 v1<=v2:", v1 <= v2)          -- 显式 __le，5.4 不会从 __lt 推导
print("#v1        =", #v1)

-- 演示：只定义 __lt、不定义 __le 时，a <= b 在 5.4 会报错
local onlyLt = setmetatable({ v = 1 }, { __lt = function(a, b) return a.v < b.v end })
local okLe, errLe = pcall(function()
  return onlyLt <= setmetatable({ v = 2 }, getmetatable(onlyLt))
end)
print("只定义 __lt 时 a<=b:", okLe, errLe)

print()
print("=== 5. __call：让表可以像函数一样调用 ===")
local callable = setmetatable({ name = "调用器" }, {
  __call = function(self, arg)
    return self.name .. " 收到: " .. tostring(arg)
  end
})
print(callable("参数"))
print(callable("另一个参数"))

print()
print("=== 6. __concat 与 .. 运算 ===")
local concatDemo = setmetatable({ v = "值" }, {
  __concat = function(a, b)
    local sa = type(a) == "table" and a.v or tostring(a)
    local sb = type(b) == "table" and b.v or tostring(b)
    return sa .. "|" .. sb
  end
})
print("表 .. 字符串:", concatDemo .. "尾部")
print("字符串 .. 表:", "头部" .. concatDemo)

print()
print("=== 7. __gc 与 __mode（弱表）===")
-- __mode 实现弱表：键或值不被强引用时自动回收
local weakValues = setmetatable({}, { __mode = "v" })
weakValues.a = {}
weakValues.b = {}
collectgarbage("collect")
print("弱值表在 GC 后:", weakValues.a, weakValues.b)   -- 可能为 nil（被回收）

local strong = {}
local weakKeys = setmetatable({}, { __mode = "k" })
weakKeys[strong] = "值"
collectgarbage("collect")
print("强引用还在，弱键表条目保留:", weakKeys[strong] ~= nil)

print("=== 9. 性能提醒：__index 有成本 ===")
-- 每次访问缺失的键都会触发函数调用。
-- 热路径上频繁访问不存在的字段会明显变慢。
local N = 200000

local plain = { x = 1 }
local withMeta = setmetatable({ x = 1 }, { __index = function() return nil end })

local t1 = os.clock()
for i = 1, N do local _ = plain.x end
local plainTime = os.clock() - t1

local t2 = os.clock()
for i = 1, N do local _ = withMeta.x end      -- 存在，不走 __index
local metaTime = os.clock() - t2

print(string.format("访问已有字段: 普通 %.4f / 元表 %.4f", plainTime, metaTime))

local t3 = os.clock()
for i = 1, N do local _ = withMeta.missing end  -- 不存在，走 __index 函数
local missTime = os.clock() - t3

print(string.format("访问缺失字段（触发 __index）: %.4f 秒", missTime))
print(string.format("触发元方法比直接命中慢约 %.1f 倍", missTime / plainTime))
__index 的两种形式（必须理解） 形式一：__index 是表——去那个表里查找。
这是继承的实现方式：setmetatable(child, { __index = parent })，子表里没有的键会去父表找。

形式二：__index 是函数——调用函数，用返回值作为结果。
这是动态生成的实现方式：可以按需计算、延迟加载、返回默认值。

性能提醒：函数形式的 __index 每次访问缺失键都会触发一次函数调用。
实测数据：触发元方法比直接命中字段慢约 3 倍。
热路径上避免频繁访问"不存在的字段"——或者用 rawset 缓存查到的值。
rawget / rawset：绕过元表 rawget(t, k) 和 rawset(t, k, v) 完全绕过元表，直接操作表本身。

用途一：检查字段是否真的存在
t.foo 可能来自 __index，而 rawget(t, "foo") 只查表本身。

用途二：在 __newindex 里真正写入
这是最容易踩的坑：如果在 __newindex 函数里写 tbl[key] = value，会再次触发 __newindex，导致无限递归。必须用 rawset。
