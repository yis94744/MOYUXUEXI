local function class(name, base)
  local c = {}
  c.__name = name
  c.__base = base
  c.__index = c

  if base then
    setmetatable(c, { __index = base })
    c.super = base
  end

  -- 类型判断：沿继承链向上查找
  function c.isInstanceOf(obj, className)
    local cls = getmetatable(obj)
    while cls do
      if cls == className or cls.__name == className then
        return true
      end
      cls = cls.__base
    end
    return false
  end

  -- 创建实例（自动调用 ctor）
  function c.new(...)
    local obj = setmetatable({}, c)
    if obj.ctor then
      obj:ctor(...)
    end
    return obj
  end

  return c
end

-- 第一级：Entity
local Entity = class("Entity")

function Entity:ctor(id, name)
  self.id = id
  self.name = name
  self.hp = 100
  self.maxHp = 100
end

function Entity:takeDamage(amount)
  self.hp = math.max(0, self.hp - amount)
  print(string.format("    %s 受到 %d 伤害，剩余 %d", self.name, amount, self.hp))
end

function Entity:getInfo()
  return string.format("%s(id=%d, hp=%d/%d)", self.name, self.id, self.hp, self.maxHp)
end

-- 第二级：Character（注意 super 的捕获）
local Character = class("Character", Entity)
local super = Character.super              -- 【关键】捕获父类

function Character:ctor(id, name, role)
  super.ctor(self, id, name)               -- 【正确】用局部变量
  self.role = role
  self.level = 1
  self.exp = 0
end

function Character:gainExp(amount)
  self.exp = self.exp + amount
  local needed = self.level * 100
  if self.exp >= needed then
    self.level = self.level + 1
    self.exp = self.exp - needed
    self.maxHp = self.maxHp + 20
    self.hp = self.maxHp
    print(string.format("    %s 升级到 %d 级！", self.name, self.level))
  end
end

function Character:getInfo()
  local baseInfo = super.getInfo(self)     -- 【正确】
  return baseInfo .. string.format(", lv=%d, role=%s", self.level, self.role)
end

-- 第三级：Player
local Player = class("Player", Character)
local superPlayer = Player.super            -- 【关键】每个类要单独捕获

function Player:ctor(id, name, role, isVip)
  superPlayer.ctor(self, id, name, role)    -- 调用 Character 的 ctor
  self.isVip = isVip
  self.gold = 0
end

function Player:addGold(amount)
  local actual = self.isVip and amount * 2 or amount
  self.gold = self.gold + actual
  return actual
end

function Player:getInfo()
  local baseInfo = superPlayer.getInfo(self)
  return baseInfo .. string.format(", gold=%d%s", self.gold, self.isVip and " [VIP]" or "")
end

local Component = class("Component")

function Component:ctor(owner)
  self.owner = owner
  self.enabled = true
end

local ListComponent = class("ListComponent", Component)
local superList = ListComponent.super

function ListComponent:ctor(owner, itemCount)
  superList.ctor(self, owner)
  self.itemCount = itemCount
  self.scrollPos = 0
end

function ListComponent:scroll(delta)
  self.scrollPos = math.max(0, self.scrollPos + delta)
  local maxScroll = self.itemCount * 100
  if self.scrollPos > maxScroll then self.scrollPos = maxScroll end
  return self.scrollPos
end

local RedDotComponent = class("RedDotComponent", Component)
local superDot = RedDotComponent.super

function RedDotComponent:ctor(owner, dotKey)
  superDot.ctor(self, owner)
  self.dotKey = dotKey
  self.visible = false
end

function RedDotComponent:refresh(isVisible)
  if self.visible ~= isVisible then
    self.visible = isVisible
    return true
  end
  return false
end

local View = class("View")

function View:ctor(name)
  self.name = name
  self.components = {}
end

function View:addComponent(componentClass, ...)
  local comp = componentClass.new(self, ...)
  self.components[#self.components + 1] = comp
  return comp
end

function View:getComponent(componentClass)
  for _, c in ipairs(self.components) do
    if componentClass.isInstanceOf(c, componentClass) then
      return c
    end
  end
  return nil
end

local ClassRegistry = { classes = {} }

function ClassRegistry.register(name, cls)
  ClassRegistry.classes[name] = cls
end

function ClassRegistry.create(name, ...)
  local cls = ClassRegistry.classes[name]
  if not cls then
    error("未注册的类: " .. tostring(name))
  end
  return cls.new(...)
end

二、⭐ 一个真实踩到的坑：self.super 无限递归

错误示范：这样写会栈溢出 我最初写这一页的示例时，就是这么写的： 
local Character = class("Character", Entity)

function Character:ctor(id, name)
  self.super.ctor(self, id, name)   -- ❌ 无限递归！
end
运行结果：stack overflow，堆栈里全是同一个 ctor 在循环调用。 看起来完全合理对不对？——"我在 Character 里，调父类 Entity 的 ctor"。 但它错了，而且错得很隐蔽。
根因：self.super 查的是"实例的类"，不是"当前方法的类" 关键理解：self 是实例，不是类。 · self.super 实际是 getmetatable(self).super · 而 getmetatable(self) 是实例所属的类（Player） · 所以 p.super 永远是 Character（Player 的父类） 本页实测验证了这一点： 
p.super 实际是: Character
-- 注意！p 是 Player 的实例，但 p.super 拿到的是 Character
于是递归发生了： · 创建 Player 实例 → 调用 Character:ctor · 里面写 self.super.ctor → self 是 p → p.super 是 Character · 结果又调用了 Character:ctor → 无限循环 什么时候会踩坑，一句话就能判断：
· 如果写 self.super 的那个类 == 实例真正的类，self.super 恰好就是它的父类 —— 碰巧是对的（两级继承直接 new 父类时就是这种情形）
· 只要实例是这个类的子类，self.super 就会指回这个类自己 —— 必然递归

本页实测（Lua 5.4.7）：
· Player.new(...)（三级）→ stack overflow，Character:ctor 被递归进入 249993 次
· Character.new(...)（两级、直接建父类实例）→ 正常运行，不崩

所以"两级侥幸不崩"只是运气——只要哪天给这个类加个子类，坑立刻爆。
正确做法：用局部变量捕获父类 每个类都要单独捕获自己的父类： 
local Character = class("Character", Entity)
local super = Character.super          -- ✅ 捕获

function Character:ctor(id, name)
  super.ctor(self, id, name)           -- ✅ 用局部变量
end

local Player = class("Player", Character)
local superPlayer = Player.super        -- ✅ 每个类单独捕获

function Player:ctor(id, name, role)
  superPlayer.ctor(self, id, name, role)
end
为什么这样对：局部变量 super 在定义方法时就已经绑定到 Character.super（= Entity），与 self 是谁无关。 本页实测：三级继承正确工作 
勇者(id=1, hp=100/100), lv=1, role=战士, gold=0 [VIP]
这一行的每一段都来自不同层级的构造函数——说明 super 调用链完全正确。

同一个 ctor 递归了将近 25 万次才撑爆栈——所以看到 "stack overflow" 时， 先怀疑"自己调了自己"，而不是先怀疑数据太大。

三、组合优于继承

组件化：避免类爆炸 问题：如果 UI 界面都靠继承，每加一个能力，组合数翻倍： ListView, RedDotView, ListViewRedDot, AnimListView, AnimRedDotListView… 解决：把能力做成可挂载的组件： 
local bagView = View.new("背包界面")
bagView:addComponent(ListComponent, 20)
bagView:addComponent(RedDotComponent, "bag.new")
本页实测输出： · 组件数量: 2 · 列表组件滚动到: 250 · 红点组件刷新: true / true(无变化) / false(消失) 注意第二次 refresh(true) 返回 false——红点组件会做状态去重，无变化时不触发 UI 刷新。这是很实用的优化。

四、类注册表：配置驱动

把"类型判断"变成"查表" 没有注册表时： 
if typeName == "Player" then new(Player, ...)
elseif typeName == "Monster" then new(Monster, ...)

每新增一种类型都要改这段代码。 有注册表时：

五、单例的状态残留 ★★

Lua 热更项目最常见的疑难 bug 问题：Lua 模块是全局缓存的，重新加载脚本时单例的状态会残留。 触发场景： · 热更新后重新执行模块代码 · 编辑器里停止播放再重新开始 · 结果：上次的数据还在，产生诡异 bug 三种解决办法： ① 提供 reset() 方法——框架启动时统一调用所有单例的 reset ② 在模块末尾返回新表——避免缓存旧表 ③ 避免在单例里存"应该有生命周期"的数据——那类数据应该属于具体对象 这一条在页 18（调试排错）会再次出现——它是 Lua 项目最容易被忽视、又最难排查的问题之一。
