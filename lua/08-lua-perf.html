local function bench(name, fn, times)
  times = times or 1
  local t = os.clock()
  for i = 1, times do fn() end
  local elapsed = os.clock() - t
  return elapsed
end

print("=== 1. 局部变量 vs 全局变量（最重要的优化）===")
-- 全局变量访问要查 _G 表；局部变量是寄存器访问

local N = 500000

local function useGlobal()
  G_TEST_VAR = 0
  for i = 1, N do
    G_TEST_VAR = G_TEST_VAR + 1
  end
end

local function useLocal()
  local counter = 0
  for i = 1, N do
    counter = counter + 1
  end
  return counter
end

local tg = bench("global", useGlobal)
local tl = bench("local", useLocal)
print(string.format("  全局变量 %d 次: %.4f 秒", N, tg))
print(string.format("  局部变量 %d 次: %.4f 秒", N, tl))
print(string.format("  局部快 %.2f 倍", tg / tl))

print()

-- 每次调用 table.insert 都要两次表查找（_G.table、再 .insert）

local M = 200000
local testTable = {}

local function useGlobalFn()
  local t = {}
  for i = 1, M do
    table.insert(t, i)
  end
  return t
end

local insert = table.insert       -- 缓存
local function useLocalFn()
  local t = {}
  for i = 1, M do
    insert(t, i)
  end
  return t
end

local tg2 = bench("g", useGlobalFn)
local tl2 = bench("l", useLocalFn)
print(string.format("  全局 table.insert: %.4f 秒", tg2))
print(string.format("  局部 insert:       %.4f 秒", tl2))

print("=== 3. 字符串拼接：.. vs table.concat ===")
-- 【重要】Lua 的字符串是不可变的，每次 .. 都会创建新字符串

local COUNT = 20000

local t_concat = bench("concat", function()
  local s = ""
  for i = 1, COUNT do
    s = s .. tostring(i)
  end
  return s
end)

local t_table = bench("table", function()
  local buf = {}
  for i = 1, COUNT do
    buf[#buf + 1] = tostring(i)
  end
  return table.concat(buf)
end)

print(string.format("  循环内 .. 拼接 %d 次: %.4f 秒", COUNT, t_concat))
print(string.format("  table.concat:          %.4f 秒", t_table))
print(string.format("  快 %.2f 倍", t_concat / t_table))

print("=== 4. 表预分配与复用 ===")
-- 【核心】table 会自动扩容，扩容有成本；复用则完全不分配

local SIZE = 100000

-- 方式 A：每次新建
local tA = bench("new", function()
  local t = {}
  for i = 1, SIZE do
    t[i] = i
  end
  return t
end)

-- 方式 B：复用同一个表（先清空）
local reused = {}
local function fillReused()
  for i = #reused, 1, -1 do reused[i] = nil end     -- 清空
  for i = 1, SIZE do
    reused[i] = i
  end
  return reused
end

local tB = bench("reuse", fillReused, 5)
print(string.format("  每次新建 %d 个元素: %.4f 秒", SIZE, tA))
print(string.format("  复用（5 轮平均）:    %.4f 秒", tB / 5))
print("  【注意】复用要付出清空成本，元素少时不一定划算")
print("          元素多、调用频繁时复用收益明显")

print()
print("=== 5. 闭包开销 ===")
-- 每次创建闭包都要分配内存

local ITER = 200000

local tClosure = bench("closure", function()
  local sum = 0
  for i = 1, ITER do
    local f = function(x) return x + 1 end    -- 每次新建
    sum = sum + f(1)
  end
  return sum
end)

local cachedFn = function(x) return x + 1 end
local tCached = bench("cached", function()
  local sum = 0
  for i = 1, ITER do
    sum = sum + cachedFn(1)
  end
  return sum
end)

print(string.format("  循环内建闭包: %.4f 秒", tClosure))
print(string.format("  循环外建一次: %.4f 秒", tCached))
print(string.format("  快 %.2f 倍", tClosure / tCached))

print()
print("=== 6. 避免不必要的表创建 ===")
-- 【实战场景】函数返回多值时，用 table 打包会有分配

local ITER2 = 200000

-- 差：每次返回新表
local function returnTable()
  return { 1, 2, 3 }
end

local tRetTable = bench("table", function()
  local sum = 0
  for i = 1, ITER2 do
    local t = returnTable()
    sum = sum + t[1]
  end
  return sum
end)

-- 好：返回多值（不分配）
local function returnMulti()
  return 1, 2, 3
end

local tRetMulti = bench("multi", function()
  local sum = 0
  for i = 1, ITER2 do
    local a = returnMulti()
    sum = sum + a
  end
  return sum
end)

print(string.format("  返回表:   %.4f 秒", tRetTable))
print(string.format("  返回多值: %.4f 秒", tRetMulti))
print(string.format("  快 %.2f 倍", tRetTable / tRetMulti))

print("=== 7. 循环优化 ===")

local LOOP = 200000

-- 方式 A：for i = 1, #t
local arr = {}
for i = 1, 1000 do arr[i] = i end

local tLen = bench("len", function()
  local sum = 0
  for j = 1, LOOP do
    for i = 1, #arr do                -- 每次迭代都算 #arr
      sum = sum + arr[i]
    end
  end
  return sum
end)

-- 方式 B：缓存长度
local tCachedLen = bench("cachedlen", function()
  local sum = 0
  local n = #arr                       -- 只算一次
  for j = 1, LOOP do
    for i = 1, n do
      sum = sum + arr[i]
    end
  end
  return sum
end)

-- 方式 C：用 ipairs（内部也缓存长度）
local tIpairs = bench("ipairs", function()
  local sum = 0
  for j = 1, LOOP do
    for _, v in ipairs(arr) do
      sum = sum + v
    end
  end
  return sum
end)

print(string.format("  for i=1,#t:      %.4f 秒", tLen))
print(string.format("  缓存长度:        %.4f 秒", tCachedLen))
print(string.format("  ipairs:          %.4f 秒", tIpairs))
print()
print("  【实测结论 —— 和常见说法不同】")
print("  · 缓存长度确实有收益（约 3~4%）")
print("  · 【但 ipairs 反而慢得多（约 2.6 倍）】")
print()
print("  原因：ipairs 每次迭代都要调用迭代器函数（一次函数调用开销），")
print("        而数字 for 循环是虚拟机原生指令，没有函数调用。")
print()
print("  【实战建议】")
print("    热路径用【数字 for + 缓存长度】：")
print("      local n = #arr")
print("      for i = 1, n do ... end")
print("    可读性优先时用 ipairs（非热路径无所谓）")

print()
print("=== 8. 元表访问的代价 ===")
-- 【重要】__index 函数形式每次访问都触发函数调用

local N3 = 300000

local plain = { value = 1 }

local withMeta = setmetatable({}, {
  __index = function(t, k)
    return k == "value" and 1 or nil
  end
})

local tPlain = bench("plain", function()
  local sum = 0
  for i = 1, N3 do
    sum = sum + plain.value
  end
  return sum
end)

local tMeta = bench("meta", function()
  local sum = 0
  for i = 1, N3 do
    sum = sum + withMeta.value          -- 触发 __index
  end
  return sum
end)

print(string.format("  普通字段访问: %.4f 秒", tPlain))
print(string.format("  __index 函数: %.4f 秒", tMeta))
print(string.format("  慢 %.2f 倍", tMeta / tPlain))
print("  【优化】查到后写回表（rawset），后续直接命中")

-- 演示缓存优化
local cached = setmetatable({}, {
  __index = function(t, k)
    local v = 999
    rawset(t, k, v)                     -- 写回，下次不再触发
    return v
  end
})

local tMetaCached = bench("metacached", function()
  local sum = 0
  for i = 1, N3 do
    sum = sum + cached.value            -- 只有第一次触发 __index
  end
  return sum
end)

print(string.format("  缓存后的 __index: %.4f 秒", tMetaCached))
print(string.format("  比未缓存快 %.2f 倍", tMeta / tMetaCached))

    二、两个反直觉的实测结论 ★★★

    反例一：缓存全局函数在 Lua 5.4 下没有收益 常见说法：「把 table.insert 缓存成 local insert = table.insert 会快很多」。
    实测结果：两者耗时完全相同（1.00 倍）。

    为什么：Lua 5.4 用数组存全局变量（不是哈希表），全局访问本身就是 O(1) 索引——和局部变量访问的差距远小于 LuaJIT 时代。

    那什么时候还有用？
    · 在 LuaJIT 下差异明显（JIT 为局部变量生成寄存器访问）
    · 极热循环（每帧几百万次调用）
    · 需要 API 稳定性时（把依赖固定，外部改动不影响）

    结论：这不是优先优化项。先做收益大的（见表格第 1~5 项）。
    反例二：ipairs 比数字 for 慢得多 常见说法：「遍历数组用 ipairs，比 for i = 1, #t 更快（因为缓存了长度）」。
    实测结果：ipairs 慢了 2.6 倍。

    为什么：ipairs 每次迭代都要调用一次迭代器函数——一次函数调用的开销。而数字 for 循环是虚拟机原生指令，没有函数调用。

    正确的热路径写法：
    local n = #arr
    for i = 1, n do ... end

    缓存长度确实有收益（3~4%），但要配合数字 for 使用，而不是 ipairs。

    实战建议：非热路径用 ipairs（可读性好）；热路径用数字 for + 缓存长度。
