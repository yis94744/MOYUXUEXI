print("=== 1. 分段计时器（最实用的工具）===")
print()

local Profiler = {}
Profiler.__index = Profiler

function Profiler.new(threshold)
  return setmetatable({
    sections = {},           -- name -> { total, count, max }
    threshold = threshold or 0.001,   -- 只报告超过阈值的
    startTimes = {},         -- 用于嵌套计时
    order = {},              -- 记录调用顺序
  }, Profiler)
end

-- 【关键】只累计统计，不每次都打印
function Profiler:begin(name)
  self.startTimes[name] = os.clock()
end

function Profiler:finish(name)
  local startT = self.startTimes[name]
  if not startT then return end

  local elapsed = os.clock() - startT
  self.startTimes[name] = nil

  local s = self.sections[name]
  if not s then
    s = { total = 0, count = 0, max = 0, min = math.huge }
    self.sections[name] = s
    self.order[#self.order + 1] = name
  end

  s.total = s.total + elapsed
  s.count = s.count + 1
  if elapsed > s.max then s.max = elapsed end
  if elapsed < s.min then s.min = elapsed end
end

-- 便捷包装
function Profiler:wrap(name, fn)
  return function(...)
    self:begin(name)
    local results = { fn(...) }
    self:finish(name)
    return table.unpack(results)
  end
end

function Profiler:report()
  print(string.format("  %-22s %-12s %-10s %-12s %s",
    "模块", "总耗时(ms)", "调用次数", "平均(ms)", "最大(ms)"))
  print("  " .. string.rep("-", 78))

  -- 按总耗时排序
  local sorted = {}
  for name, s in pairs(self.sections) do
    sorted[#sorted + 1] = { name = name, s = s }
  end
  table.sort(sorted, function(a, b) return a.s.total > b.s.total end)

  for _, item in ipairs(sorted) do
    local s = item.s
    print(string.format("  %-22s %-12.3f %-10d %-12.4f %.3f",
      item.name, s.total * 1000, s.count, s.total * 1000 / s.count, s.max * 1000))
  end
end

function Profiler:reset()
  self.sections = {}
  self.order = {}
end

-- 【有问题的实现】每个敌人都单独调用 API
local function updateEnemies_SLOW(enemies, deltaTime)
  local count = 0
  for i = 1, #enemies do
    local e = enemies[i]
    e.x = e.x + e.speedX * deltaTime
    e.y = e.y + e.speedY * deltaTime
    -- 【问题】每次一个跨语言调用
    CSharpAPI.setPosition(e.x, e.y, 0)
    count = count + 1
  end
  return count
end

-- 【优化后】批量处理
local function updateEnemies_FAST(enemies, deltaTime)
  local batch = {}
  for i = 1, #enemies do
    local e = enemies[i]
    e.x = e.x + e.speedX * deltaTime
    e.y = e.y + e.speedY * deltaTime
    batch[i] = { e.x, e.y, 0 }
  end
  -- 【关键】只调用一次（真实项目里是一次批量 API）
  CSharpAPI.callCount = CSharpAPI.callCount + 1
  return #batch
end

local N = 5000

-- 差：循环内 .. 拼接

local function measure(fn, data)
  local t = os.clock()
  fn(data)
  return (os.clock() - t) * 1000
end

-- O(n) 的实现
local function linearSearch(data, target)
  for i = 1, #data do
    if data[i] == target then return i end
  end
  return nil
end

-- 建立索引（一次）
local function buildIndex(data)
  local idx = {}
  for i = 1, #data do
    idx[data[i]] = i
  end
  return idx
end

print(string.format("  %-14s %-20s %-20s %s", "数据量", "遍历查找(ms)", "索引查找(ms)", "差距"))
print("  " .. string.rep("-", 68))

for _, size in ipairs({ 1000, 5000, 20000 }) do
  local data = {}
  for i = 1, size do data[i] = i * 3 end

  local idx = buildIndex(data)

  -- 查 100 次
  local t5 = os.clock()
  for _ = 1, 100 do
    linearSearch(data, size * 3)
  end
  local scanMs = (os.clock() - t5) * 1000

  local t6 = os.clock()
  for _ = 1, 100 do
    local _ = idx[size * 3]
  end
  local idxMs = (os.clock() - t6) * 1000

  print(string.format("  %-14d %-20.2f %-20.2f %.0fx",
    size, scanMs, idxMs, idxMs > 0 and scanMs / idxMs or 0))
end

    二、Lua 性能的四大瓶颈 ★★★

    瓶颈症状定位方法
    跨语言调用过多CPU 高但 Lua 代码简单统计每秒调用次数
    GC 频繁规律性卡顿看 GC Alloc
    算法复杂度问题数据量增大后突然卡分段计时
    字符串拼接日志或 UI 更新处卡页 08 的第二点
    分段计时：最快定位瓶颈的方法 当 Profiler 看不清时，用手动计时：
    local t1 = os.clock()
    doSomething()
    local dt = os.clock() - t1
    if dt > 0.001 then print("doSomething 耗时: " .. dt) end

    要点：
    · 只在超过阈值时打印——避免日志本身成为瓶颈
    · 按模块打点，不要按函数打点（否则开销太大）
    · 累计统计比单次更有意义

    这个方法看似原始，但在 Lua 里是最有效的——因为它不依赖任何 Profiler 支持。

    三、跨语言边界的统计方法

    给跨语言调用加计数器 问题：你不知道每秒发生了多少次 Lua→C# 调用。
    做法：包一层计数：
    local callCount = 0
    local function wrap(fn)
      return function(...)
        callCount = callCount + 1
        return fn(...)
      end
    end

    每秒打印一次计数，就能看出：
    · 正常情况下应该是几百次
    · 如果有几万次，说明热路径上在做密集的跨语言调用

    这是发现"性能黑洞"最直接的方法。

    四、优化收益的优先级

    优先级优化项典型收益工作量
    1消除跨语言调用（批量/缓存）极大中
    2字符串拼接改 table.concat17~30 倍小
    3局部变量化2 倍小
    4避免循环内建闭包3.8 倍小
    5消除 GC 分配消除卡顿中
    6缓存表长度3~4%小
    7算法优化看场景大
    8分帧执行消除单帧尖峰中
    为什么把算法优化排在第 7 因为它的改动量最大、风险最高。

    前面的 1~6 项通常能把性能提升 2~5 倍，而且改动小、风险低。
    只有当这些都不够时，才考虑重写算法。

    而且很多时候，"算法慢"只是表象——真正的问题是在循环里做了跨语言调用。
    先检查 1~6 项，往往就能解决问题。
