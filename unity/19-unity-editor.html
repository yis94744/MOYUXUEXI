    一、为什么编辑器扩展值得学

    编辑器扩展不是"锦上添花"，而是解决三个真实问题。

    using UnityEngine;
#if UNITY_EDITOR
using UnityEditor;
#endif

// ============================================================
//  编辑器扩展：让工具提升团队效率
// ============================================================
//  为什么编辑器扩展值得学？
//    1. 它能把"重复的手工操作"变成一键完成
//    2. 它能防止策划/美术配错数据
//    3. 【它是深入理解 Unity 的最佳途径】——因为你要操作编辑器本身

// ============================================================
//  1. 自定义 Inspector
// ============================================================
public class EnemyConfigInspectorTarget : MonoBehaviour
{
    [Range(1, 9999)] public int maxHp = 100;
    [Range(0f, 100f)] public float moveSpeed = 3f;
    [Range(0f, 100f)] public float attack = 10f;
    public string displayName = "史莱姆";
    public bool isBoss = false;
    [TextArea(2, 5)] public string description;
}

#if UNITY_EDITOR
[CustomEditor(typeof(EnemyConfigInspectorTarget))]
public class EnemyConfigInspectorTargetEditor : Editor
{
    public override void OnInspectorGUI()
    {
        var t = (EnemyConfigInspectorTarget)target;

        // 绘制默认的 Inspector
        DrawDefaultInspector();

        EditorGUILayout.Space(10);

        // 加一个"常用操作"按钮
        if (GUILayout.Button("恢复默认值"))
        {
            Undo.RecordObject(t, "Reset Enemy Config");
            t.maxHp = 100;
            t.moveSpeed = 3f;
            t.attack = 10f;
            t.displayName = "史莱姆";
            EditorUtility.SetDirty(t);
        }

        // 加一个数据校验提示
        if (t.attack > t.maxHp)
        {
            EditorGUILayout.HelpBox("攻击力大于血量，这可能不合法", MessageType.Warning);
        }

        if (t.isBoss && t.maxHp < 1000)
        {
            EditorGUILayout.HelpBox("Boss 的血量通常应该更高", MessageType.Info);
        }
    }
}
#endif

// ============================================================
//  2. CustomPropertyDrawer：让字段显示得更直观
// ============================================================
[System.Serializable]
public struct MinMaxRange
{
    public float min;
    public float max;
}

#if UNITY_EDITOR
[CustomPropertyDrawer(typeof(MinMaxRange))]
public class MinMaxRangeDrawer : PropertyDrawer
{
    public override void OnGUI(Rect position, SerializedProperty property, GUIContent label)
    {
        EditorGUI.BeginProperty(position, label, property);

        var minProp = property.FindPropertyRelative("min");
        var maxProp = property.FindPropertyRelative("max");

        // 用一个 MinMaxSlider 代替两个输入框
        float min = minProp.floatValue;
        float max = maxProp.floatValue;
        EditorGUI.MinMaxSlider(position, label, ref min, ref max, 0f, 100f);
        minProp.floatValue = min;
        maxProp.floatValue = max;

        EditorGUI.EndProperty();
    }
}
#endif

// ============================================================
//  3. EditorWindow：自制工具窗口
// ============================================================
#if UNITY_EDITOR
public class BatchToolWindow : EditorWindow
{
    private int _count = 10;
    private float _spacing = 2f;

    [MenuItem("工具/批量生成物体 %#g")]      // %#g = Ctrl+Shift+G
    public static void ShowWindow()
    {
        GetWindow<BatchToolWindow>("批量生成");
    }

    private void OnGUI()
    {
        GUILayout.Label("批量生成设置", EditorStyles.boldLabel);

        _count = EditorGUILayout.IntSlider("数量", _count, 1, 100);
        _spacing = EditorGUILayout.Slider("间隔", _spacing, 0.5f, 10f);

        if (GUILayout.Button("生成"))
        {
            GenerateObjects();
        }

        if (GUILayout.Button("清理场景中的 Test_ 物体"))
        {
            CleanupTestObjects();
        }
    }

    private void GenerateObjects()
    {
        for (int i = 0; i < _count; i++)
        {
            var go = GameObject.CreatePrimitive(PrimitiveType.Cube);
            go.name = "Test_" + i;
            go.transform.position = new Vector3(i * _spacing, 0f, 0f);

            // 【重要】注册到 Undo 系统，支持 Ctrl+Z
            Undo.RegisterCreatedObjectUndo(go, "Generate Objects");
        }
        Debug.Log("生成了 " + _count + " 个物体");
    }

    private void CleanupTestObjects()
    {
        var all = FindObjectsOfType<GameObject>();
        int removed = 0;
        foreach (var go in all)
        {
            if (go.name.StartsWith("Test_"))
            {
                Undo.DestroyObjectImmediate(go);
                removed++;
            }
        }
        Debug.Log("清理了 " + removed + " 个物体");
    }
}
#endif

//  5. 资产导入处理器
// ============================================================
#if UNITY_EDITOR
public class TextureImportPostprocessor : AssetPostprocessor
{
    // 匹配路径：所有在 Textures/UI 下的贴图
    private void OnPreprocessTexture()
    {
        if (!assetPath.Contains("/Textures/UI/")) return;

        var importer = (TextureImporter)assetImporter;

        // 统一设置：UI 贴图用 Sprite 类型
        importer.textureType = TextureImporterType.Sprite;
        importer.spriteImportMode = SpriteImportMode.Single;

        // 移动端用 ASTC 压缩
        importer.textureCompression = TextureImporterCompression.CompressedHQ;

        Debug.Log("自动处理 UI 贴图: " + assetPath);
    }
}
#endif

// ============================================================
//  编辑器扩展的四个最佳实践
// ============================================================
//  1. 【必须用 Undo】任何修改都要能 Ctrl+Z，否则美术会恨你
//  2. 【用 #if UNITY_EDITOR 包裹】编辑器代码不能进包体
//  3. 【放在 Editor 目录】或加 asmdef，避免打包报错
//  4. 【给快捷键】常用的操作加 MenuItem 快捷键，效率翻倍
    问题没有工具时有工具时
    重复操作手动点 100 次一键完成
    配错数据运行时才发现编辑器里就拦住
    团队效率每个人都踩同样的坑工具把经验固化
    编辑器扩展的四个层次 ① 自定义 Inspector——让字段显示更清晰、加校验
    ② PropertyDrawer——让某个类型显示得更直观
    ③ EditorWindow——自制工具窗口，批量操作
    ④ AssetPostprocessor——资源导入时自动处理

    从 ① 开始做起，慢慢往 ④ 走。

    二、四个必须遵守的规则

    规则一：必须用 Undo 任何修改都要能 Ctrl+Z。
    用 Undo.RecordObject 记录修改前的状态，Undo.RegisterCreatedObjectUndo 登记新建对象。

    不这么做的后果是：美术点错一次按钮，整个场景白改了。他们下次就不会用你的工具了。
    规则二：编辑器代码不能进包体 编辑器专用代码必须用 #if UNITY_EDITOR 包裹，或者放在 Editor 目录下。

    否则打包时会报错——因为 UnityEditor 命名空间在运行时不存在。

    最常见的编译错误：The type or namespace name UnityEditor could not be found
    规则三：注意性能 OnInspectorGUI 会每帧调用多次（鼠标移动、重绘都会触发）。
    · 不要在里做昂贵计算
    · 不要用 FindObjectsOfType 之类遍历场景的 API
    · 需要缓存的数据存到字段里
    规则四：改完要 SetDirty 修改字段后必须调 EditorUtility.SetDirty(target)，否则改动不会保存到磁盘。
    表现是：改了、看起来生效了，但关掉 Unity 再打开就没了。

    三、工具链的全局视角

    编辑器扩展只是工具链的一环。完整的工程工具链还包括：

    环节工具解决的问题
    代码规范EditorConfig / 分析器命名、格式统一
    配置表Excel → ScriptableObject / JSON策划改数据不用找程序
    资源检查AssetPostprocessor / 自定义检查贴图格式错、命名不规范
    打包BuildPipeline / 命令行构建一键出包，可 CI 集成
    测试Unity Test Framework关键逻辑自动化验证
    版本控制Git + LFS / UnityYAMLMerge大文件与场景合并
    工具的价值在于"固化经验" 每个项目都会踩一些坑——贴图格式不对、资源命名混乱、场景里多余物件……
    没有工具时：每次都要靠人记住、靠 review 发现。
    有工具时：把这些检查写成脚本，问题在提交前就被拦住。

    判断标准很简单：同一个错误如果犯过两次，就该写个工具防止第三次。
