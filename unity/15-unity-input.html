    一、两套输入系统

    Unity 有两套并存的输入系统，这不是"新旧替代"的关系，而是"简单版与功能版"的关系。选错了会在项目中后期付出代价。

    using UnityEngine;

// ============================================================
//  输入系统：旧系统与 Input System 对比
// ============================================================
// 【重要】Unity 有两套输入系统，可以共存但不能混用同一个按键
//
//   旧系统：Input.GetAxis / GetKey     —— 简单，但难以重映射
//   新系统：InputAction / InputSystem  —— 功能强，包体积更大
//
// 本文件演示旧系统（新系统需要安装 package，且用不同的 API）

public class LegacyInput : MonoBehaviour
{
    [SerializeField] private float _moveSpeed = 5f;
    [SerializeField] private float _rotateSpeed = 180f;

    private void Update()
    {
        // ---------- 三种按键检测方式 ----------

        // 方式 1：GetKey —— 按住时一直为 true
        bool holding = Input.GetKey(KeyCode.W);

        // 方式 2：GetKeyDown —— 只在按下的那一帧为 true
        bool pressedThisFrame = Input.GetKeyDown(KeyCode.Space);

        // 方式 3：GetKeyUp —— 只在松开的那一帧为 true
        bool releasedThisFrame = Input.GetKeyUp(KeyCode.Space);

        // ---------- 轴输入：自带平滑 ----------
        // GetAxis 会做插值（有惯性），GetAxisRaw 直接返回 -1/0/1
        float h = Input.GetAxis("Horizontal");      // 平滑，适合移动
        float vRaw = Input.GetAxisRaw("Vertical");  // 立即，适合菜单

        // ---------- 鼠标 ----------
        float mouseX = Input.GetAxis("Mouse X");
        float mouseY = Input.GetAxis("Mouse Y");

        // 注意：鼠标位置是屏幕坐标（左下角为原点）
        Vector3 mousePos = Input.mousePosition;
        bool leftClick = Input.GetMouseButton(0);      // 0=左 1=右 2=中
        bool leftDown = Input.GetMouseButtonDown(0);

        // ---------- 移动与旋转 ----------
        Vector3 dir = new Vector3(h, 0f, vRaw);
        if (dir.sqrMagnitude > 1f) dir.Normalize();
        transform.position += dir * (_moveSpeed * Time.deltaTime);

        transform.Rotate(Vector3.up, mouseX * _rotateSpeed * Time.deltaTime);

        Debug.Log(holding + " " + pressedThisFrame + " " + releasedThisFrame);
        Debug.Log(mouseY + " " + mousePos + " " + leftClick + " " + leftDown);

        // ---------- 触屏（旧系统方式）----------
        HandleTouch();
    }

    private void HandleTouch()
    {
        if (Input.touchCount == 0) return;

        // 遍历所有触点（支持多点触控）
        for (int i = 0; i < Input.touchCount; i++)
        {
            Touch t = Input.GetTouch(i);

            switch (t.phase)
            {
                case TouchPhase.Began:
                    Debug.Log("手指按下 " + t.position);
                    break;
                case TouchPhase.Moved:
                    Debug.Log("移动了 " + t.deltaPosition);
                    break;
                case TouchPhase.Ended:
                    Debug.Log("抬起");
                    break;
            }
        }
    }

    // ---------- 陀螺仪与加速度计 ----------
    private void HandleSensors()
    {
        // 需要设备支持
        Vector3 accel = Input.acceleration;
        Vector3 gyro = Input.gyro.rotationRate;

        // 注意：陀螺仪需要先启用
        Input.gyro.enabled = true;

        Debug.Log(accel + " " + gyro);
    }
}

// ============================================================
//  输入重映射：旧系统的痛点
// ============================================================
// 旧系统的按键和轴是【在 Project Settings 里配置的】，
// 运行时虽然可以通过 InputManager 修改，但非常麻烦。
//
// 新系统（Input System）的 InputActionAsset 原生支持：
//   · 运行时重映射（Rebinding）
//   · 多设备并存（键鼠 + 手柄）
//   · 动作与按键解耦
//
// 【选型建议】
//   · 简单项目 / 原型        -> 旧系统够用
//   · 需要重映射 / 多手柄 / 主机平台 -> 新系统
//   · 已有项目              -> 不要为了"先进"而重写输入层

// ============================================================
//  条件编译：同时支持两套系统
// ============================================================
#if ENABLE_INPUT_SYSTEM
// 使用新系统
// using UnityEngine.InputSystem;
#endif

public class InputAdapter : MonoBehaviour
{
    // 通过宏定义做适配层，业务代码不关心底层用哪套
    public Vector2 GetMoveInput()
    {
#if ENABLE_INPUT_SYSTEM && !ENABLE_LEGACY_INPUT_MANAGER
        // 新版：需要 InputAction 引用
        return Vector2.zero;
#else
        return new Vector2(
            Input.GetAxisRaw("Horizontal"),
            Input.GetAxisRaw("Vertical"));
#endif
    }
}

    APIInput.GetAxis / GetKeyInputAction / 回调
    上手极简单需要学习
    重映射很难原生支持
    多设备手动处理自动切换

    移动端够用更强（触屏手势）

    二、旧系统的关键细节

    下面这些细节看起来简单但经常被搞错。

    GetAxis 与 GetAxisRaw 的区别 GetAxis 带平滑插值——输入 1 时返回值会逐渐从 0 涨到 1。这适合角色移动（手感更顺滑）。
    GetAxisRaw 立即返回 -1/0/1。这适合菜单导航、需要精确响应的场景。

    常见错误：用 GetAxis 做 UI 导航，导致按键响应"慢半拍"。
    输入必须放在 Update 这是页 11 讲过的内容，但在输入上尤其重要：
    FixedUpdate 一帧可能执行 0 次或多次，而 Input.GetKeyDown 只在按下的那一帧为 true。
    把它放在 FixedUpdate 里会导致——有时能检测到按键，有时丢失。

    这类 bug 极难复现，因为取决于帧率与物理步长的巧合。
    鼠标位置的坐标系 Input.mousePosition 返回的是屏幕坐标，左下角为原点（不是左上角）。
    而且它是 Vector3，z 值始终是 0——不要用它直接当世界坐标用（见页 26 讲的 ScreenToWorldPoint 陷阱）。

    三、音频系统

    Unity 的音频上手极简单——挂一个 AudioSource、拖一个 clip 就能响。但要做好听感，需要理解几个参数。

    属性范围说明
    volume0 ~ 1线性音量
    pitch0.1 ~ 31 是原速；0.95~1.05 用于制造变化
    loopbool循环播放
    playOnAwakeboolBGM 常用；音效要关掉
    spatialBlend0 ~ 10 = 2D，1 = 3D；中间值是混合
    rolloffMode枚举3D 衰减曲线，Linear 最可控
    minDistance米此距离内保持最大音量，常用 1~3
    maxDistance米超出后不再衰减，常用 20~40
    priority0 ~ 2550 是最高优先级（很多人搞反）
    dopplerLevel0 ~ 5设为 0 关闭多普勒效应
    两个反直觉的设计 ① priority 数字越小优先级越高。0 是最高，255 是最低。BGM 应设成较低优先级（如 128），重要音效设 0。

    ② dopplerLevel 默认是 1，会产生多普勒效应。高速移动的物体（子弹、飞车）会因此出现怪异的音调变化。绝大多数游戏应该直接设为 0。
    三个经验值 随机音调范围：0.95 ~ 1.05。再大就会明显跑调，再小听不出区别。
    3D 音效距离：minDistance 设 1~3 米，maxDistance 设 20~40 米。
    音量转 dB：20 * log10(线性值)。线性 0.5 约等于 -6dB。人类听感是对数的，所以音量滑条通常要转成 dB 才符合直觉。
