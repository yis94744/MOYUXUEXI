    一、三代资源方案的演进

    资源管理方案经历了三代。每一代都解决了上一代的问题，也带来了新的复杂度。

    using UnityEngine;
using System.Collections.Generic;

// ============================================================
//  资源管理：从 Resources 到 Addressables 的三代演进
// ============================================================
//  第一代 Resources.Load      —— 简单，但全量打包，不可热更
//  第二代 AssetBundle         —— 灵活，但依赖管理要自己做
//  第三代 Addressables        —— 自动依赖、可远程、引用计数
//
// 本文件演示三种方案的写法与取舍。

public class ResourcesDemo : MonoBehaviour
{
    // ============================================================
    //  第一代：Resources（不推荐）
    // ============================================================
    private void LoadFromResources()
    {
        // 【致命问题】Resources 目录下所有资源都会被【无条件】打进包里，
        // 无论有没有用到。这意味着：
        //   · 包体增大（哪怕是废弃资源也会被打包）
        //   · 启动变慢（启动时要建立索引）
        //   · 【无法热更新】（在包里，改不了）
        GameObject prefab = Resources.Load<GameObject>("Prefabs/Enemy");
        if (prefab != null)
        {
            Instantiate(prefab);
        }

        // 卸载：只能统一卸载，无法精确控制
        // Resources.UnloadUnusedAssets();   // 触发 GC，有性能开销

        // 【官方立场】Unity 明确不推荐使用 Resources 目录做资源管理
    }
}

// ============================================================
//  第二代：AssetBundle
// ============================================================
public class BundleDemo : MonoBehaviour
{
    private AssetBundle _bundle;

    private void LoadBundle()
    {
        // 加载 AB 包（本地用 LoadFromFile，远程用 UnityWebRequest）
        string path = System.IO.Path.Combine(Application.streamingAssetsPath, "enemy");
        _bundle = AssetBundle.LoadFromFile(path);

        if (_bundle == null)
        {
            Debug.LogError("Bundle 加载失败");
            return;
        }

        // 从包里加载具体资源
        GameObject prefab = _bundle.LoadAsset<GameObject>("Enemy");
        if (prefab != null) Instantiate(prefab);

        // 【关键】卸载策略（这是最容易搞错的地方）
        //   Unload(false)：只卸载包的结构，已加载的资源【保留】
        //   Unload(true)：连已加载的资源一起卸载（会导致场景物体变白）
        // _bundle.Unload(false);
    }

    // ---------- AB 包的三大痛点 ----------
    // 1. 【依赖管理】A 包依赖 B 包，必须自己维护依赖表并先加载 B
    // 2. 【重复资源】同一个贴图被多个包引用，会被打进多个包（包体膨胀）
    // 3. 【引用计数】资源被谁引用了、什么时候能卸载，全要自己管
}

// ============================================================
//  第三代：Addressables（推荐）
// ============================================================
public class AddressableDemo : MonoBehaviour
{
    // 用 async / await 的方式（需要 UniTask 或 Unity 2021+ 的 AsyncOperationHandle）
    private async void LoadAsync()
    {
        // 按地址加载，不需要知道它在哪个包里
        var handle = UnityEngine.AddressableAssets.Addressables
            .LoadAssetAsync<GameObject>("Assets/Prefabs/Enemy.prefab");

        await handle.Task;

        if (handle.Status == UnityEngine.ResourceManagement.AsyncOperations.AsyncOperationStatus.Succeeded)
        {
            Instantiate(handle.Result);
        }

        // 【重要】引用计数：加载一次，就要释放一次
        UnityEngine.AddressableAssets.Addressables.Release(handle);
    }

    // ---------- Addressables 解决的三个问题 ----------
    // 1. 自动依赖管理：声明了依赖，系统自动先加载
    // 2. 自动去重：同一个资源不会重复打包
    // 3. 引用计数：Load 加一，Release 减一，减到零才卸载

    // ---------- 但仍然要注意 ----------
    // · 必须严格配对 Load / Release，漏了会内存泄漏
    // · 首次加载有延迟（要读 catalog），需要预加载
    // · 远程加载要处理失败重试
}

// ============================================================
//  资源管理选型表
// ============================================================
//  方案            上手难度   热更   依赖管理   包体影响
//  ----------------------------------------------------------
//  Resources       极低      不支持  无        全量打包（差）
//  AssetBundle     高        支持    手动（难） 可控（好）
//  Addressables    中        支持    自动（好） 可控（好）
//
//  【结论】新项目一律用 Addressables。
//          小项目 / 原型可以用 Resources，但不要等到项目大了再迁移——
//          迁移成本远高于一开始就用对。

// ============================================================
//  资源加载的性能要点
// ============================================================
public class AssetPerf : MonoBehaviour
{
    // 1. 【异步加载】大资源必须异步，否则卡帧
    private void LoadAsyncPattern()
    {
        // 用协程或 async 加载，配合加载界面
        // StartCoroutine(LoadRoutine());
    }

    // 2. 【预加载】在加载界面把常用资源先加载好
    // 3. 【对象池】频繁创建销毁的对象用池（见页 11）
    // 4. 【及时释放】场景切换时释放不再用的资源
    // 5. 【图集】UI 小图合并成大图，减少 Draw Call 与加载次数
    // 6. 【压缩格式】纹理用 ASTC（移动端）/ DXT（PC），不要用无压缩
}

    Resources极低不支持无全量打包（差）
    AssetBundle高支持手动（难）可控（好）
    Addressables中支持自动（好）可控（好）

    二、Resources 为什么被官方弃用

    Resources 的三个致命问题 ① 全量打包：Resources 目录下所有资源都会被无条件打进包里，无论有没有用到。
    · 废弃资源也会被打包 → 包体膨胀
    · 启动时要建立索引 → 启动变慢

    ② 无法热更新：资源在包里，改了必须重新发版。这对需要频繁更新的手游是致命的。

    ③ 内存无法精确控制：只能 UnloadUnusedAssets 统一卸载，而它本身会触发 GC，有性能开销。

    Unity 官方明确不推荐使用 Resources 目录做资源管理。

    三、AssetBundle 的三大痛点

    AssetBundle 解决了热更新，但把复杂度全部转嫁给了开发者。

    痛点一：依赖管理 A 包依赖 B 包，你必须自己维护依赖表，并保证加载 A 之前先加载 B。
    漏了就会看到——贴图变白、材质丢失、模型变成紫色方块。
    痛点二：重复资源 同一个贴图被多个包引用时，会被打进多个包。
    这会导致包体膨胀，而且同一个资源在内存里有多份副本。

    解决方法是做"依赖包"提取公共资源——但那个配置表非常难维护。
    痛点三：引用计数 资源被谁引用了、什么时候能卸载，全部要自己管。
    卸载早了 → 场景物体变白；卸载晚了 → 内存爆掉。

    Unload(false) 与 Unload(true) 的区别是经典面试题。

    四、Addressables 解决什么

    Addressables 的三个优势 ① 自动依赖管理：声明了依赖，系统自动先加载。不用再手写依赖表。
    ② 自动去重：同一个资源不会重复打包。
    ③ 引用计数：Load 加一，Release 减一，减到零才卸载。卸载时机自动管理。

    但仍要注意：
    · 必须严格配对 Load / Release，漏了会内存泄漏
    · 首次加载有延迟（要读 catalog），需要预加载
    · 远程加载要处理失败重试
    迁移成本提示 小项目 / 原型可以用 Resources，但不要等到项目大了再迁移——迁移成本远高于一开始就用对。
    新项目一律用 Addressables。
    已有项目：如果包体还能接受，不要为了"先进"而重写资源层。

    五、热更新的完整流程

    热更新是上线手游的标配能力。它的流程是：

    步骤做什么关键点
    1. 资源版本服务器维护资源版本号与清单客户端启动时比对
    2. 差异下载只下载变化的资源包增量更新，不是全量
    3. 校验MD5 / 大小校验防止下载损坏
    4. 落盘写入持久化目录persistentDataPath
    5. 切换下次启动使用新资源支持回滚
    热更新最容易出事故的三个点 ① 下载中断：玩家在弱网下载到一半退出。必须支持断点续传，且保证半包不会被使用。

    ② 版本不一致：玩家 A 更新了、玩家 B 没更新，两人联机会出问题。需要做版本校验与强制更新。

    ③ 无法回滚：新版本有严重 bug 时，必须能快速回退到上一版本。
    业界标准做法：保留上一版本的完整资源，出问题时切换回去。

    这一条是"能不能上线的分界线"——没有回滚方案的热更，等于把线上项目放在没有刹车的车上。
