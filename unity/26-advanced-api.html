    一、零分配物理查询 ★

    这是收益最大的一类优化。物理查询在游戏中调用极其频繁（射线检测、范围检测、碰撞预测），而 Alloc 版本每次都会分配一个数组——每帧几十次调用就是每帧几十个数组。

    分配版本零分配版本说明
    Physics.RaycastAllPhysics.RaycastNonAlloc多个命中结果
    Physics.SphereCastAllPhysics.SphereCastNonAlloc球形投射多个结果
    Physics.BoxCastAllPhysics.BoxCastNonAlloc盒形投射多个结果
    Physics.CapsuleCastAllPhysics.CapsuleCastNonAlloc胶囊投射多个结果
    Physics.OverlapSpherePhysics.OverlapSphereNonAlloc球形范围内的碰撞体
    Physics.OverlapBoxPhysics.OverlapBoxNonAlloc盒形范围内的碰撞体
    Physics.OverlapCapsulePhysics.OverlapCapsuleNonAlloc胶囊范围内的碰撞体
     
    using UnityEngine;

// ============================================================
//  零分配物理查询：Alloc 版本 vs NonAlloc 版本
// ============================================================
public class NonAllocDemo : MonoBehaviour
{
    // 预分配缓冲区，全程复用（这是关键）
    private readonly RaycastHit[] _hits = new RaycastHit[16];
    private readonly Collider[] _overlaps = new Collider[32];
    private readonly Collider[] _boxHits = new Collider[16];
    private readonly Collider[] _capsuleHits = new Collider[16];
    private readonly RaycastHit[] _sphereHits = new RaycastHit[8];

    private void Update()
    {
        // ---------- 射线：NonAlloc ----------
        // 返回命中数量，而不是数组；数组预先传进去
        int count = Physics.RaycastNonAlloc(
            transform.position,      // 起点
            Vector3.forward,         // 方向
            _hits,                   // 结果缓冲区
            100f,                    // 最大距离
            ~0,                      // 层掩码（~0 = 所有层）
            QueryTriggerInteraction.Ignore);

        // 缓冲区满的警告：结果可能被截断
        if (count == _hits.Length)
        {
            Debug.LogWarning("射线缓冲可能已满，考虑增大数组");
        }

        for (int i = 0; i < count; i++)
        {
            Debug.Log(_hits[i].collider.name + " 距离 " + _hits[i].distance);
        }

        // ---------- 球形范围：NonAlloc ----------
        int sphereCount = Physics.OverlapSphereNonAlloc(
            transform.position, 5f, _overlaps);

        for (int i = 0; i < sphereCount; i++)
        {
            // 处理范围内的碰撞体
        }

        // ---------- 盒形范围 ----------
        int boxCount = Physics.OverlapBoxNonAlloc(
            transform.position,              // 中心
            new Vector3(1f, 1f, 1f),         // 半尺寸（注意是半尺寸！）
            _boxHits,
            transform.rotation,
            ~0,
            QueryTriggerInteraction.Ignore);

        // ---------- 胶囊范围 ----------
        int capCount = Physics.OverlapCapsuleNonAlloc(
            transform.position + Vector3.up,        // 起点
            transform.position - Vector3.up,        // 终点
            0.5f,                                   // 半径
            _capsuleHits);

        // ---------- 球形投射 ----------
        int sphereCastCount = Physics.SphereCastNonAlloc(
            transform.position,
            0.5f,                                   // 球半径
            Vector3.forward,
            _sphereHits,
            50f);

        Debug.Log(count + sphereCount + boxCount + capCount + sphereCastCount);
    }

    // ---------- 对比：Alloc 版本的问题 ----------
    private void AllocVersion()
    {
        // 每次调用都分配一个新数组 —— 每帧执行就是每帧分配
        RaycastHit[] hits = Physics.RaycastAll(transform.position, Vector3.forward, 100f);

        // 而且 RaycastAll 的顺序是「未排序」的，NonAlloc 版本也一样
        // 需要最近的命中要自己排序
        System.Array.Sort(hits, (a, b) => a.distance.CompareTo(b.distance));
    }

    // ---------- 要「最近的命中」其实不用数组 ----------
    private void ClosestOnly()
    {
        // 单个 Raycast 就已经按距离返回最近命中，零分配
        if (Physics.Raycast(transform.position, Vector3.forward, out RaycastHit hit, 100f))
        {
            Debug.Log(hit.collider.name + " " + hit.point);
        }
    }
}
    NonAlloc 的三个必知细节 ① 返回的是数量，不是数组。你必须自己用 for (int i = 0; i < count; i++) 遍历缓冲区。
    ② 缓冲区可能不够。当返回值等于数组长度时，说明结果被截断了——必须检查并给出警告。
    ③ 结果顺序不确定。NonAlloc 和 Alloc 版本都不保证按距离排序。要最近的命中，直接用单个 Physics.Raycast 更划算（它本身就返回最近命中）。
    还有一个容易踩的坑：OverlapBoxNonAlloc 的第二个参数是半尺寸（halfExtents），不是完整尺寸。传 Vector3.one 得到的是边长 2 的盒子。

    二、Transform 进阶 API

    Transform 的属性访问每次都要跨越 C# 与 C++ 的边界，这有真实开销。Unity 后来加了一批"批量"API 专门解决这个问题——但很多人不知道它们存在。

    不推荐写法推荐写法收益
    t.position = p; t.rotation = r;t.SetPositionAndRotation(p, r)一次边界调用代替两次
    var p = t.position; var r = t.rotation;t.GetPositionAndRotation(out p, out r)同上
    t.localPosition = ...; t.localRotation = ...t.SetLocalPositionAndRotation(...)同上
    每帧访问 Camera.mainAwake 里缓存到字段省掉 FindGameObjectWithTag
    用 GetComponent 判断非空TryGetComponent(out T x)不存在时不产生分配
     
    using UnityEngine;

// ============================================================
//  Transform 进阶：性能与精度
// ============================================================
public class AdvancedTransform : MonoBehaviour
{
    [SerializeField] private Transform _target;

    private void Update()
    {
        // ---------- 一秒设置位置+旋转（避免两次属性访问）----------
        // 旧写法：transform.position = p; transform.rotation = r;
        // 每次访问 position/rotation 都涉及 C++ <-> C# 边界调用
        transform.SetPositionAndRotation(_target.position, _target.rotation);
        transform.SetLocalPositionAndRotation(
            new Vector3(0f, 0f, 0f), Quaternion.identity);

        // ---------- 一次性读取位置+旋转 ----------
        transform.GetPositionAndRotation(out Vector3 worldPos, out Quaternion worldRot);
        transform.GetLocalPositionAndRotation(out Vector3 localPos, out Quaternion localRot);

        Debug.Log(worldPos + " " + worldRot + " " + localPos + " " + localRot);

        // ---------- 坐标空间转换（成对比记忆）----------
        // 局部点 -> 世界点
        Vector3 worldPoint = transform.TransformPoint(new Vector3(0f, 1f, 0f));
        // 世界点 -> 局部点
        Vector3 localPoint = transform.InverseTransformPoint(worldPoint);

        // 方向（不受缩放和位置影响）
        Vector3 worldDir = transform.TransformDirection(Vector3.forward);
        Vector3 localDir = transform.InverseTransformDirection(worldDir);

        // 向量（受缩放影响）
        Vector3 worldVec = transform.TransformVector(Vector3.right);
        Vector3 localVec = transform.InverseTransformVector(worldVec);

        Debug.Log(worldPoint + " " + localPoint + " " + worldDir + " " + localDir);
        Debug.Log(worldVec + " " + localVec);
    }

    // ---------- 父级关系操作 ----------
    private void HierarchyOps()
    {
        // 设置父物体：第二个参数 true = 保持世界坐标不变
        transform.SetParent(_target, true);

        // 脱离父物体
        transform.SetParent(null);

        // 层级顺序（决定 UI 覆盖关系、渲染顺序）
        transform.SetAsFirstSibling();
        transform.SetAsLastSibling();
        transform.SetSiblingIndex(2);
        int idx = transform.GetSiblingIndex();

        // 解除所有子物体的父子关系
        transform.DetachChildren();

        // 判断关系
        bool isChild = transform.IsChildOf(_target);

        Debug.Log(idx + " " + isChild);
    }

    // ---------- 查找子物体 ----------
    private void Finding()
    {
        // Find：只找直接子物体，路径式查找
        Transform t1 = transform.Find("Head");
        Transform t2 = transform.Find("Body/Arm/Hand");

        // 递归查找（Unity 没有内置递归查找，要自己写）
        Transform t3 = FindRecursive(transform, "Hand");

        Debug.Log(t1 + " " + t2 + " " + t3);
    }

    private static Transform FindRecursive(Transform parent, string name)
    {
        for (int i = 0; i < parent.childCount; i++)
        {
            Transform child = parent.GetChild(i);
            if (child.name == name) return child;
            Transform r = FindRecursive(child, name);
            if (r != null) return r;
        }
        return null;
    }

    // ---------- 绕点旋转 / 朝向 ----------
    private void RotationOps()
    {
        // 绕某点旋转（适合做轨道相机、卫星运动）
        transform.RotateAround(_target.position, Vector3.up, 90f * Time.deltaTime);

        // 朝向目标（世界空间的上方向）
        transform.LookAt(_target, Vector3.up);
    }
}
    TransformPoint 与 TransformDirection 的区别 TransformPoint：把点从局部空间转到世界空间——受位置、旋转、缩放影响。
    TransformDirection：把方向从局部转到世界——只受旋转影响，不受位置和缩放影响。
    TransformVector：把向量从局部转到世界——受旋转和缩放影响，不受位置影响。
    怎么记：点在空间里有位置，方向只是朝向，向量有长度。搞混会导致"位置偏移被重复计算"这类难查的 bug。

    三、组件使用技巧

    using UnityEngine;

// ============================================================
//  组件使用技巧：查找、缓存、批量操作
// ============================================================
public class ComponentTips : MonoBehaviour
{
    // ---------- 技巧 1：优先用 TryGetComponent，不用 GetComponent + null 判断 ----------
    private Rigidbody _rb;

    private void Awake()
    {
        // ❌ 旧写法：GetComponent 在组件不存在时会分配（用于错误信息）
        // if (GetComponent<Rigidbody>() != null) { }

        // ✅ TryGetComponent：不存在时不分配、不抛异常
        if (TryGetComponent(out _rb))
        {
            _rb.mass = 1f;
        }

        // ✅ 支持接口类型（Unity 2019.2+），这是解耦的关键
        if (TryGetComponent(out IDamageable damageable))
        {
            damageable.TakeDamage(0f);
        }
    }

    // ---------- 技巧 2：GetComponent 缓存的正确位置 ----------
    private Animator _animator;

    private void Start()
    {
        // Awake/Start 里查一次，之后一直用字段
        _animator = GetComponentInChildren<Animator>();

        // ❌ 绝对不要在 Update 里反复 GetComponent / Camera.main /
        //    GameObject.Find / FindObjectOfType —— 全都有查找开销
    }

    // ---------- 技巧 3：用 GetComponentsInChildren 一次性收集 ----------
    private Renderer[] _renderers;

    private void CollectOnce()
    {
        // 第二个参数 includeInactive：是否包含未激活的子物体
        _renderers = GetComponentsInChildren<Renderer>(true);
    }

    // ---------- 技巧 4：即使不用 List 参数，也有零分配重载 ----------
    private readonly System.Collections.Generic.List<Collider> _buffer =
        new System.Collections.Generic.List<Collider>(32);

    private void NoAllocCollect()
    {
        // GetComponents 有 List 重载，复用同一个 List 就不会分配
        GetComponentsInChildren(_buffer);
        GetComponents(_buffer);
    }

    // ---------- 技巧 5：组件启用/禁用用 enabled，不是 SetActive ----------
    private void Toggle()
    {
        // enabled 只关闭这个组件的逻辑，物体和其他组件照常
        _animator.enabled = false;

        // SetActive 会关闭整个物体（含所有组件、子物体、协程）
        // gameObject.SetActive(false);
    }
}

public interface IDamageable
{
    void TakeDamage(float amount);
}
    关于 GetComponent 的性能，说清楚一件事 网上常说 GetComponent 很慢，但实际情况是有条件的：
    ① 组件不存在时的分配：旧版 GetComponent 在找不到组件时会构造错误信息字符串。TryGetComponent 解决了这个问题。
    ② 每帧重复查找：即使单次很快，每帧调用几千次也会累积成瓶颈。在 Awake 里缓存一次是标准做法。
    结论：不要因为"性能"就把所有 GetComponent 都提前缓存到 Awake，那会拖慢启动；只需把热路径上的找出来缓存。

    四、相机与坐标转换

    using UnityEngine;

// ============================================================
//  相机与坐标转换（游戏里高频需求）
// ============================================================
public class CameraAnd2D : MonoBehaviour
{
    [SerializeField] private Camera _camera;
    [SerializeField] private Transform _worldObject;

    private Camera Cam
    {
        get
        {
            // 缓存 Camera.main：它内部是 FindGameObjectWithTag，有开销
            if (_camera == null) _camera = Camera.main;
            return _camera;
        }
    }

    private void Update()
    {
        // ---------- 屏幕坐标 -> 世界坐标 ----------
        // 用途：鼠标点击选择、拖拽
        if (Input.GetMouseButtonDown(0))
        {
            Vector3 screenPos = Input.mousePosition;

            // 2D 游戏：直接用 z 值表示与相机的距离
            Vector3 world = Cam.ScreenToWorldPoint(screenPos);

            // 3D 游戏：先发一条射线（更准确）
            Ray ray = Cam.ScreenPointToRay(screenPos);
            if (Physics.Raycast(ray, out RaycastHit hit, 100f))
            {
                Debug.Log("点到了 " + hit.collider.name + " 在 " + hit.point);
            }

            // 注意：ScreenToWorldPoint 结果在相机近裁剪面上，z 不是 0
            Debug.Log(world);
        }

        // ---------- 世界坐标 -> 屏幕坐标 ----------
        // 用途：头顶血条、UI 跟随、屏幕外指示箭头
        Vector3 sp = Cam.WorldToScreenPoint(_worldObject.position);

        // z > 0 表示在相机前方；z < 0 表示在相机背后（要特殊处理）
        if (sp.z > 0f)
        {
            // 用 sp.x, sp.y 设置 UI 位置
        }

        // ---------- 视口坐标（0~1 归一化）----------
        // 用途：判断是否在屏幕内、自适应布局
        Vector3 vp = Cam.WorldToViewportPoint(_worldObject.position);
        bool onScreen = vp.x > 0f && vp.x < 1f && vp.y > 0f && vp.y < 1f && vp.z > 0f;

        Vector3 backToWorld = Cam.ViewportToWorldPoint(new Vector3(0.5f, 0.5f, 10f));

        Debug.Log(onScreen + " " + backToWorld);
    }
}
    ScreenToWorldPoint 的 z 值陷阱 这不是 bug，是透视投影的数学结果。对于透视相机，屏幕上的一个点对应的是一条射线，而不是一个确定的世界点。
    ScreenToWorldPoint 返回的是近裁剪面上的那个点——所以它的 z 值不是 0，也不等于相机到目标的距离。
    正确做法：① 2D 正交相机把传入的 screenPos.z 设成相机到平面的距离；② 3D 透视相机用 ScreenPointToRay 加射线检测，这才是准确的做法。

    五、音频实用技巧

    using UnityEngine;

// ============================================================
//  音频系统的实用技巧
// ============================================================
public class AudioTips : MonoBehaviour
{
    [SerializeField] private AudioSource _bgm;
    [SerializeField] private AudioSource _sfx;
    [SerializeField] private AudioClip[] _stepClips;

    private int _lastStepIndex = -1;

    // ---------- 技巧 1：PlayOneShot 播放不打断的音效 ----------
    public void PlaySfx(AudioClip clip, float volume = 1f)
    {
        // PlayOneShot 可以在同一个 AudioSource 上叠加播放
        // 适合：脚步、射击、UI 点击等短音效
        _sfx.PlayOneShot(clip, volume);
    }

    // ---------- 技巧 2：避免重复播放同一音效（随机 + 去重）----------
    public void PlayRandomStep()
    {
        if (_stepClips == null || _stepClips.Length == 0) return;

        // 避免连续两次播同一个，听感更自然
        int idx = Random.Range(0, _stepClips.Length);
        if (_stepClips.Length > 1 && idx == _lastStepIndex)
        {
            idx = (idx + 1) % _stepClips.Length;
        }
        _lastStepIndex = idx;

        // 随机音调让重复音效不单调（0.95~1.05 是常用范围）
        _sfx.pitch = Random.Range(0.95f, 1.05f);
        _sfx.PlayOneShot(_stepClips[idx]);
    }

    // ---------- 技巧 3：音量是对数感知，不是线性的 ----------
    public void SetVolume(float slider01)
    {
        // 直接赋值 slider01 会感觉「前面变化太快、后面听不出来」
        // 用 Log10 转换更符合听感：0.5 -> -6dB
        float db = Mathf.Log10(Mathf.Max(slider01, 0.0001f)) * 20f;
        _bgm.volume = slider01;   // AudioSource.volume 本身是线性 0~1
        Debug.Log("线性 " + slider01 + " 约等于 " + db + " dB");
    }

    // ---------- 技巧 4：AudioSource 的循环与淡入淡出 ----------
    public void PlayBgm(AudioClip clip)
    {
        _bgm.clip = clip;
        _bgm.loop = true;
        _bgm.volume = 0f;
        _bgm.Play();
        // 之后在 Update 里把 volume 逐渐加到目标值，做淡入
    }

    // ---------- 技巧 5：暂停游戏时音效也要停 ----------
    // Time.timeScale = 0 时：
    //   - AudioSource.Pause() / UnPause() 手动控制
    //   - 或用 AudioListener.pause = true 全局暂停
    public void OnGamePaused(bool paused)
    {
        AudioListener.pause = paused;
    }

    // ---------- 技巧 6：3D 音效的距离设置 ----------
    public void Configure3DSound()
    {
        _sfx.spatialBlend = 1f;          // 0 = 2D，1 = 3D
        _sfx.rolloffMode = AudioRolloffMode.Linear;
        _sfx.minDistance = 2f;           // 小于此距离保持最大音量
        _sfx.maxDistance = 30f;          // 大于此距离不再衰减（配合 Linear）
        _sfx.dopplerLevel = 0f;          // 关闭多普勒，避免移动时音调怪异
    }
}
    音频的三个经验数字 随机音调范围：0.95 ~ 1.05。再大就会明显跑调，再小听不出区别。
    3D 音效距离：minDistance 通常设 1 到 3 米，maxDistance 设 20 到 40 米。超过 maxDistance 后音量不再衰减，所以要配合 Linear 衰减模式使用。
    音量转 dB：20 * log10(线性值)。线性 0.5 约等于 -6dB，线性 0.1 约等于 -20dB。人类听感是对数的，所以音量滑条通常要转成 dB 才符合直觉。
