    一、协程：分帧执行的方法

    协程是 Unity 特有的机制。它的本质是：一段可以"暂停"和"恢复"的方法，由 Unity 的更新循环驱动。它不是线程——协程全程在主线程上跑，只是把一段逻辑拆到多帧执行。

    using System.Collections;
using System.Collections.Generic;
using UnityEngine;

// ============================================================
//  协程基础：yield 的全部用法
// ============================================================
public class CoroutineDemo : MonoBehaviour
{
    private Coroutine _handle;

    private void Start()
    {
        _handle = StartCoroutine(MyRoutine());
        // StopCoroutine(_handle);
        // StopAllCoroutines();
    }

    private IEnumerator MyRoutine()
    {
        Debug.Log("第一帧");

        yield return null;                                  // 下一帧
        yield return new WaitForSeconds(1f);                // 1 秒（受 timeScale 影响）
        yield return new WaitForSecondsRealtime(1f);        // 真实 1 秒
        yield return new WaitForFixedUpdate();              // 下一个 FixedUpdate 之后
        yield return new WaitForEndOfFrame();               // 本帧渲染结束后

        yield return new WaitUntil(() => Input.GetKeyDown(KeyCode.Space));
        yield return new WaitWhile(() => transform.position.y > 0f);

        yield return StartCoroutine(OtherRoutine());        // 等另一个协程结束

        yield break;                                        // 立即结束
    }

    private IEnumerator OtherRoutine()
    {
        yield return null;
    }
}

// ============================================================
//  陷阱 1：物体被禁用后协程停止，且不会恢复
// ============================================================
public class CoroutineTrap1 : MonoBehaviour
{
    public void Disable()
    {
        gameObject.SetActive(false);   // 所有协程立刻停止，且不会自动恢复
    }
}

// ============================================================
//  陷阱 2：不能在 try-catch 里 yield

public class CoroutineTrap2 : MonoBehaviour

    // 编译错误：不能在 try-catch 中 yield
    // private IEnumerator Bad()

// ============================================================
//  陷阱 3：WaitForSeconds 每次都 new 会产生 GC
// ============================================================
public static class WaitCache
{
    private static readonly Dictionary<float, WaitForSeconds> _cache
        = new Dictionary<float, WaitForSeconds>();

    public static WaitForSeconds Get(float seconds)
    {
        // 量化到 0.05 秒，避免浮点键过多
        float key = Mathf.Round(seconds * 20f) / 20f;
        if (!_cache.TryGetValue(key, out WaitForSeconds w))
        {
            w = new WaitForSeconds(key);
            _cache[key] = w;
        }
        return w;
    }
}

public class CoroutineTrap3 : MonoBehaviour
{
    private IEnumerator Fire()
    {
        while (true)
        {
            yield return WaitCache.Get(0.1f);   // 复用，不再每次 new
        }
    }
}

    yield 指令完整对照表

    yield 指令恢复时机受 timeScale 影响典型用途
    yield return null下一帧的 Update 之后否逐帧逻辑、等待一帧
    new WaitForSeconds(t)t 秒后（缩放时间）是冷却、延时
    new WaitForSecondsRealtime(t)t 秒后（真实时间）否暂停菜单里的动画
    new WaitForFixedUpdate()下一个 FixedUpdate 之后是物理相关
    new WaitForEndOfFrame()本帧渲染结束后否截图、读像素
    new WaitUntil(pred)条件为 true 后否等待某个状态
    new WaitWhile(pred)条件为 false 后否等待某个状态结束
    StartCoroutine(other)该协程结束后—协程串行
    yield break立即结束协程—提前退出
    WaitForEndOfFrame 在编辑器里的一个坑 当 Game 视图不可见时（窗口被最小化、或切到其他标签页），WaitForEndOfFrame 不会触发。
    表现是：你的截图协程挂住了，直到切回 Game 视图才继续。这是"截图功能在编辑器里偶尔卡住"的经典原因。
    真机上没有这个问题。所以截图相关功能一定要在真机上验证。

    二、协程的四个陷阱

    陷阱 1：物体被禁用，协程停止且不恢复

    这是最隐蔽的协程 bug gameObject.SetActive(false) 会立即停止该对象上所有协程，而且重新激活后不会自动恢复。
    典型场景：你做"角色复活"，把角色 SetActive(false) 再 SetActive(true)，结果复活动画的协程再也不执行了。
    解决方式：① 不用 SetActive，改用其他方式隐藏；② 在 OnEnable 里重新启动协程；③ 用状态机管理。

    陷阱 2：不能在 try-catch 里 yield

    这是 C# 语言的限制（不是 Unity 的）。yield return 不能出现在 try 块里带 catch 的场景。解决方式是把风险代码提取成独立方法。

    陷阱 3：WaitForSeconds 每次 new 都产生 GC

    这是页 07 清单里的第 6 条。高频循环的协程（比如每 0.1 秒执行一次）长时间运行会产生可观的垃圾。

    陷阱 4：宿主销毁后协程的残留引用

    using System.Collections;
using UnityEngine;

// 协程宿主销毁时的行为
public class CoroutineOwner : MonoBehaviour
{
    private void Start()
    {
        StartCoroutine(Routine());
    }

    private IEnumerator Routine()
    {
        while (true)
        {
            yield return new WaitForSeconds(1f);

            // 宿主销毁后协程会停止，这一行不会执行
            // 所以直接访问 this 的成员是"相对安全"的
            Debug.Log(transform.position);
        }
    }
}

// 但如果是"别的对象"持有协程呢？
public class CoroutineHolder : MonoBehaviour
{
    // 防御性写法：每轮都检查目标是否还活着
    public IEnumerator SafeRoutine(GameObject target)
    {
        while (target != null)      // 用 Unity 的 == 判断（能识别"假 null"）
        {
            yield return new WaitForSeconds(1f);
            if (target == null) yield break;
            target.transform.Rotate(Vector3.up, 10f);
        }
    }
}

    三、async/await：真正的异步

    协程解决的是"分帧执行"，但它不能真正并行——因为协程还在主线程上。如果你要读一个 100MB 的文件、或者发一个网络请求，协程依然会卡住主线程。这时需要 async/await 和真正的后台线程。

    Unity 里 async/await 的铁律 在线程池线程上调用任何 Unity API 都会抛异常。错误信息是 "xxx can only be called from the main thread"。
    Transform、GameObject、Time——这些都不是线程安全的。
    记住：await 之后你在哪个线程，取决于你 await 的是什么。
    await 的对象await 之后的线程能否直接调 Unity API
    Task.Run(() => ...)线程池线程不能
    Task.Delay(n)主线程（默认上下文）可以
    await task.ConfigureAwait(false)线程池线程不能
    UniTask 的 UniTask.Delay主线程可以
    UniTask 的 SwitchToThreadPool()线程池线程不能
    UniTask 的 SwitchToMainThread()主线程可以
     
    using System.Threading;
using System.Threading.Tasks;
using UnityEngine;

// ============================================================
//  async/await 在 Unity 中的正确用法
// ============================================================
public class AsyncDemo : MonoBehaviour
{
    private CancellationTokenSource _cts;

    private void Start()
    {
        _cts = new CancellationTokenSource();
        _ = LoadAsync(_cts.Token);
    }

    private void OnDestroy()
    {
        // 对象销毁时取消任务，避免回调到已销毁对象
        _cts?.Cancel();
        _cts?.Dispose();
    }

    private async Task LoadAsync(CancellationToken token)
    {
        // await 之后可能回到线程池线程，不能直接调用 Unity API
        string json = await Task.Run(() =>
            System.IO.File.ReadAllText("data.json"), token);

        if (token.IsCancellationRequested) return;
        if (this == null) return;      // 对象可能已被销毁

        // 回到主线程后再调用 Unity API
        Debug.Log(json.Length);
    }
}

// ============================================================
//  错误示范：三种典型崩溃
// ============================================================
public class AsyncWrong : MonoBehaviour
{
    // 错误 1：在线程池线程上调用 Unity API
    private async void Wrong1()
    {
        await Task.Run(() =>
        {
            // 崩溃：Unity API 只能在主线程调用
            // Debug.Log(transform.position);
        });
    }

    // 错误 2：async void 的异常无法被捕获
    private void Wrong2()
    {
        // CallAsyncVoid();   // 里面抛异常会直接崩溃，无法 try-catch
    }
    private async void CallAsyncVoid()
    {
        await Task.Delay(100);
        throw new System.Exception("这个异常会直接让程序崩溃");
    }

    // 错误 3：对象销毁后继续执行
    private async Task Wrong3()
    {
        await Task.Delay(5000);     // 等 5 秒
        transform.position = Vector3.zero;   // 此时对象可能已被销毁 -> MissingReferenceException
    }

    // 正确的替代方案
    private async Task Right3(CancellationToken token)
    {
        await Task.Delay(5000, token);
        if (this == null || token.IsCancellationRequested) return;
        transform.position = Vector3.zero;
    }
}

    三个典型崩溃场景

    1. 在线程池线程上调 Unity API → 抛 UnityException: ... can only be called from the main thread。
    2. 用 async void → 里面的异常无法被任何 try-catch 捕获，直接崩溃。除了事件处理器，一律用 async Task。
    3. 对象销毁后 await 回来继续执行 → MissingReferenceException。必须用 CancellationToken 或在 await 后检查 this == null。

    四、UniTask：Unity 的现代异步方案

    原生 Task 在 Unity 里有几个问题：每次分配、容易用错线程、和 Unity 的 PlayerLoop 集成不好。UniTask 是社区的事实标准解决方案，它解决了这些问题。

    维度 Coroutine async/await + Task UniTask
    GC 分配 每次 StartCoroutine 有少量 较多 Task 是 class 零分配 基于 struct
    返回值 不能返回 可以 可以
    异常传播 只能 Debug.Log 传播到 await 处 传播
    取消机制 StopCoroutine CancellationToken CancellationToken
    线程安全 主线程 可能切线程 主线程
    依赖 内置 内置 需安装包
    适用 简单延时、序列动画 IO 密集、网络 现代 Unity 项目首选
    协程还是 UniTask？给一个明确的建议 简单场景用协程：等 1 秒、做一个序列动画、每隔 N 帧执行一次。这些用协程最直观，不值得引入额外依赖。
    复杂场景用 UniTask：需要返回值、需要异常处理、需要取消、需要和 Task/网络库交互。
    不要混用。一个文件里既有 StartCoroutine 又有 async UniTask，会让"这段代码什么时候执行完"变得极难追踪。按模块统一。
    迁移策略：新项目直接用 UniTask；老项目不要批量重写协程，只在新增的复杂异步逻辑上用 UniTask。
