    一、优化优先级链

    渲染优化必须按优先级做。下面这张表按"收益/难度"排序——从下往上做的都是浪费。

    using UnityEngine;

// ============================================================
//  渲染优化：从 Draw Call 到 Overdraw
// ============================================================
//  渲染优化有明确的优先级。按收益从高到低：
//    1. 减少 Draw Call（合批）
//    2. 减少 Overdraw（半透明、粒子）
//    3. 降低分辨率与阴影质量
//    4. 简化 Shader
//    5. LOD 与剔除

public class RenderOptimize : MonoBehaviour
{
    [SerializeField] private Renderer[] _renderers;
    [SerializeField] private MaterialPropertyBlock _mpb;

    private void Awake()
    {
        if (_mpb == null) _mpb = new MaterialPropertyBlock();
    }

    // ============================================================
    //  技巧 1：MaterialPropertyBlock —— 改颜色不破坏合批
    // ============================================================
    private void SetColorByPropertyBlock(Renderer r, Color c)
    {
        // 【关键】用 MPB 不会克隆材质，因此【不破坏 SRP Batcher 之外的合批】
        // 但注意：MPB 会打断 SRP Batcher（URP 下的主力合批方式）
        r.GetPropertyBlock(_mpb);
        _mpb.SetColor("_BaseColor", c);          // URP 用 _BaseColor
        _mpb.SetColor("_Color", c);              // Built-in 用 _Color
        r.SetPropertyBlock(_mpb);
    }

    // ============================================================
    //  技巧 2：关闭不必要的阴影
    // ============================================================
    private void OptimizeShadows()
    {
        foreach (var r in _renderers)
        {
            // 小物件、地面装饰的阴影收益很低，但开销不小
            r.shadowCastingMode = UnityEngine.Rendering.ShadowCastingMode.Off;
            r.receiveShadows = false;
        }
    }

    // ============================================================
    //  技巧 3：用 isVisible 跳过不可见物体的逻辑
    // ============================================================
    private void Update()
    {
        foreach (var r in _renderers)
        {
            // isVisible 表示"被任何相机看到"
            // 不可见的物体可以跳过昂贵的逻辑（动画更新、特效计算）
            if (!r.isVisible) continue;

            // 只对可见物体做重活
        }
    }

    // ============================================================
    //  技巧 4：控制 Overdraw
    // ============================================================
    // Overdraw = 同一个像素被绘制多次。
    //   常见来源：全屏半透明 UI、粒子特效、地面贴花
    //   移动端上，Overdraw 是 GPU 的主要瓶颈之一
    //
    // 优化手段：
    //   · 减少全屏半透明层数（背景 + 前景遮罩 + 特效 = 3 层 Overdraw）
    //   · 粒子数量与尺寸要克制（大颗粒粒子杀死 GPU）
    //   · 用不透明材质代替半透明（能不透明就不透明）
    //   · UI 拆 Canvas（见页 14），减少重建与重绘

    // ============================================================
    //  技巧 5：相机剔除设置
    // ============================================================
    private void OptimizeCamera(Camera cam)
    {
        // farClipPlane：能小就小。过大影响深度精度（Z-Fighting）
        cam.farClipPlane = 200f;

        // nearClipPlane：不要太小。太小会严重影响深度精度
        cam.nearClipPlane = 0.3f;

        // cullingMask：只渲染需要的层
        cam.cullingMask = ~0;

        // 关闭不需要的功能
        cam.allowHDR = false;         // 移动端通常关掉
        cam.allowMSAA = false;        // 移动端通常关掉

        // useOcclusionCulling：遮挡剔除（需要烘焙）
        cam.useOcclusionCulling = true;
    }
}

// ============================================================
//  渲染性能优化优先级表
// ============================================================
//  优先级   手段                     收益      难度
//  ----------------------------------------------------------
//    1    减少 Draw Call（合批）      极高      中
//    2    减少 Overdraw              高        中
//    3    降低阴影质量/距离           高        低
//    4    纹理压缩（ASTC/DXT）        高        低
//    5    关闭不必要的后处理           高        低
//    6    LOD Group                  中        中
//    7    遮挡剔除（烘焙）            中        高
//    8    简化 Shader                中        高
//    9    GPU Instancing             中        中
//
//  【第一条永远是】先用 Frame Debugger 和 Profiler 定位瓶颈，
//                不要盲目优化。

// ============================================================
//  画质与性能的取舍（移动端）
// ============================================================
public class MobileQualityTiers : MonoBehaviour
{
    // 按机型分级（低配 / 中配 / 高配）
    public void ApplyTier(int tier)
    {
        switch (tier)
        {
            case 0:   // 低配
                QualitySettings.shadows = ShadowQuality.Disable;
                QualitySettings.shadowDistance = 0f;
                QualitySettings.antiAliasing = 0;
                Application.targetFrameRate = 30;
                break;

            case 1:   // 中配
                QualitySettings.shadows = ShadowQuality.HardOnly;
                QualitySettings.shadowDistance = 30f;
                QualitySettings.antiAliasing = 0;
                Application.targetFrameRate = 60;
                break;

            case 2:   // 高配
                QualitySettings.shadows = ShadowQuality.All;
                QualitySettings.shadowDistance = 60f;
                QualitySettings.antiAliasing = 2;
                Application.targetFrameRate = 60;
                break;
        }
    }
}

    1减少 Draw Call（合批）极高中
    2减少 Overdraw高中
    3降低阴影质量/距离高低
    4纹理压缩（ASTC/DXT）高低
    5关闭不必要的后处理高低

    8简化 Shader中高

    第一条永远是：先测量 不要凭感觉优化。渲染问题分两种，用错工具会找错方向：
    · CPU 瓶颈（Draw Call 太多）→ 看 Profiler 的 Camera.Render 耗时
    · GPU 瓶颈（Overdraw 严重）→ 看 Profiler 的 GPU 模块，或用 Frame Debugger

    把 GPU 问题当 CPU 问题优化，会白忙一场。

    二、MaterialPropertyBlock 的取舍

    MPB 不是万能的 MaterialPropertyBlock 的价值：改个别物体的颜色时不克隆材质。

    但它会打断 SRP Batcher——这是 URP 下的主力合批方式（见页 10）。

    所以：
    · Built-in 管线：MPB 通常有正面收益
    · URP/HDRP：如果原本能走 SRP Batcher，用 MPB 反而可能变慢

    结论：用 Frame Debugger 验证效果，别凭印象。

    三、画质分级策略

    移动端必须做机型分级——因为设备性能差异可以到 10 倍。让高端机跑满画质、低端机保证流畅。

    档位阴影抗锯齿目标帧率
    低配关闭030
    中配HardOnly，30 米060
    高配All，60 米2x60
    分级的判断依据 不要只用机型名判断（同款机型不同地区版本差异很大）。
    更可靠的做法：
    · 首次启动跑一个基准测试（渲染 100 个三角形测帧率）
    · 结合内存与 GPU 型号综合判断
    · 允许玩家手动调整——这是最稳妥的兜底
