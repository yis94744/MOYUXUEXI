一、为什么需要 Job System

现代手机有 6~8 个核心，但普通 Unity 代码只用其中一个。Job System 的意义就是把能并行的计算分到多个核上。

场景单线程耗时8 核理论加速
10000 个单位的移动计算10 ms约 1.5 ms
大规模射线检测8 ms约 1.2 ms
网格生成 / 程序化地形20 ms约 3 ms
寻路批量计算15 ms约 2.5 ms
三个必须知道的前提 ① 只有能并行的任务才有收益。互相依赖的计算没法并行。如果每个单位都依赖前一个的结果，那 Job 帮不了你。

② Job 里不能访问托管对象。不能 new 类、不能用 List<T>、不能抛异常、不能调 Unity API（Transform 等）。只能用 NativeContainer 和值类型。

③ 加速比不是线性的。8 核不会给你 8 倍——有调度开销、内存带宽限制、以及不能并行的部分。

二、Job System 的四个概念

理解这四件事就能用 Job System。

using Unity.Collections;
using Unity.Jobs;
using Unity.Burst;
using Unity.Mathematics;
using UnityEngine;

// ============================================================
//  Job System：多线程并行计算
// ============================================================
// 【核心限制】Job 里只能用值类型与 NativeContainer，
//            不能用 class、List<T>、抛异常、或调用 Unity API。

// ============================================================
//  1. IJob：单线程任务
// ============================================================
[BurstCompile]
public struct AddJob : IJob
{
    public float A;
    public float B;
    public NativeArray<float> Result;      // 需要用 NativeArray 回传结果

    public void Execute()
    {
        Result[0] = A + B;
    }
}

// ============================================================
//  2. IJobParallelFor：并行处理数组
// ============================================================
[BurstCompile]
public struct MoveJob : IJobParallelFor
{
    // 输入数据
    [ReadOnly] public NativeArray<float3> Positions;
    [ReadOnly] public NativeArray<float> Speeds;

    // 输出数据
    [WriteOnly] public NativeArray<float3> Results;

    public float DeltaTime;

    // 这个方法的 index 会被自动分配到多个线程上并行执行
    public void Execute(int index)
    {
        float3 pos = Positions[index];
        float speed = Speeds[index];

        // 简单的移动计算
        pos.y += speed * DeltaTime;
        Results[index] = pos;
    }
}

// ============================================================
//  3. 坐标变换 Job（实战常见场景）
// ============================================================
[BurstCompile]
public struct TransformJob : IJobParallelFor
{
    [ReadOnly] public NativeArray<float3> LocalPositions;
    [ReadOnly] public float4x4 LocalToWorld;

    [WriteOnly] public NativeArray<float3> WorldPositions;

    public void Execute(int index)
    {
        // 用 math 库（Burst 友好的数学库，比 Mathf 快）
        float4 local = new float4(LocalPositions[index], 1f);
        float4 world = math.mul(LocalToWorld, local);
        WorldPositions[index] = world.xyz;
    }
}

// ============================================================
//  4. 网格生成 Job（程序化生成的典型应用）
// ============================================================
[BurstCompile]
public struct MeshGenerateJob : IJobParallelFor
{
    public int Width;
    public float Scale;

    [WriteOnly] public NativeArray<float3> Vertices;

    public void Execute(int index)
    {
        int x = index % Width;
        int z = index / Width;

        // 用噪声生成高度（这里用简化的正弦组合代替真噪声）
        float h = math.sin(x * Scale) * math.cos(z * Scale) * 2f;

        Vertices[index] = new float3(x, h, z);
    }
}

// ============================================================
//  5. 使用示例：完整的 Job 生命周期
// ============================================================
public class JobDemo : MonoBehaviour
{
    private const int COUNT = 10000;

    private NativeArray<float3> _positions;
    private NativeArray<float3> _results;
    private NativeArray<float> _speeds;

    private void Start()
    {
        // 【第一步】分配 NativeContainer
        // Allocator 的三种选择：
        //   Persistent  —— 长期存在，手动释放（最常用）
        //   Temp        —— 一帧内有效，帧末自动释放
        //   TempJob     —— 四帧内有效，用于 Job 之间传递
        _positions = new NativeArray<float3>(COUNT, Allocator.Persistent);
        _results = new NativeArray<float3>(COUNT, Allocator.Persistent);
        _speeds = new NativeArray<float>(COUNT, Allocator.Persistent);

        for (int i = 0; i < COUNT; i++)
        {
            _positions[i] = new float3(i % 100, 0f, i / 100);
            _speeds[i] = 1f + (i % 5) * 0.5f;
        }

        RunJob();
    }

    private void RunJob()
    {
        var job = new MoveJob
        {
            Positions = _positions,
            Speeds = _speeds,
            Results = _results,
            DeltaTime = Time.deltaTime,
        };

        // 【第二步】调度 Job
        // 第二个参数是 batch size（每批处理多少个）
        // 【经验值】太小 -> 调度开销大；太大 -> 负载不均
        //            32 ~ 128 是常见范围
        JobHandle handle = job.Schedule(COUNT, 64);

        // 【第三步】等待完成
        // 【重要】Complete() 会阻塞主线程直到 Job 完成。
        //        如果在同一帧立即 Complete，就失去了并行的意义。
        //        正确做法是让 Job 跑一帧，下一帧再 Complete。
        handle.Complete();

        // 此时可以安全读取 _results
        Debug.Log("第一个结果: " + _results[0]);
    }

    // ============================================================
    //  异步模式：Job 跨帧执行（推荐）
    // ============================================================
    private JobHandle _handle;
    private bool _jobRunning;

    private void Update()
    {
        // 上一帧的 Job 完成了没？
        if (_jobRunning && _handle.IsCompleted)
        {
            _handle.Complete();
            _jobRunning = false;

            // 这里处理结果
            // 然后用上一帧的结果开始新的 Job
            StartNextJob();
        }
    }

    private void StartNextJob()
    {
        var job = new MoveJob
        {
            Positions = _positions,
            Speeds = _speeds,
            Results = _results,
            DeltaTime = Time.deltaTime,
        };

        _handle = job.Schedule(COUNT, 64);
        _jobRunning = true;
    }

    // ============================================================
    //  【必须】释放 NativeContainer
    // ============================================================
    private void OnDestroy()
    {
        // 先等 Job 完成，否则会报错
        _handle.Complete();

        // 用 IsCreated 检查，避免重复释放
        if (_positions.IsCreated) _positions.Dispose();
        if (_results.IsCreated) _results.Dispose();
        if (_speeds.IsCreated) _speeds.Dispose();
    }

    // ============================================================
    //  性能测试：单线程 vs 多线程
    // ============================================================
    public void Benchmark()
    {
        // ---------- 单线程 ----------
        var sw = System.Diagnostics.Stopwatch.StartNew();

        for (int i = 0; i < COUNT; i++)
        {
            float3 pos = _positions[i];
            pos.y += _speeds[i] * 0.016f;
            _results[i] = pos;
        }

        sw.Stop();
        long singleThreadMs = sw.ElapsedMilliseconds;

        // ---------- 多线程 ----------
        sw.Restart();

        var job = new MoveJob
        {
            Positions = _positions,
            Speeds = _speeds,
            Results = _results,
            DeltaTime = 0.016f,
        };

        job.Schedule(COUNT, 64).Complete();

        sw.Stop();
        long jobMs = sw.ElapsedMilliseconds;

        Debug.Log($"单线程: {singleThreadMs} ms, Job: {jobMs} ms");
        // 【注意】Job 有调度开销，数据量小时可能反而更慢
    }
}

// ============================================================
//  Job 的限制清单（必须记住）
// ============================================================
//  ❌ 不能用 class / string / List<T> / Dictionary
//  ❌ 不能抛异常（Burst 编译时无法编译 try-catch）
//  ❌ 不能调用 Unity API（Transform、GameObject、Debug.Log 都不行）
//  ❌ 不能有静态可变状态（多线程会竞争）
//  ❌ 不能有引用类型的字段
//
//  ✅ 可以用 struct / enum / NativeArray / math 库
//  ✅ 可以用 [ReadOnly] / [WriteOnly] 标记加速访问
//  ✅ 可以用 [BurstCompile] 获得 5~20 倍加速

// ============================================================
//  Allocator 的选择
// ============================================================
//  Allocator.Persistent  —— 长期存在，必须手动 Dispose（最常用）
//  Allocator.Temp        —— 一帧内有效，不能传给 Job
//  Allocator.TempJob     —— 四帧内有效，可以传给 Job（有警告）
//
//  【常见错误】用 TempJob 分配长期使用的数组 -> 四帧后自动释放 -> 崩溃
概念作用关键限制
IJob单个任务在一个线程上执行
IJobParallelFor并行处理数组每次处理一个索引
NativeArray<T>Job 之间共享数据必须手动 Dispose
JobHandle任务依赖与完成Complete() 会阻塞主线程
NativeContainer 的泄漏问题 NativeArray 不会被 GC 回收——它是非托管内存。
忘记 Dispose() 会报错：
A Native Collection has not been disposed, resulting in a memory leak.

三个防护措施：
① 用 using 语句（NativeArray 实现了 IDisposable）
② 开启 Jobs → Leak Detection（编辑器下会精确报出泄漏位置）
③ 在 OnDestroy 里兜底释放

三、Burst 编译器

Burst 是什么 Burst 是一个把 IL 编译成高度优化的机器码的编译器。
它使用 LLVM 后端，能做：
· SIMD 向量化（一次处理 4/8 个数据）
· 循环展开与内联
· 去除边界检查（在有保证的情况下）

实测效果：纯数学运算（矩阵乘法、向量运算）通常快 5~20 倍。

限制：Burst 编译的代码里不能用异常、不能用引用类型、不能用虚函数。

四、ECS 的核心思想

ECS 不是"更好的 MonoBehaviour"，而是完全不同的思维方式。

面向对象（MonoBehaviour）数据导向（ECS）
数据位置散落在各对象里连续内存，按组件分组
代码组织按对象按行为
缓存命中差（指针跳转）优（顺序访问）
并行友好差（依赖复杂）优（数据无依赖）
上手难度低高
适合绝大多数游戏逻辑海量同质单位
什么时候值得用 DOTS 值得用的场景（单位数 10000+）：
· 大规模 RTS / 塔防 / 模拟经营
· 弹幕游戏（几千发子弹）
· 程序化生成的世界
· 大规模 AI 群体

不值得用的场景：
· 普通 RPG / 动作游戏（几十个敌人）
· UI、剧情、任务系统——这些用 DOTS 是自找麻烦
· 团队不熟悉数据导向设计时

结论：DOTS 是"特化工具"，不是"升级版 Unity"。绝大多数项目用不上。
