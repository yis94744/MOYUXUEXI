14 · UI 系统

UI 是玩家接触最多的东西，也是性能问题最容易被忽视的地方——因为它不报错，只是悄悄吃掉帧率。这一页讲 UGUI 的两大核心：多分辨率适配（锚点）与性能优化（Canvas 重建）。

这一页在路线中的位置 上一页（13）动画系统讲的是"角色怎么动"。这一页讲"界面怎么显示"——两者经常配合（动画事件触发 UI 效果）。
关键衔接：页 11 讲的生命周期在 UI 里同样重要——OnEnable 订阅事件、OnDisable 取消订阅的模式，在 UI 面板的开关上用得最多。
下一页（15）输入与音频。UI 的按钮点击本质上是输入系统的一部分，两页会衔接。
难度提示：锚点的概念需要动手试——光看表格记不住，一定要在编辑器里拖着看效果。

一、RectTransform 与锚点 ★

锚点是 UGUI 最核心也最容易搞错的概念。理解它只需要三个参数：anchorMin、anchorMax、pivot。

using UnityEngine;

// ============================================================
//  RectTransform 与锚点：多分辨率适配的核心
// ============================================================
// 这是 UGUI 最容易搞错的部分。核心概念只有三个：
//   anchorMin / anchorMax  —— 锚点（相对父物体的比例，0~1）
//   pivot                  —— 轴心（自身旋转/缩放的中心，0~1）
//   anchoredPosition       —— 相对锚点的偏移
//
// 记住：anchoredPosition 是相对「锚点区域」的偏移，不是相对父物体中心。
public class UguiAnchor : MonoBehaviour
{
    [SerializeField] private RectTransform _panel;
    [SerializeField] private RectTransform _icon;

    private void Start()
    {
        // ============================================================
        //  四种常见布局的正确设置方式
        // ============================================================

        // ---------- 1. 铺满父容器（背景、遮罩）----------
        SetStretch(_panel);

        // ---------- 2. 固定大小居中（弹窗、图标）----------
        SetCenterFixed(_icon, new Vector2(200f, 200f));

        // ---------- 3. 贴在某个角（关闭按钮、头像）----------
        SetTopRight(_icon, new Vector2(-20f, -20f), new Vector2(48f, 48f));

        // ---------- 4. 横向拉伸 + 固定高度（血条、进度条）----------
        SetHorizontalStretch(_panel, 300f, 20f, -30f);

        // ---------- 5. 按比例定位（血条跟随 3D 物体）----------
        // 下面这个在 Update 里调用
    }

    // ============================================================
    //  通用设置方法
    // ============================================================

    /// 铺满父容器
    public static void SetStretch(RectTransform rt, float padding = 0f)
    {
        rt.anchorMin = Vector2.zero;
        rt.anchorMax = Vector2.one;
        rt.pivot = new Vector2(0.5f, 0.5f);
        rt.offsetMin = new Vector2(padding, padding);          // left, bottom
        rt.offsetMax = new Vector2(-padding, -padding);        // -right, -top
    }

    /// 固定大小并居中
    public static void SetCenterFixed(RectTransform rt, Vector2 size)
    {
        rt.anchorMin = new Vector2(0.5f, 0.5f);
        rt.anchorMax = new Vector2(0.5f, 0.5f);
        rt.pivot = new Vector2(0.5f, 0.5f);
        rt.anchoredPosition = Vector2.zero;
        rt.sizeDelta = size;
    }

    /// 贴右上角（offset 用负值表示向内偏移）
    public static void SetTopRight(RectTransform rt, Vector2 offset, Vector2 size)
    {
        rt.anchorMin = new Vector2(1f, 1f);
        rt.anchorMax = new Vector2(1f, 1f);
        rt.pivot = new Vector2(1f, 1f);      // 轴心也放右上角，缩放时才不会跑偏
        rt.anchoredPosition = offset;
        rt.sizeDelta = size;
    }

    /// 横向拉伸、固定高度、贴底部
    public static void SetHorizontalStretch(RectTransform rt, float height,
                                            float bottomOffset, float sideMargin)
    {
        rt.anchorMin = new Vector2(0f, 0f);
        rt.anchorMax = new Vector2(1f, 0f);
        rt.pivot = new Vector2(0.5f, 0f);

        // 左右留边距
        rt.offsetMin = new Vector2(sideMargin, bottomOffset);
        rt.offsetMax = new Vector2(-sideMargin, bottomOffset + height);
    }

    // ============================================================
    //  世界坐标转 UI 坐标（血条、伤害数字跟随）
    // ============================================================
    private Camera _cam;

    private void LateUpdate()
    {
        FollowWorldTarget();
    }

    private void FollowWorldTarget()
    {
        if (_cam == null) _cam = Camera.main;

        Vector3 worldPos = _icon.position;   // 这里应该是要跟随的 3D 物体位置

        // 世界坐标 -> 屏幕坐标
        Vector3 screenPos = _cam.WorldToScreenPoint(worldPos);

        // 在相机背后：隐藏 UI（否则会显示在屏幕另一侧的错误位置）
        if (screenPos.z < 0f)
        {
            _icon.gameObject.SetActive(false);
            return;
        }
        _icon.gameObject.SetActive(true);

        // 屏幕坐标 -> UI 局部坐标
        // 必须用屏幕坐标而不是世界坐标，这是最常见的错误
        RectTransform canvasRect = _icon.GetComponentInParent<Canvas>().transform as RectTransform;
        if (RectTransformUtility.ScreenPointToLocalPointInRectangle(
                canvasRect, screenPos, _cam, out Vector2 localPoint))
        {
            _icon.anchoredPosition = localPoint;
        }
    }
}

// ============================================================
//  锚点速查表
// ============================================================
//  需求                       anchorMin     anchorMax     pivot
//  铺满                        (0,0)         (1,1)        (0.5,0.5)
//  居中固定                    (0.5,0.5)     (0.5,0.5)    (0.5,0.5)
//  贴左上                      (0,1)         (0,1)        (0,1)
//  贴右上                      (1,1)         (1,1)        (1,1)
//  贴左下                      (0,0)         (0,0)        (0,0)
//  贴右下                      (1,0)         (1,0)        (1,0)
//  横向拉伸+贴底               (0,0)         (1,0)        (0.5,0)
//  纵向拉伸+贴左               (0,0)         (0,1)        (0,0.5)
//
//  规律：
//    anchorMin == anchorMax  -> 锚点是一个「点」，sizeDelta 就是实际大小
//    anchorMin != anchorMax  -> 锚点是一个「区域」，sizeDelta 是相对区域的增量
//    pivot 决定「旋转和缩放的中心」，也影响 anchoredPosition 的基准点
锚点的核心规律（一句话记住） anchorMin == anchorMax 时，锚点是一个点，sizeDelta 就是元素的实际大小。
anchorMin != anchorMax 时，锚点是一个区域，sizeDelta 是相对区域的增量（正值变大、负值变小）。

另一个关键点：pivot 决定"旋转和缩放的中心"，也影响 anchoredPosition 的基准。
贴角元素一定要把 pivot 也设成对应的角——否则做缩放动画时元素会"跑偏"。
血条跟随 3D 物体：三个必须处理的细节 ① 用世界坐标转屏幕坐标，不要直接赋值世界坐标。UI 在 Canvas 坐标系里，3D 物体在世界坐标系里，必须先转换。
② 相机背后的物体要隐藏。WorldToScreenPoint 返回的 z < 0 表示物体在相机背后，此时屏幕坐标是镜像的——不处理的话血条会出现在屏幕另一侧的错误位置。
③ 放在 LateUpdate 里。要等 3D 物体这一帧移动完再更新 UI 位置，否则血条会比物体慢一帧（这正是页 11 讲的 LateUpdate 的用途）。

二、Canvas 重建：UGUI 性能问题的根源 ★★★

这是 UI 性能优化的第一课，也是最容易被忽视的。很多项目"UI 一刷新就掉帧"，根因就在这里。

using UnityEngine;
using UnityEngine.UI;

// ============================================================
//  Canvas 重建机制 —— UGUI 性能问题的根源
// ============================================================
// 【核心原理】
// Canvas 把子元素合批成一个网格（Mesh）。
// 当任何一个子元素发生变化（位置、颜色、文本内容），
// 整个 Canvas 的网格都要重新生成 —— 这叫「重建（Rebuild）」。
//
// 后果：一个细节变化（比如分数跳动）会导致
//       整个 Canvas 下几百个元素全部重新构建。
//
// 解决办法：【拆分 Canvas】
//   把变化频繁的元素单独放在一个 Canvas 下，
//   把静态元素放在另一个 Canvas 下。
public class CanvasRebuildDemo : MonoBehaviour
{
    // ============================================================
    //  错误示范：所有 UI 塞在一个 Canvas 里
    // ============================================================
    //  Hierarchy
    //    Canvas (大)
    //      ├─ 背包格子 x 100      （几乎不变）
    //      ├─ 技能图标 x 20       （偶尔变）
    //      ├─ 分数文本            （【每帧都在变】）
    //      └─ 倒计时文本          （【每帧都在变】）
    //
    //  结果：分数一变，上面 120 个元素全部重建。
    //        低端机上表现为「UI 一刷新就掉帧」。

    // ============================================================
    //  正确做法：按「变化频率」拆分 Canvas
    // ============================================================
    //  Hierarchy
    //    Canvas_Static (静态层)
    //      ├─ 背包格子 x 100
    //      └─ 技能图标 x 20
    //    Canvas_Dynamic (动态层)   <- 单独的 Canvas 组件
    //      ├─ 分数文本
    //      └─ 倒计时文本
    //
    //  这样分数变化只重建 Canvas_Dynamic，静态层完全不受影响。

    [SerializeField] private Canvas _staticCanvas;
    [SerializeField] private Canvas _dynamicCanvas;

    // ============================================================
    //  优化 1：只在值变化时更新文本
    // ============================================================
    private Text _scoreText;      // 注意：实际项目推荐用 TMP_Text
    private int _lastScore = int.MinValue;

    public void UpdateScore(int score)
    {
        // 【关键】值没变就不更新。
        // 直接每帧 text = score.ToString() 会导致每帧重建，
        // 而且 ToString() 还会产生 GC 分配（见页 07）。
        if (score == _lastScore) return;
        _lastScore = score;

        _scoreText.text = score.ToString();
    }

    // ============================================================
    //  优化 2：用 CanvasGroup 控制显隐，而不是 SetActive
    // ============================================================
    // SetActive(false/true) 会导致 Canvas 布局重算 + 重建。
    // CanvasGroup 只是改渲染参数，成本低得多。
    public void HidePanel(CanvasGroup group)
    {
        group.alpha = 0f;
        group.blocksRaycasts = false;   // 同时禁用点击
        group.interactable = false;
    }

    public void ShowPanel(CanvasGroup group)
    {
        group.alpha = 1f;
        group.blocksRaycasts = true;
        group.interactable = true;
    }

    // ============================================================
    //  优化 3：禁用不必要的 Raycast Target
    // ============================================================
    // Image / Text 默认勾选 Raycast Target，
    // 意味着每次点击都要对所有 UI 元素做射线检测。
    // 纯装饰性的图片和文本应该【取消勾选】。
    public void DisableRaycastOnDecorations(Graphic[] decorations)
    {
        foreach (var g in decorations)
        {
            g.raycastTarget = false;
        }
    }

    // ============================================================
    //  优化 4：避免 LayoutGroup 的每帧重算
    // ============================================================
    // HorizontalLayoutGroup / VerticalLayoutGroup / ContentSizeFitter
    // 会在子元素变化时触发【整棵布局树重算】，开销很大。
    //
    // 对策：
    //   1. 动态列表用「手动定位」代替 LayoutGroup
    //   2. 确实需要时，用 LayoutRebuilder 手动触发而不是每帧自动
    public void ManualRebuild(RectTransform rt)
    {
        // 只在需要时手动重建，而不是让 LayoutGroup 自动每帧检查
        LayoutRebuilder.ForceRebuildLayoutImmediate(rt);
    }

    // ============================================================
    //  性能诊断：检测重建
    // ============================================================
    // 方法 1：Profiler -> CPU -> 搜索 Canvas.SendWillRenderCanvases
    // 方法 2：在 Scene 视图右上角开启「Canvas Rebuild」可视化
    //         （可以直观看到哪些 Canvas 在重建）
    // 方法 3：Window -> Analysis -> Frame Debugger 看合批情况
}

// ============================================================
//  拆 Canvas 的判断标准
// ============================================================
//  元素类型                    是否单独拆
//  ----------------------------------------------------------
//  每秒都在变的（分数、倒计时） 【必须拆】
//  鼠标/触屏交互的（按钮悬停）   【建议拆】
//  频繁增删的（背包格子、列表）  【建议拆】
//  基本不变的（背景、边框）     不用拆
//
//  【注意】不要拆得太碎。
//  每个 Canvas 都会打断合批，产生额外的 Draw Call。
//  经验值：一个界面拆 2~4 个 Canvas 比较合理。
//
//  【补充】Unity 2020+ 的「Nested Canvas」：
//  子 Canvas 有自己的网格，但会和父 Canvas 一起渲染。
//  它比独立 Canvas 轻量，但仍能隔离重建。
Canvas 重建的原理（必须理解） Canvas 会把所有子元素合批成一个网格（Mesh）。当任何一个子元素发生变化（位置、颜色、文本内容、启用状态），整个 Canvas 的网格都要重新生成。

后果：一个分数文本每帧变化，会导致它所在 Canvas 下几百个元素全部重建。
在低端机上，这表现为"UI 一刷新就掉帧"。

解决办法：按"变化频率"拆分 Canvas——把频繁变化的元素单独放在一个 Canvas 下。
拆 Canvas 的经验值
元素类型是否单独拆原因
每秒都在变的（分数、倒计时）必须拆否则一直在重建静态元素
鼠标 / 触屏交互的（按钮悬停）建议拆悬停会触发重建
频繁增删的（背包格子、列表）建议拆增删会改变布局
基本不变的（背景、边框）不用拆不会触发重建
但不要拆得太碎。每个 Canvas 都会打断合批，产生额外的 Draw Call。
经验值：一个界面拆 2~4 个 Canvas 比较合理。

补充：Unity 2020+ 的嵌套 Canvas（Nested Canvas）有自己的网格，但会和父 Canvas 一起渲染。它比独立 Canvas 轻量，但仍能隔离重建——是拆分的更好选择。

三、UGUI 优化清单

#优化项做法收益
1拆 Canvas按变化频率分组极高
2只在值变化时更新文本加值比较，避免每帧赋值高
3取消装饰元素的 Raycast Target纯装饰的 Image / Text 取消勾选高
4用 CanvasGroup 代替 SetActive控制 alpha / blocksRaycasts中
5避免 LayoutGroup 每帧重算动态列表手动定位，或手动触发重建中
6图集（Atlas）小图合并成大图，减少 Draw Call中
7用 TMP 代替 TextTextMeshPro 更清晰、性能更好中
8Overdraw 优化减少全屏半透明图层叠加低
Raycast Target 是最容易被忽略的优化 Image 和 Text 默认都勾选 Raycast Target。这意味着每次点击 / 触摸，UI 系统都要对该元素做一次射线检测。

一个复杂的界面可能有几百个 UI 元素，其中大部分是纯装饰（背景图、边框、图标、说明文字），它们根本不需要响应点击。

做法：选中所有装饰性元素，取消 Raycast Target 勾选。
这是纯粹的配置工作，不写一行代码，收益却很直接。

四、TextMeshPro 的注意事项

TMP 的 SetText 不解析格式说明符 很多人以为 SetText 的数值重载会像 string.Format 那样支持格式串，但实际上不会。

我反汇编了 Unity.TextMeshPro.dll 确认过：SetText 有 20 个重载（含 string、StringBuilder、char[]，以及 1 到 8 个 float 参数的版本）。
但在处理 float 参数的实现里，字符常量只有 {（123）和 }（125）——没有冒号（58）。

所以 SetText("{0:F1}", 3.14159f) 不会被识别为格式串，冒号后面的内容会被原样输出。

正确做法：
• 需要格式化就先用 ToString("F1") 转成字符串，再传给 SetText；
• 或者直接用 SetText(string) 重载配合插值字符串（但注意插值字符串会产生 GC 分配）。
TMP 的零分配更新方案 如果文本每帧都在变（比如血量百分比），ToString() 会产生 GC 分配（见页 07）。
三种更好的方案：
① 值没变就不更新——最简单也最有效；
② 用 StringBuilder + SetText(StringBuilder) 重载——复用同一个 Builder，零分配；
③ 用图片代替文本——数字显示用图集（传统手游的常见做法）。
