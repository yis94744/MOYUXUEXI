06 · 泛型、反射与现代特性

前五页你学的是"怎么写代码"。从这一页开始，你学的是怎么写"能被别人复用的代码"。泛型让你写一次就能适配所有类型；反射让你能在运行时"看见"自己的代码结构；特性让你能给代码贴标签；异常让你能优雅地处理意外。

这一页在路线中的位置 上一页（05）你学会了用委托和事件解耦模块。但那些代码往往是为某一个具体类型写死的。
这一页解决"复用"问题：泛型让容器/管理器不用为每种类型重写一遍；反射和特性让"注册系统"这类重复工作变成自动的。
下一页（07）会把这些知识用在性能上——泛型不装箱、反射有代价、特性有运行时开销，这些都是 GC 分配的来源。
换句话说：这一页教你"强大的工具"，下一页教你"这些工具的账单"。

一、泛型：写一次，适配所有类型

假设你要写一个对象池。不用泛型的话，你有两个选择：

  • 方案 A：为每种类型写一个池子——BulletPool、EnemyPool、EffectPool…代码重复到让人崩溃。
  • 方案 B：用一个 Stack<object>——但每次取出都要强制转换，而且值类型会装箱（还记得页 03 讲的装箱吗）。

泛型是第三个选项：写一次代码，编译时为每种类型生成专用版本，既没有重复，也没有装箱。

泛型类与泛型方法

using UnityEngine;
using System;
using System.Collections.Generic;

// ============================================================
//  泛型类：把类型变成参数
// ============================================================
public class ObjectPool<T> where T : class
{
    private readonly Stack<T> _pool = new Stack<T>();
    private readonly Func<T> _factory;
    private readonly Action<T> _onGet;
    private readonly Action<T> _onRelease;

    public ObjectPool(Func<T> factory, Action<T> onGet = null,
                      Action<T> onRelease = null, int prewarm = 0)
    {
        _factory = factory ?? throw new ArgumentNullException(nameof(factory));
        _onGet = onGet;
        _onRelease = onRelease;

        for (int i = 0; i < prewarm; i++) _pool.Push(_factory());
    }

    public T Get()
    {
        T item = _pool.Count > 0 ? _pool.Pop() : _factory();
        _onGet?.Invoke(item);
        return item;
    }

    public void Release(T item)
    {
        _onRelease?.Invoke(item);
        _pool.Push(item);
    }

    public int Count { get { return _pool.Count; } }
}

// ============================================================
//  泛型约束的完整写法
// ============================================================
public class ConstraintDemo
{
    // struct 约束：只能是值类型
    public static T Max<T>(T a, T b) where T : struct, IComparable<T>
    {
        return a.CompareTo(b) > 0 ? a : b;
    }

    // class + new() 约束
    public static T Create<T>() where T : class, new()
    {
        return new T();
    }

    // 多个接口约束
    public static void Process<T>(T item) where T : IDisposable, IComparable<T>
    {
        item.Dispose();
    }
}

// ============================================================
//  泛型单例基类（Unity 中最常用的泛型模式）
// ============================================================
public abstract class MonoSingleton<T> : MonoBehaviour where T : MonoSingleton<T>
{
    private static T _instance;

    public static T Instance
    {
        get
        {
            if (_instance != null) return _instance;

            _instance = FindObjectOfType<T>();
            if (_instance == null)
            {
                var go = new GameObject(typeof(T).Name);
                _instance = go.AddComponent<T>();
            }
            return _instance;
        }
    }

    protected virtual void Awake()
    {
        if (_instance != null && _instance != this)
        {
            Destroy(gameObject);     // 场景里出现重复实例时销毁后来的
            return;
        }
        _instance = (T)this;
        DontDestroyOnLoad(gameObject);
    }

    protected virtual void OnDestroy()
    {
        if (_instance == this) _instance = null;   // 必须清理，否则留下"假 null"
    }
}

public class AudioManager : MonoSingleton<AudioManager>
{
    public void PlayBgm(string name) { }
}
注意 where T : class 这个约束 上面的对象池加了 where T : class，意思是"T 必须是引用类型"。
为什么？因为池子里用 Stack<T> 存储，如果 T 是值类型，取出时会有语义问题——值类型是拷贝语义，池化它对性能帮助很小，反而增加复杂度。
约束不只是"限制"，更是一种"文档"：它明确告诉使用者"这个类为引用类型设计"。

泛型约束全表

约束含义典型用途
where T : struct必须是值类型（不含 Nullable）数学运算、比较
where T : class必须是引用类型对象池、缓存
where T : notnull不可为空（C# 8）字典键
where T : unmanaged非托管类型（C# 7.3）可用于指针、Burst
where T : new()必须有公开无参构造必须放在约束列表最后
where T : BaseClass必须是某类或其派生组件查找
where T : IInterface必须实现接口可写多个
where T : U与另一类型参数的关系裸类型约束

实战：泛型单例基类

这是 Unity 中最经典的泛型应用。每个 Manager 都要写一遍单例逻辑太蠢了，用泛型基类一次解决。

上面的 MonoSingleton<T> 代码就是完整实现。几个关键点：

  • where T : MonoSingleton<T> —— 这个"自引用约束"是CRTP 模式，让基类能知道具体子类型。
  • _instance = (T)this; —— 因为约束保证了 T 是自己的派生类，转换必然成功。
  • OnDestroy 里必须置空 —— 否则场景切换后留下"假 null"（页 11 会详细讲）。
  • DontDestroyOnLoad 是否该加 —— 取决于你的需求。全站管理器需要，场景内管理器不需要。
Unity 的 API 版本提醒 FindObjectOfType<T>() 从 Unity 2023.1 起被标记为过时。
新 API 是 FindFirstObjectByType<T>()（返回第一个）和 FindAnyObjectByType<T>()（更快，任意一个）。
2020–2022 用 FindObjectOfType 即可；写新项目建议用新 API 并加版本宏。

二、反射：让代码"看见"自己

反射是在运行时检查和操作类型的能力。听起来很抽象，但你在 Unity 里其实一直在享受它的好处：

  • Inspector 能显示你的字段 —— 那是反射在读字段元数据。
  • 你在类上加 [SerializeField] 就生效 —— 那是反射在读特性。
  • Unity 能找到所有标了 [MenuItem] 的方法并放进菜单 —— 那也是反射。

但你自己用反射时，要清楚它的代价。

using UnityEngine;
using System;
using System.Reflection;

public class ReflectTarget
{
    public int PublicField = 1;
    private int _privateField = 2;
    public int PublicProperty { get; set; }
    public static string StaticField = "s";

    public void PublicMethod(int x) { Debug.Log("called " + x); }
    private void PrivateMethod() { Debug.Log("private"); }
}

public class ReflectionDemo : MonoBehaviour
{
    private void Start()
    {
        Type t = typeof(ReflectTarget);

        // ============ 基本信息 ============
        Debug.Log(t.FullName);                 // 完整类型名
        Debug.Log(t.Name);                     // 简单名
        Debug.Log(t.IsClass);
        Debug.Log(t.BaseType?.Name);
        Debug.Log(t.Assembly.GetName().Name);

        // ============ 字段 ============
        const BindingFlags Flags = BindingFlags.Public
                                 | BindingFlags.NonPublic
                                 | BindingFlags.Instance;

        foreach (FieldInfo f in t.GetFields(Flags))
        {
            Debug.Log(f.Name + " : " + f.FieldType.Name);
        }

        // ============ 创建实例 + 调用方法 ============
        object instance = Activator.CreateInstance(t);

        MethodInfo method = t.GetMethod("PublicMethod");
        method.Invoke(instance, new object[] { 5 });

        MethodInfo priv = t.GetMethod("PrivateMethod",
            BindingFlags.NonPublic | BindingFlags.Instance);
        priv.Invoke(instance, null);

        // ============ 读写私有字段 ============
        FieldInfo privField = t.GetField("_privateField", Flags);
        Debug.Log(privField.GetValue(instance));     // 2
        privField.SetValue(instance, 99);

        // ============ 静态成员 ============
        FieldInfo sf = t.GetField("StaticField");
        Debug.Log(sf.GetValue(null));                // 静态字段传 null
    }
}

Unity 中的两个典型用途

用途一：用特性标记 + 反射批量注册系统。想象你有 20 个游戏系统（战斗、背包、任务…），每次新增一个都要手动去改一个注册列表 —— 这是典型的重复劳动。用特性标记 + 反射扫描就能全自动。

using UnityEngine;
using System;
using System.Reflection;

// 为单文件编译补上被引用的类型（原示例中它定义在 reflection.cs）
public class ReflectTarget
{
    public int PublicField = 1;
    public void PublicMethod(int x) { }
}

// ============================================================
//  反射的性能代价与三种优化方案
// ============================================================
public class FastReflection
{
    // 反射 Invoke 比直接调用慢 100~1000 倍，且产生 GC（装箱 + object[] 分配）

    // ---------- 方案一：缓存 MemberInfo ----------
    private static readonly MethodInfo CachedMethod =
        typeof(ReflectTarget).GetMethod("PublicMethod");

    // ---------- 方案二：缓存委托（推荐，接近直接调用的性能）----------
    private static Action<ReflectTarget, int> _fastAction;

    static FastReflection()
    {
        MethodInfo m = typeof(ReflectTarget).GetMethod("PublicMethod");
        _fastAction = (Action<ReflectTarget, int>)Delegate.CreateDelegate(
            typeof(Action<ReflectTarget, int>), m);
    }

    public static void CallFast(ReflectTarget t)
    {
        _fastAction(t, 5);
    }

    // ---------- 方案三：表达式树（可支持更复杂的场景）----------
    // ---------- 方案四：IL Emit（最强但最复杂，IL2CPP 下受限）----------
}

// ============================================================
//  实战：用特性标记 + 反射批量注册系统
// ============================================================
[AttributeUsage(AttributeTargets.Class)]
public class GameSystemAttribute : Attribute
{
    public int Order { get; }
    public GameSystemAttribute(int order = 0) { Order = order; }
}

[GameSystem(10)]
public class CombatSystem { }

[GameSystem(20)]
public class SaveSystem { }

public static class SystemRegistry
{
    public static System.Collections.Generic.List<Type> DiscoverAll()
    {
        var result = new System.Collections.Generic.List<Type>();
        Assembly asm = typeof(SystemRegistry).Assembly;

        foreach (Type t in asm.GetTypes())
        {
            if (!t.IsClass || t.IsAbstract) continue;

            var attr = t.GetCustomAttribute<GameSystemAttribute>();
            if (attr != null) result.Add(t);
        }

        result.Sort((a, b) =>
            a.GetCustomAttribute<GameSystemAttribute>().Order
                .CompareTo(b.GetCustomAttribute<GameSystemAttribute>().Order));

        return result;
    }
}
IL2CPP 的重要限制 IL2CPP 是 AOT（提前编译）。泛型实例只在编译期被引用过的情况下才会生成代码。
如果某个泛型组合只在反射里用到，AOT 阶段可能不生成 → 运行时抛 ExecutionEngineException。
三个解决办法：
① 在代码里显式实例化一次（写个用不到的 dummy 调用）；
② 加 [UnityEngine.Scripting.Preserve] 特性；
③ 在 link.xml 中保留类型/程序集（页 22 详解）。

三、特性：给代码贴标签

特性（Attribute）是元数据——它不改变代码行为，只是"贴个标签"，等别人（框架或你自己的代码）来读这个标签。

Unity 常用特性作用备注
[SerializeField]私有字段暴露到 Inspector最常用
[SerializeReference]真正的多态序列化Unity 2019.3+
[FormerlySerializedAs("old")]字段改名后保留已存数据强烈推荐
[HideInInspector]public 字段不在 Inspector 显示但仍序列化
[Range(min,max)]数值滑块仅 float/int
[Min(v)]最小值硬限制Unity 2019.1+
[Tooltip("...")]鼠标悬停提示
[Header("...")]Inspector 分组标题
[RequireComponent(typeof(T))]自动添加依赖组件类特性
[DisallowMultipleComponent]禁止同物体挂多个类特性
[ExecuteAlways]编辑模式也执行生命周期Unity 2018.1+
[CreateAssetMenu]右键创建 ScriptableObject类特性
[ContextMenu("名称")]组件右上角菜单方法特性
[UnityEngine.Scripting.Preserve]防止 IL2CPP 剥离见上面 IL2CPP 部分
[FormerlySerializedAs] 是被严重低估的特性 场景：你把字段 _speed 改名成 _moveSpeed。
所有已经保存了数据的预制体和场景里，这个字段的值会丢失——因为 Unity 是按字段名匹配数据的。
加上 [FormerlySerializedAs("_speed")] 后，Unity 会先尝试读新名字，读不到再读旧名字，数据就保住了。
规则：任何会影响序列化的字段改名，都要加这个特性。上线后改名不带它，等于把策划配的数据全清空。

四、异常处理

异常是"出了意料之外的事"时的标准处理方式。但在 Unity 里，滥用异常比不用异常更糟——因为异常不仅慢，还可能中断一帧的执行。

using UnityEngine;
using System;

public class ExceptionDemo : MonoBehaviour
{
    private void Start()
    {
        try
        {
            GameObject go = GameObject.Find("NotExist");
            go.SetActive(false);                    // NullReferenceException
        }
        catch (NullReferenceException e)
        {
            Debug.LogError("空引用：" + e.Message);
        }
        catch (Exception e) when (e.Message.Contains("timeout"))   // 异常筛选器（C# 6+）
        {
            Debug.LogError("超时：" + e.Message);
        }
        catch (Exception e)
        {
            Debug.LogException(e);                  // 保留堆栈
            throw;                                  // 重新抛出（保留原始堆栈）
        }
        finally
        {
            // 无论是否异常都会执行，用于释放资源
        }
    }
}

异常触发场景

Unity 中异常处理的四条实践建议 ① 用 TryXxx 模式表达"预期内的失败"，不要用异常。
比如"文件不存在"是预期内的，返回 false 就好；"文件存在但格式损坏"才是异常。
② 不要用异常做流程控制。异常涉及堆栈展开，性能很差。靠异常实现的逻辑只能算 bug。
③ 不要 catch (Exception) 后什么都不做。吞掉异常是 bug 的温床——出问题时你连线索都没有。
④ 在 Job System / Burst 中不能使用异常。Burst 编译的代码里任何 try/catch 都会编译失败（页 18 详讲）。

五、现代 C# 特性与 Unity 支持矩阵

C# 一直在演进，但Unity 的 C# 版本是被锁定的——它由 Unity 内置的编译器决定，你无法自由升级。这导致一个尴尬局面：网上很多 C# 教程教的语法，在 Unity 里根本用不了。

Unity 的 C# 版本对照表（务必对照项目版本）

Unity 版本内置 C# 版本.NET API 兼容级别关键新增
2017.xC# 6.NET 3.5 / 4.x 可选字符串插值、表达式体、nameof
2018.3C# 7.3.NET 4.x Equivalentout var、元组、模式匹配、局部函数
2019.4 LTSC# 7.3.NET Standard 2.0 / 4.x同上
2020.2C# 8.0.NET Standard 2.1 / 4.xswitch 表达式、??=、using 声明
2021.2+C# 9.0.NET Standard 2.1record、init、目标类型 new
Unity 6（6000.x）C# 9.0.NET Standard 2.1仍是 C# 9
三个必须记住的版本限制 ① C# 8 的默认接口实现（DIM）：老版本编译期就报错，2021.3 起能编译、能运行，但仍不建议依赖。Unity 2020.2 及更早的官方文档写明该特性会直接编译报错；2021.3.21f1c1 实测能编译并正常调用，但 IL2CPP 后端与不同版本的支持并不一致（页 04 有完整说明）。
② C# 10/11 的新特性 Unity 完全不支持。包括文件作用域命名空间、record struct、原始字符串 """..."""、required 成员。
③ 不要在 Unity 项目里用 -langversion:latest。把语言版本调到 11 并不会让新语法可用，反而可能编译失败。
写代码前先确认「这个语法我的 Unity 版本支持吗」，能省下大量试错时间。

using UnityEngine;
using System;
using System.Collections.Generic;

//  C# 6：字符串插值、表达式体、nameof、?.、when
// ============================================================
public class ModernFeatures : MonoBehaviour
{

    private int _level = 7;

    {

        string s2 = $"Level {_level:D3}";

        Debug.Log(nameof(_speed));            // "_speed"

        // C# 6：空条件运算符
        Transform t = null;
        Vector3? pos = t?.position;

        Debug.Log(s1 + s2 + pos);
    }

    // C# 7：out 变量内联声明
    private void TryParseDemo()
    {

        {
            Debug.Log(value);
        }
    }

    // C# 7：元组返回多个值
    private (bool success, int value) TryGet()
    {
        return (true, 42);
    }

    // C# 7：模式匹配

    {
        switch (obj)

    private void StaticLambda()
    {
        var list = new List<int> { 3, 1, 2 };
        list.Sort(static (x, y) => x.CompareTo(y));
    }
}

// ============================================================
//  C# 9：record 与 init —— 必须先补 IsExternalInit
// ============================================================
// Unity 的 .NET Standard 2.1 参考程序集里【没有】IsExternalInit，
// 而 C# 9 的 init / record 都依赖它。
// 不补这个 shim 会报错：
//   error CS0518: 预定义类型 System.Runtime.CompilerServices.IsExternalInit 未定义或导入
//
// 解决办法：在项目里自己定义一次（放在任意 .cs 文件、任意命名空间外即可，
// 全项目只需要一份，重复定义会报 CS0101 类型重复）。
namespace System.Runtime.CompilerServices
{
    internal static class IsExternalInit { }
}

public record PlayerProfile(string Name, int Level);

public class RecordDemo : MonoBehaviour
{
    private void Start()
    {
        var p1 = new PlayerProfile("Hero", 5);
        var p2 = new PlayerProfile("Hero", 5);

        Debug.Log(p1 == p2);          // True —— record 是值相等
        Debug.Log(p1.ToString());     // PlayerProfile { Name = Hero, Level = 5 }

        // with 表达式：基于现有实例创建修改后的副本
        var p3 = p1 with { Level = 6 };
        Debug.Log(p3);

        // 解构
        var (name, level) = p1;
        Debug.Log(name + " " + level);
    }
}
关于 record 在 Unity 中的一个限制 record 不能被 Unity 序列化——它不是 [System.Serializable]，编译器生成的字段是只读 init 属性。
所以 record 适合做纯逻辑层的不可变数据、编辑器工具、DTO，但不要用在需要 Inspector 显示的地方。
还有一个必踩的坑：Unity 的 .NET Standard 2.1 参考程序集里没有 System.Runtime.CompilerServices.IsExternalInit，而 record 和 init 都依赖它。不补就会在 Unity 编译管线里直接报 error CS0518: Predefined type 'System.Runtime.CompilerServices.IsExternalInit' is not defined or imported（我在 2021.3.21f1c1 下复现过）。
解决办法：在项目里自己声明一次即可 —— namespace System.Runtime.CompilerServices { internal static class IsExternalInit { } }。全项目只需要一份。本节示例代码里已经带上了这个 shim。
好消息是：补上 shim 之后，record 在 Mono 和 IL2CPP 下都能正常用（Unity 2021.2+）。
