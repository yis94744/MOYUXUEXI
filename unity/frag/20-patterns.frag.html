20 · 架构与设计模式

设计模式最大的误区是以为"用得多就是架构好"。真相恰恰相反——绝大多数模式在你的项目里都用不上。这一页只讲五个真正高频的模式，并告诉你什么时候不该用。

一、五个真正会用的模式

下面这五个模式覆盖了游戏开发中90% 的架构需求。其余的模式要么用不上，要么就是这五个的变体。

using UnityEngine;
using System.Collections.Generic;

// ============================================================
//  游戏架构与设计模式：五个真正会用的模式
// ============================================================
//  设计模式的误区：以为"用得多"就是"架构好"。
//  真相是——【绝大多数模式在这个项目里都用不上】。
//  下面这五个是真正高频的。

// ============================================================
//  1. 单例（Singleton）—— 用但要小心
// ============================================================
// 【适用】全局唯一且必须随时访问的系统：音频、存档、事件总线
// 【不适用】业务逻辑模块 —— 那会让依赖关系失控

public abstract class MonoSingleton<T> : MonoBehaviour where T : MonoSingleton<T>
{
    private static T _instance;
    private static bool _applicationQuitting = false;

    public static T Instance
    {
        get
        {
            // 【关键】应用退出时不要再创建新对象
            if (_applicationQuitting) return null;

            if (_instance != null) return _instance;

            _instance = FindFirstObjectByType<T>();
            if (_instance == null)
            {
                var go = new GameObject(typeof(T).Name);
                _instance = go.AddComponent<T>();
            }
            return _instance;
        }
    }

    protected virtual void Awake()
    {
        if (_instance != null && _instance != this)
        {
            Destroy(gameObject);
            return;
        }
        _instance = (T)this;
    }

    protected virtual void OnDestroy()
    {
        // 【必须清理】否则会留下"假 null"（见页 11）
        if (_instance == this) _instance = null;
    }

    protected virtual void OnApplicationQuit()
    {
        _applicationQuitting = true;
    }
}

// 使用示例
public class AudioManager : MonoSingleton<AudioManager>
{
    public void PlayBgm(string name) { }
}

// ============================================================
//  2. 观察者 / 事件总线 —— 解耦的核心
// ============================================================
// 【适用】跨模块通信、UI 与逻辑解耦
// 【注意】必须成对订阅/取消订阅（见页 05）

public static class EventBus
{
    // 用字典支持多种事件类型
    private static readonly Dictionary<System.Type, System.Delegate> _handlers
        = new Dictionary<System.Type, System.Delegate>();

    public static void Subscribe<T>(System.Action<T> handler) where T : struct
    {
        var type = typeof(T);
        if (_handlers.TryGetValue(type, out var d))
            _handlers[type] = System.Delegate.Combine(d, handler);
        else
            _handlers[type] = handler;
    }

    public static void Unsubscribe<T>(System.Action<T> handler) where T : struct
    {
        var type = typeof(T);
        if (!_handlers.TryGetValue(type, out var d)) return;

        var result = System.Delegate.Remove(d, handler);
        if (result == null) _handlers.Remove(type);   // 【关键】空了就移除，避免内存滞留
        else _handlers[type] = result;
    }

    public static void Publish<T>(T evt) where T : struct
    {
        if (_handlers.TryGetValue(typeof(T), out var d) && d is System.Action<T> action)
        {
            action.Invoke(evt);
        }
    }

    // 场景切换时清理（防止残留订阅）
    public static void Clear() { _handlers.Clear(); }
}

// 事件定义（用 struct 避免 GC，见页 07）
public struct PlayerDamagedEvent
{
    public int Amount;
    public int RemainingHp;
}

public struct ScoreChangedEvent
{
    public int NewScore;
}

// 使用示例
public class EventDemo : MonoBehaviour
{
    private void OnEnable()
    {
        EventBus.Subscribe<PlayerDamagedEvent>(OnDamaged);
    }

    private void OnDisable()
    {
        EventBus.Unsubscribe<PlayerDamagedEvent>(OnDamaged);   // 必须取消
    }

    private void OnDamaged(PlayerDamagedEvent e)
    {
        Debug.Log($"受伤 {e.Amount}，剩余 {e.RemainingHp}");
    }

    private void Test()
    {
        EventBus.Publish(new PlayerDamagedEvent { Amount = 10, RemainingHp = 90 });
    }
}

// ============================================================
//  3. 状态机 —— 见页 16（算法站）
// ============================================================
// 这里给一个最小的枚举式状态机，说明它在架构里的位置

public enum GameState { Loading, MainMenu, Playing, Paused, GameOver }

public class GameStateMachine : MonoBehaviour
{
    public GameState Current { get; private set; } = GameState.Loading;

    // 状态变化事件（让 UI 等模块响应）
    public event System.Action<GameState, GameState> StateChanged;

    public void ChangeState(GameState next)
    {
        if (Current == next) return;         // 【去重】

        var prev = Current;
        OnExit(prev);
        Current = next;
        OnEnter(next);

        StateChanged?.Invoke(prev, next);
    }

    private void OnEnter(GameState s)
    {
        switch (s)
        {
            case GameState.Playing: Time.timeScale = 1f; break;
            case GameState.Paused:  Time.timeScale = 0f; break;   // 暂停用 timeScale
        }
    }

    private void OnExit(GameState s)
    {
        // 状态退出时清理（见算法站页 16）
    }
}

// ============================================================
//  4. 对象池 —— 见页 11
// ============================================================
// 这里展示"通用对象池管理器"，比页 11 的版本更适合项目使用

public class PoolManager : MonoSingleton<PoolManager>
{
    private readonly Dictionary<GameObject, Queue<GameObject>> _pools
        = new Dictionary<GameObject, Queue<GameObject>>();

    public GameObject Spawn(GameObject prefab, Vector3 pos, Quaternion rot)
    {
        if (!_pools.TryGetValue(prefab, out var queue))
        {
            queue = new Queue<GameObject>();
            _pools[prefab] = queue;
        }

        GameObject go;
        if (queue.Count > 0)
        {
            go = queue.Dequeue();
            go.transform.SetPositionAndRotation(pos, rot);
            go.SetActive(true);
        }
        else
        {
            go = Instantiate(prefab, pos, rot, transform);
        }
        return go;
    }

    public void Despawn(GameObject prefab, GameObject instance)
    {
        if (instance == null) return;
        instance.SetActive(false);
        instance.transform.SetParent(transform, false);
        _pools[prefab].Enqueue(instance);
    }
}

// ============================================================
//  5. 命令模式 —— 用于回放、撤销、网络同步
// ============================================================
// 【适用】需要记录/重放操作的系统（录像、撤销、帧同步）

public interface ICommand
{
    void Execute();
    void Undo();
}

public class MoveCommand : ICommand
{
    private readonly Transform _target;
    private readonly Vector3 _delta;
    private Vector3 _previous;

    public MoveCommand(Transform target, Vector3 delta)
    {
        _target = target;
        _delta = delta;
    }

    public void Execute()
    {
        _previous = _target.position;
        _target.position += _delta;
    }

    public void Undo()
    {
        _target.position = _previous;
    }
}

public class CommandInvoker
{
    private readonly Stack<ICommand> _history = new Stack<ICommand>();

    public void Execute(ICommand cmd)
    {
        cmd.Execute();
        _history.Push(cmd);
    }

    public void Undo()
    {
        if (_history.Count == 0) return;
        _history.Pop().Undo();
    }
}

// ============================================================
//  什么该用模式、什么不该
// ============================================================
//  【该用】
//    · 单例：音频、存档、事件总线（全局唯一且必须随时访问）
//    · 观察者：UI 与逻辑解耦、跨模块通信
//    · 状态机：游戏流程、角色状态、UI 界面
//    · 对象池：频繁创建销毁的对象
//    · 命令：需要回放/撤销/同步的操作
//
//  【不该用】
//    · 只有两个类的"工厂模式" —— 直接 new 就行
//    · 为一个按钮写的"策略模式" —— 直接 if 就行
//    · 只有一层继承的"抽象工厂" —— 过度设计
//    · 【任何"以后可能会用"的模式】—— 到时候再说
//
//  【判断标准】
//    如果去掉这个模式，代码更短、更好懂，那就去掉它。

二、什么时候不该用模式

模式滥用的四个信号 ① 只有两个类的"工厂"：直接 new 就行，多一层工厂只是多一层看不懂。

② 为一个按钮写的"策略模式"：直接 if 就行。

③ 只有一层继承的"抽象工厂"：这是典型的过度设计。

④ "以后可能会用"的任何模式：等到真的要用时再重构。过早抽象比不抽象更糟——因为你抽象错了方向。
唯一的判断标准 如果去掉这个模式，代码更短、更好懂，那就去掉它。

设计模式的目的是降低复杂度。如果一个模式让代码变复杂了，那它用错了——不是它不好，是场景不对。

记住：模式是"发现"的，不是"套用"的。你写代码写到最后发现"这不就是观察者模式吗"，而不是一开始就想"我要用观察者模式"。

三、依赖方向：架构的本质

所有架构问题的本质都是依赖方向。依赖关系决定了改一处会不会牵连全局。

架构依赖方向后果
面条式互相引用改一处崩三处
分层上层依赖下层改底层影响上层
依赖倒置都依赖抽象可替换、可测试
游戏项目的分层建议 典型的分层：
表现层（UI、特效、动画）
逻辑层（战斗、背包、任务）
数据层（配置、存档、网络）
基础设施（资源、音频、输入）

规则：上层可以依赖下层，下层不能依赖上层。
比如"战斗逻辑"不应该直接引用"UI 血条"——应该发事件，让 UI 自己订阅（观察者模式）。

检验方法：试着单独编译"逻辑层"，如果必须带着 UI 一起编译，说明依赖方向错了。
