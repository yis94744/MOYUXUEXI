05 · 委托、事件与 LINQ

这三个东西是让代码"解耦"的核心工具。没有它们，你的 Player 类会直接引用 UI、音效、成就、统计等一大堆系统；有了它们，Player 只管发"我受伤了"这个事实，谁关心谁自己去听。但它们也是GC 分配的重灾区——这一页同时讲"怎么用"和"怎么用得不掉帧"。

阶段目标 约 1.5 周，35 小时。结束时你能：用事件重构一个紧耦合的模块、说清闭包捕获什么时候产生堆分配、知道 LINQ 在什么场景下不该用。

一、委托与事件

委托：类型安全的方法引用

using UnityEngine;

// ============================================================
//  自定义委托类型
// ============================================================
public delegate void DamageHandler(int amount, string source);

public class Health
{
    public DamageHandler OnDamage;

    public void ApplyDamage(int amount, string source)
    {
        OnDamage?.Invoke(amount, source);   // ?. 保证无订阅者时不抛异常
    }
}

public class DelegateDemo : MonoBehaviour
{
    private void LogDamage(int amount, string source)
    {
        Debug.Log(amount + " from " + source);
    }

    private void Start()
    {
        var h = new Health();

        h.OnDamage += LogDamage;                                // 方法组
        h.OnDamage += (a, s) => Debug.Log("lambda " + a);       // Lambda

        h.ApplyDamage(10, "fire");      // 两个处理器依次执行

        h.OnDamage -= LogDamage;        // 移除
        h.OnDamage = null;              // 清空
    }
}
内置委托签名典型用途
Actionvoid ()无参无返回
Action<T1,…>void (T1,…)最多 16 个参数
Func<TResult>TResult ()有返回值
Predicate<T>bool (T)List.Find 等
Comparison<T>int (T,T)List.Sort
UnityActionvoid ()UnityEvent 专用
多播委托的返回值只取最后一个 Func<int> f = () => 1; f += () => 2; f += () => 3; 调用 f() 返回 3。
想拿到全部结果，用 f.GetInvocationList() 遍历——但注意这会产生数组分配，不要放热路径。

事件：委托的安全封装

event 关键字把委托字段保护起来：外部只能 += / -=，不能赋值、不能直接调用。这是"发布-订阅"模式在 C# 里的标准实现。

using UnityEngine;

public class PlayerHealth : MonoBehaviour
{
    [SerializeField] private float _maxHp = 100f;
    private float _hp;

    // 事件声明：外部只能订阅/取消订阅
    public event System.Action<float, float> HpChanged;   // (current, max)
    public event System.Action Died;

    private void Awake() { _hp = _maxHp; }

    public void TakeDamage(float amount)
    {
        if (_hp <= 0f) return;

        _hp = Mathf.Max(0f, _hp - amount);
        HpChanged?.Invoke(_hp, _maxHp);      // ?. 保证无订阅者时不抛异常

        if (_hp <= 0f) Died?.Invoke();
    }
}

// ============================================================
//  订阅方：UI 血条
// ============================================================
public class HpBar : MonoBehaviour
{
    [SerializeField] private PlayerHealth _health;
    [SerializeField] private UnityEngine.UI.Image _fill;

    // 必须成对订阅/取消订阅！这是 Unity 最常见的泄漏来源
    private void OnEnable()  { _health.HpChanged += HandleHpChanged; }
    private void OnDisable() { _health.HpChanged -= HandleHpChanged; }

    private void HandleHpChanged(float current, float max)
    {
        _fill.fillAmount = max <= 0f ? 0f : current / max;
    }
}
为什么必须成对取消订阅 事件列表里保存着订阅者的委托，委托持有订阅者实例的引用。
如果 UI 对象被 Destroy 了但没取消订阅，那么：
① 它无法被 GC 回收（内存泄漏）；
② 事件再次触发时会调用到已销毁的对象，抛 MissingReferenceException。
规则：在 OnEnable 里订阅，就在 OnDisable 里取消。用这对回调能自动处理"对象被禁用"和"对象被销毁"两种情况。

C# event vs UnityEvent

维度 C# event UnityEvent
Inspector 配置 不可见 可视化连线
性能 更快 直接委托调用 稍慢 内部列表 + 虚调用
序列化 不序列化 可序列化（编辑器里保存绑定）
参数限制 任意泛型 最多 4 个参数
移除监听 直接 -= 传 lambda 时无法移除
适用 代码内部模块解耦、高频事件 策划配置、UI 按钮、美术连线 
using UnityEngine;
using UnityEngine.Events;

public class Interactable : MonoBehaviour
{
    // 可在 Inspector 里配置，不需要写代码
    public UnityEvent OnInteract;
    public UnityEvent<int> OnScore;              // 泛型版（2020.1+ 可在 Inspector 配置）
    public UnityEvent<string, float> OnMessage;

    private void OnMouseDown()
    {
        OnInteract?.Invoke();
    }
}

// ============================================================
//  代码绑定的正确姿势
// ============================================================
public class InteractBinder : MonoBehaviour
{
    [SerializeField] private Interactable _target;
    private UnityAction _cachedHandler;      // 关键：保存引用才能移除

    private void OnEnable()
    {
        // 错误：传 lambda 后无法移除（每次都是新对象）
        // _target.OnInteract.AddListener(() => Debug.Log("hi"));

        // 正确：先存到变量，再添加
        _cachedHandler = () => Debug.Log("interacted");
        _target.OnInteract.AddListener(_cachedHandler);
    }

    private void OnDisable()
    {
        if (_target == null) return;

        if (_cachedHandler != null)
        {
            _target.OnInteract.RemoveListener(_cachedHandler);
            _cachedHandler = null;
        }
    }
}

二、闭包：Lambda 的隐蔽成本 ★

Lambda 很方便，但捕获外部变量时会生成一个隐藏的类实例——每次都分配堆内存。这是 Unity 里"代码看起来很干净但每帧掉帧"的常见原因——它正是 GC 分配十大来源里的第 5 条。

闭包分配规则（必须记住） ① 不捕获任何外部状态（只用静态方法或字面量）→ 编译器会缓存委托实例，零分配。
② 捕获了局部变量 → 生成闭包类实例，每次执行都分配。
③ 捕获了 this（访问实例字段或调用实例方法）→ 同样生成包装对象，每次执行都分配。
所以 Update 里写的 lambda 是 GC 的头号来源之一。 
using System;
using System.Collections.Generic;
using UnityEngine;

public class ClosureDemo : MonoBehaviour
{
    private void Start()
    {
        // ============ for 循环变量捕获（经典 bug）============
        var actions = new List<Action>();
        for (int i = 0; i < 3; i++)
        {
            actions.Add(() => Debug.Log(i));   // 全部输出 3
        }
        foreach (var a in actions) a();

        // 修复：引入局部副本
        actions.Clear();
        for (int i = 0; i < 3; i++)
        {
            int captured = i;
            actions.Add(() => Debug.Log(captured));   // 0, 1, 2
        }
        foreach (var a in actions) a();

        // foreach 的迭代变量在 C# 5+ 是每次迭代独立拷贝，没有这个问题
        actions.Clear();
        foreach (int item in new[] { 0, 1, 2 })
        {
            actions.Add(() => Debug.Log(item));       // 0, 1, 2
        }
        foreach (var a in actions) a();
    }
}
using UnityEngine;
using System.Collections.Generic;

public class ClosurePerf : MonoBehaviour
{
    private List<Enemy> _enemies = new List<Enemy>();
    private float _scale = 1.5f;

    // 每帧分配：lambda 捕获了 this（因为读了实例字段 _scale）
    private void BadUpdate()
    {
        _enemies.Sort((a, b) => (a.Hp * _scale).CompareTo(b.Hp * _scale));   // 每次都是新的闭包对象
    }

    // 零分配：静态比较器，编译器把实例缓存到静态字段
    private static readonly System.Comparison<Enemy> HpComparison =
        (a, b) => a.Hp.CompareTo(b.Hp);

    private void GoodUpdate()
    {
        _enemies.Sort(HpComparison);
    }

    // 另一个零分配写法：lambda 不捕获任何外部状态（含 this），
    // 编译器会把它提升到方法外的缓存字段，只在第一次分配一次
    private void NoCaptureUpdate()
    {
        _enemies.Sort((a, b) => a.Hp.CompareTo(b.Hp));   // 实测 0 B/次
    }

    // C# 9 的 static lambda（Unity 2021.2+）：编译期保证不捕获
    private void StaticLambdaUpdate()
    {
        _enemies.Sort(static (a, b) => a.Hp.CompareTo(b.Hp));
    }
}

public class Enemy
{
    public float Hp;
}

三、LINQ：能力与代价

LINQ 让代码极其可读，代价是每次查询都会分配对象——这也是 GC 来源清单里的一条。关键不是"用不用"，而是知道在什么地方不能用。

using System.Linq;
using System.Collections.Generic;
using UnityEngine;

public class LinqDemo : MonoBehaviour
{
    public class Enemy
    {
        public int Id;
        public string Name;
        public float Hp;
        public int Level;
        public bool IsBoss;
    }

    private List<Enemy> _enemies = new List<Enemy>();

    private void Start()
    {
        // ---------- 筛选与投影 ----------
        List<Enemy> alive = _enemies.Where(e => e.Hp > 0f).ToList();
        string[] names = _enemies.Select(e => e.Name).ToArray();

        // ---------- 排序 ----------
        List<Enemy> sorted = _enemies.OrderBy(e => e.Level)
                                     .ThenByDescending(e => e.Hp)
                                     .ToList();

        // ---------- 聚合 ----------
        int totalLevel = _enemies.Sum(e => e.Level);
        float avgHp = _enemies.Average(e => e.Hp);
        int bossCount = _enemies.Count(e => e.IsBoss);

        // ---------- 查找 ----------
        Enemy first = _enemies.FirstOrDefault(e => e.IsBoss);
        bool anyAlive = _enemies.Any(e => e.Hp > 0f);

        // ---------- 分组 ----------
        foreach (var g in _enemies.GroupBy(e => e.Level))
        {
            Debug.Log(g.Key + ": " + g.Count());
        }

        // ============ 延迟执行（必须理解）============
        var query = _enemies.Where(e => e.Hp > 0f);   // 此时没有执行！
        _enemies.Add(new Enemy { Hp = 1f });          // 修改源集合
        foreach (var e in query) { }                  // 遍历时才执行，看到最新的集合

        // 想快照就加 ToList
        var snapshot = _enemies.Where(e => e.Hp > 0f).ToList();

        Debug.Log(alive.Count + names.Length + sorted.Count + totalLevel + avgHp + bossCount);
        Debug.Log(first + " " + anyAlive + " " + snapshot.Count);
    }
}
延迟执行：既是特性也是陷阱 var query = _enemies.Where(e => e.Hp > 0f); —— 这一行不执行任何筛选，只是构造了一个"查询描述"。
真正执行发生在你遍历它的时候。所以：
① 中间修改了源集合，遍历时看到的是最新的集合；
② 同一个 query 遍历两次会执行两次；
③ 源集合在遍历过程中被修改，会抛 InvalidOperationException。
想让它变成快照，加 .ToList() / .ToArray()。

LINQ 的性能代价

using System.Linq;
using UnityEngine;

public class LinqPerf : MonoBehaviour
{
    private System.Collections.Generic.List<Transform> _enemies
        = new System.Collections.Generic.List<Transform>();

    // ❌ 每帧 LINQ：多次 GC 分配（迭代器对象、闭包、结果集合）
    private void BadUpdate()
    {
        Transform target = _enemies
            .Where(e => e != null)
            .OrderBy(e => (e.position - transform.position).sqrMagnitude)
            .FirstOrDefault();

        if (target != null) Attack(target);
    }

    // ✅ 手写循环：零 GC，速度更快
    private Transform FindNearest()
    {
        Transform best = null;
        float bestSqr = float.MaxValue;
        Vector3 myPos = transform.position;

        for (int i = 0; i < _enemies.Count; i++)
        {
            Transform e = _enemies[i];
            if (e == null) continue;

            float d2 = (e.position - myPos).sqrMagnitude;
            if (d2 < bestSqr) { bestSqr = d2; best = e; }
        }
        return best;
    }

    private void Attack(Transform t) { }
}
LINQ 使用策略 放心用：初始化、编辑器工具、关卡加载、存档读写。这些地方可读性远比性能重要。
不要用：Update、FixedUpdate、物理回调、渲染相关、任何每帧执行且数据量大的地方。
折中方案：如果确实需要 LINQ 语义又要性能，可以看 ZLinq、StructLinq 这类零分配实现，或者自己写几个常用的扩展方法。
最重要的一点：先用 Profiler 确认这里真的是瓶颈，再决定要不要改。绝大多数项目的卡顿不在 LINQ 上。
