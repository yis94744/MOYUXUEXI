    一、依赖注入：后端架构的地基 ★★★

    依赖注入（DI）的核心：类不自己创建依赖，而是从外部接收。

    using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;

namespace BackendSamples.AspNetCore
{
    // ============================================================
    // 1. 依赖注入（DI）：后端架构的地基
    // ============================================================
    // 【为什么需要】如果每个类都自己 new 依赖，会出现：
    //   · 无法替换实现（测试时改不了数据库）
    //   · 无法管理生命周期（每次 new 一个连接池）
    //   · 依赖关系散落各处，改一处崩一片

    public interface IInventoryRepository
    {
        int GetCount(long userId, int itemId);
        void Add(long userId, int itemId, int count);
    }

    // 内存实现（生产环境换成数据库实现，业务代码不用改）
    public class InMemoryInventoryRepository : IInventoryRepository
    {
        private readonly Dictionary<(long, int), int> _data = new();

        public int GetCount(long userId, int itemId) =>
            _data.TryGetValue((userId, itemId), out int c) ? c : 0;

        public void Add(long userId, int itemId, int count)
        {
            _data.TryGetValue((userId, itemId), out int cur);
            _data[(userId, itemId)] = cur + count;
        }
    }

    // 业务服务：只依赖【接口】，不依赖具体实现
    public class InventoryService
    {
        private readonly IInventoryRepository _repo;

        // 【关键】通过构造函数注入，而不是自己 new
        public InventoryService(IInventoryRepository repo)
        {
            _repo = repo ?? throw new ArgumentNullException(nameof(repo));
        }

        public bool TryPickup(long userId, int itemId, int count)
        {
            if (count <= 0) return false;
            _repo.Add(userId, itemId, count);
            return true;
        }

        public int Query(long userId, int itemId) => _repo.GetCount(userId, itemId);
    }

    // ============================================================
    // 2. 三种生命周期
    // ============================================================
    // 【最容易踩的坑】选错生命周期会导致：
    //   · 内存泄漏（Singleton 持有请求数据）
    //   · 并发问题（Singleton 里存可变状态）
    //   · 性能问题（Scoped 被当成 Transient 频繁创建）

    public class SingletonService
    {
        public int Counter;
        public SingletonService() => Console.WriteLine("    [创建] SingletonService");
    }

    public class ScopedService
    {
        public int Counter;
        public ScopedService() => Console.WriteLine("    [创建] ScopedService");
    }

    public class TransientService
    {
        public int Counter;
        public TransientService() => Console.WriteLine("    [创建] TransientService");
    }

    // 简化的 DI 容器（演示用，生产用内置的 ServiceProvider）
    public class SimpleContainer
    {
        private readonly Dictionary<Type, object> _singletons = new();
        private readonly Dictionary<Type, Func<object>> _factories = new();

        public void AddSingleton<T>(T instance) => _singletons[typeof(T)] = instance;

        public void AddScoped<T>(Func<T> factory) where T : class =>
            _factories[typeof(T)] = () => factory();

        public T Get<T>() where T : class
        {
            var type = typeof(T);
            if (_singletons.TryGetValue(type, out var s)) return (T)s;
            if (_factories.TryGetValue(type, out var f)) return (T)f();
            throw new InvalidOperationException("未注册: " + type.Name);
        }
    }

    // ============================================================
    // 3. 中间件：请求管道的每一环
    // ============================================================
    // 【本质】一个中间件 = 一个 Func<Request, Func<Task>, Task>
    //   它可以：① 在下一个中间件前后做事情  ② 短路（不调用 next）

    public class HttpContext
    {
        public string Path { get; set; } = "";
        public Dictionary<string, string> Headers { get; } = new();
        public int StatusCode { get; set; } = 200;
        public string Body { get; set; } = "";
        public DateTime StartTime { get; set; }
        public long UserId { get; set; }
    }

    public delegate Task RequestDelegate(HttpContext ctx);

    // 中间件 1：记录耗时
    public class TimingMiddleware
    {
        private readonly RequestDelegate _next;
        public TimingMiddleware(RequestDelegate next) => _next = next;

        public async Task InvokeAsync(HttpContext ctx)
        {
            ctx.StartTime = DateTime.Now;
            await _next(ctx);   // 调用下一个中间件
            var ms = (DateTime.Now - ctx.StartTime).TotalMilliseconds;
            Console.WriteLine($"    [耗时] {ctx.Path} -> {ctx.StatusCode} ({ms:F1}ms)");
        }
    }

    // 中间件 2：认证（可以短路）
    public class AuthMiddleware
    {
        private readonly RequestDelegate _next;
        public AuthMiddleware(RequestDelegate next) => _next = next;

        public async Task InvokeAsync(HttpContext ctx)
        {
            // 白名单
            if (ctx.Path == "/login" || ctx.Path == "/health")
            {
                await _next(ctx);
                return;
            }

            // 【关键】没有 token 直接短路，不往下走
            if (!ctx.Headers.ContainsKey("token"))
            {
                ctx.StatusCode = 401;
                ctx.Body = "未登录";
                Console.WriteLine($"    [认证失败] {ctx.Path}");
                return;
            }

            ctx.UserId = 1001;   // 从 token 解析出来
            await _next(ctx);
        }
    }

    // 中间件 3：业务处理（终点）
    public class EndpointMiddleware
    {
        private readonly RequestDelegate? _next;
        public EndpointMiddleware(RequestDelegate? next = null) => _next = next;

        public Task InvokeAsync(HttpContext ctx)
        {
            ctx.StatusCode = 200;
            ctx.Body = ctx.Path switch
            {
                "/health" => "OK",
                "/login" => "{\"token\":\"abc123\"}",
                "/inventory" => "{\"items\":[{\"id\":1001,\"count\":5}]}",
                _ => "{\"msg\":\"hello\"}"
            };
            return Task.CompletedTask;
        }
    }

    // 管道构建器
    public class PipelineBuilder
    {
        private readonly List<Func<RequestDelegate, RequestDelegate>> _middlewares = new();

        public PipelineBuilder Use(Func<RequestDelegate, RequestDelegate> mw)
        {
            _middlewares.Add(mw);
            return this;
        }

        public RequestDelegate Build(RequestDelegate terminal)
        {
            // 【关键】从后往前包装，最后注册的最先执行
            RequestDelegate pipeline = terminal;
            for (int i = _middlewares.Count - 1; i >= 0; i--)
            {
                pipeline = _middlewares[i](pipeline);
            }
            return pipeline;
        }
    }

    // ============================================================
    // 4. 配置系统：分层与覆盖
    // ============================================================
    // 【规则】后加载的覆盖先加载的
    //   appsettings.json -> appsettings.{Env}.json -> 环境变量 -> 命令行

    public class AppConfig
    {
        private readonly Dictionary<string, string> _values = new();

        public void Load(Dictionary<string, string> source)
        {
            foreach (var kv in source) _values[kv.Key] = kv.Value;
        }

        public string Get(string key, string def = "") =>
            _values.TryGetValue(key, out var v) ? v : def;

        public int GetInt(string key, int def = 0) =>
            int.TryParse(Get(key), out int v) ? v : def;

        public bool GetBool(string key, bool def = false) =>
            bool.TryParse(Get(key), out bool v) ? v : def;
    }

    // ============================================================
    // 演示
    // ============================================================
    public static class Demo
    {
        public static async Task Run()
        {
            Console.WriteLine("=== 1. 依赖注入 ===");
            var container = new SimpleContainer();
            container.AddSingleton<IInventoryRepository>(new InMemoryInventoryRepository());
            container.AddScoped(() => new InventoryService(container.Get<IInventoryRepository>()));

            var inv = container.Get<InventoryService>();
            inv.TryPickup(1001, 5001, 10);
            inv.TryPickup(1001, 5001, 5);
            Console.WriteLine($"  玩家 1001 的 5001 数量: {inv.Query(1001, 5001)}");
            Console.WriteLine("  【关键】业务代码只依赖 IInventoryRepository 接口");
            Console.WriteLine("          换数据库实现不用改 InventoryService");

            Console.WriteLine();
            Console.WriteLine("=== 2. 三种生命周期 ===");
            Console.WriteLine("  Singleton: 全局唯一，第一次获取时创建");
            var s1 = new SingletonService();
            var s2 = s1;   // 单例：同一实例
            s1.Counter++;
            Console.WriteLine($"    s1.Counter={s1.Counter}, s2.Counter={s2.Counter}（同一实例）");

            Console.WriteLine("  Scoped: 每个请求一个，请求内共享");
            var sc1 = new ScopedService();
            sc1.Counter++;
            Console.WriteLine($"    同请求内两个服务共享: {sc1.Counter}");

            Console.WriteLine("  Transient: 每次获取都新建");
            var t1 = new TransientService();
            var t2 = new TransientService();
            t1.Counter++;
            Console.WriteLine($"    t1.Counter={t1.Counter}, t2.Counter={t2.Counter}（不同实例）");

            Console.WriteLine();
            Console.WriteLine("  【选择依据】");
            Console.WriteLine("    Singleton -> 无状态的服务（配置、缓存客户端、日志）");
            Console.WriteLine("    Scoped    -> 有请求级状态的（数据库上下文 DbContext）");
            Console.WriteLine("    Transient -> 轻量无状态（工具类）");

            Console.WriteLine();
            Console.WriteLine("=== 3. 中间件管道 ===");
            var builder = new PipelineBuilder();
            builder.Use(next => ctx => new TimingMiddleware(next).InvokeAsync(ctx));
            builder.Use(next => ctx => new AuthMiddleware(next).InvokeAsync(ctx));
            var pipeline = builder.Build(ctx => new EndpointMiddleware().InvokeAsync(ctx));

            Console.WriteLine("  请求 1: /health（白名单，不需要 token）");
            await pipeline(new HttpContext { Path = "/health" });

            Console.WriteLine();
            Console.WriteLine("  请求 2: /inventory（无 token，被拦截）");
            await pipeline(new HttpContext { Path = "/inventory" });

            Console.WriteLine();
            Console.WriteLine("  请求 3: /inventory（带 token，正常）");
            var ctx3 = new HttpContext { Path = "/inventory" };
            ctx3.Headers["token"] = "abc123";
            await pipeline(ctx3);
            Console.WriteLine($"    返回: {ctx3.Body}");

            Console.WriteLine();
            Console.WriteLine("  【关键】中间件的顺序决定行为");
            Console.WriteLine("    Timing 在最外层 -> 能统计到认证失败的耗时");
            Console.WriteLine("    Auth 在 Endpoint 前面 -> 能拦住未认证请求");

            Console.WriteLine();
            Console.WriteLine("=== 4. 配置系统 ===");
            var cfg = new AppConfig();
            cfg.Load(new Dictionary<string, string>
            {
                ["Database:Host"] = "localhost",
                ["Database:Port"] = "5432",
                ["Cache:Enabled"] = "true",
            });
            // 环境变量覆盖
            cfg.Load(new Dictionary<string, string> { ["Database:Host"] = "prod-db" });

            Console.WriteLine($"  Database:Host = {cfg.Get("Database:Host")}（被环境变量覆盖）");
            Console.WriteLine($"  Database:Port = {cfg.GetInt("Database:Port")}");
            Console.WriteLine($"  Cache:Enabled = {cfg.GetBool("Cache:Enabled")}");
            Console.WriteLine("  【规则】后加载的覆盖先加载的");
            Console.WriteLine("          json -> json.{Env} -> 环境变量 -> 命令行");
        }
    }
}

{
    public static async Task Main() => await BackendSamples.AspNetCore.Demo.Run();
}

    二、三种生命周期：选错就是灾难 ★★★

    三者的区别（最常考）
    生命周期何时创建典型用途
    Singleton第一次请求时创建，全局唯一配置、缓存客户端、日志
    Scoped每个请求一个DbContext、请求级上下文
    Transient每次获取都新建轻量无状态工具类
    经典错误：把 DbContext 注册成 Singleton——
    · DbContext 不是线程安全的
    · 多个请求并发使用同一个实例 → 数据错乱
    · 长期不释放 → 内存泄漏 + 连接池耗尽

    正确做法：AddDbContext 默认就是 Scoped。
    一个隐蔽的坑：Singleton 捕获了 Scoped 场景：
    class MySingleton
    {
      public MySingleton(MyDbContext db) { ... }
    }
    Singleton 依赖 Scoped → 这个 Scoped 会活到应用结束

    后果：
    · DbContext 被长期持有 → 追踪的对象越来越多，内存暴涨
    · 数据可能读到旧值（因为缓存了旧实体）

    框架默认行为：开发环境会直接抛异常（Cannot consume scoped service from singleton）。
    如果确实需要：用 IServiceScopeFactory 手动创建作用域。

    三、中间件：请求管道的每一环 ★★

    中间件的本质 一个中间件就是一个函数：
    Task Invoke(HttpContext ctx)

    它能做三件事：
    ① 在下一个中间件之前做事（如计时开始、记录请求）
    ② 在之后做事（如计时结束、改响应）
    ③ 短路——不调用 next，直接返回

    本页实测验证：
    · /health 走白名单 → 通过
    · /inventory 无 token → 被 Auth 短路，返回 401
    · /inventory 带 token → 走到 Endpoint

    顺序很重要：Timing 放最外层，才能统计到认证失败的耗时。
    常见中间件的顺序 官方推荐顺序（从上到下）：
    1. 异常处理（要能捕获所有后续异常）
    2. HTTPS 重定向
    3. 静态文件（命中就直接返回，不往下走）
    4. 路由
    5. 认证（你是谁）
    6. 授权（你能不能做）
    7. 端点

    放错顺序的典型后果：
    · 异常处理放最后 → 前面中间件的异常捕获不到
    · 授权放在认证之前 → 拿不到用户身份

    四、配置系统

    分层覆盖规则 加载顺序（后面的覆盖前面的）：
    ① appsettings.json
    ② appsettings.{Environment}.json
    ③ 环境变量
    ④ 命令行参数

    本页实测：
    Database:Host 在 json 里是 localhost
    环境变量覆盖后变成 prod-db

    实战价值：同一份代码，不同环境用不同配置。
    · 开发：连本地数据库
    · 生产：环境变量注入真实地址（不写进代码库）
