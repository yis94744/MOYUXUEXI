    using System;
using System.Collections.Generic;

namespace BackendSamples.Frontend

    // ============================================================
    // 1. 前端框架选型（写给后端看）
    // ============================================================
    // 【从后端视角看前端】
    //   前端本质是「把后端数据渲染成界面 + 收集用户输入」

    public static class FrameworkChoice
    {
        public static readonly (string Framework, string Pros, string Cons, string Use)[] All =
        {
            ("Vue 3",   "上手快、中文文档好", "生态比 React 小", "【运营后台首选】"),
            ("React",   "生态最大、灵活",     "概念多、需要更多决策", "复杂交互、国际化"),
            ("Svelte",  "编译期优化、代码少", "生态小", "小型项目"),
            ("纯 HTML+JS", "零依赖",          "复杂页面难维护", "极简工具页"),
            ("ASP.NET Razor", "后端一把梭",  "交互能力弱", "【最省事】表单型后台"),
        };

        // 【对后端最重要的判断】要不要引入构建工具
        public static string Recommendation(int pageCount, bool needRichInteraction)
        {
            if (pageCount <= 3 && !needRichInteraction)
                return "用 Razor / 服务端渲染，不要引入前端框架";
            if (pageCount <= 10 && !needRichInteraction)
                return "Vue 3 + CDN（不构建）";
            return "Vue 3 / React + Vite";
        }
    }

    // ============================================================
    // 2. 组件化思维（后端也能理解）
    // ============================================================
    // 【类比】前端组件 ≈ 后端的「可复用模块」
    //   输入（props）-> 处理 -> 输出（渲染）

    public class Component
    {
        public string Name = "";
        public Dictionary<string, object> Props = new();
        public List<Component> Children = new();

        public Component(string name) => Name = name;

        public Component WithProp(string key, object value)
        {
            Props[key] = value;
            return this;
        }

        public Component Add(Component child)
        {
            Children.Add(child);
            return this;
        }

        // 模拟渲染
        public string Render(int depth = 0)
        {
            var indent = new string(' ', depth * 2);
            var props = Props.Count > 0
                ? " " + string.Join(" ", Props.Select(kv => $"{kv.Key}=\"{kv.Value}\""))
                : "";

            var sb = new System.Text.StringBuilder();
            sb.AppendLine($"{indent}&lt;{Name}{props}&gt;");
            foreach (var c in Children) sb.Append(c.Render(depth + 1));
            sb.AppendLine($"{indent}&lt;/{Name}&gt;");
            return sb.ToString();
        }
    }

    // ============================================================
    // 4. 一个典型的分页列表：前后端怎么配合
    // ============================================================
    // 【后端该给什么】前端要什么就给什么，不要让它拼多次请求

    public class PageResult<T>
    {
        public List<T> Items = new();
        public int Total;
        public int Page;
        public int Size;

        public int TotalPages => (Total + Size - 1) / Size;
        public bool HasPrev => Page > 1;
        public bool HasNext => Page < TotalPages;
    }

    public class PlayerAdminItem
    {
        public long Id;
        public string Name = "";
        public int Level;
        public int Gold;
        public string LastLogin = "";
        public bool Banned;
    }

    // 【反例】只返回列表，前端还要再请求总数
    public class BadApi
    {
        public List<PlayerAdminItem> GetPlayers_Bad(int page, int size)
        {
            // 前端拿到列表后，还要再调一次 /count 才知道总页数
            return new List<PlayerAdminItem>();
        }
    }

    // 【正例】一次返回列表 + 总数 + 分页信息
    public class GoodApi
    {
        private readonly List<PlayerAdminItem> _all = new();

        public GoodApi()
        {
            for (int i = 1; i <= 55; i++)
            {
                _all.Add(new PlayerAdminItem
                {
                    Id = i,
                    Name = $"玩家{i}",
                    Level = (i % 50) + 1,
                    Gold = i * 100,
                    LastLogin = DateTime.UtcNow.AddHours(-i).ToString("MM-dd HH:mm"),
                    Banned = i % 17 == 0,
                });
            }
        }

        public PageResult<PlayerAdminItem> GetPlayers(int page, int size)
        {
            var r = new PageResult<PlayerAdminItem>
            {
                Total = _all.Count,
                Page = page,
                Size = size,
            };
            r.Items = _all.Skip((page - 1) * size).Take(size).ToList();
            return r;
        }

        // 【关键】筛选条件也要一次给全
        public PageResult<PlayerAdminItem> Search(string? keyword, bool? banned, int page, int size)
        {
            var q = _all.AsEnumerable();

            if (!string.IsNullOrWhiteSpace(keyword))
                q = q.Where(p => p.Name.Contains(keyword));

            if (banned.HasValue)
                q = q.Where(p => p.Banned == banned.Value);

            var list = q.ToList();
            var r = new PageResult<PlayerAdminItem>
            {
                Total = list.Count,
                Page = page,
                Size = size,
            };
            r.Items = list.Skip((page - 1) * size).Take(size).ToList();
            return r;
        }
    }

    // ============================================================
    // 5. 前端状态管理（后端需要理解的）
    // ============================================================
    // 【核心问题】前端有「状态」：登录用户、列表数据、加载中、错误
    //   状态管理错了 -> 界面错乱（显示旧数据、重复请求）

    public class ListState
    {
        public bool Loading;
        public string? Error;
        public List<PlayerAdminItem> Items = new();
        public int Total;
        public int Page = 1;

        public bool IsEmpty => !Loading && Error == null && Items.Count == 0;

        public string Describe()
        {
            if (Loading) return "加载中...";
            if (Error != null) return "错误: " + Error;
            if (IsEmpty) return "暂无数据";
            return $"共 {Total} 条，第 {Page} 页，当前显示 {Items.Count} 条";
        }
    }

    二、选型的实用判断 ★★

    本页给出的三条建议 按页面数和交互复杂度判断：

    场景推荐
    ≤3 页 + 简单表单Razor（服务端渲染）
    ≤10 页 + 简单交互Vue 3 + CDN（不构建）
    复杂后台Vue 3 / React + Vite
    给后端的建议：如果只是做个 GM 后台，Razor 就够了。
    不要为了「技术先进」引入整套前端工程——
    · 要维护 node_modules
    · 要配构建流程
    · 团队成员还得会前端

    判断标准：这个后台会有多少人用？用多久？

    三、分页接口：一次给全 ★★★

    反例：只返回列表 错误设计：
    GET /api/players?page=1 → [{...}, {...}]
    前端不知道总共有多少页！
    只能再调 GET /api/players/count → 2 次请求

    问题：
    · 两次请求之间数据可能变了 → 页码对不上
    · 前端要多写一次请求逻辑
    · 首屏慢一倍

    本页实测的正确做法：
    { items: [...], total: 55, page: 1, size: 10 }
    10 条，共 55 条，6 页
    有上一页: False，有下一页: True
    第 6 页（最后一页）: 5 条，有下一页: False

    原则：前端渲染一屏所需的所有数据，一次给全。
    筛选也一样 错误：先按名字筛，再按状态筛（两次请求）
    正确：所有筛选条件一次传
    GET /api/players?keyword=玩家1&banned=true&page=1

    本页实测：
    搜「玩家1」: 结果 11 条，本页 5 条
    只看已封禁: 结果 3 条（玩家17 / 玩家34 / 玩家51）

    后端要注意：
    · total 要反映筛选后的总数（不是全表）
    · 空条件要正确处理
    · 索引要覆盖筛选字段（页 07）

    四、前端状态：四种情况都要处理

    本页实现的 ListState 前端列表有四种状态：
    加载中...
    错误: 网络超时
    暂无数据
    共 55 条，第 1 页，当前显示 3 条

    后端要注意什么：
    · 加载中 → 前端显示骨架屏（后端要快）
    · 错误 → 返回明确的错误码和消息（页 03）
    · 空数据 → 返回空数组，不是 404
    · 正常 → 带全分页信息

    常见错误：查不到数据时返回 404 → 前端当成错误处理 → 显示报错而不是「暂无数据」。
