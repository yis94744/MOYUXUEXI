    一、房间状态机：不能乱跳

    using System;
using System.Collections.Generic;
using System.Linq;

namespace BackendSamples.Battle
{
    // ============================================================
    // 1. 战斗房间的生命周期
    // ============================================================

    public enum RoomState
    {
        Waiting,    // 等待玩家
        Loading,    // 加载中
        Playing,    // 战斗中
        Settling,   // 结算中
        Closed,     // 已关闭
    }

    public class PlayerInRoom
    {
        public long PlayerId;
        public string Name = "";
        public bool Connected = true;
        public bool Ready;
        public int Hp = 100;
        public int KillCount;
        public DateTime? DisconnectedAt;
    }

    public class BattleRoom
    {
        public string RoomId = Guid.NewGuid().ToString("N").Substring(0, 8);
        public RoomState State = RoomState.Waiting;
        public List<PlayerInRoom> Players = new();
        public DateTime CreatedAt = DateTime.UtcNow;
        public DateTime? StartedAt;
        public DateTime? EndedAt;
        public int FrameId;

        // 【关键】结算幂等标记
        public bool Settled;

        public const int MaxPlayers = 4;
        public const int ReconnectGraceSec = 60;

        public int AliveCount => Players.Count(p => p.Hp > 0);
        public bool IsFull => Players.Count >= MaxPlayers;
        public bool AllReady => Players.Count > 0 && Players.All(p => p.Ready);

        public override string ToString() =>
            $"房间{RoomId} 状态={State} 玩家={Players.Count}/{MaxPlayers} 存活={AliveCount}";
    }

    // ============================================================
    // 2. 战斗服：房间管理
    // ============================================================

    public class BattleServer
    {
        private readonly Dictionary<string, BattleRoom> _rooms = new();
        private int _totalSettled;

        public BattleRoom CreateRoom()
        {
            var room = new BattleRoom();
            _rooms[room.RoomId] = room;
            Console.WriteLine($"    创建房间 {room.RoomId}");
            return room;
        }

        public (bool Ok, string Msg) Join(BattleRoom room, long playerId, string name)
        {
            // 【关键】状态检查：战斗中不能加入
            if (room.State != RoomState.Waiting)
                return (false, $"房间状态为 {room.State}，不能加入");

            if (room.IsFull)
                return (false, "房间已满");

            if (room.Players.Any(p => p.PlayerId == playerId))
                return (false, "已在房间中");

            room.Players.Add(new PlayerInRoom { PlayerId = playerId, Name = name });
            Console.WriteLine($"    玩家 {name}({playerId}) 加入，当前 {room.Players.Count}/{BattleRoom.MaxPlayers}");
            return (true, "加入成功");
        }

        public (bool Ok, string Msg) Start(BattleRoom room)
        {
            if (room.State != RoomState.Waiting) return (false, "状态不对");
            if (room.Players.Count < 2) return (false, "至少需要2人");
            if (!room.AllReady) return (false, "还有玩家没准备");

            room.State = RoomState.Playing;
            room.StartedAt = DateTime.UtcNow;
            room.FrameId = 0;
            Console.WriteLine($"    房间 {room.RoomId} 开始战斗");
            return (true, "开始成功");
        }

        // 推进一帧
        public void Tick(BattleRoom room)
        {
            if (room.State != RoomState.Playing) return;
            room.FrameId++;

            // 检查战斗结束条件
            if (room.AliveCount <= 1)
            {
                Console.WriteLine($"    战斗结束：存活 {room.AliveCount} 人，共 {room.FrameId} 帧");
                room.State = RoomState.Settling;
                room.EndedAt = DateTime.UtcNow;
            }
        }

        // 掉线处理
        public void OnDisconnect(BattleRoom room, long playerId)
        {
            var p = room.Players.FirstOrDefault(x => x.PlayerId == playerId);
            if (p == null) return;

            p.Connected = false;
            p.DisconnectedAt = DateTime.UtcNow;
            Console.WriteLine($"    玩家 {p.Name} 掉线，宽限 {BattleRoom.ReconnectGraceSec} 秒");
        }

        // 重连处理
        public bool OnReconnect(BattleRoom room, long playerId, DateTime now)
        {
            var p = room.Players.FirstOrDefault(x => x.PlayerId == playerId);
            if (p == null || p.Connected) return false;

            double offline = (now - (p.DisconnectedAt ?? now)).TotalSeconds;
            if (offline > BattleRoom.ReconnectGraceSec)
            {
                Console.WriteLine($"    玩家 {p.Name} 离线 {offline:F0} 秒，超过宽限期");
                return false;
            }

            p.Connected = true;
            p.DisconnectedAt = null;
            Console.WriteLine($"    玩家 {p.Name} 重连成功（离线 {offline:F0} 秒）");
            return true;
        }

        // ============================================================
        // 3. 结算：必须幂等 ★★★
        // ============================================================
        public (bool Ok, string Msg) Settle(BattleRoom room)
        {
            if (room.State != RoomState.Settling)
                return (false, $"状态 {room.State} 不能结算");

            // 【关键】幂等检查
            if (room.Settled)
            {
                Console.WriteLine($"    房间 {room.RoomId} 已结算过，跳过");
                return (true, "已结算");
            }

            Console.WriteLine($"  结算房间 {room.RoomId}:");
            var winner = room.Players.OrderByDescending(p => p.KillCount).FirstOrDefault();
            foreach (var p in room.Players)
            {
                var isWinner = p == winner;
                Console.WriteLine($"    玩家 {p.Name}: 击杀 {p.KillCount}，{(isWinner ? "胜利" : "失败")}，发奖 {(isWinner ? 100 : 20)} 金币");
            }

            room.Settled = true;
            room.State = RoomState.Closed;
            _totalSettled++;
            return (true, "结算完成");
        }

        public int RoomCount => _rooms.Count;
        public int SettledCount => _totalSettled;
    }

    // ============================================================
    // 4. 战斗校验：防作弊的关键
    // ============================================================
    // 【核心问题】客户端上报"我赢了，杀了10个"，服务端信吗？
    //   不信。必须校验。

    public class BattleValidator
    {
        public class BattleReport
        {
            public long PlayerId;
            public int Kills;
            public int Deaths;
            public int DamageDealt;
            public int DamageTaken;
            public int DurationSec;
        }

        public static (bool Valid, string Reason) Validate(BattleReport r, int roomDurationSec)
        {
            // 【校验1】击杀数不能超过时间允许的上限
            //   假设最快 3 秒杀一个
            int maxPossibleKills = roomDurationSec / 3 + 1;
            if (r.Kills > maxPossibleKills)
                return (false, $"击杀数 {r.Kills} 超过理论上限 {maxPossibleKills}");

            // 【校验2】死亡数逻辑
            if (r.Kills > 0 && r.Deaths > r.Kills * 2 + 5)
                return (false, "死亡数异常（可能是刷分）");

            // 【校验3】伤害与击杀要匹配
            if (r.Kills > 0 && r.DamageDealt < r.Kills * 50)
                return (false, $"伤害 {r.DamageDealt} 与击杀 {r.Kills} 不匹配");

            // 【校验4】受伤与死亡
            if (r.Deaths > 0 && r.DamageTaken == 0)
                return (false, "有死亡但没受伤，异常");

            // 【校验5】时长
            if (r.DurationSec < roomDurationSec - 5 || r.DurationSec > roomDurationSec + 30)
                return (false, $"战斗时长 {r.DurationSec}s 与房间 {roomDurationSec}s 不符");

            return (true, "校验通过");
        }
    }

    // ============================================================
    // 演示
    // ============================================================
    public static class Demo
    {
        public static void Run()
        {
            Console.WriteLine("=== 1. 房间生命周期 ===");
            var server = new BattleServer();
            var room = server.CreateRoom();

            Console.WriteLine("  三个玩家加入:");
            server.Join(room, 1001, "玩家A");
            server.Join(room, 1002, "玩家B");
            server.Join(room, 1003, "玩家C");
            Console.WriteLine($"    {room}");

            Console.WriteLine();
            Console.WriteLine("  尝试开始（还没人准备）:");
            var (ok, msg) = server.Start(room);
            Console.WriteLine($"    结果: {ok}, {msg}");

            Console.WriteLine("  全员准备后:");
            foreach (var p in room.Players) p.Ready = true;
            (ok, msg) = server.Start(room);
            Console.WriteLine($"    结果: {ok}, {msg}");
            Console.WriteLine($"    {room}");

            Console.WriteLine();
            Console.WriteLine("  战斗中不能加入:");
            server.Join(room, 1004, "玩家D");

            Console.WriteLine();
            Console.WriteLine("=== 2. 推进战斗 ===");
            room.Players[0].KillCount = 1;
            room.Players[1].KillCount = 2;
            room.Players[2].Hp = 0;

            for (int i = 0; i < 5; i++)
            {
                server.Tick(room);
            }
            Console.WriteLine($"    当前帧: {room.FrameId}，{room}");

            Console.WriteLine();
            Console.WriteLine("=== 3. 掉线与重连 ===");
            var now = DateTime.UtcNow;
            server.OnDisconnect(room, 1002);

            Console.WriteLine("  30 秒后重连（在宽限期内）:");
            server.OnReconnect(room, 1002, now.AddSeconds(30));

            Console.WriteLine("  再掉线，90 秒后才回来（超过宽限期）:");
            server.OnDisconnect(room, 1002);
            server.OnReconnect(room, 1002, now.AddSeconds(120));

            Console.WriteLine();
            Console.WriteLine("=== 4. 结算（幂等）===");
            room.Players[1].Hp = 0;
            room.Players[2].Hp = 0;

            // 【关键】必须让 Tick 检测到战斗结束，状态才会进入 Settling
            Console.WriteLine("  推最后一帧，触发结束检测:");
            server.Tick(room);
            Console.WriteLine($"    房间状态: {room.State}");

            Console.WriteLine();
            Console.WriteLine("  第一次结算:");
            server.Settle(room);

            Console.WriteLine();
            Console.WriteLine("  重复调用结算（模拟重试）:");
            server.Settle(room);
            Console.WriteLine($"    已结算房间数: {server.SettledCount}（没重复发奖）");

            Console.WriteLine();
            Console.WriteLine("=== 5. 战斗结算校验（防作弊）===");
            var cases = new[]
            {
                (name: "正常", report: new BattleValidator.BattleReport
                    { PlayerId = 1, Kills = 5, Deaths = 2, DamageDealt = 800, DamageTaken = 300, DurationSec = 60 }),

                (name: "击杀数异常", report: new BattleValidator.BattleReport
                    { PlayerId = 2, Kills = 100, Deaths = 0, DamageDealt = 10000, DamageTaken = 0, DurationSec = 30 }),

                (name: "有死亡但没受伤", report: new BattleValidator.BattleReport
                    { PlayerId = 3, Kills = 3, Deaths = 2, DamageDealt = 500, DamageTaken = 0, DurationSec = 45 }),

                (name: "时长不符", report: new BattleValidator.BattleReport
                    { PlayerId = 4, Kills = 2, Deaths = 1, DamageDealt = 300, DamageTaken = 150, DurationSec = 999 }),
            };

            foreach (var c in cases)
            {
                var (valid, reason) = BattleValidator.Validate(c.report, roomDurationSec: 45);
                Console.WriteLine($"    [{c.name,-14}] {(valid ? "通过" : "拒绝: " + reason)}");
            }

            Console.WriteLine();
            Console.WriteLine("  【关键】客户端上报的数据一律不可信");
            Console.WriteLine("          必须用「理论上限」做交叉校验");
        }
    }
}

public class Program
{
    public static void Main() => BackendSamples.Battle.Demo.Run();
}
    状态能做什么不能做什么
    Waiting加入、准备、开始不能推进帧
    Loading加载资源不能结算
    Playing推进帧不能加入
    Settling结算不能推进
    Closed—什么都不能做

    二、状态转移必须显式触发 ★★

    一个真实的 bug（我写的时候踩到了） 我最初这样写：
    room.Players[1].Hp = 0;
    room.Players[2].Hp = 0;
    server.Settle(room);
    结果：结算没执行，已结算房间数 = 0

    原因：
    · 房间状态还是 Playing
    · Settle 要求 Settling 状态
    · 状态不会自动转移——必须调 Tick 检测"存活数 ≤ 1"

    修正：
    server.Tick(room); // 触发结束检测
    // 输出: 战斗结束：存活 1 人，共 6 帧
    // 输出: 房间状态: Settling
    server.Settle(room); // 现在才成功

    教训：状态机必须有"谁负责推进"的明确约定。

    三、断线重连：战斗中的特殊处理 ★★★

    宽限期设计 本页实测：
    玩家 B 掉线，宽限 60 秒
    30 秒后重连 → 重连成功（离线 30 秒）
    再掉线，120 秒才回来 → 超过宽限期，失败

    宽限期内要做什么：
    · 保留玩家在房间里（不删除）
    · 广播「掉线中」状态给其他人
    · 继续推进战斗（角色原地不动或 AI 托管）
    · 缓存期间的事件（页 12）

    超过宽限期：
    · 踢出房间
    · 如果是 1v1 → 判负
    · 如果是团队 → 让队友继续

    四、结算幂等：重复调用只发一次奖 ★★★

    为什么结算必须幂等 触发结算的路径可能有多条：
    · 战斗正常结束
    · 超时强制结束
    · 服务器重启后恢复
    · 客户端重试上报

    如果结算不幂等：玩家可能收到多份奖励。

    本页实测：
    推最后一帧，触发结束检测:
      战斗结束：存活 1 人，共 6 帧
      房间状态: Settling
    第一次结算:
      玩家A: 击杀 1，失败，发奖 20 金币
      玩家B: 击杀 2，胜利，发奖 100 金币
      玩家C: 击杀 0，失败，发奖 20 金币
    重复调用结算（模拟重试）:
      已结算房间数: 1（没重复发奖）

    注意顺序：先 Tick 检测结束 → 状态变 Settling → 才能 Settle。
    直接 Settle 会被拒（状态还是 Playing）——这就是第二节那个真实 bug。

    实现方式：
    · room.Settled 标记
    · 或数据库唯一索引 (roomId)
    · 多实例部署必须用数据库方案

    五、结算校验：客户端数据不可信 ★★★

    本页实现的五条校验 问题：客户端上报「我杀了 100 个」，服务端信吗？
    不信。必须用「理论上限」交叉校验。

    本页实测（1 组正常 + 3 组拒绝）：
    [正常    ] 通过
    [击杀数异常 ] 拒绝: 击杀数 100 超过理论上限 16
    [有死亡但没受伤] 拒绝: 有死亡但没受伤，异常
    [时长不符  ] 拒绝: 战斗时长 999s 与房间 45s 不符

    五条校验逻辑：
    ① 击杀数 ≤ 时长 / 最快击杀间隔（30秒最多杀10个）
    ② 死亡数 ≤ 击杀数的合理倍数
    ③ 有击杀就必须有伤害输出
    ④ 有死亡就必须有受伤
    ⑤ 上报时长要和房间时长吻合

    这些校验能用极低成本拦掉大部分低级作弊。
