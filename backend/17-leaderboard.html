    一、为什么不能用「每次排序」★★★

    排序方案的致命问题 简单做法：players.OrderByDescending(p => p.Score)

    问题：
    · 每次查询都要排 10 万条 → 几百毫秒
    · 每次有玩家分数变化 → 又要重排
    · 100 个玩家同时更新 → 排序 100 次

    正确做法：跳表（SkipList）
    · 插入时就维护好顺序
    · 插入/删除/查排名都是 O(log n)
    · 取前 100 名是 O(log n + k)

    Redis 的 ZSet 就是这么实现的——所以排行榜应该用 ZSet。

    本页实测对比（1 万条数据）：
    跳表插入 10000 条: 6.0 ms
    跳表取前 100: 0.010 ms
    跳表查排名: 0.179 ms
    排序（一次性）: 5.5 ms 
    using System;
using System.Collections.Generic;
using System.Linq;

namespace BackendSamples.Leaderboard
{
    // ============================================================
    // 1. 排行榜的核心数据结构：跳表 / 有序集合
    // ============================================================
    // 【为什么不用排序数组】
    //   每次更新都要 O(n) 移动 -> 10 万玩家疯狂更新 -> 崩
    //
    // 【Redis ZSet 的实现】跳表（SkipList）
    //   插入/删除/查询都是 O(log n)
    //   范围查询（取前100名）也是 O(log n + k)

    public class SkipListNode
    {
        public double Score;
        public string Member = "";
        public SkipListNode[] Forward;
        public SkipListNode(int level) => Forward = new SkipListNode[level];
    }

    public class SkipList
    {
        private const int MaxLevel = 16;
        private readonly SkipListNode _head = new(MaxLevel);
        private int _level = 1;
        private readonly Random _rnd = new(42);

        public int Count { get; private set; }

        // 【关键】分数 + 成员名，分数相同按名字排序（保证顺序稳定）
        private static int Compare(double s1, string m1, double s2, string m2)
        {
            int c = s2.CompareTo(s1);   // 【降序】分数高的在前
            return c != 0 ? c : string.CompareOrdinal(m1, m2);
        }

        public void Add(double score, string member)
        {
            var update = new SkipListNode[MaxLevel];
            var cur = _head;

            for (int i = _level - 1; i >= 0; i--)
            {
                while (cur.Forward[i] != null &&
                       Compare(cur.Forward[i].Score, cur.Forward[i].Member, score, member) < 0)
                {
                    cur = cur.Forward[i];
                }
                update[i] = cur;
            }

            // 已存在则更新
            var next = cur.Forward[0];
            if (next != null && next.Member == member && next.Score == score)
                return;

            int newLevel = RandomLevel();
            if (newLevel > _level)
            {
                for (int i = _level; i < newLevel; i++) update[i] = _head;
                _level = newLevel;
            }

            var node = new SkipListNode(newLevel) { Score = score, Member = member };
            for (int i = 0; i < newLevel; i++)
            {
                node.Forward[i] = update[i].Forward[i];
                update[i].Forward[i] = node;
            }
            Count++;
        }

        private int RandomLevel()
        {
            int level = 1;
            while (_rnd.Next(2) == 1 && level < MaxLevel) level++;
            return level;
        }

        // 取前 N 名
        public List<(int Rank, string Member, double Score)> TopN(int n)
        {
            var result = new List<(int, string, double)>();
            var cur = _head.Forward[0];
            int rank = 1;

            while (cur != null && rank <= n)
            {
                result.Add((rank, cur.Member, cur.Score));
                cur = cur.Forward[0];
                rank++;
            }

            return result;
        }

        // 查某人的排名
        public int? GetRank(string member)
        {
            var cur = _head.Forward[0];
            int rank = 1;
            while (cur != null)
            {
                if (cur.Member == member) return rank;
                cur = cur.Forward[0];
                rank++;
            }
            return null;
        }

        // 取某人的分数
        public double? GetScore(string member)
        {
            var cur = _head.Forward[0];
            while (cur != null)
            {
                if (cur.Member == member) return cur.Score;
                cur = cur.Forward[0];
            }
            return null;
        }
    }

    // ============================================================
    // 2. 排行榜的四个常见需求
    // ============================================================

    public class LeaderboardService
    {
        private readonly SkipList _all = new();
        private readonly Dictionary<long, double> _scores = new();

        // 更新分数
        public void UpdateScore(long playerId, double score)
        {
            _scores[playerId] = score;
            _all.Add(score, playerId.ToString());
        }

        // 【需求1】全服前 N 名
        public List<(int Rank, long PlayerId, double Score)> GetTopN(int n)
        {
            return _all.TopN(n).Select(x =>
                (x.Rank, long.Parse(x.Member), x.Score)).ToList();
        }

        // 【需求2】我的排名（玩家最关心的）
        public (int? Rank, double Score) GetMyRank(long playerId)
        {
            double score = _scores.TryGetValue(playerId, out var s) ? s : 0;
            int? rank = _all.GetRank(playerId.ToString());
            return (rank, score);
        }

        // 【需求3】我附近的排名（前后各 N 名）
        //   【关键】玩家看不到自己的排名会流失
        public List<(int Rank, long PlayerId, double Score)> GetAroundMe(long playerId, int range = 3)
        {
            var top = _all.TopN(_all.Count);
            var idx = top.FindIndex(x => x.Member == playerId.ToString());
            if (idx < 0) return new();

            int start = Math.Max(0, idx - range);
            int end = Math.Min(top.Count - 1, idx + range);

            return top.Skip(start).Take(end - start + 1).Select(x =>
                (x.Rank, long.Parse(x.Member), x.Score)).ToList();
        }

        // 【需求4】分段奖励
        public string GetRewardTier(int rank)
        {
            return rank switch
            {
                1 => "冠军奖励",
                <= 3 => "前三奖励",
                <= 10 => "前十奖励",
                <= 100 => "前百奖励",
                _ => "参与奖",
            };
        }

        public int Total => _all.Count;
    }

    // ============================================================
    // 3. 性能对比：跳表 vs 排序
    // ============================================================

    public static class PerfComparison
    {
        public static void Run()
        {
            const int N = 10000;
            var rnd = new Random(42);
            var scores = Enumerable.Range(1, N)
                .Select(i => (Id: i, Score: rnd.NextDouble() * 10000))
                .ToList();

            Console.WriteLine($"  数据量: {N} 个玩家");

            // 方式1：跳表（插入时就维护好顺序）
            var sw = System.Diagnostics.Stopwatch.StartNew();
            var sl = new SkipList();
            foreach (var (id, score) in scores) sl.Add(score, id.ToString());
            sw.Stop();
            Console.WriteLine($"  跳表插入 {N} 条:     {sw.Elapsed.TotalMilliseconds:F1} ms");

            sw.Restart();
            var top = sl.TopN(100);
            sw.Stop();
            Console.WriteLine($"  跳表取前 100:       {sw.Elapsed.TotalMilliseconds:F3} ms");

            sw.Restart();
            var rank = sl.GetRank("5000");
            sw.Stop();
            Console.WriteLine($"  跳表查排名:         {sw.Elapsed.TotalMilliseconds:F3} ms");

            // 方式2：每次排序（错误做法）
            sw.Restart();
            var sorted = scores.OrderByDescending(x => x.Score).ToList();
            sw.Stop();
            Console.WriteLine();
            Console.WriteLine($"  排序（一次性）:      {sw.Elapsed.TotalMilliseconds:F1} ms");
            Console.WriteLine("  【问题】每次有玩家分数变化，都要重新排一次");
            Console.WriteLine("          100 个玩家同时更新 = 排序 100 次");

            Console.WriteLine();
            Console.WriteLine("  【结论】跳表把「插入 + 查询」都做成 O(log n)");
            Console.WriteLine("          Redis 的 ZSet 就是这么实现的");
        }
    }

    {
        public static void Run()
        {
            Console.WriteLine("=== 1. 排行榜基本操作 ===");
            var lb = new LeaderboardService();

            Console.WriteLine("  10 个玩家更新分数:");
            var players = new (long Id, double Score)[]
            {
                (1001, 9500), (1002, 8800), (1003, 9200), (1004, 7600),
                (1005, 9800), (1006, 8100), (1007, 8900), (1008, 9900),
                (1009, 7000), (1010, 8500),

            Console.WriteLine($"    总人数: {lb.Total}");

            Console.WriteLine();
            Console.WriteLine("  玩家 1004 看自己的排名:");
            var (r, s) = lb.GetMyRank(1004);
            Console.WriteLine($"    排名 {r}，分数 {s:F0}");

            Console.WriteLine();
            Console.WriteLine("  玩家 1004 附近的排名（前后各 2 名）:");
            foreach (var (rank, pid, score) in lb.GetAroundMe(1004, 2))
            {
                var mark = pid == 1004 ? " <<< 我" : "";
                Console.WriteLine($"    第{rank}名  玩家{pid}  {score:F0}{mark}");
            }
            Console.WriteLine();
            Console.WriteLine("  【关键】玩家最关心「我附近的人」，而不是第一名");
            Console.WriteLine("          只给前10名会让普通玩家失去动力");

            Console.WriteLine();
            Console.WriteLine("=== 2. 分数更新（已有玩家）===");
            Console.WriteLine("  玩家 1004 从 7600 涨到 9700:");

            var (r2, s2) = lb.GetMyRank(1004);
            Console.WriteLine($"    新排名 {r2}，分数 {s2:F0}");

            Console.WriteLine("    · 好友榜每次查时实时算（数据量小）");

        }
    }

    二、玩家最关心的是「我附近」★★★

    只给前 10 名会流失玩家 心理：
    · 前 10 名永远是那几个大佬
    · 普通玩家看自己的排名在第 8 万名 → 毫无参与感

    本页实现：GetAroundMe
    · 显示我 + 前后各 2 名
    · 玩家看到「再打 200 分就能超过上面的人」

    本页实测输出（玩家 1004 排名第 9，显示前后各 2 名）：
    第7名 玩家1010 8500
    第8名 玩家1006 8100
    第9名 玩家1004 7600 <<< 我
    第10名 玩家1009 7000

    这是留存设计，不只是技术问题。

    三、分数更新与并列排名

    两个细节 ① 同一玩家重复更新
    · 应该是更新分数，不是新增一条
    · 本页跳表实现了这个检查

    ② 分数相同时怎么排
    · 本页用成员名做次级排序
    · Compare(s1, m1, s2, m2)：先比分数，相同比名字
    · 保证顺序稳定（否则同一分数每次查询顺序不同）

    游戏里的常见需求：分数相同 → 先达到的排前面
    → 需要额外存达到时间做次级排序。

    四、榜单类型与赛季设计

    类型刷新用途
    全服总榜永久累积历史最高分
    赛季榜每赛季重置赛季奖励
    周榜每周重置活跃度激励
    好友榜实时社交比较
    公会榜按公会聚合公会战
    赛季重置怎么做 错误做法：遍历所有玩家清零 → 10 万人耗时很长。

    正确做法：换 key
    rank:s1 → rank:s2
    · 新赛季写入新 key
    · 旧 key 归档（用于历史查询）
    · 零成本切换

    发奖励怎么做：
    · 结算时读取旧 key 的前 N 名
    · 通过邮件发放
    · 发奖要幂等（页 15）

    本章动手任务

    本章过关标准：
    ① 知道排行榜该用跳表/ZSet，不能用每次排序；
    ② 能实现跳表的插入与前 N 查询；
    ③ 知道要显示「我附近」而不是只有前 10 名；
    ④ 能处理并列排名与重复更新；
    ⑤ 知道赛季重置用「换 key」而不是遍历清零。

    第 ① 和 第 ③ 条最关键——**前者是性能，后者是留存**。
