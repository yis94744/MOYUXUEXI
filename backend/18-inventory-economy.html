    一、并发扣减的四种方案 ★★★

    错误示范：先读后写 var w = Get(playerId);
    if (w.Gold < amount) return false;
    w.Gold -= amount; // ← 危险

    问题：两步之间可能被其他请求插入。
    · 请求A 读到 1000
    · 请求B 读到 1000
    · 两个都扣 300 → 结果 400（应该 400，实际正确）
    · 但如果两个都扣 900 → 余额变成 100 或 -800

    这就是「丢失更新」——真实的生产事故来源。
    四种正确方案
    方案原理适用
    加锁lock / 分布式锁单机，或必须串行
    乐观锁版本号 CAS，冲突重试读多写少
    条件 UPDATEWHERE gold >= ?最常用，最简单
    队列串行化同一玩家的操作进同一队列复杂但彻底
    本页实测了前三种。
    推荐：条件 UPDATE（数据库保证原子，不需要重试）。 
    using System;
using System.Collections.Generic;
using System.Linq;

namespace BackendSamples.Inventory
{
    // ============================================================
    // 1. 并发扣减的三种方案
    // ============================================================
    // 【核心问题】玩家同时点两次购买，金币会不会扣两次？

    public class PlayerWallet
    {
        public long PlayerId;
        public int Gold;
        public int Version;   // 乐观锁版本号
    }

    public class WalletService
    {
        private readonly Dictionary<long, PlayerWallet> _db = new();
        private readonly object _lock = new();

        public WalletService()
        {
            _db[1001] = new PlayerWallet { PlayerId = 1001, Gold = 1000, Version = 1 };
            _db[1002] = new PlayerWallet { PlayerId = 1002, Gold = 500, Version = 1 };
        }

        public PlayerWallet? Get(long id) => _db.TryGetValue(id, out var w) ? w : null;

        // ---- 方案一：无条件扣减（错误！）----
        // 【问题】先读后写，两步之间可能被其他请求插入
        public bool Deduct_Bad(long playerId, int amount)
        {
            var w = Get(playerId);
            if (w == null || w.Gold < amount) return false;

            // 【危险】这里如果另一个请求改了 w.Gold，就会覆盖
            w.Gold -= amount;
            return true;
        }

        // ---- 方案二：加锁（单机可行，多机不行）----
        public bool Deduct_Lock(long playerId, int amount)
        {
            lock (_lock)
            {
                var w = Get(playerId);
                if (w == null || w.Gold < amount) return false;
                w.Gold -= amount;
                return true;
            }
        }

        // ---- 方案三：乐观锁（推荐）----
        // 【原理】UPDATE ... WHERE version = 旧版本
        //   影响行数为 0 说明被别人改过 -> 重试
        public bool Deduct_Optimistic(long playerId, int amount, int maxRetry = 3)
        {
            for (int retry = 0; retry < maxRetry; retry++)
            {
                var w = Get(playerId);
                if (w == null || w.Gold < amount) return false;

                int oldVersion = w.Version;

                // 【模拟】CAS 操作：只有版本没变才更新
                lock (_lock)
                {
                    if (w.Version != oldVersion)
                        continue;   // 版本变了，重试

                    w.Gold -= amount;
                    w.Version++;
                    return true;
                }
            }

            Console.WriteLine($"    重试 {maxRetry} 次仍失败（并发太高）");
            return false;
        }

        // ---- 方案四：条件 UPDATE（最简单有效）----
        // SQL: UPDATE wallet SET gold = gold - 100 WHERE id = ? AND gold >= 100
        //   数据库保证原子性，影响行数为 0 就说明钱不够
        public bool Deduct_Conditional(long playerId, int amount)
        {
            lock (_lock)
            {
                var w = Get(playerId);
                // 【关键】判断和扣减在同一个原子操作里
                if (w == null || w.Gold < amount) return false;
                w.Gold -= amount;
                return true;
            }
        }
    }

    // ============================================================
    // 2. 道具流水：每一笔变更都要记录
    // ============================================================
    // 【为什么需要流水】
    //   · 玩家投诉"我的道具没了" -> 可以查
    //   · 发现刷道具漏洞 -> 可以追溯
    //   · 对账 -> 收入支出要平衡

    public class ItemFlow
    {
        public long Id;
        public long PlayerId;
        public int ItemId;
        public int Delta;          // 正数=获得，负数=消耗
        public string Reason = ""; // 原因
        public string RefId = "";  // 关联ID（订单号/任务ID）
        public DateTime Time = DateTime.UtcNow;

        public override string ToString() =>
            $"[{Time:HH:mm:ss}] 玩家{PlayerId} 道具{ItemId} {(Delta > 0 ? "+" : "")}{Delta} ({Reason})";
    }

    public class InventoryService
    {
        private readonly Dictionary<(long, int), int> _items = new();
        private readonly List<ItemFlow> _flows = new();
        private long _flowId;

        public int GetCount(long playerId, int itemId) =>
            _items.TryGetValue((playerId, itemId), out int c) ? c : 0;

        // 【关键】变更道具必须写流水
        public bool AddItem(long playerId, int itemId, int count, string reason, string refId = "")
        {
            if (count <= 0) return false;

            var key = (playerId, itemId);
            _items.TryGetValue(key, out int cur);
            _items[key] = cur + count;

            _flows.Add(new ItemFlow
            {
                Id = ++_flowId,
                PlayerId = playerId,
                ItemId = itemId,
                Delta = count,
                Reason = reason,
                RefId = refId,
            });

            return true;
        }

        public bool RemoveItem(long playerId, int itemId, int count, string reason, string refId = "")
        {
            if (count <= 0) return false;

            var key = (playerId, itemId);
            _items.TryGetValue(key, out int cur);
            if (cur < count)
            {
                Console.WriteLine($"    道具不足: 有 {cur}, 需要 {count}");
                return false;
            }

            _items[key] = cur - count;

            _flows.Add(new ItemFlow
            {
                Id = ++_flowId,
                PlayerId = playerId,
                ItemId = itemId,
                Delta = -count,
                Reason = reason,
                RefId = refId,
            });

            return true;
        }

        public IEnumerable<ItemFlow> GetFlows(long playerId) =>
            _flows.Where(f => f.PlayerId == playerId);

        // 【对账】检查流水总和是否等于当前库存
        public (bool Balanced, int Computed, int Actual) Reconcile(long playerId, int itemId)
        {
            int computed = _flows
                .Where(f => f.PlayerId == playerId && f.ItemId == itemId)
                .Sum(f => f.Delta);
            int actual = GetCount(playerId, itemId);
            return (computed == actual, computed, actual);
        }

        public int FlowCount => _flows.Count;
    }

    // ============================================================
    // 3. 防刷：同一笔奖励不能领两次
    // ============================================================

    public class RewardClaimer
    {
        private readonly HashSet<string> _claimed = new();
        private readonly InventoryService _inv;
        private readonly WalletService _wallet;

        public RewardClaimer(InventoryService inv, WalletService wallet)
        {
            _inv = inv;
            _wallet = wallet;
        }

        // 【关键】用业务唯一键防重复领取
        public (bool Ok, string Msg) Claim(long playerId, string rewardKey, int gold, int itemId, int count)
        {
            // 【核心】业务唯一键 = 玩家 + 奖励标识
            string key = playerId + ":" + rewardKey;

            if (_claimed.Contains(key))
            {
                // 【重要】返回成功，不是失败
                return (true, "已经领取过了");
            }

            _claimed.Add(key);

            // 发奖
            if (gold > 0)
            {
                var w = _wallet.Get(playerId);
                if (w != null) w.Gold += gold;
            }
            if (itemId > 0 && count > 0)
            {
                _inv.AddItem(playerId, itemId, count, "claim:" + rewardKey, rewardKey);
            }

            return (true, "领取成功");
        }

        public int ClaimedCount => _claimed.Count;
    }

    // ============================================================
    // 4. 防刷检测：异常行为识别
    // ============================================================

    public class AntiCheatMonitor
    {
        private readonly Dictionary<long, List<(DateTime Time, int Amount)>> _history = new();

        public void Record(long playerId, int amount)
        {
            if (!_history.TryGetValue(playerId, out var list))
            {
                list = new List<(DateTime, int)>();
                _history[playerId] = list;
            }
            list.Add((DateTime.UtcNow, amount));
        }

        // 【检测1】单位时间收益过高
        public bool CheckEarningRate(long playerId, int windowSec = 60, int threshold = 10000)
        {
            if (!_history.TryGetValue(playerId, out var list)) return false;

            var now = DateTime.UtcNow;
            int total = list
                .Where(x => (now - x.Time).TotalSeconds <= windowSec)
                .Sum(x => x.Amount);

            if (total > threshold)
            {
                Console.WriteLine($"    【异常】玩家{playerId} 在 {windowSec}s 内获得 {total}（阈值 {threshold}）");
                return true;
            }
            return false;
        }

        // 【检测2】操作频率过高
        public bool CheckActionRate(long playerId, int windowSec = 10, int threshold = 50)
        {
            if (!_history.TryGetValue(playerId, out var list)) return false;

            var now = DateTime.UtcNow;
            int count = list.Count(x => (now - x.Time).TotalSeconds <= windowSec);

            if (count > threshold)
            {
                Console.WriteLine($"    【异常】玩家{playerId} 在 {windowSec}s 内操作 {count} 次（阈值 {threshold}）");
                return true;
            }
            return false;
        }

        public void Clear(long playerId) => _history.Remove(playerId);
    }

    二、道具流水：每一笔都要记 ★★★

    为什么流水是必须的 三个真实场景：
    ① 玩家投诉"我的装备没了" → 查流水，看到某次分解操作
    ② 发现刷道具漏洞 → 查流水，找到异常来源
    ③ 对账 → 流水总和 == 当前库存？

    本页实测的对账：
    流水: +10(任务) +5(购买) -3(合成) = +12（另有 1 笔 -100 因不足被拒，不入账）
    对账: 流水合计=12, 实际库存=12, 平衡=True

    流水必须包含：
    · 玩家ID、道具ID、变化量
    · 原因（任务/购买/合成）
    · 关联ID（订单号、任务ID）
    · 时间戳

    没有流水的系统，出了问题只能瞎猜。

    三、防重复领取：业务唯一键 ★★★

    本页的实现与实测 核心：用「玩家ID + 奖励标识」做唯一键。
    string key = playerId + ":" + rewardKey;
    if (_claimed.Contains(key)) return (true, "已经领取过了");

    本页实测：
    第1次: 领取成功，金币=500
    第2次: 已经领取过了，金币=500
    第3次: 已经领取过了，金币=500
    实际领取次数: 1

    关键细节：
    · 重复请求返回成功（不是失败）
    · 奖励标识要包含时间维度（daily_login_2026_10_21）
    · 多实例部署要用数据库唯一索引

    四、异常检测：发现刷子

    三条检测规则 本页实现并实测：
    【异常】玩家2001 在 60s 内获得 20000（阈值 10000）
    玩家2002 是否异常: False

    规则一：单位时间收益上限
    · 正常玩家 1 分钟能赚多少？
    · 超过 3 倍就标记

    规则二：操作频率
    · 人类手速有上限（约 10 次/秒）
    · 超过就可能是脚本

    规则三：收益与难度匹配
    · 新手段位打出满级装备 → 异常

    注意：检测到异常不要立刻封号——先标记，人工复核。
