    一、统一响应格式

    所有接口返回同一种结构——客户端才能用一套逻辑处理。

    using System;
using System.Collections.Generic;
using System.Threading.Tasks;

namespace BackendSamples.ApiDesign
{
    // ============================================================
    // 1. 统一响应格式
    // ============================================================
    // 【为什么需要】客户端要能用一套逻辑处理所有接口
    //   成功：{ code: 0, data: {...}, msg: "" }
    //   失败：{ code: 1001, data: null, msg: "余额不足" }

    public class ApiResponse<T>
    {
        public int Code { get; set; }
        public string Msg { get; set; } = "";
        public T Data { get; set; }

        public static ApiResponse<T> Ok(T data) =>
            new ApiResponse<T> { Code = 0, Msg = "", Data = data };

        public static ApiResponse<T> Fail(int code, string msg) =>
            new ApiResponse<T> { Code = code, Msg = msg, Data = default };
    }

    // ============================================================
    // 2. 错误码设计
    // ============================================================
    // 【关键】错误码要分段，一眼能看出问题类别
    //   0        成功
    //   1xxx     参数/客户端错误（重试无用）
    //   2xxx     权限/认证错误
    //   3xxx     业务错误（余额不足、道具不足）
    //   4xxx     系统错误（重试可能有用）
    //   5xxx     第三方服务错误

    public static class ErrorCode
    {
        public const int Success = 0;

        // 1xxx 参数错误
        public const int InvalidParam = 1001;
        public const int MissingParam = 1002;

        // 2xxx 权限
        public const int Unauthorized = 2001;
        public const int TokenExpired = 2002;
        public const int Forbidden = 2003;

        // 3xxx 业务
        public const int InsufficientGold = 3001;
        public const int ItemNotFound = 3002;
        public const int InventoryFull = 3003;
        public const int AlreadyClaimed = 3004;

        // 4xxx 系统
        public const int InternalError = 4001;
        public const int ServiceBusy = 4002;
    }

    // ============================================================
    // 3. HTTP 状态码 vs 业务错误码
    // ============================================================
    // 【重要】两者是不同层的东西，都要用
    //
    //   HTTP 状态码：传输层语义（200/400/401/404/500）
    //     给「网关、负载均衡、监控」看
    //     401 会自动触发某些客户端的重新登录
    //
    //   业务错误码：业务层语义（3001 金币不足）
    //     给「客户端业务逻辑」看
    //
    // 【常见错误】全部返回 200，用 code 区分
    //   → 问题：监控看不到错误率、网关无法做熔断
    //
    // 【推荐做法】
    //   客户端参数错误 → 400 + code 1001
    //   未登录         → 401 + code 2001
    //   余额不足       → 200 + code 3001（业务失败不是协议错误）
    //   服务器异常     → 500 + code 4001

    public static class HttpStatusGuide
    {
        public static string Explain(int statusCode)
        {
            return statusCode switch
            {
                200 => "成功",
                201 => "已创建",
                204 => "成功但无内容",
                400 => "请求参数错误",
                401 => "未认证",
                403 => "已认证但无权限",
                404 => "资源不存在",
                409 => "冲突（重复提交）",
                422 => "语义错误（能解析但不合法）",
                429 => "请求过于频繁",
                500 => "服务器内部错误",
                502 => "上游服务错误",
                503 => "服务不可用",
                _ => "其他"
            };
        }
    }

    // ============================================================
    // 4. 幂等：后端最重要的一条设计
    // ============================================================
    // 【问题】客户端网络抖动，同一个「领奖」请求发了 3 次
    //   → 如果不做幂等，玩家领了 3 份奖励
    //
    // 【为什么客户端改不了】
    //   客户端重试是「必要行为」——不能要求玩家网络永不抖动

    public class IdempotencyService
    {
        // 【方案一】唯一键去重
        //   客户端每次请求带一个 requestId
        //   服务端记录已处理的 requestId
        private readonly Dictionary<string, string> _processed = new();

        public string Handle(string requestId, Func<string> action)
        {
            // 已经处理过，直接返回上次的结果
            if (_processed.TryGetValue(requestId, out var cached))
                return cached;

            var result = action();
            _processed[requestId] = result;
            return result;
        }

        // 【方案二】数据库唯一索引
        //   CREATE UNIQUE INDEX uk_order ON orders(user_id, order_id)
        //   重复插入会抛异常，捕获后当作成功
        //
        // 【方案三】状态机 + 版本号
        //   UPDATE orders SET status = 2 WHERE id = 1 AND status = 1
        //   影响行数为 0 说明已经被处理过

        public int ProcessedCount => _processed.Count;
    }

    // ============================================================
    // 5. 实战：一个完整的领奖接口
    // ============================================================

    public class RewardService
    {
        private readonly HashSet<string> _claimed = new();
        private int _gold;

        // 领取任务奖励
        // 返回: (成功?, 错误码, 提示, 当前金币)
        public (bool ok, int code, string msg, int gold) ClaimReward(
            long userId, long questId, string requestId)
        {
            // ① 参数校验
            if (userId <= 0) return (false, ErrorCode.InvalidParam, "用户ID非法", _gold);
            if (questId <= 0) return (false, ErrorCode.InvalidParam, "任务ID非法", _gold);

            // ② 幂等检查（关键！）
            //   同一个 requestId 重复请求，直接返回当前状态
            string key = userId + ":" + questId;
            if (_claimed.Contains(key))
            {
                // 【注意】这里返回 ok=true，因为「已经领过」对客户端来说是成功
                //   如果返回失败，客户端会以为自己没领到，一直重试
                return (true, ErrorCode.AlreadyClaimed, "已经领取过了", _gold);
            }

            // ③ 业务逻辑
            _gold += 100;
            _claimed.Add(key);

            // ④ 返回结果
            return (true, ErrorCode.Success, "领取成功", _gold);
        }

        public int Gold => _gold;
    }

    // ============================================================
    // 6. 演示
    // ============================================================
    public static class Demo
    {
        public static void Run()
        {
            Console.WriteLine("=== 3. HTTP 状态码说明 ===");
            foreach (int code in new[] { 200, 400, 401, 403, 409, 429, 500 })
            {
                Console.WriteLine($"  {code}: {HttpStatusGuide.Explain(code)}");
            }

            Console.WriteLine();
            Console.WriteLine("=== 4. 幂等演示 ===");
            var svc = new RewardService();

            // 模拟客户端重试 3 次
            Console.WriteLine("  客户端连续发送 3 次领奖请求（同一 requestId）:");
            for (int i = 1; i <= 3; i++)
            {
                var (ok, code, msg, gold) = svc.ClaimReward(1001, 2001, "req-abc");
                Console.WriteLine($"    第 {i} 次: ok={ok}, code={code}, msg={msg}, 金币={gold}");
            }

            Console.WriteLine();
            Console.WriteLine("  【关键】金币只涨了 100，不是 300");
            Console.WriteLine("          这就是幂等的价值");

            Console.WriteLine();
            Console.WriteLine("=== 5. 统一响应演示 ===");
            var okResp = ApiResponse<string>.Ok("数据内容");
            Console.WriteLine($"  成功: code={okResp.Code}, data={okResp.Data}");

            var failResp = ApiResponse<string>.Fail(ErrorCode.InsufficientGold, "金币不足");
            Console.WriteLine($"  失败: code={failResp.Code}, msg={failResp.Msg}");
        }
    }
}

public class Program
{
    public static void Main() => BackendSamples.ApiDesign.Demo.Run();
}
    字段作用示例
    code业务错误码（0 = 成功）0 / 3001
    msg给用户看的提示"金币不足"
    data业务数据{...} 或 null
    可选 requestId链路追踪出问题时定位日志

    二、错误码分段设计 ★★

    错误码不带分段 = 后期灾难 反例：错误码从 1 开始随意追加
    · 1 = 参数错误、2 = 未登录、3 = 金币不足、4 = 系统错误……
    · 加到 200 个之后，没人记得 137 是什么

    正确做法：分段
    段含义客户端行为
    0成功—
    1xxx参数/客户端错误不要重试
    2xxx权限/认证跳登录
    3xxx业务错误提示用户
    4xxx系统错误可以重试
    5xxx第三方服务稍后重试
    好处：客户端可以按段位统一处理——code / 1000 === 2 就跳登录，不需要枚举所有错误码。

    三、幂等：后端最重要的一条设计 ★★★

    先看一个真实的事故场景 场景：玩家点「领取任务奖励」
    · 网络抖动，请求超时
    · 客户端自动重试（这是正常行为）
    · 实际上第一次请求已经成功了，只是响应丢包
    · 玩家领了 2 份奖励

    更糟的情况：
    · 弱网下客户端重试 5 次 → 领 5 份
    · 有人用工具刷接口 → 无限刷

    本页实测输出验证了正确做法：
    第 1 次: ok=True, code=0, msg=领取成功, 金币=100
    第 2 次: ok=True, code=3004, msg=已经领取过了, 金币=100
    第 3 次: ok=True, code=3004, msg=已经领取过了, 金币=100
    金币只涨了 100，不是 300。
    三种幂等实现方案 方案一：唯一键去重（本页代码）
    客户端每次请求带 requestId，服务端记录已处理的 ID
    简单，但要考虑 requestId 存储的过期清理

    方案二：数据库唯一索引（最可靠）
    CREATE UNIQUE INDEX uk_claim ON claims(user_id, quest_id)
    重复插入直接抛异常，捕获后当作「已领过」
    优点：数据库保证，多实例部署也不会重复

    方案三：状态机 + 版本号
    UPDATE orders SET status = 2 WHERE id = 1 AND status = 1
    影响行数为 0 == 已经被处理过
    优点：不需要额外存储，天然支持并发
    一个容易做错的细节 「已经领过」应该返回成功还是失败？

    本页代码返回了 ok=True——这是对的。

    为什么：
    · 如果返回失败，客户端会以为「没领到」→ 继续重试
    · 重试又失败 → 死循环
    · 玩家看到「领取失败」但实际上已领 → 客服投诉

    正确语义：「这个操作的目标状态已经达成了」= 成功。
    用一个单独的 code 告诉客户端「是重复请求」，但 ok 是 true。

    四、HTTP 状态码怎么用

    场景HTTP 状态码业务码
    正常成功2000
    参数格式错误4001001
    未登录/Token 失效4012001
    已登录但无权限4032003
    金币不足2003001
    服务器异常5004001
    请求过频4294002
    业务失败用 200 还是 4xx 这是一个有争议的话题，两种做法都有实践：

    做法 A：全部 200，用 code 区分（本页推荐）
    · 优点：客户端只处理一种响应形态
    · 缺点：监控看不到错误率，网关无法基于状态码熔断

    做法 B：区分传输错误与业务错误
    · 4xx 只用于协议层错误（参数、认证）
    · 业务失败（金币不足）用 200 + code
    · 推荐这个——因为「金币不足」不是接口出了问题，是正常业务结果

    关键不是选哪个，而是团队统一。

    本章动手任务

    本章过关标准：
    ① 能设计出统一的响应格式；
    ② 能按分段设计错误码，并说明每段客户端该怎么处理；
    ③ 能解释幂等是什么、为什么必需；
    ④ 能实现至少一种幂等方案，并实测验证；
    ⑤ 知道「已处理」应返回成功而非失败。
