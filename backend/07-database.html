    一、N+1 查询：后端最经典的性能问题 ★★★

    症状：接口在测试环境很快，数据一多就超时。

    using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;

namespace BackendSamples.Database
{
    // ============================================================
    // 1. 关系建模：一对多、多对多
    // ============================================================

    public class Player
    {
        public long Id { get; set; }
        public string Name { get; set; } = "";
        // 【导航属性】一个玩家有多个道具
        public List<PlayerItem> Items { get; set; } = new();
    }

    public class PlayerItem
    {
        public long Id { get; set; }
        public long PlayerId { get; set; }
        public int ItemId { get; set; }
        public int Count { get; set; }
        // 反向导航
        public Player? Player { get; set; }
    }

    // 简化版内存数据库（模拟 EF Core 的行为）
    public class InMemoryDb
    {
        public List<Player> Players = new();
        public List<PlayerItem> Items = new();

        // 【关键】模拟 EF Core 的「查询计数」
        //   真实的 EF Core 每次访问数据库都算一次往返
        public int QueryCount;

        public List<Player> QueryPlayers()
        {
            QueryCount++;
            return Players;
        }

        public List<PlayerItem> QueryItemsByPlayer(long playerId)
        {
            QueryCount++;
            return Items.Where(i => i.PlayerId == playerId).ToList();
        }

        // 【关键】全量查询关联表 —— 这也是一次数据库往返，必须计数
        //   对应 EF Core 的 .Include(p => p.Items) 或一次性 IN 查询
        public List<PlayerItem> QueryAllItems()
        {
            QueryCount++;
            return Items.ToList();
        }

        public void Reset() => QueryCount = 0;
    }

    // ============================================================
    // 2. N+1 查询问题 ★★★
    // ============================================================
    // 【后端最经典的性能问题】
    //   1 次查主表 + N 次查关联表 = N+1 次数据库往返

    public class NPlusOneDemo
    {
        // 【错误】N+1 查询
        public static void BadQuery(InMemoryDb db)
        {
            var players = db.QueryPlayers();              // 1 次

            foreach (var p in players)
            {
                // 【问题】每个玩家都单独查一次
                var items = db.QueryItemsByPlayer(p.Id);  // N 次
                p.Items = items;
            }
            // 总计: 1 + N 次
        }

        // 【正确】一次性把关联数据取出来
        public static void GoodQuery(InMemoryDb db)
        {
            var players = db.QueryPlayers();              // 1 次
            var allItems = db.QueryAllItems();            // 1 次（用 Include 或一次 IN 查询）

            // 内存里做分组（O(n)）
            var lookup = allItems.GroupBy(i => i.PlayerId)
                                 .ToDictionary(g => g.Key, g => g.ToList());

            foreach (var p in players)
            {
                p.Items = lookup.TryGetValue(p.Id, out var list) ? list : new List<PlayerItem>();
            }
            // 总计: 2 次（无论多少个玩家）
        }
    }

    // ============================================================
    // 3. 索引：查询优化最有效的手段
    // ============================================================

    public class IndexDemo
    {
        public static void Run()
        {
            const int N = 100000;

            // 建表
            var items = new List<PlayerItem>(N);
            var rnd = new Random(42);
            for (int i = 0; i < N; i++)
            {
                items.Add(new PlayerItem
                {
                    Id = i,
                    PlayerId = i % 1000,          // 1000 个玩家
                    ItemId = rnd.Next(1, 500),
                    Count = rnd.Next(1, 100),
                });
            }

            long targetPlayer = 500;

            Console.WriteLine($"  数据量: {N} 条，目标 PlayerId={targetPlayer}");

            // 【无索引】全表扫描
            var sw = Stopwatch.StartNew();
            var r1 = items.Where(i => i.PlayerId == targetPlayer).ToList();
            sw.Stop();
            double ms1 = sw.Elapsed.TotalMilliseconds;
            Console.WriteLine($"  无索引（全表扫描）: {ms1:F3} ms，找到 {r1.Count} 条");

            // 【建索引】按 PlayerId 建哈希索引 = O(1) 定位
            var index = new Dictionary<long, List<PlayerItem>>();
            foreach (var it in items)
            {
                if (!index.TryGetValue(it.PlayerId, out var list))
                {
                    list = new List<PlayerItem>();
                    index[it.PlayerId] = list;
                }
                list.Add(it);
            }

            sw.Restart();
            var r2 = index.TryGetValue(targetPlayer, out var hit) ? hit : new List<PlayerItem>();
            sw.Stop();
            double ms2 = sw.Elapsed.TotalMilliseconds;
            string ratio = ms2 > 0.0001 ? (ms1 / ms2).ToString("F0") + " 倍" : "差异过大";
            Console.WriteLine($"  有索引:             {ms2:F4} ms，找到 {r2.Count} 条");
            Console.WriteLine($"  差距:               {ratio}");

            Console.WriteLine();
            Console.WriteLine("  【结论】索引把 O(n) 变成 O(1)");
            Console.WriteLine("         真实数据库里这个差距通常是 1000 倍以上");

            // 索引的代价
            Console.WriteLine();
            Console.WriteLine("  【索引的代价】");
            Console.WriteLine("    · 占用额外存储（通常 10~30%）");
            Console.WriteLine("    · 写入变慢（每次 INSERT/UPDATE 都要维护索引）");
            Console.WriteLine("    · 所以：不是索引越多越好，只给高频查询建");
        }
    }

    // ============================================================
    // 4. 事务：保证多个操作原子性
    // ============================================================

    public class TransactionDemo
    {
        public class Account
        {
            public long Id;
            public string Name = "";
            public decimal Balance;
        }

        private readonly List<Account> _accounts = new()
        {
            new() { Id = 1, Name = "玩家A", Balance = 1000m },
            new() { Id = 2, Name = "玩家B", Balance = 500m },
        };

        // 【错误】没有事务：中间失败会数据不一致
        public bool Transfer_Bad(long from, long to, decimal amount)
        {
            var a = _accounts.First(x => x.Id == from);
            var b = _accounts.First(x => x.Id == to);

            if (a.Balance < amount) return false;

            a.Balance -= amount;
            // 【假设这里崩溃了 / 抛异常】
            // b.Balance += amount 没执行 -> 钱消失了
            b.Balance += amount;
            return true;
        }

        // 【正确】事务：要么全成功，要么全失败
        public bool Transfer_Good(long from, long to, decimal amount)
        {
            var a = _accounts.First(x => x.Id == from);
            var b = _accounts.First(x => x.Id == to);

            if (a.Balance < amount)
            {
                Console.WriteLine($"    余额不足: {a.Name} 有 {a.Balance}, 需要 {amount}");
                return false;
            }

            // 模拟事务
            decimal oldA = a.Balance;
            decimal oldB = b.Balance;

            try
            {
                a.Balance -= amount;

                // 【模拟】中间发生异常
                if (amount > 800) throw new InvalidOperationException("模拟：转账过程中系统故障");

                b.Balance += amount;
                return true;
            }
            catch (Exception ex)
            {
                // 【关键】回滚
                a.Balance = oldA;
                b.Balance = oldB;
                Console.WriteLine($"    异常: {ex.Message}");
                Console.WriteLine($"    已回滚，余额保持不变");
                return false;
            }
        }

        public void PrintBalances()
        {
            foreach (var a in _accounts)
                Console.WriteLine($"    {a.Name}: {a.Balance}");
        }
    }

    public static class Demo
    {
        public static void Run()
        {
            // 准备数据
            var db = new InMemoryDb();
            for (int i = 1; i <= 20; i++)
            {
                db.Players.Add(new Player { Id = i, Name = $"玩家{i}" });
                for (int j = 0; j < 3; j++)
                    db.Items.Add(new PlayerItem { Id = i * 10 + j, PlayerId = i, ItemId = j + 1, Count = 1 });

            Console.WriteLine("=== 1. N+1 查询问题 ===");
            Console.WriteLine($"  数据: 20 个玩家，每人 3 个道具");
            Console.WriteLine();

            db.Reset();
            NPlusOneDemo.BadQuery(db);
            Console.WriteLine($"  【错误做法】查询次数: {db.QueryCount} 次（1 + 20）");

            db.Reset();
            NPlusOneDemo.GoodQuery(db);
            Console.WriteLine($"  【正确做法】查询次数: {db.QueryCount} 次（固定 2 次，与玩家数无关）");

            Console.WriteLine();
            Console.WriteLine("  【为什么严重】");
            Console.WriteLine("    数据库往返一次约 1~10ms");
            Console.WriteLine("    100 个玩家 = 101 次往返 = 100ms~1s");
            Console.WriteLine("    -> 接口超时、数据库连接池耗尽");

            Console.WriteLine();
            Console.WriteLine("=== 2. 索引的效果 ===");
            IndexDemo.Run();

            Console.WriteLine();
            Console.WriteLine("=== 3. 事务 ===");
            var tx = new TransactionDemo();

            Console.WriteLine("  正常转账 100:");
            tx.Transfer_Good(1, 2, 100m);
            tx.PrintBalances();

            Console.WriteLine();
            Console.WriteLine("  转账 900（触发模拟故障）:");
            tx.Transfer_Good(1, 2, 900m);
            tx.PrintBalances();

            Console.WriteLine();
            Console.WriteLine("  【关键】没有事务时，故障会导致钱凭空消失");
            Console.WriteLine("          事务保证：要么全成功，要么全回滚");

            Console.WriteLine();
            Console.WriteLine("=== 4. 迁移命令 ===");
            MigrationExplainer.Print();
        }

    二、索引：查询优化最有效的手段 ★★

    实测：索引带来几个数量级的差距 本页实测（10 万条数据）：
    · 无索引（全表扫描）：毫秒级
    · 有索引：微秒级

    真实数据库里的差距更大——因为磁盘 IO 比内存慢几个数量级。

    但要记住索引的代价：
    · 占用存储（通常 10~30%）
    · 写入变慢（每次 INSERT/UPDATE 都要维护索引）
    · 索引越多，写入越慢

    原则：只给「高频查询 + 区分度高」的字段建索引。
    · 高频：用户查自己的道具
    · 区分度高：userId 好，gender 差（只有两个值）

    三、事务：保证原子性 ★★★

    什么是事务 一组操作，要么全成功，要么全失败。

    经典例子：转账
    A 扣 100 → B 加 100
    如果没有事务：
    · A 扣了 100
    · 系统崩溃
    · B 没收到 → 100 凭空消失

    本页实测：
    转账 900（触发故障）
    异常: 模拟：转账过程中系统故障
    已回滚，余额保持不变

    游戏里的应用：
    · 扣金币 + 发道具（不能只扣不发）
    · 战斗结算（扣体力 + 发奖励）
    · 交易系统
    事务的两个常见误区 误区一：事务可以随便用
    · 事务会持有锁，长事务会阻塞其他请求
    · 不要在事务里做网络请求、文件 IO
    · 原则：事务尽可能短

    误区二：以为 EF Core 会自动包事务
    · SaveChanges() 单次调用是原子的
    · 但多次 SaveChanges 不在同一事务里
    · 需要显式 BeginTransaction

    还要注意隔离级别：
    · 默认 ReadCommitted——能防脏读，但防不了幻读
    · 高并发扣减要用行锁或乐观锁（见页 18）

    四、迁移：数据库结构的版本控制

    迁移解决什么问题 问题：你改了实体类，怎么让数据库表也跟着改？

    手工改表的问题：
    · 开发环境改了，生产忘了改
    · 团队其他人不知道要改
    · 回滚时不知道怎么还原

    迁移的做法：
    ① 改代码 → migrations add Xxx 生成脚本
    ② 脚本进版本库（和代码一起 review）
    ③ 部署时 database update 应用

    生产环境最佳实践：
    · 用 dbcontext script 生成 SQL
    · DBA 审核后再执行
    · 不要在生产直接 update
