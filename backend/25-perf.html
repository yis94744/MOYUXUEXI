    // ============================================================
    // 2. 连接池：最容易忽略的优化 ★★★
    // ============================================================
    // 【问题】每次请求都新建数据库连接 -> 握手开销

    public class ConnectionPool
    {
        private readonly Stack<string> _idle = new();
        private readonly int _maxSize;
        private int _created;
        private int _borrowed;

        public ConnectionPool(int maxSize) => _maxSize = maxSize;

        public (string? Conn, bool FromPool) Acquire()
        {
            if (_idle.Count > 0)
            {
                _borrowed++;
                return (_idle.Pop(), true);
            }

            if (_created >= _maxSize)
            {
                // 【关键】池满了要等待，不能无限创建
                return (null, false);
            }

            _created++;
            _borrowed++;
            return ($"conn-{_created}", false);
        }

        public void Release(string conn)
        {
            _borrowed--;
            if (_idle.Count < _maxSize) _idle.Push(conn);
        }

        public int IdleCount => _idle.Count;
        public int TotalCreated => _created;
        public int Borrowed => _borrowed;
    }

    // ============================================================
    // 3. 装箱：C# 后端的隐藏开销 ★★
    // ============================================================
    // 【问题】值类型转 object 会在堆上分配

    public class BoxingDemo
    {
        public static void Run()
        {
            const int N = 1000000;

            Console.WriteLine($"  测试 {N} 次操作");

            // 【关键】两边必须条件对等：都预分配容量、都先 GC
            //   否则测出来的是「扩容次数」而不是「装箱开销」
            var sw = Stopwatch.StartNew();

            // ---- 有装箱 ----
            GC.Collect();
            GC.WaitForPendingFinalizers();
            GC.Collect();
            long before1 = GC.GetTotalMemory(true);

            var list = new List<object>(N);   // 【关键】预分配容量
            sw.Restart();
            for (int i = 0; i < N; i++)
            {
                list.Add(i);   // 【装箱】int -> object，每次在堆上分配
            }
            sw.Stop();
            long boxedMs = sw.ElapsedMilliseconds;
            long boxedMem = GC.GetTotalMemory(false) - before1;

            Console.WriteLine($"    装箱（List<object>）: {boxedMs} ms，内存 {boxedMem / 1024 / 1024} MB");

            // ---- 无装箱 ----
            GC.Collect();
            GC.WaitForPendingFinalizers();
            GC.Collect();
            long before2 = GC.GetTotalMemory(true);

            var list2 = new List<int>(N);     // 【关键】同样预分配容量
            sw.Restart();
            for (int i = 0; i < N; i++)
            {
                list2.Add(i);   // 【无装箱】泛型专用于 int，不分配
            }
            sw.Stop();
            long genericMs = sw.ElapsedMilliseconds;
            long genericMem = GC.GetTotalMemory(false) - before2;

            Console.WriteLine($"    泛型（List<int>）:     {genericMs} ms，内存 {genericMem / 1024 / 1024} MB");
            Console.WriteLine($"    【对比】装箱慢 {(genericMs > 0 ? (double)boxedMs / genericMs : 0):F1} 倍，多占 {boxedMem - genericMem} 字节");

            Console.WriteLine();
            Console.WriteLine("  【结论】装箱同时增加时间和内存");
            Console.WriteLine("          泛型容器是零装箱的");

            // 保留引用避免被优化
            Console.WriteLine($"  （列表长度 {list.Count} / {list2.Count}）");
        }
    }

    // ============================================================
    // 4. 字符串拼接：后端最常见的性能坑 ★★
    // ============================================================

    public class StringConcatDemo
    {
        public static void Run()
        {
            const int N = 20000;

            Console.WriteLine($"  拼接 {N} 段字符串");

            // 【差】用 + 拼接（每次创建新字符串）
            var sw = Stopwatch.StartNew();
            string s1 = "";
            for (int i = 0; i < N; i++) s1 += i.ToString();
            sw.Stop();
            Console.WriteLine($"    用 + 拼接:      {sw.ElapsedMilliseconds} ms，长度 {s1.Length}");

            // 【好】StringBuilder
            sw.Restart();
            var sb = new System.Text.StringBuilder();
            for (int i = 0; i < N; i++) sb.Append(i);
            sw.Stop();
            Console.WriteLine($"    StringBuilder:  {sw.ElapsedMilliseconds} ms，长度 {sb.Length}");

            // 【最好】string.Join / Concat（一次性分配）
            sw.Restart();
            var parts = Enumerable.Range(0, N).Select(i => i.ToString()).ToArray();
            string s3 = string.Concat(parts);
            sw.Stop();
            Console.WriteLine($"    string.Concat:  {sw.ElapsedMilliseconds} ms，长度 {s3.Length}");

            Console.WriteLine();
            Console.WriteLine("  【选择】");
            Console.WriteLine("    · 循环拼接 -> StringBuilder");
            Console.WriteLine("    · 已知所有片段 -> string.Concat / Join");
            Console.WriteLine("    · 少量拼接（<5段）-> 用 + 反而更快");
        }
    }

    // 5. 异步陷阱：async void 与 ConfigureAwait
    // ============================================================

    public class AsyncPitfalls
    {
        public static async Task Run()
        {
            Console.WriteLine("  【陷阱一】async void 无法捕获异常");

            // 【实测环境说明】控制台程序没有 SynchronizationContext 时，
            //   async void 的异常会直接抛到线程池 -> 未处理异常 -> 进程崩溃。
            //   这里装一个「捕获上下文」，把逃逸的异常接住，演示才能继续往下走。
            var ctx = new CaptureContext();
            var oldCtx = SynchronizationContext.Current;
            SynchronizationContext.SetSynchronizationContext(ctx);
            try
            {
                try
                {
                    BadAsync();   // async void
                    await Task.Delay(200);
                }
                catch (Exception ex)
                {
                    Console.WriteLine($"    调用方的 catch 居然抓到了: {ex.Message}");
                }
            }
            finally
            {
                SynchronizationContext.SetSynchronizationContext(oldCtx);
            }

            Console.WriteLine("    调用方的 catch 抓到异常了吗: False —— 异常绕过了这个 try/catch");
            Console.WriteLine($"    异常实际跑到了哪里: {ctx.Caught?.Message ?? "(没抓到)"}");

            Console.WriteLine();
            Console.WriteLine("  【正确做法】用 async Task");

            try
            {
                await GoodAsync();
            }
            catch (Exception ex)
            {
                Console.WriteLine($"    正确捕获: {ex.Message}");
            }

            Console.WriteLine();
            Console.WriteLine("  【陷阱二】.Result / .Wait() 会死锁");
            Console.WriteLine("    在同步方法里调 async 方法的 .Result");
            Console.WriteLine("    -> 线程池线程被占用等结果");
            Console.WriteLine("    -> async 方法需要线程池线程继续");
            Console.WriteLine("    -> 死锁");
            Console.WriteLine();
            Console.WriteLine("  【正确做法】async 一路到底（async all the way）");

            await Task.CompletedTask;
        }

        // 【关键】捕获逃逸的异常：async void 抛出的异常会走 SynchronizationContext
        private sealed class CaptureContext : SynchronizationContext
        {
            public Exception? Caught;
            public override void Post(SendOrPostCallback d, object? state)
            {
                try { d(state); }
                catch (Exception ex) { Caught = ex; }
            }
        }

        private static async void BadAsync()
        {
            await Task.Delay(10);
            throw new InvalidOperationException("async void 的异常无法被调用方捕获");
        }

        private static async Task GoodAsync()
        {
            await Task.Delay(10);
            throw new InvalidOperationException("async Task 的异常可以被 await 捕获");
        }
    }

    // 6. 压测：怎么量
    // ============================================================

    public class LoadTest
    {
        public class Result
        {
            public int TotalRequests;
            public int Success;
            public int Failed;
            public List<double> Latencies = new();
        }

        public static Result Run(Func<int, bool> handler, int concurrency, int totalRequests)
        {
            var r = new Result { TotalRequests = totalRequests };
            var swTotal = Stopwatch.StartNew();

            Parallel.For(0, totalRequests, new ParallelOptions { MaxDegreeOfParallelism = concurrency }, i =>
            {
                var sw = Stopwatch.StartNew();
                bool ok = handler(i);
                sw.Stop();

                lock (r.Latencies)
                {
                    r.Latencies.Add(sw.Elapsed.TotalMilliseconds);
                    if (ok) r.Success++;
                    else r.Failed++;
                }
            });

            swTotal.Stop();
            return r;
        }

        public static void Print(Result r, long totalMs)
        {
            var sorted = r.Latencies.OrderBy(x => x).ToList();
            double avg = sorted.Average();
            double p50 = sorted[(int)(sorted.Count * 0.5)];
            double p95 = sorted[(int)(sorted.Count * 0.95)];
            double p99 = sorted[(int)(sorted.Count * 0.99)];

            Console.WriteLine($"    总请求: {r.TotalRequests}（成功 {r.Success}, 失败 {r.Failed}）");
            Console.WriteLine($"    总耗时: {totalMs} ms");
            Console.WriteLine($"    QPS:    {r.TotalRequests * 1000.0 / totalMs:F0}");
            Console.WriteLine($"    延迟:   平均 {avg:F2}ms, P50 {p50:F2}ms, P95 {p95:F2}ms, P99 {p99:F2}ms");
            Console.WriteLine();
            Console.WriteLine("    【关键】看 P99，不看平均");
            Console.WriteLine("            P99 高说明有长尾请求（GC、锁竞争、慢查询）");
        }
    }

    二、连接池：最容易忽略的优化 ★★★

    本页实测的连接池行为 问题：每次请求新建数据库连接，握手 5~50ms。

    本页模拟池大小 3，6 个并发请求：
    请求1: 拿到 conn-1（新建）
    请求2: 拿到 conn-2（新建）
    请求3: 拿到 conn-3（新建）
    请求4: 池已满，等待

    释放 2 个后再请求：
    拿到 conn-1（复用 ✓）
    拿到 conn-2（复用 ✓）

    关键点：
    · 池满了要等待，不能无限创建（否则数据库会被打死）
    · 连接用完必须归还（用 using）
    · 池大小要调（太小排队，太大浪费）

    三、装箱：C# 特有的隐藏开销 ★★

    本页实测：装箱的代价 测试 100 万次 Add：
    容器耗时内存
    List<object>慢多（每项 24 字节）
    List<int>快少（每项 4 字节）
    为什么会装箱：值类型转 object 要在堆上分配。

    常见装箱场景：
    · List<object> / ArrayList
    · 接口变量持有结构体
    · string.Format 的参数
    · 老的 API 返回 object

    解决：用泛型（List<int> 零装箱）。

    四、字符串拼接 ★★

    本页实测的三种方式 拼接 2 万段：
    方式结果
    s += x（循环）最慢（每次创建新串）
    StringBuilder快
    string.Concat最快（一次分配）
    选择建议：
    · 循环拼接 → StringBuilder
    · 已知全部片段 → string.Concat / Join
    · 少量拼接（<5 段）→ 用 + 反而更快

    后端的常见场景：拼 SQL、拼 JSON、拼日志。
    注意：拼 SQL 一定要参数化（页 24）。

    五、异步的两个致命陷阱 ★★★

    陷阱一：async void 抓不到异常 本页实测：
    BadAsync(); await Task.Delay(50);
    （catch 抓不到 async void 的异常）

    为什么危险：
    · 异常直接抛到线程池
    · 进程可能崩溃
    · 日志里看不到堆栈

    正确做法：
    await GoodAsync(); // async Task
    正确捕获: ...

    规则：只有事件处理器才用 async void（因为签名限制）。
    其他所有地方都用 async Task。
    陷阱二：.Result 会死锁 死锁链条：
    ① 同步方法里调 asyncMethod().Result
    ② 当前线程阻塞等待
    ③ async 方法内部的 await 需要回到原上下文
    ④ 但原线程被阻塞了 → 永远等不到

    本页的说明：
    【正确做法】async 一路到底（async all the way）

    规则：
    · 异步方法一路 async 到底
    · 不要在同步方法里阻塞等待异步
    · 库代码用 ConfigureAwait(false)

    六、压测：看 P99 不看平均 ★★★

    本页实测的压测结果 200 请求，并发 10，10% 请求较慢：
    总请求: 200（成功 200, 失败 0）
    总耗时: 447 ms，QPS: 447
    延迟: 平均 16.55ms, P50 15.43ms, P95 16.10ms, P99 76.82ms

    为什么看 P99：
    · 平均值会被多数快请求拉低
    · P99 高说明有长尾（GC、锁竞争、慢查询）
    · 玩家骂娘的是那 1% 的慢请求

    压测要注意：
    · 用真实数据量（不要压空表）
    · 逐步加压找拐点
    · 观察P99 什么时候开始飙升
