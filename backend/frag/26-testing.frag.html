using System;
using System.Collections.Generic;
using System.Linq;

namespace BackendSamples.Testing
{
    // ============================================================
    // 1. 三层测试：单元 / 集成 / 端到端
    // ============================================================

    public static class TestLayers
    {
        public static readonly (string Layer, string Scope, string Speed, string Stability, string Ratio)[] All =
        {
            ("单元测试",   "一个函数/类",       "毫秒",   "稳定",  "70%"),
            ("集成测试",   "多组件+真实依赖",   "秒",     "偶发失败", "20%"),
            ("端到端",     "完整链路",          "十秒+",  "易碎",  "10%"),
        };

        // 【测试金字塔】底层多、上层少
        public static string ExplainPyramid() =>
            "单元 70% + 集成 20% + 端到端 10%（倒金字塔是反模式）";
    }

    // ============================================================
    // 2. 一个可测试的类：依赖注入是关键
    // ============================================================

    public interface IClock
    {
        DateTime Now { get; }
    }

    public class SystemClock : IClock
    {
        public DateTime Now => DateTime.UtcNow;
    }

    // 【可测试】时间从外部注入
    public class FakeClock : IClock
    {
        public DateTime Now { get; set; } = new DateTime(2026, 10, 21, 12, 0, 0, DateTimeKind.Utc);
        public void Advance(TimeSpan t) => Now += t;
    }

    public interface IPlayerRepo
    {
        int? GetGold(long playerId);
        void SetGold(long playerId, int gold);
    }

    public class FakePlayerRepo : IPlayerRepo
    {
        private readonly Dictionary<long, int> _data = new();
        public void Seed(long id, int gold) => _data[id] = gold;
        public int? GetGold(long id) => _data.TryGetValue(id, out int g) ? g : null;
        public void SetGold(long id, int g) => _data[id] = g;
    }

    // 【核心】业务类只依赖接口
    public class DailyRewardService
    {
        private readonly IPlayerRepo _repo;
        private readonly IClock _clock;
        private readonly Dictionary<long, DateTime> _lastClaim = new();

        public DailyRewardService(IPlayerRepo repo, IClock clock)
        {
            _repo = repo;
            _clock = clock;
        }

        public (bool Ok, string Msg) ClaimDaily(long playerId, int amount)
        {
            var gold = _repo.GetGold(playerId);
            if (gold == null) return (false, "玩家不存在");

            // 【可测】用注入的 clock，而不是 DateTime.Now
            var today = _clock.Now.Date;

            if (_lastClaim.TryGetValue(playerId, out var last) && last.Date == today)
                return (false, "今天已经领过了");

            _repo.SetGold(playerId, gold.Value + amount);
            _lastClaim[playerId] = _clock.Now;

            return (true, $"领取成功，当前金币 {gold.Value + amount}");
        }
    }

    // ============================================================
    // 3. 一个极简的测试框架
    // ============================================================

    public class MiniTest
    {
        private int _passed;
        private int _failed;
        private readonly List<string> _failures = new();

        public void Assert(bool condition, string name)
        {
            if (condition) { _passed++; Console.WriteLine($"    PASS  {name}"); }
            else { _failed++; _failures.Add(name); Console.WriteLine($"    FAIL  {name}"); }
        }

        public void AssertEqual<T>(T expected, T actual, string name)
        {
            bool ok = EqualityComparer<T>.Default.Equals(expected, actual);
            Assert(ok, ok ? name : $"{name}（期望 {expected}，实际 {actual}）");
        }

        public void Report()
        {
            Console.WriteLine();
            Console.WriteLine($"  结果: 通过 {_passed}，失败 {_failed}");
            if (_failed > 0)
            {
                Console.WriteLine("  失败清单:");
                foreach (var f in _failures) Console.WriteLine($"    - {f}");
            }
        }
    }

    // ============================================================
    // 4. 编写测试：好的测试长什么样
    // ============================================================

    public static class TestExamples
    {
        public static void Run()
        {
            var t = new MiniTest();

            // ---- 正常路径 ----
            {
                var repo = new FakePlayerRepo();
                var clock = new FakeClock();
                repo.Seed(1001, 500);

                var svc = new DailyRewardService(repo, clock);
                var (ok, msg) = svc.ClaimDaily(1001, 100);

                t.Assert(ok, "正常领取：成功");
                t.AssertEqual(600, repo.GetGold(1001), "正常领取：金币加了 100");
            }

            // ---- 重复领取 ----
            {
                var repo = new FakePlayerRepo();
                var clock = new FakeClock();
                repo.Seed(1001, 500);

                var svc = new DailyRewardService(repo, clock);
                svc.ClaimDaily(1001, 100);
                var (ok, msg) = svc.ClaimDaily(1001, 100);

                t.Assert(!ok, "重复领取：被拒绝");
                t.AssertEqual(600, repo.GetGold(1001), "重复领取：金币没变");
            }

            // ---- 跨天可以再领 ----
            {
                var repo = new FakePlayerRepo();
                var clock = new FakeClock();
                repo.Seed(1001, 500);

                var svc = new DailyRewardService(repo, clock);
                svc.ClaimDaily(1001, 100);
                clock.Advance(TimeSpan.FromDays(1));
                var (ok, msg) = svc.ClaimDaily(1001, 100);

                t.Assert(ok, "跨天领取：成功");
                t.AssertEqual(700, repo.GetGold(1001), "跨天领取：金币累计 200");
            }

            // ---- 边界：玩家不存在 ----
            {
                var repo = new FakePlayerRepo();
                var clock = new FakeClock();
                var svc = new DailyRewardService(repo, clock);

                var (ok, msg) = svc.ClaimDaily(9999, 100);
                t.Assert(!ok, "玩家不存在：拒绝");
            }

            // ---- 边界：金额为 0 ----
            {
                var repo = new FakePlayerRepo();
                var clock = new FakeClock();
                repo.Seed(1001, 500);
                var svc = new DailyRewardService(repo, clock);

                var (ok, _) = svc.ClaimDaily(1001, 0);
                t.Assert(ok, "金额 0：允许（业务上可能不需要，但代码不崩）");
                t.AssertEqual(500, repo.GetGold(1001), "金额 0：金币没变");
            }

            t.Report();
        }
    }

二、可测试性来自设计 ★★★

三个让代码「测不了」的设计 ① 内部 new 依赖
var repo = new SqlRepo(); // 测试时要连真数据库
解决：依赖注入

② 直接用 DateTime.Now
var today = DateTime.Now.Date; // 测试结果不可重复
解决：注入 IClock

③ 静态方法/单例
Database.Instance.Query(...) // 换不掉
解决：改成实例 + 接口

本页实现的：
DailyRewardService(IPlayerRepo repo, IClock clock)
两个依赖都是接口，测试时换成 Fake。
假时钟的价值 本页实现的 FakeClock：
public DateTime Now { get; set; }
public void Advance(TimeSpan t) => Now += t;

能测什么：
· ClaimDaily(1001, 100) → 成功
· 立即再领 → 被拒绝
· clock.Advance(1天) → 再领成功

如果直接用 DateTime.Now，这个测试根本没法写。

四、后端测试的重点

测什么为什么重要
金额计算错了会赔钱
权限判断错了会越权
幂等逻辑错了会重复发奖
状态机转移错了会卡死
边界值负数、0、超大值
并发同时请求是否正确
五个测试好习惯
习惯为什么
一个测试只测一件事失败时能定位
测试名说清场景看名字知道测什么
覆盖边界条件bug 都在边界
测试之间不依赖可单独运行、可乱序
用假时钟不用 Now结果可重复
反例：Test1、Test2 这种命名 —— 三个月后你不知道它们在测什么。
