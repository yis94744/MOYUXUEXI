一、参数的五个来源

同一个参数名，放在不同位置意思完全不同——这是新手最容易混的地方。

using System;
using System.Collections.Generic;
using System.Linq;
using System.Text.RegularExpressions;

namespace BackendSamples.RoutingBinding
{
    // ============================================================
    // 1. 参数从哪来（模型绑定）
    // ============================================================
    // 【最容易搞混】同一个名字，放在不同位置意思完全不同
    //   /api/users/1001            -> 路由参数
    //   /api/users?id=1001         -> 查询参数
    //   POST body {"id":1001}      -> 请求体
    //   Header: X-UserId: 1001     -> 请求头

    public class BindSource
    {
        public string Name;
        public string Description;
        public string Example;
    }

    public static class BindSources
    {
        public static readonly BindSource[] All =
        {
            new() { Name = "[FromRoute]", Description = "URL 路径段", Example = "/users/{id}" },
            new() { Name = "[FromQuery]", Description = "查询字符串", Example = "?page=1&size=20" },
            new() { Name = "[FromBody]", Description = "请求体（JSON）", Example = "{\"name\":\"x\"}" },
            new() { Name = "[FromHeader]", Description = "请求头", Example = "X-Token: abc" },
            new() { Name = "[FromForm]", Description = "表单（文件上传）", Example = "multipart/form-data" },
        };
    }

    // ============================================================
    // 2. 路由约束：在路由层就拒绝非法请求
    // ============================================================
    // 【价值】非法请求不进业务代码，减少无谓的数据库查询

    public static class RouteConstraints
    {
        public static readonly (string Constraint, string Meaning, string Example)[] All =
        {
            ("int", "整数", "{id:int}"),
            ("long", "长整数", "{id:long}"),
            ("guid", "GUID", "{id:guid}"),
            ("alpha", "只含字母", "{name:alpha}"),
            ("min(1)", "最小值", "{page:min(1)}"),
            ("max(100)", "最大值", "{size:max(100)}"),
            ("range(1,100)", "范围", "{page:range(1,100)}"),
            ("length(2,20)", "长度", "{name:length(2,20)}"),
            ("regex(...)", "正则", "{code:regex(^[A-Z]{3}$)}"),
        };
    }

    // 简化版路由匹配（演示原理）
    public class SimpleRouter
    {
        private readonly List<(string Pattern, Func<Dictionary<string, string>, string> Handler)> _routes = new();

        public void MapGet(string pattern, Func<Dictionary<string, string>, string> handler)
            => _routes.Add((pattern, handler));

        public string? Dispatch(string path)
        {
            foreach (var (pattern, handler) in _routes)
            {
                var params_ = Match(pattern, path);
                if (params_ != null) return handler(params_);
            }
            return null;
        }

        // 极简版路径匹配：支持 {name} 和 {name:int}
        private static Dictionary<string, string>? Match(string pattern, string path)
        {
            var pParts = pattern.Trim('/').Split('/');
            var aParts = path.Trim('/').Split('/');
            if (pParts.Length != aParts.Length) return null;

            var result = new Dictionary<string, string>();

            for (int i = 0; i < pParts.Length; i++)
            {
                var p = pParts[i];
                var a = aParts[i];

                if (p.StartsWith("{") && p.EndsWith("}"))
                {
                    var inner = p.Substring(1, p.Length - 2);
                    var parts = inner.Split(':');
                    var name = parts[0];
                    var constraint = parts.Length > 1 ? parts[1] : null;

                    // 约束检查
                    if (constraint == "int" && !int.TryParse(a, out _)) return null;
                    if (constraint == "guid" && !Guid.TryParse(a, out _)) return null;
                    if (constraint == "alpha" && !Regex.IsMatch(a, "^[A-Za-z]+$")) return null;

                    result[name] = a;
                }
                else if (p != a)
                {
                    return null;   // 字面量不匹配
                }
            }

            return result;
        }
    }

    // ============================================================
    // 3. 参数验证：后端必须做的事
    // ============================================================
    // 【铁律】客户端传什么都要校验，不能信

    public class ValidationResult
    {
        public bool Ok;
        public List<string> Errors = new();

        public static ValidationResult Success() => new() { Ok = true };
        public static ValidationResult Fail(params string[] errs) =>
            new() { Ok = false, Errors = errs.ToList() };
    }

    public class Validator
    {
        private readonly List<string> _errors = new();

        public Validator Require(bool condition, string message)
        {
            if (!condition) _errors.Add(message);
            return this;
        }

        public Validator Range(long value, long min, long max, string field)
        {
            if (value < min || value > max)
                _errors.Add($"{field} 必须在 {min}~{max} 之间，实际 {value}");
            return this;
        }

        public Validator Length(string? s, int min, int max, string field)
        {
            int len = s?.Length ?? 0;
            if (len < min || len > max)
                _errors.Add($"{field} 长度必须在 {min}~{max} 之间，实际 {len}");
            return this;
        }

        public ValidationResult Build() =>
            _errors.Count == 0 ? ValidationResult.Success() : ValidationResult.Fail(_errors.ToArray());
    }

    // ============================================================
    // 4. 分页参数：一个容易做错的典型
    // ============================================================
    // 【问题】客户端传 page=0 或 size=999999 会导致：
    //   · page=0 -> 负数偏移，SQL 报错
    //   · size 太大 -> 全表扫描，服务雪崩

    public class PageRequest
    {
        private const int MaxSize = 100;
        private const int DefaultSize = 20;

        public int Page { get; private set; } = 1;
        public int Size { get; private set; } = DefaultSize;

        public static PageRequest Parse(int page, int size)
        {
            var r = new PageRequest();

            // 【关键】所有边界都要夹紧，不能让非法值穿透
            r.Page = page < 1 ? 1 : page;

            r.Size = size switch
            {
                <= 0 => DefaultSize,
                > MaxSize => MaxSize,     // 防止一次拉太多
                _ => size
            };

            return r;
        }

        public int Skip => (Page - 1) * Size;

        public override string ToString() => $"page={Page}, size={Size}, skip={Skip}";
    }

    // ============================================================
    // 演示
    // ============================================================
    public static class Demo
    {
        public static void Run()
        {
            Console.WriteLine("=== 1. 参数来源 ===");
            foreach (var s in BindSources.All)
            {
                Console.WriteLine($"  {s.Name,-14} {s.Description,-16} {s.Example}");
            }
            Console.WriteLine();
            Console.WriteLine("  【实战建议】");
            Console.WriteLine("    简单类型用 [FromRoute] / [FromQuery]");
            Console.WriteLine("    复杂对象用 [FromBody]");
            Console.WriteLine("    不要在一个方法里混用多个 [FromBody]（只能有一个）");

            Console.WriteLine();
            Console.WriteLine("=== 2. 路由约束 ===");
            foreach (var (c, m, e) in RouteConstraints.All)
            {
                Console.WriteLine($"  {c,-14} {m,-12} {e}");
            }

            Console.WriteLine();
            Console.WriteLine("  实测路由匹配:");
            var router = new SimpleRouter();
            router.MapGet("/users/{id:int}", p => $"用户详情: id={p["id"]}");
            router.MapGet("/users/{name:alpha}", p => $"按名字查: name={p["name"]}");
            router.MapGet("/health", _ => "OK");

            foreach (var path in new[] { "/users/1001", "/users/tom", "/health", "/users/abc123" })
            {
                var r = router.Dispatch(path);
                Console.WriteLine($"    {path,-18} -> {(r ?? "404 无匹配")}");
            }
            Console.WriteLine("  【关键】/users/abc123 不匹配 int 约束，也不匹配 alpha（含数字）");
            Console.WriteLine("          所以 404 —— 非法请求不会进业务代码");

            Console.WriteLine();
            Console.WriteLine("=== 3. 参数验证 ===");
            var cases = new[]
            {
                (name: "正常", userId: 1001L, count: 5, nickname: "勇者"),
                (name: "ID非法", userId: -1L, count: 5, nickname: "勇者"),
                (name: "数量超限", userId: 1001L, count: 99999, nickname: "勇者"),
                (name: "昵称为空", userId: 1001L, count: 5, nickname: ""),
            };

            foreach (var c in cases)
            {
                var v = new Validator()
                    .Require(c.userId > 0, "userId 必须大于 0")
                    .Range(c.count, 1, 999, "count")
                    .Length(c.nickname, 1, 20, "nickname")
                    .Build();

                if (v.Ok)
                    Console.WriteLine($"  [{c.name,-10}] 通过");
                else
                    Console.WriteLine($"  [{c.name,-10}] 失败: {string.Join("; ", v.Errors)}");
            }

            Console.WriteLine();
            Console.WriteLine("  【铁律】客户端可以绕过所有前端校验");
            Console.WriteLine("          服务端必须再校验一次");

            Console.WriteLine();
            Console.WriteLine("=== 4. 分页边界处理 ===");
            var pages = new[] { (1, 20), (0, 20), (1, 0), (1, 99999), (5, 50) };
            foreach (var (p, s) in pages)
            {
                var req = PageRequest.Parse(p, s);
                Console.WriteLine($"  输入 (page={p}, size={s}) -> {req}");
            }

            Console.WriteLine();
            Console.WriteLine("  【关键】size=99999 被夹到 100，防止一次拉全表");
            Console.WriteLine("          page=0 被夹到 1，防止负数偏移");
        }
    }
}

public class Program
{
    public static void Main() => BackendSamples.RoutingBinding.Demo.Run();
}
来源写法典型场景
路由/users/{id}资源标识
查询串?page=1分页、筛选
请求体{"name":"x"}POST/PUT 的数据
请求头X-Token: abc认证、追踪 ID
表单multipart/form-data文件上传

二、路由约束：在入口就拒绝 ★★

为什么要用路由约束 不加约束：
/api/users/{id}
请求 /api/users/abc → 进入业务代码 → 解析失败 → 查了一次数据库才发现参数错

加约束：
/api/users/{id:int}
请求 /api/users/abc → 路由层直接 404，不进业务代码

本页实测输出：
/users/1001 -> 用户详情: id=1001（匹配 int）
/users/tom -> 按名字查: name=tom（匹配 alpha）
/users/abc123 -> 404 无匹配（不匹配 int，也不匹配 alpha）

价值：非法请求在入口就被挡住，节省数据库和 CPU。

三、参数验证：后端必须做的事 ★★★

铁律：客户端可以绕过一切前端校验 这不是「防君子」的问题——而是基本工程纪律。

绕过方式：
· 抓包工具直接改请求
· Postman / curl 直接调接口
· 反编译客户端看你调了什么接口

所以服务端必须校验：
· 类型——传了字符串给 int 字段
· 范围——数量传了 -1 或 999999
· 长度——昵称传了 10 万字
· 格式——手机号传了乱码
· 业务规则——金币不够却要买

本页实测：四种输入，非法的一个都没通过。
验证的三种实现方式 ① 手写（本页代码）
简单直接，适合小项目

② DataAnnotations 特性
[Required] [Range(1,100)] [StringLength(20)]
框架自动验证，返回 400
缺点：错误信息格式化不够灵活

③ FluentValidation（推荐）
RuleFor(x => x.Count).InclusiveBetween(1, 999);
可读性好、支持复杂规则、错误信息可定制

不管用哪种：校验必须在服务端执行，且要在业务逻辑之前。

四、分页：一个容易做错的典型

两个边界不处理会出事故 问题一：size 太大
?size=999999 → 一次查出全表 → 数据库和内存都被打爆
本页实测：size=99999 被夹到 100。

问题二：page 为 0 或负数
?page=0 → skip = (0-1)*size = 负数 → SQL 报错
本页实测：page=0 被夹到 1。

本页输出验证：
(page=0, size=20) -> page=1, size=20, skip=0
(page=1, size=99999) -> page=1, size=100, skip=0

原则：所有外部输入都要「夹紧」，不能让非法值穿透到数据库。
