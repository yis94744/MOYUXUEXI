using System;
using System.Collections.Generic;
using System.Linq;

namespace BackendSamples.Admin
{
    // ============================================================
    // 1. 运营后台的本质：CRUD + 权限 + 审计
    // ============================================================
    // 【和玩家接口的区别】
    //   玩家接口：高频、要性能、要防作弊
    //   GM 后台：低频、要安全、要留痕

    public enum GmRole
    {
        Viewer,      // 只能看
        Operator,    // 能操作（发奖、封号）
        Admin,       // 能改配置、加人
        SuperAdmin,  // 什么都能做
    }

    public static class RolePermission
    {
        public static readonly (GmRole Role, string CanDo, string Cannot)[] All =
        {
            (GmRole.Viewer,     "查询玩家、查日志",           "任何修改操作"),
            (GmRole.Operator,   "发奖、封号、改体力",         "改配置、加GM"),
            (GmRole.Admin,      "改活动配置、加Operator",     "加Admin、改权限"),
            (GmRole.SuperAdmin, "一切",                      "—"),
        };

        public static bool Can(GmRole role, string action) => action switch
        {
            "view"            => true,
            "grant_item"      => role >= GmRole.Operator,
            "ban_player"      => role >= GmRole.Operator,
            "edit_config"     => role >= GmRole.Admin,
            "add_gm"          => role >= GmRole.Admin,
            "change_role"     => role >= GmRole.SuperAdmin,
            _ => false,
        };
    }

    // ============================================================
    // 2. 审计日志：谁在什么时候做了什么
    // ============================================================
    // 【为什么必须】
    //   · 内部人员误操作 -> 能追溯
    //   · 内鬼刷道具 -> 能查出来
    //   · 合规审计 -> 必须留痕

    public class AuditLog
    {
        public long Id;
        public string Operator = "";
        public GmRole Role;
        public string Action = "";
        public string Target = "";
        public string Detail = "";
        public DateTime Time = DateTime.UtcNow;
        public string Ip = "";

        public override string ToString() =>
            $"[{Time:HH:mm:ss}] {Operator}({Role}) {Action} {Target} | {Detail}";
    }

    public class AuditService
    {
        private readonly List<AuditLog> _logs = new();
        private long _id;

        public void Record(string op, GmRole role, string action, string target, string detail, string ip = "127.0.0.1")
        {
            _logs.Add(new AuditLog
            {
                Id = ++_id,
                Operator = op,
                Role = role,
                Action = action,
                Target = target,
                Detail = detail,
                Ip = ip,
            });
        }

        public IEnumerable<AuditLog> All => _logs;
        public IEnumerable<AuditLog> ByOperator(string op) => _logs.Where(l => l.Operator == op);
        public int Count => _logs.Count;
    }

    // ============================================================
    // 3. GM 接口：权限检查 + 审计 + 幂等
    // ============================================================

    public class GmService
    {
        private readonly AuditService _audit;
        private readonly Dictionary<long, int> _playerGold = new();
        private readonly HashSet<long> _banned = new();
        private readonly HashSet<string> _processedOps = new();

        public GmService(AuditService audit)
        {
            _audit = audit;
            for (long i = 1; i <= 10; i++) _playerGold[i] = 1000;
        }

        public int GetGold(long playerId) => _playerGold.TryGetValue(playerId, out int g) ? g : 0;
        public bool IsBanned(long playerId) => _banned.Contains(playerId);

        // 【GM 发奖】三个关键点
        public (bool Ok, string Msg) GrantGold(
            string op, GmRole role, long playerId, int amount, string reason, string opId)
        {
            // ① 权限检查
            if (!RolePermission.Can(role, "grant_item"))
                return (false, $"权限不足（{role} 不能发奖）");

            // ② 参数校验（防止手滑多打一个 0）
            if (amount <= 0)
                return (false, "数量必须大于 0");
            if (amount > 100000)
                return (false, "单次发放不能超过 10 万（请分批）");

            // ③ 幂等（防止重复点击提交）
            if (_processedOps.Contains(opId))
                return (true, "已处理过（幂等）");
            _processedOps.Add(opId);

            // 执行
            if (!_playerGold.ContainsKey(playerId))
                return (false, "玩家不存在");

            _playerGold[playerId] += amount;

            // ④ 审计
            _audit.Record(op, role, "grant_gold", $"玩家{playerId}", $"+{amount} ({reason})");

            return (true, $"成功，玩家余额 {_playerGold[playerId]}");
        }

        // 【GM 封号】
        public (bool Ok, string Msg) BanPlayer(
            string op, GmRole role, long playerId, string reason, int days)
        {
            if (!RolePermission.Can(role, "ban_player"))
                return (false, "权限不足");

            if (string.IsNullOrWhiteSpace(reason))
                return (false, "封号必须填写原因");

            // 【关键】封号前要二次确认（本 demo 用一个规则模拟）
            if (days > 365)
                return (false, "超过 365 天需要 SuperAdmin 审批");

            _banned.Add(playerId);
            _audit.Record(op, role, "ban_player", $"玩家{playerId}", $"{days}天: {reason}");
            return (true, $"玩家 {playerId} 已封禁 {days} 天");
        }

        // 【危险操作】删档
        public (bool Ok, string Msg) DeletePlayer(
            string op, GmRole role, long playerId, string confirmCode)
        {
            // 【关键】危险操作需要二次确认码
            if (role != GmRole.SuperAdmin)
                return (false, "只有 SuperAdmin 能删档");

            if (confirmCode != "DELETE-" + playerId)
                return (false, "确认码错误（应为 DELETE-玩家ID）");

            _playerGold.Remove(playerId);
            _audit.Record(op, role, "delete_player", $"玩家{playerId}", "永久删除");
            return (true, "已删除");
        }
    }

    // ============================================================
    // 4. 批量操作：最容易出事的地方
    // ============================================================
    // 【场景】给 1000 个玩家发补偿
    // 【风险】操作失误 -> 发错人 / 发错量

    public class BatchOperation
    {
        public class BatchResult
        {
            public int Total;
            public int Success;
            public int Failed;
            public List<string> Errors = new();
        }

        // 【安全做法】先预览，再执行
        public static BatchResult Preview(List<long> playerIds, int amount)
        {
            var r = new BatchResult { Total = playerIds.Count };

            Console.WriteLine($"  预览批量发奖:");
            Console.WriteLine($"    目标人数: {playerIds.Count}");
            Console.WriteLine($"    每人金额: {amount}");
            Console.WriteLine($"    总额: {(long)playerIds.Count * amount}");
            Console.WriteLine($"    前 5 个目标: {string.Join(", ", playerIds.Take(5))}");

            return r;
        }

        // 分批执行 + 失败记录
        public static BatchResult Execute(
            List<long> playerIds, int amount, int batchSize,
            Func<long, int, (bool Ok, string Msg)> action)
        {
            var r = new BatchResult { Total = playerIds.Count };

            for (int i = 0; i < playerIds.Count; i += batchSize)
            {
                var batch = playerIds.Skip(i).Take(batchSize).ToList();
                Console.WriteLine($"    处理第 {i / batchSize + 1} 批（{batch.Count} 人）");

                foreach (var pid in batch)
                {
                    var (ok, msg) = action(pid, amount);
                    if (ok) r.Success++;
                    else
                    {
                        r.Failed++;
                        r.Errors.Add($"玩家{pid}: {msg}");
                    }
                }
            }

            return r;
        }
    }

    // 5. 后台的四个必备功能

    public static class MustHave
    {
        public static readonly (string Feature, string Why, string Risk)[] All =
        {
            ("玩家查询",   "客服要用",           "不能直接看敏感信息"),
            ("发奖/扣款",  "补偿、活动",         "【必须审计】"),
            ("封号/解封",  "违规处理",           "【必须填原因】"),
            ("日志查询",   "排查问题",           "注意脱敏"),
            ("配置修改",   "活动调整",           "要审批流"),
            ("权限管理",   "团队协作",           "最小权限原则"),
        };
    }

    // 演示
    // ============================================================
    public static class Demo
    {
        public static void Run()
        {
            Console.WriteLine("=== 1. 角色与权限 ===");
            foreach (var (role, can, cannot) in RolePermission.All)
            {
                Console.WriteLine($"    {role,-12} 能做: {can,-24} 不能: {cannot}");
            }

            Console.WriteLine();
            Console.WriteLine("  权限检查测试:");
            foreach (var (role, action) in new[]
            {
                (GmRole.Viewer, "view"),
                (GmRole.Viewer, "grant_item"),
                (GmRole.Operator, "grant_item"),
                (GmRole.Operator, "edit_config"),
                (GmRole.Admin, "edit_config"),
                (GmRole.Admin, "change_role"),
            })
            {
                var ok = RolePermission.Can(role, action);
                Console.WriteLine($"    {role,-12} 执行 {action,-14} -> {(ok ? "允许" : "拒绝")}");
            }

            Console.WriteLine();
            Console.WriteLine("=== 2. GM 发奖（权限+校验+幂等+审计）===");
            var audit = new AuditService();
            var gm = new GmService(audit);

            Console.WriteLine("  Viewer 尝试发奖:");
            var (ok1, msg1) = gm.GrantGold("小明", GmRole.Viewer, 1, 1000, "测试", "op-001");
            Console.WriteLine($"    {msg1}");

            Console.WriteLine();
            Console.WriteLine("  Operator 正常发奖:");
            var (ok2, msg2) = gm.GrantGold("小红", GmRole.Operator, 1, 500, "补偿", "op-002");
            Console.WriteLine($"    {msg2}");

            Console.WriteLine();
            Console.WriteLine("  重复提交（同一个 opId）:");
            var (ok3, msg3) = gm.GrantGold("小红", GmRole.Operator, 1, 500, "补偿", "op-002");
            Console.WriteLine($"    {msg3}（余额没变，防重复）");

            Console.WriteLine();
            Console.WriteLine("  数量超出上限（手滑打多了 0）:");
            var (ok4, msg4) = gm.GrantGold("小红", GmRole.Operator, 1, 2000000, "补偿", "op-003");
            Console.WriteLine($"    {msg4}");

            Console.WriteLine();
            Console.WriteLine("=== 3. 封号 ===");
            Console.WriteLine("  不填原因:");
            var (b1, bm1) = gm.BanPlayer("小红", GmRole.Operator, 2, "", 7);
            Console.WriteLine($"    {bm1}");

            Console.WriteLine("  正常封号:");
            var (b2, bm2) = gm.BanPlayer("小红", GmRole.Operator, 2, "使用外挂", 7);
            Console.WriteLine($"    {bm2}");

            Console.WriteLine();
            Console.WriteLine("=== 4. 危险操作（删档）===");
            Console.WriteLine("  Operator 尝试删档:");
            var (d1, dm1) = gm.DeletePlayer("小红", GmRole.Operator, 3, "DELETE-3");
            Console.WriteLine($"    {dm1}");

            Console.WriteLine("  SuperAdmin 但确认码错误:");
            var (d2, dm2) = gm.DeletePlayer("老板", GmRole.SuperAdmin, 3, "delete-3");
            Console.WriteLine($"    {dm2}");

            Console.WriteLine("  SuperAdmin 确认码正确:");
            var (d3, dm3) = gm.DeletePlayer("老板", GmRole.SuperAdmin, 3, "DELETE-3");
            Console.WriteLine($"    {dm3}");

            Console.WriteLine();
            Console.WriteLine("=== 5. 审计日志 ===");
            Console.WriteLine($"  共 {audit.Count} 条记录:");
            foreach (var l in audit.All)
            {
                Console.WriteLine($"    {l}");
            }

            Console.WriteLine();
            Console.WriteLine("  【关键】所有修改操作都要留痕");
            Console.WriteLine("          出事时能查清楚是谁做的");

            Console.WriteLine();
            Console.WriteLine("=== 6. 批量操作：先预览再执行 ===");
            var targets = Enumerable.Range(1, 10).Select(i => (long)i).ToList();
            BatchOperation.Preview(targets, 1000);

            Console.WriteLine();
            Console.WriteLine("  执行（每批 3 人）:");
            var result = BatchOperation.Execute(targets, 1000, 3,
                (pid, amount) => gm.GrantGold("运营", GmRole.Operator, pid, amount, "全服补偿", $"batch-{pid}"));

            Console.WriteLine($"    成功 {result.Success}，失败 {result.Failed}");
            foreach (var e in result.Errors)
            {
                Console.WriteLine($"      {e}");
            }

            Console.WriteLine();
            Console.WriteLine("  【危险】玩家 10 不存在，失败被记录");
            Console.WriteLine("          批量操作必须能容忍部分失败");

            Console.WriteLine();
            Console.WriteLine("=== 7. 后台必备功能 ===");
            foreach (var (f, why, risk) in MustHave.All)
            {
                Console.WriteLine($"    {f,-12} 用途: {why,-16} 风险: {risk}");
            }
        }
    }

频率极高低
关注点性能、防作弊安全、可追溯

二、四级权限模型 ★★

本页实现的角色与权限
角色能做不能做
Viewer查询任何修改
Operator发奖、封号改配置
Admin改配置、加人改权限
SuperAdmin一切—
本页实测的权限拦截：
Viewer   执行 view       -> 允许
Viewer   执行 grant_item -> 拒绝
Operator 执行 grant_item -> 允许
Operator 执行 edit_config -> 拒绝
Admin   执行 edit_config -> 允许
Admin   执行 change_role -> 拒绝（只有 SuperAdmin 能）

原则：最小权限——能不给就不给。

三、GM 发奖的四个关键点 ★★★

本页实测的四道防线 ① 权限检查
Viewer 尝试发奖 -> 权限不足

② 参数校验（防手滑）
数量 2000000 -> 拒绝，单次不能超过 10 万
这是真实需求——多打一个 0 就会造成事故

③ 幂等（防重复点击）
同一 opId 重复提交 -> 已处理过，余额没变
运营手抖点两次提交是常事

④ 审计
[03:20:15] 小红(Operator) grant_gold 玩家1 | +500 (补偿)
（时间戳取运行时时刻，格式是 HH:mm:ss）
出事时能查清楚是谁做的

四个缺一不可。

四、危险操作要二次确认 ★★

本页的删档保护 三道保护：
① 只有 SuperAdmin 能做
② 必须提供确认码（DELETE-玩家ID）
③ 记录完整审计

本页实测：
Operator 尝试删档 -> 只有 SuperAdmin 能删档
SuperAdmin 确认码错误 -> 确认码错误
SuperAdmin 确认码正确 -> 已删除

为什么需要确认码：防止误点。
· 界面上「删除」按钮和「发奖」按钮挨着
· 手一抖就点了删除
· 要求输入玩家ID才能删 → 必须动脑子，不会误触

五、批量操作：最危险的地方 ★★★

本页的「先预览再执行」 风险：给 1000 个玩家发补偿，条件写错了 → 全服多发

本页实现的安全流程：
预览: 目标人数 10，每人 1000，总额 10000
前 5 个目标: 1, 2, 3, 4, 5
→ 确认无误后再执行

执行要分批 + 容错：
处理第 1 批（3 人）
处理第 2 批（3 人）
成功 9，失败 1
玩家3: 玩家不存在

关键：部分失败不能中断整个批量——
要记录失败清单，事后单独处理。
