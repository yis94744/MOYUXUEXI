    public static class DeployModes
    {
        public static readonly (string Mode, string How, string Pros, string Cons)[] All =
        {
            ("直接在服务器跑", "dotnet publish + systemd", "简单、无额外依赖", "环境不一致"),
            ("Docker",         "镜像 + 容器",             "环境一致、好回滚", "要学 Docker"),
            ("PaaS / 云托管",  "上传代码，平台托管",       "省运维",           "贵、受制于人"),
        };

        public static string Recommend(int teamSize, bool hasOps)
        {
            if (teamSize <= 2 && !hasOps) return "PaaS（省时间）";
            if (teamSize <= 10) return "Docker + 单机/少量机器";
            return "K8s（规模化）";
        }
    }

    // ============================================================
    // 2. 发布的原子性：不能出现「一半新一半旧」
    // ============================================================

    public class DeploySimulator
    {
        public class Version
        {
            public string Tag = "";
            public bool Healthy;
            public DateTime DeployedAt = DateTime.UtcNow;
            public int RequestCount;
        }

        private readonly List<Version> _history = new();

        public DeploySimulator()
        {
            // 【关键】初始版本也要进历史，否则第一次部署后「回滚」找不到 v1.0
            var initial = new Version { Tag = "v1.0", Healthy = true };
            _history.Add(initial);
            Current = initial;
        }

        public Version Current { get; private set; } = new Version { Tag = "(未初始化)", Healthy = false };

        public (bool Ok, string Msg) Deploy(string tag, bool healthy)
        {
            Console.WriteLine($"    部署 {tag}...");

            var newV = new Version { Tag = tag, Healthy = healthy };

            if (!healthy)
            {
                // 【关键】健康检查失败，不切换
                Console.WriteLine($"    健康检查失败，保持 {Current.Tag}");
                _history.Add(newV);
                return (false, "部署失败，已回滚");
            }

            var old = Current;
            Current = newV;
            _history.Add(newV);
            Console.WriteLine($"    健康检查通过，{old.Tag} -> {newV.Tag}");
            return (true, "部署成功");
        }

        public (bool Ok, string Msg) Rollback()
        {
            // 找「当前版本之前」最近的一个健康版本
            int idx = _history.FindLastIndex(v => v.Tag == Current.Tag);
            var prev = idx > 0
                ? _history.Take(idx).LastOrDefault(v => v.Healthy)
                : null;
            if (prev == null) return (false, "没有可回滚的版本");

            var old = Current;
            Current = prev;
            Console.WriteLine($"    回滚: {old.Tag} -> {prev.Tag}");
            return (true, "回滚成功");
        }

        public IReadOnlyList<Version> History => _history;
    }

    // ============================================================
    // 3. 滚动发布 vs 蓝绿 vs 金丝雀
    // ============================================================

    public static class ReleaseStrategy
    {
        public static readonly (string Name, string How, string Pros, string Cons)[] All =
        {
            ("滚动发布", "逐台替换",       "资源省",             "【新老版本共存】"),
            ("蓝绿部署", "两套环境切换",   "回滚快",             "要双倍资源"),
            ("金丝雀",   "先放 1% 流量",   "风险最小",           "要多监控"),
        };

        // 模拟金丝雀
        public class CanaryRelease
        {
            private Version? _stable;
            private Version? _canary;
            private int _canaryPercent;
            private readonly Random _rnd = new(42);

            public class Version { public string Tag = ""; public int Errors; public int Total; }

            public void Start(string stableTag, string canaryTag, int canaryPercent)
            {
                _stable = new Version { Tag = stableTag };
                _canary = new Version { Tag = canaryTag };
                _canaryPercent = canaryPercent;
                Console.WriteLine($"    金丝雀发布: {stableTag} 为主，{canaryTag} 接收 {canaryPercent}% 流量");
            }

            public string Route()
            {
                var v = _rnd.Next(100) < _canaryPercent ? _canary! : _stable!;
                v.Total++;
                return v.Tag;
            }

            public void RecordError(string tag)
            {
                if (tag == _canary!.Tag) _canary.Errors++;
                else _stable!.Errors++;
            }

            // 【关键】对比错误率
            public (bool ShouldPromote, string Reason) Evaluate()
            {
                double stableRate = _stable!.Total == 0 ? 0 : (double)_stable.Errors / _stable.Total;
                double canaryRate = _canary!.Total == 0 ? 0 : (double)_canary.Errors / _canary.Total;

                Console.WriteLine($"    稳定版: {_stable.Total} 请求, {_stable.Errors} 错误 ({stableRate * 100:F1}%)");
                Console.WriteLine($"    金丝雀: {_canary.Total} 请求, {_canary.Errors} 错误 ({canaryRate * 100:F1}%)");

                // 【规则】金丝雀错误率不能明显高于稳定版
                if (canaryRate > stableRate * 2 + 0.01)
                    return (false, $"金丝雀错误率 {canaryRate * 100:F1}% 过高，建议回滚");

                return (true, "错误率正常，可以逐步放量");
            }
        }
    }

    // ============================================================
    // 4. 健康检查：部署的基础
    // ============================================================

    public class HealthChecker
    {
        public class CheckResult
        {
            public string Name = "";
            public bool Ok;
            public string Detail = "";
        }

        // 【区分】存活探针 vs 就绪探针
        //   liveness  -- 进程还活着吗？（挂了就重启）
        //   readiness -- 能接流量吗？（依赖还没就绪就不接）

        public List<CheckResult> RunChecks()
        {
            var results = new List<CheckResult>();

            // 检查数据库
            results.Add(new CheckResult
            {
                Name = "database",
                Ok = true,
                Detail = "连接正常，延迟 2ms",
            });

            // 检查 Redis
            results.Add(new CheckResult
            {
                Name = "redis",
                Ok = true,
                Detail = "连接正常",
            });

            // 检查磁盘
            var diskFreePercent = 35;   // 模拟
            results.Add(new CheckResult
            {
                Name = "disk",
                Ok = diskFreePercent > 10,
                Detail = $"剩余 {diskFreePercent}%",
            });

            // 检查依赖服务
            results.Add(new CheckResult
            {
                Name = "payment_service",
                Ok = true,
                Detail = "响应 45ms",
            });

            return results;
        }

        public bool IsHealthy(List<CheckResult> results) => results.All(r => r.Ok);
    }

    public static class Pipeline
    {
        public static readonly (string Stage, string What, string FailAction)[] All =
        {
            ("1. 构建",     "编译 + 单测",         "阻止发布"),
            ("2. 静态扫描", "代码规范、安全漏洞",  "严重问题阻止"),
            ("3. 打包",     "生成镜像/产物",       "阻止发布"),
            ("4. 部署测试", "部署到测试环境",      "阻止发布"),
            ("5. 集成测试", "端到端测试",          "阻止发布"),
            ("6. 部署生产", "灰度发布",            "自动回滚"),
            ("7. 监控",     "错误率、延迟",        "异常告警"),
        };

        public static void Print()
        {
            foreach (var (stage, what, fail) in All)
            {
                Console.WriteLine($"    {stage,-12} {what,-24} 失败: {fail}");
            }
        }
    }

            Console.WriteLine();

            Console.WriteLine();
            Console.WriteLine("  部署 v1.1（健康）:");
            var (ok1, msg1) = dep.Deploy("v1.1", healthy: true);
            Console.WriteLine($"    结果: {msg1}");

            Console.WriteLine();
            Console.WriteLine("  部署 v1.2（不健康）:");
            var (ok2, msg2) = dep.Deploy("v1.2", healthy: false);
            Console.WriteLine($"    结果: {msg2}，当前版本仍是 {dep.Current.Tag}");

            Console.WriteLine();
            Console.WriteLine("  回滚:");
            var (ok3, msg3) = dep.Rollback();
            Console.WriteLine($"    结果: {msg3}，当前版本 {dep.Current.Tag}");

            Console.WriteLine();
            Console.WriteLine("  【关键】部署失败必须自动保持旧版本");
            Console.WriteLine("          不能出现「一半新一半旧」");

            Console.WriteLine();
            Console.WriteLine("=== 3. 三种发布策略 ===");
            foreach (var (name, how, pros, cons) in ReleaseStrategy.All)
            {
                Console.WriteLine($"    {name,-10} {how,-16} 优点: {pros,-14} 缺点: {cons}");
            }

            Console.WriteLine();
            Console.WriteLine("  金丝雀演练（1% 流量）:");
            var canary = new ReleaseStrategy.CanaryRelease();
            canary.Start("v2.0", "v2.1", 10);

            var rnd = new Random(42);
            for (int i = 0; i < 100; i++)
            {
                var tag = canary.Route();
                // 模拟：金丝雀版本有 5% 错误率，稳定版 1%
                double errRate = tag == "v2.1" ? 0.05 : 0.01;
                if (rnd.NextDouble() < errRate) canary.RecordError(tag);
            }

            var (promote, reason) = canary.Evaluate();
            Console.WriteLine($"    决策: {(promote ? "继续放量" : "回滚")} — {reason}");

            Console.WriteLine();
            Console.WriteLine("  【金丝雀的价值】");
            Console.WriteLine("    · 只有 10% 用户受影响");
            Console.WriteLine("    · 能在扩大影响前发现问题");
            Console.WriteLine("    · 大厂的标准做法");

            Console.WriteLine();
            Console.WriteLine("=== 4. 健康检查 ===");
            var hc = new HealthChecker();
            var checks = hc.RunChecks();

            Console.WriteLine("  各类检查:");
            foreach (var c in checks)
            {
                var mark = c.Ok ? "[OK]" : "[FAIL]";
                Console.WriteLine($"    {mark,-7} {c.Name,-18} {c.Detail}");
            }

            Console.WriteLine();
            Console.WriteLine($"  整体健康: {hc.IsHealthy(checks)}");

            Console.WriteLine();
            Console.WriteLine("  【两类探针的区别】");
            Console.WriteLine("    liveness  -> 进程活着吗？挂了就重启");
            Console.WriteLine("    readiness -> 能接流量吗？依赖没就绪就不接");
            Console.WriteLine("    【常见错误】把 liveness 做成检查数据库");
            Console.WriteLine("      -> 数据库抖动会导致所有 pod 被重启");

            Console.WriteLine();
            Console.WriteLine("=== 5. CI/CD 流水线 ===");
            Pipeline.Print();

            Console.WriteLine();
            Console.WriteLine("  【最小可用 CI】");
            Console.WriteLine("    1. push 触发构建");
            Console.WriteLine("    2. 跑单元测试");
            Console.WriteLine("    3. 打包镜像");
            Console.WriteLine("    4. 部署到测试环境");
            Console.WriteLine("    5. 手动点「发布生产」");
        }

二、部署必须原子 ★★★

本页实测的部署流程 正确流程：
当前版本: v1.0
部署 v1.1（健康）-> v1.0 -> v1.1
部署 v1.2（不健康）-> 健康检查失败，保持 v1.1
回滚: v1.1 -> v1.0，结果: 回滚成功，当前版本 v1.0

关键点：
① 先启动新版本，健康检查通过后才切流量
② 健康检查失败 → 不切换，旧版本继续服务
③ 保留旧版本（随时能回滚）

反例：先停旧版本，再启动新版本 → 中间有一段服务不可用。

三、三种发布策略 ★★

对比与选择
策略做法优缺点
滚动发布逐台替换资源省，但新老版本共存
蓝绿部署两套环境切换回滚快，但要双倍资源
金丝雀先放 1~10% 流量风险最小，大厂标配
本页实测的金丝雀逻辑（10% 流量给金丝雀，注入故障）：
稳定版: 80 请求, 0 错误 (0.0%)
金丝雀: 20 请求, 14 错误 (70.0%)
决策: 回滚 — 金丝雀错误率 70.0% 过高，建议回滚

判断规则：金丝雀错误率超过稳定版的 2 倍就回滚
价值：只有 10% 用户受影响，能在扩大影响前发现问题。

四、健康检查：两类探针别搞混 ★★★

liveness vs readiness
探针回答的问题失败后的动作
liveness进程还活着吗？重启容器
readiness能接流量吗？摘掉流量，不重启
本页实测的检查项：
[OK] database      连接正常，延迟 2ms
[OK] redis         连接正常
[OK] disk          剩余 35%
[OK] payment_service 响应 45ms
整体健康: True

最经典的错误：把 liveness 做成检查数据库
→ 数据库抖动一下 → 所有 pod 全部被重启 → 雪崩

正确做法：
· liveness 只检查自己能响应（HTTP 200）
· readiness 才检查依赖是否就绪
