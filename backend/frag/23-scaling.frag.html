    public class StatefulServer
    {
        // 【错误】状态存在进程内存
        private readonly Dictionary<string, long> _sessions = new();
        public int LocalSessionCount => _sessions.Count;

        public void Login(string token, long playerId) => _sessions[token] = playerId;

        public long? GetPlayer(string token) =>
            _sessions.TryGetValue(token, out long id) ? id : null;

        // 【问题】扩展到 2 台机器后：
        //   玩家在 A 机登录，请求被负载均衡到 B 机 -> 查不到 session
    }

    public class StatelessServer
    {
        // 【正确】状态放外部存储（Redis）
        private readonly FakeRedis _redis;
        public StatelessServer(FakeRedis redis) => _redis = redis;

        public void Login(string token, long playerId) => _redis.Set("session:" + token, playerId.ToString());

        public long? GetPlayer(string token)
        {
            var v = _redis.Get("session:" + token);
            return v != null ? long.Parse(v) : null;
        }

        // 【关键】任何一台机器都能服务任何请求
    }

    public class FakeRedis
    {
        private readonly Dictionary<string, string> _data = new();
        public void Set(string k, string v) => _data[k] = v;
        public string? Get(string k) => _data.TryGetValue(k, out var v) ? v : null;
        public int Count => _data.Count;
    }

    // ============================================================
    // 3. 负载均衡算法
    // ============================================================

    public class LoadBalancer
    {
        private readonly List<string> _servers;
        private int _roundRobinIndex;
        private readonly Dictionary<string, int> _connCount = new();

        public LoadBalancer(List<string> servers)
        {
            _servers = servers;
            foreach (var s in servers) _connCount[s] = 0;
        }

        // 【算法1】轮询：简单，不关心服务器负载
        public string RoundRobin()
        {
            var s = _servers[_roundRobinIndex % _servers.Count];
            _roundRobinIndex++;
            return s;
        }

        // 【算法2】最少连接：把请求给当前连接最少的
        public string LeastConnections()
        {
            var s = _servers.OrderBy(x => _connCount[x]).First();
            _connCount[s]++;
            return s;
        }

        public void Release(string server) => _connCount[server] = Math.Max(0, _connCount[server] - 1);

        // 【算法3】一致性哈希：同一玩家固定到同一台（有状态服务用）
        public string ConsistentHash(string key)
        {
            // 【简化】用哈希取模
            int hash = 0;
            foreach (char c in key) hash = (hash * 31 + c) % 1000000;
            return _servers[Math.Abs(hash) % _servers.Count];
        }

        // 【一致性哈希的问题】增加/删除服务器时，大量 key 会重新映射
        public static string ExplainMigration(int oldCount, int newCount)
        {
            double migrated = 1.0 - (double)Math.Min(oldCount, newCount) / Math.Max(oldCount, newCount);
            return $"从 {oldCount} 台扩到 {newCount} 台，约 {migrated * 100:F0}% 的 key 会迁移";
        }

        public Dictionary<string, int> Stats => new(_connCount);
    }

    // ============================================================
    // 4. 限流：保护自己不被压垮
    // ============================================================

    public class RateLimiter
    {
        // ---- 算法1：固定窗口（简单但有边界问题）----
        private DateTime _windowStart = DateTime.UtcNow;
        private int _windowCount;
        private readonly int _limitPerWindow;
        private readonly int _windowSec;

        public RateLimiter(int limitPerWindow, int windowSec = 1)
        {
            _limitPerWindow = limitPerWindow;
            _windowSec = windowSec;
        }

        public bool TryFixedWindow(DateTime now)
        {
            if ((now - _windowStart).TotalSeconds >= _windowSec)
            {
                _windowStart = now;
                _windowCount = 0;
            }

            if (_windowCount >= _limitPerWindow) return false;
            _windowCount++;
            return true;
        }

        // 【问题】固定窗口的边界：0.9秒发满 + 1.1秒又发满 = 2倍流量

        // ---- 算法2：滑动窗口（更平滑）----
        private readonly Queue<DateTime> _requests = new();

        public bool TrySlidingWindow(DateTime now)
        {
            // 清掉过期的
            while (_requests.Count > 0 && (now - _requests.Peek()).TotalSeconds >= _windowSec)
                _requests.Dequeue();

            if (_requests.Count >= _limitPerWindow) return false;
            _requests.Enqueue(now);
            return true;
        }

        // ---- 算法3：令牌桶（允许突发）----
        private double _tokens;
        private DateTime _lastRefill = DateTime.UtcNow;
        private readonly double _capacity;
        private readonly double _refillPerSec;

        public RateLimiter(double capacity, double refillPerSec)
        {
            _capacity = capacity;
            _refillPerSec = refillPerSec;
            _tokens = capacity;
        }

        public bool TryTokenBucket(DateTime now, int cost = 1)
        {
            // 补充令牌
            var elapsed = (now - _lastRefill).TotalSeconds;
            _tokens = Math.Min(_capacity, _tokens + elapsed * _refillPerSec);
            _lastRefill = now;

            if (_tokens < cost) return false;
            _tokens -= cost;
            return true;
        }

        public double RemainingTokens => _tokens;
    }

    // 5. 熔断与降级
    // ============================================================

    public class CircuitBreaker
    {
        public enum State { Closed, Open, HalfOpen }

        private State _state = State.Closed;
        private int _failureCount;
        private DateTime _openedAt;
        private readonly int _threshold;
        private readonly int _openSeconds;

        public CircuitBreaker(int threshold = 5, int openSeconds = 10)
        {
            _threshold = threshold;
            _openSeconds = openSeconds;
        }

        public State CurrentState => _state;

        public (bool Allowed, string Reason) TryCall(DateTime now)
        {
            switch (_state)
            {
                case State.Closed:
                    return (true, "正常调用");

                case State.Open:
                    // 熔断中：等一段时间后尝试半开
                    if ((now - _openedAt).TotalSeconds >= _openSeconds)
                    {
                        _state = State.HalfOpen;
                        return (true, "半开：试探调用");
                    }
                    return (false, $"熔断中（{_openSeconds - (now - _openedAt).TotalSeconds:F0}s 后重试）");

                case State.HalfOpen:
                    return (true, "半开：试探调用");

                default:
                    return (false, "未知状态");
            }
        }

        public void RecordSuccess()
        {
            _failureCount = 0;
            _state = State.Closed;
        }

        public void RecordFailure(DateTime now)
        {
            _failureCount++;

            if (_state == State.HalfOpen || _failureCount >= _threshold)
            {
                _state = State.Open;
                _openedAt = now;
                _failureCount = 0;
            }
        }
    }

    public class Sharding
    {
        // 【为什么分】单表 5000 万行后，索引效率急剧下降
        public static readonly (string Strategy, string How, string Pros, string Cons)[] Strategies =
        {
            ("按用户ID取模", "id % N", "分布均匀", "扩容要迁移"),

            ("一致性哈希",   "哈希环", "扩容影响小", "实现复杂"),
            ("按业务",       "玩家/订单/日志分库", "隔离性好", "不能跨库join"),
        };

        // 【关键问题】分表后怎么查「全服排名」？

        {
            return "各分库分别查前 100 -> 汇总再排 -> 取前 100（两阶段查询）";
        }
    }

            Console.WriteLine("    如果是 3 台扩到 100 台，97% 都要迁移！");
            Console.WriteLine("    -> 所以要用「一致性哈希环」，只迁移相邻的一段");

            Console.WriteLine("  【熔断的价值】");
            Console.WriteLine("    · 下游服务挂了，不要一直重试（会雪崩）");
            Console.WriteLine("    · 直接返回兜底数据，保护调用方");
            Console.WriteLine("    · 等下游恢复后自动半开试探");

二、无状态化：水平扩展的前提 ★★★

Session 存本机 = 无法水平扩展 本页实测的对比：

【错误】Session 存进程内存
Login("token-abc", 1001) -> 本机 1 条
扩到 2 台后 → 玩家请求打到另一台 → 查不到 session → 被踢下线

【正确】Session 存 Redis
Login 写入 Redis -> 任意机器都能查到
GetPlayer("token-abc") = 1001

除了 Session，还有哪些状态要外置：
· 上传的临时文件 → 对象存储
· 缓存的本地数据 → Redis
· 定时任务的状态 → 数据库（否则每台都跑一遍）
· WebSocket 连接 → 这个没法外置（见页 11，网关要特殊处理）

三、负载均衡的三种算法 ★★

本页实现的三种 ① 轮询（Round Robin）
srv-1, srv-2, srv-3, srv-1, srv-2, ...
· 简单，不关心负载
· 适合机器性能相同、请求耗时均匀

② 最少连接
· 把请求给当前连接最少的
· 适合请求耗时差异大

③ 一致性哈希
· player-1001 永远到同一台
· 有状态服务必须用这个

扩容的代价（本页计算）：
3 台扩到 4 台：约 25% 的 key 会迁移
3 台扩到 100 台：约 97% 都要迁移！
→ 本页代码用的是「哈希取模」（不是真正的一致性哈希环），所以扩容时几乎全量重映射
→ 真实系统用「一致性哈希环 + 虚拟节点」，只迁移相邻一段（约 1/N）

四、限流的三种算法 ★★

固定窗口的边界问题 本页实测：限 5 次/秒，10 个请求
· 固定窗口：通过 5 个
· 滑动窗口：通过 5 个
· 令牌桶：通过 4 个（桶容量 5，但首个请求要先产生令牌）
一次突发看不出前两者的差别，但边界场景差别很大

固定窗口的经典问题：
· 0.9 秒内发 5 次（通过）
· 1.0 秒窗口重置
· 1.1 秒内又发 5 次（通过）
→ 0.2 秒内实际通过了 10 次 = 2 倍突发

三种算法对比：
算法优点缺点
固定窗口最简单边界突发
滑动窗口平滑要存时间戳
令牌桶允许突发参数要调

五、熔断器：保护自己 ★★★

三个状态的流转 本页实测：
第 1 次调用: 正常调用
第 2 次调用: 正常调用
第 3 次调用: 正常调用
第 4 次调用（已熔断）: 熔断中（4s 后重试）
10 秒后（转半开）: 半开：试探调用，状态=HalfOpen

状态行为
Closed正常调用，统计失败
Open直接拒绝，不调用下游
HalfOpen放少量请求试探
为什么需要熔断：
· 下游挂了 → 调用方一直等待超时 → 线程池耗尽
· 雪崩：一个服务挂，拖垮整个系统

熔断后做什么：
· 返回兜底数据（缓存、默认值）
· 或明确告知用户"稍后再试"

六、分库分表

什么时候该分 判断标准：
· 单表 > 5000 万行 → 考虑分表
· 单库磁盘 > 1TB → 考虑分库
· 写入 QPS > 5000 → 考虑分库

四种策略：
· 按 ID 取模——均匀，但扩容要迁移
· 按范围——好扩容，但热点集中
· 一致性哈希——扩容影响小，实现复杂
· 按业务——隔离性好，但不能跨库 join

最难的问题：分表后怎么查「全服排名」？
→ 各分库查前 100 → 汇总再排 → 取前 100（两阶段查询）
