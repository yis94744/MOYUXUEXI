一、缓存模式：Cache Aside

最常用的缓存模式只有两条规则：读时回填，写时删除。

using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;

namespace BackendSamples.Cache
{
    // ============================================================
    // 1. 缓存三种模式
    // ============================================================

    public class PlayerData
    {
        public long Id;
        public string Name = "";
        public int Level;
        public int Gold;

        public override string ToString() => $"#{Id} {Name} Lv{Level} 金币{Gold}";
    }

    // 模拟「慢速数据库」
    public class SlowDatabase
    {
        private readonly Dictionary<long, PlayerData> _data = new();
        public int QueryCount;
        public int WriteCount;

        public SlowDatabase()
        {
            for (long i = 1; i <= 100; i++)
            {
                _data[i] = new PlayerData { Id = i, Name = $"玩家{i}", Level = (int)(i % 50) + 1, Gold = (int)(i * 100) };
            }
        }

        public PlayerData? Get(long id)
        {
            QueryCount++;
            Thread.Sleep(2);   // 【模拟】数据库往返 2ms
            return _data.TryGetValue(id, out var p) ? p : null;
        }

        public void Save(PlayerData p)
        {
            WriteCount++;
            _data[p.Id] = p;
        }
    }

    // 模拟缓存
    public class FakeCache
    {
        private readonly Dictionary<string, (object Value, DateTime ExpireAt)> _store = new();
        public int Hits, Misses;

        public T? Get<T>(string key)
        {
            if (_store.TryGetValue(key, out var entry))
            {
                if (entry.ExpireAt > DateTime.UtcNow)
                {
                    Hits++;
                    return (T)entry.Value;
                }
                _store.Remove(key);   // 过期
            }
            Misses++;
            return default;
        }

        public void Set<T>(string key, T value, TimeSpan ttl) =>
            _store[key] = (value, DateTime.UtcNow + ttl);

        public bool Remove(string key) => _store.Remove(key);

        public int Count => _store.Count;
    }

    // ============================================================
    // 2. 三种缓存模式
    // ============================================================

    public class CachePatterns
    {
        private readonly SlowDatabase _db;
        private readonly FakeCache _cache;

        public CachePatterns(SlowDatabase db, FakeCache cache)
        {
            _db = db;
            _cache = cache;
        }

        // 【模式一】Cache Aside（旁路缓存）—— 最常用
        //   读：先查缓存，没有就查库并回填
        //   写：先更新库，再删除缓存
        public PlayerData? GetPlayer_CacheAside(long id)
        {
            string key = "player:" + id;

            // 1. 查缓存
            var cached = _cache.Get<PlayerData>(key);
            if (cached != null) return cached;

            // 2. 缓存没有，查数据库
            var p = _db.Get(id);
            if (p == null) return null;

            // 3. 回填缓存
            _cache.Set(key, p, TimeSpan.FromMinutes(5));
            return p;
        }

        public void UpdatePlayer_CacheAside(PlayerData p)
        {
            // 【关键】先更新数据库，再删缓存
            _db.Save(p);
            _cache.Remove("player:" + p.Id);

            // 【为什么是删而不是更新】
            //   更新缓存可能出现「两个请求同时更新，顺序错乱」
            //   删除更简单：下次读时自然回填最新值
        }

        // 【模式二】Read Through
        //   应用只和缓存打交道，缓存负责查库（需缓存中间件支持）
        //   本 demo 用包装体现

        // 【模式三】Write Behind（写回）
        //   写只写缓存，异步批量刷库
        //   优点：写入快
        //   缺点：缓存挂了会丢数据
        public void WriteBehind(PlayerData p)
        {
            _cache.Set("player:" + p.Id, p, TimeSpan.FromMinutes(5));
            // 【问题】如果进程崩溃，这个数据就丢了
            //   所以只适合「丢了也不致命」的数据（如排行榜分数）
        }
    }

    // ============================================================
    // 3. 三大缓存问题：穿透、击穿、雪崩 ★★★
    // ============================================================

    public class CacheProblems
    {
        private readonly SlowDatabase _db;
        private readonly FakeCache _cache;

        public CacheProblems(SlowDatabase db, FakeCache cache)
        {
            _db = db;
            _cache = cache;
        }

        // ---- 问题一：缓存穿透 ----
        // 【场景】查询一个「根本不存在」的 ID
        //   缓存没有 -> 查库 -> 库里也没有 -> 不缓存
        //   -> 下次再查还是查库
        // 【危害】大量请求直接打到数据库
        public PlayerData? Get_NoProtection(long id)
        {
            var cached = _cache.Get<PlayerData>("player:" + id);
            if (cached != null) return cached;

            var p = _db.Get(id);      // 不存在时返回 null
            if (p != null) _cache.Set("player:" + id, p, TimeSpan.FromMinutes(5));
            // 【问题】p == null 时不缓存 -> 每次都查库
            return p;
        }

        // 【方案一】缓存空值（短 TTL）
        public PlayerData? Get_CacheNull(long id)
        {
            string key = "player:" + id;

            // 用特殊标记表示「确认不存在」
            var cached = _cache.Get<object>(key);
            if (cached != null)
            {
                if (cached is NullMarker) return null;   // 命中空值缓存
                return (PlayerData)cached;
            }

            var p = _db.Get(id);
            if (p != null)
                _cache.Set(key, p, TimeSpan.FromMinutes(5));
            else
                _cache.Set(key, new NullMarker(), TimeSpan.FromSeconds(30));   // 【关键】短 TTL

            return p;
        }

        // 【方案二】布隆过滤器（本 demo 简化实现）
        public class BloomFilter
        {
            private readonly bool[] _bits;
            private readonly int _hashCount;

            public BloomFilter(int size = 10000, int hashCount = 3)
            {
                _bits = new bool[size];
                _hashCount = hashCount;
            }

            public void Add(long id)
            {
                foreach (int h in Hashes(id))
                    _bits[h] = true;
            }

            public bool MightExist(long id)
            {
                foreach (int h in Hashes(id))
                    if (!_bits[h]) return false;   // 【确定不存在】
                return true;   // 【可能存在】（有误判率）
            }

            private IEnumerable<int> Hashes(long id)
            {
                for (int i = 0; i < _hashCount; i++)
                    yield return (int)Math.Abs((id * (i * 31 + 17)) % _bits.Length);
            }
        }

        private readonly BloomFilter _bloom = new();

        public void InitBloom()
        {
            for (long i = 1; i <= 100; i++) _bloom.Add(i);
        }

        public PlayerData? Get_Bloom(long id)
        {
            // 【关键】先过布隆过滤器
            if (!_bloom.MightExist(id))
            {
                // 确定不存在，直接返回，不查缓存也不查库
                return null;
            }

            var cached = _cache.Get<PlayerData>("player:" + id);
            if (cached != null) return cached;

            var p = _db.Get(id);
            if (p != null) _cache.Set("player:" + id, p, TimeSpan.FromMinutes(5));
            return p;
        }

        // ---- 问题二：缓存击穿 ----
        // 【场景】某个「热点 key」过期瞬间，大量并发请求同时打到数据库
        // 【与穿透的区别】击穿是「数据存在但缓存刚过期」

        private readonly Dictionary<string, SemaphoreSlim> _locks = new();

        public PlayerData? Get_WithLock(long id)
        {
            string key = "player:" + id;

            var cached = _cache.Get<PlayerData>(key);
            if (cached != null) return cached;

            // 【关键】只让一个请求去查库，其他等待
            var sem = _locks.TryGetValue(key, out var s) ? s : (_locks[key] = new SemaphoreSlim(1, 1));
            sem.Wait();
            try
            {
                // 双重检查：可能别人已经回填了
                cached = _cache.Get<PlayerData>(key);
                if (cached != null) return cached;

                var p = _db.Get(id);
                if (p != null) _cache.Set(key, p, TimeSpan.FromMinutes(5));
                return p;
            }
            finally
            {
                sem.Release();
            }
        }

        // ---- 问题三：缓存雪崩 ----
        // 【场景】大量 key 在同一时刻过期 -> 全部请求打到数据库
        // 【常见原因】启动时批量预热，TTL 都一样
        // 【方案】TTL 加随机抖动

        public void SetWithJitter(string key, object value, TimeSpan baseTtl)
        {
            // 【关键】在基础 TTL 上加随机偏移
            var jitter = TimeSpan.FromSeconds(new Random().Next(-60, 60));
            _cache.Set(key, value, baseTtl + jitter);
        }
    }

    public class NullMarker { }

    // ============================================================
    // 4. 分布式锁
    // ============================================================
    // 【场景】多台服务器同时处理同一个玩家的请求
    //   进程内的 lock 不管用（跨进程）
    //   Redis 的 SETNX 可以

    public class DistributedLock
    {
        private readonly Dictionary<string, string> _redis = new();   // 模拟 Redis
        private readonly object _sync = new();

        // 【关键】SET key value NX EX seconds
        //   NX: 不存在才设置（原子操作）
        //   EX: 自动过期（防死锁）
        public bool TryAcquire(string key, string token, int expireSeconds)
        {
            lock (_sync)
            {
                if (_redis.ContainsKey(key)) return false;   // 已被占用
                _redis[key] = token;
                return true;
            }
        }

        public void Release(string key, string token)
        {
            lock (_sync)
            {
                // 【关键】校验 token，防止释放别人的锁
                if (_redis.TryGetValue(key, out var owner) && owner == token)
                    _redis.Remove(key);
            }
        }

        // 【常见错误】直接 DEL，不校验 owner
        public void Release_Bad(string key)
        {
            lock (_sync)
            {
                // 如果锁已过期并被别人获取，这里会删掉别人的锁
                _redis.Remove(key);
            }
        }

        public bool IsLocked(string key)
        {
            lock (_sync) return _redis.ContainsKey(key);
        }
    }

操作流程关键点
读查缓存 → 没有则查库 → 回填先缓存后数据库
写更新数据库 → 删除缓存删而不是更新

二、三大缓存问题 ★★★

问题一：缓存穿透（查不存在的数据） 场景：
· 请求查 ID=999999（根本不存在）
· 缓存没有 → 查库 → 库里也没有 → 不缓存
· 下次同样的请求 → 还是查库

危害：恶意请求可以轻松打垮数据库。

本页实测对比：
· 无防护：5 次请求 → 5 次数据库查询
· 缓存空值：5 次请求 → 1 次数据库查询
· 布隆过滤器：5 次请求 → 0 次数据库查询

两种方案的选择：
· 缓存空值：简单，但会占内存（TTL 要短，30 秒）
· 布隆过滤器：内存极小，但有误判（说"存在"可能不存在）
问题二：缓存击穿（热点 key 过期） 与穿透的区别：
· 穿透 = 数据根本不存在
· 击穿 = 数据存在，但缓存刚过期

场景：某个热门商品/活动，缓存过期瞬间
· 1000 个并发请求同时发现缓存没了
· 1000 个请求同时查库 → 数据库瞬间被打满

解决方案：加锁 + 双重检查
· 只让一个请求去查库
· 其他请求等待
· 本页实现了 Get_WithLock

另一种方案：逻辑过期——缓存永不过期，但值里带时间戳，后台异步刷新。
问题三：缓存雪崩（大量 key 同时过期） 场景：
· 服务启动时批量预热，所有 key 的 TTL 都一样
· 到点后全部同时失效
· 所有请求一起打到数据库 → 雪崩

解决方案：TTL 加随机抖动
ttl = 基础TTL + Random(-60, 60) 秒
本页实测：三个 key 的过期时间被错开。

其他方案：
· 多级缓存（本地 + Redis）
· 熔断降级（数据库压力大时直接返回兜底数据）

三、分布式锁：多台服务器如何互斥 ★★

为什么进程内的 lock 不够 场景：
· 玩家在 A 服务器和 B 服务器同时发起「扣金币」
· lock 只在单进程内有效
· 两台服务器各扣一次 → 扣了两次

Redis 分布式锁的核心：
SET key token NX EX 30
· NX——不存在才设置（原子操作）
· EX——自动过期（防止进程崩溃导致死锁）

本页实测的关键设计：
服务器 A 尝试加锁: 结果: True（锁定成功）
服务器 B 尝试加同一个锁: 结果: False（被拒绝）
服务器 B 试图释放锁（用错误的 token）: 锁还在吗: True（是的，token 不匹配）
服务器 A 释放锁: 锁还在吗: False

为什么必须校验 token：如果锁已过期被别人获取，直接 DEL 会删掉别人的锁。

四、Redis 数据结构选型

类型用途游戏场景
String简单值、计数器玩家金币、在线人数
Hash对象字段玩家属性（避免序列化整个对象）
List队列、时间线聊天记录、邮件列表
Set去重、集合运算今日活跃玩家、好友去重
ZSet排序排行榜（页 17）
Bitmap位统计签到记录（每天1位）
HyperLogLog基数统计UV 统计（内存极小）
一个典型的选型错误 错误做法：把玩家整个对象序列化成 JSON 存 String
SET player:1001 "{\"id\":1001,\"level\":50,\"gold\":9999,...}"
问题：
· 只想改金币，也要读出整个对象、改、再写回
· 并发下容易互相覆盖

正确做法：用 Hash
HSET player:1001 level 50 gold 9999
HINCRBY player:1001 gold 100 ← 原子自增，不会覆盖

原则：需要单独更新字段的地方，用 Hash 不用 String。
