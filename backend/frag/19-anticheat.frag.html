    // 3. 服务端校验的三种模式
    // ============================================================

    public class ServerValidation
    {
        // ---- 模式一：完全服务端计算（最安全）----
        public static int CalculateDamage_Server(int attack, int defense, double critRate, Random rng)
        {
            // 【关键】随机的种子在服务端
            bool crit = rng.NextDouble() < critRate;
            int damage = Math.Max(1, attack - defense);
            return crit ? damage * 2 : damage;
        }

        // ---- 模式二：客户端计算 + 服务端校验（性能优化）----
        public static (bool Ok, string Reason) ValidateDamageReport(
            int reportedDamage, int attack, int defense, int targetHp)
        {
            // 【校验1】不能超过理论上限
            int maxPossible = (attack * 3);   // 假设最高 3 倍暴击
            if (reportedDamage > maxPossible)
                return (false, $"伤害 {reportedDamage} 超过上限 {maxPossible}");

            // 【校验2】不能超过目标血量
            if (reportedDamage > targetHp)
                return (false, "伤害超过目标血量（溢出）");

            // 【校验3】不能是负数或零（除非特殊技能）
            if (reportedDamage <= 0)
                return (false, "伤害非法");

            return (true, "通过");
        }

        // ---- 模式三：抽样复核（降低开销）----
        //   只对部分操作做完整计算，其余信任客户端
        public static bool ShouldFullAudit(long playerId, int reportCount, double auditRate = 0.1)
        {
            // 【策略】高风险玩家提高抽样率
            //   本 demo 用简单的取模模拟
            return (playerId + reportCount) % (int)(1 / auditRate) == 0;
        }
    }

    // ============================================================
    // 4. 移动速度校验：最常见的外挂
    // ============================================================

    public class MovementValidator
    {
        public class Position
        {
            public double X, Y, Z;
            public DateTime Time = DateTime.UtcNow;
        }

        // 【核心】根据时间差和最大速度，判断移动是否合理
        public static (bool Ok, string Reason, double Speed) Validate(
            Position from, Position to, double maxSpeed)
        {
            double dt = (to.Time - from.Time).TotalSeconds;
            if (dt <= 0) return (false, "时间差非法", 0);

            double dist = Math.Sqrt(
                Math.Pow(to.X - from.X, 2) +
                Math.Pow(to.Y - from.Y, 2) +
                Math.Pow(to.Z - from.Z, 2));

            double speed = dist / dt;

            // 【关键】留一定容差（网络抖动、卡顿补偿）
            const double tolerance = 1.5;   // 50% 容差

            if (speed > maxSpeed * tolerance)
            {
                return (false,
                    $"速度 {speed:F1} 超过上限 {maxSpeed * tolerance:F1}",
                    speed);
            }

            return (true, "正常", speed);
        }
    }

    // 5. 异常行为检测：多层防御
    // ============================================================

    public class BehaviorDetector
    {
        private readonly Dictionary<long, PlayerBehavior> _stats = new();

        public class PlayerBehavior
        {
            public long PlayerId;
            public int TotalKills;
            public int TotalDeaths;
            public int HeadshotCount;
            public int TotalShots;
            public int TotalHits;
            public int Matches;
            public double AvgReactionMs;
        }

        public void RecordMatch(long playerId, int kills, int deaths, int headshots, int shots, int hits, double reactionMs)
        {
            if (!_stats.TryGetValue(playerId, out var s))
            {
                s = new PlayerBehavior { PlayerId = playerId };
                _stats[playerId] = s;
            }

            s.TotalKills += kills;
            s.TotalDeaths += deaths;
            s.HeadshotCount += headshots;
            s.TotalShots += shots;
            s.TotalHits += hits;
            s.Matches++;

            // 移动平均
            s.AvgReactionMs = (s.AvgReactionMs * (s.Matches - 1) + reactionMs) / s.Matches;
        }

        // 【检测1】爆头率异常
        public (bool Suspicious, string Reason) CheckHeadshotRate(long playerId)
        {
            if (!_stats.TryGetValue(playerId, out var s) || s.TotalShots < 100)
                return (false, "样本不足");

            double headshotRate = (double)s.HeadshotCount / Math.Max(1, s.TotalHits);

            // 职业选手爆头率约 40~50%，超过 80% 就很可疑
            if (headshotRate > 0.8)
                return (true, $"爆头率 {headshotRate * 100:F1}%（正常<50%）");

            return (false, $"爆头率 {headshotRate * 100:F1}%");
        }

        // 【检测2】命中率异常
        public (bool Suspicious, string Reason) CheckAccuracy(long playerId)
        {
            if (!_stats.TryGetValue(playerId, out var s) || s.TotalShots < 100)
                return (false, "样本不足");

            double accuracy = (double)s.TotalHits / s.TotalShots;

            if (accuracy > 0.9)
                return (true, $"命中率 {accuracy * 100:F1}%（正常<70%）");

            return (false, $"命中率 {accuracy * 100:F1}%");
        }

        // 【检测3】反应时间异常（自瞄的特征）
        public (bool Suspicious, string Reason) CheckReactionTime(long playerId)
        {
            if (!_stats.TryGetValue(playerId, out var s) || s.Matches < 5)
                return (false, "样本不足");

            // 人类反应极限约 150ms，职业选手约 200ms
            if (s.AvgReactionMs < 100)
                return (true, $"平均反应 {s.AvgReactionMs:F0}ms（人类极限>150ms）");

            return (false, $"平均反应 {s.AvgReactionMs:F0}ms");
        }

        // 【检测4】K/D 比异常
        public (bool Suspicious, string Reason) CheckKdRatio(long playerId)
        {
            if (!_stats.TryGetValue(playerId, out var s) || s.Matches < 10)
                return (false, "样本不足");

            double kd = s.TotalDeaths == 0 ? s.TotalKills : (double)s.TotalKills / s.TotalDeaths;

            // 顶级选手约 3~5
            if (kd > 10)
                return (true, $"K/D {kd:F1}（正常<5）");

            return (false, $"K/D {kd:F1}");
        }

        public PlayerBehavior? GetStats(long playerId) =>
            _stats.TryGetValue(playerId, out var s) ? s : null;
    }

    public static class Punishment
    {
        public static readonly (string Level, string Action, string When)[] All =
        {
            ("观察",   "标记，不干预",     "轻度可疑"),
            ("限制",   "降低匹配优先级",   "中度可疑"),
            ("隔离",   "进作弊者匹配池",   "高度可疑"),
            ("封禁",   "禁止登录",         "确认作弊"),
            ("回滚",   "撤销非法收益",     "确认刷道具"),
        };

        // 【关键】不能自动封禁，要人工复核
        public static string Decide(int suspicionScore) => suspicionScore switch
        {
            < 30 => "正常，继续观察",
            < 60 => "标记，加入监控列表",
            < 80 => "限制匹配，人工复核",
            _ => "紧急：暂停账号，立即人工审核",
        };

        public static void Print()
        {
            foreach (var (level, action, when) in All)
                Console.WriteLine($"    {level,-6} {action,-16} {when}");
        }
    }

二、什么必须服务端算 ★★★

判断标准：改了会怎样 问自己：如果玩家能任意修改这个结果，会怎样？

逻辑放哪改了会怎样
伤害计算服务端秒杀一切
掉落判定服务端必出稀有
金币变动服务端无限金币
技能冷却服务端无 CD
移动速度服务端校验加速挂
界面/特效客户端只影响自己
一句话：所有影响其他玩家或经济系统的，都必须在服务端。

三、三种校验模式

模式一：完全服务端计算（最安全） 做法：客户端只发操作意图，服务端算一切。

优点：无法作弊
缺点：服务器压力大（要跑所有逻辑）

本页实现：
CalculateDamage_Server(attack, defense, critRate, rng)
· 随机数种子在服务端
· 客户端拿不到真正的暴击判定

适用：经济相关、竞技对战
模式二：客户端算 + 服务端校验 做法：客户端算，服务端检查是否合理。

本页实测的三条校验：
正常伤害（攻击 100，防御 20）: 服务端计算 80，校验通过
作弊客户端上报 99999: 拒绝，伤害 99999 超过上限 300
上报伤害超过目标血量: 拒绝，伤害 5000 超过上限 300

校验逻辑：
· 不能超过理论上限（攻击×3）
· 不能超过目标血量
· 不能是负数/零

适用：非经济、性能敏感的场景
模式三：抽样复核 做法：只对部分操作做完整计算。
· 普通玩家：10% 抽样
· 可疑玩家：100% 审核

优点：兼顾性能与安全

关键：让玩家不知道什么时候会被查——威慑效果。

四、移动速度校验：最常见的外挂 ★★

本页实测的三种情况 校验原理：速度 = 距离 / 时间差
超过最大速度的一定倍数 → 异常

本页实测：
正常移动（1秒走5米）: True, 速度 5.0
加速挂（1秒走50米）: False, 速度 49.8 超过上限 9.0
瞬移（时间差0）: False, 时间差非法

为什么要留容差：
· 网络抖动——包延迟导致时间差不准
· 客户端卡顿——累积了移动距离后一次性上报
· 本页用 1.5 倍容差（即在 5m/s 上限下，超过 7.5m/s 才判异常）

但容差不能太大——否则外挂可以利用。

五、行为异常检测：多指标交叉 ★★★

四个检测指标（本页实测） 本页对比了正常玩家和疑似外挂：

指标正常外挂阈值
爆头率3.3%23.2%>20%
命中率30%95%>90%
反应时间280ms45ms<150ms（人类极限）
K/D0.625.0>5
关键原则：单一指标异常不足以定性
· 手感好的玩家可能爆头率高
· 但要四个指标同时异常 → 基本确定

人类极限参考：
· 反应时间 > 150ms（神经传导极限）
· 职业选手爆头率 40~50%
· 职业选手命中率 40~60%
处置策略：不要自动封号 本页的处置等级：
可疑度 20: 正常，继续观察
可疑度 45: 标记，加入监控
可疑度 70: 限制匹配，人工复核
可疑度 90: 暂停账号，立即审核

铁律：
· 误封的代价 > 放过一个作弊者
· 高可疑度先暂停，不要直接封
· 必须有申诉渠道
· 封号要走人工审核
