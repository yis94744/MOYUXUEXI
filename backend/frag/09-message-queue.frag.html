一、消息队列解决什么

using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;

namespace BackendSamples.MessageQueue
{
    // ============================================================
    // 1. 为什么需要消息队列
    // ============================================================
    // 【场景】玩家击杀 BOSS，要做 5 件事：
    //   ① 发奖励到背包  ② 更新任务进度  ③ 写战斗日志
    //   ④ 更新排行榜    ⑤ 发邮件通知
    //
    // 【同步做法的问题】
    //   · 总耗时 = 5 件事之和（玩家等 500ms）
    //   · 任何一件失败 -> 整个流程失败
    //   · 无法应对突发流量

    public class Message
    {
        public string Id = Guid.NewGuid().ToString("N");
        public string Topic = "";
        public string Payload = "";
        public int RetryCount;
    }

    // ============================================================
    // 2. 三种投递语义
    // ============================================================

    public enum DeliverySemantic
    {
        AtMostOnce,    // 最多一次：可能丢，但不会重复
        AtLeastOnce,   // 至少一次：不会丢，但可能重复【最常用】
        ExactlyOnce,   // 恰好一次：理论上最优，但实现代价极高
    }

    public static class SemanticsExplainer
    {
        public static string Describe(DeliverySemantic s) => s switch
        {
            DeliverySemantic.AtMostOnce => "可能丢、不重复（适合日志、埋点）",
            DeliverySemantic.AtLeastOnce => "不丢、可能重复（适合业务消息，需要幂等消费）",
            DeliverySemantic.ExactlyOnce => "不丢不重（实现复杂，通常靠幂等模拟）",
            _ => ""
        };
    }

    // ============================================================
    // 3. 简化版消息队列（演示核心机制）
    // ============================================================

    public class SimpleQueue
    {
        private readonly Queue<Message> _queue = new();
        private readonly List<Message> _deadLetter = new();       // 死信队列
        private readonly Dictionary<string, int> _consumed = new(); // 幂等：已消费的消息 ID
        private readonly int _maxRetry;
        private readonly object _sync = new();

        public int Delivered, Consumed, Duplicated, Failed;

        public SimpleQueue(int maxRetry = 3) => _maxRetry = maxRetry;

        public void Publish(Message msg)
        {
            lock (_sync) { _queue.Enqueue(msg); Delivered++; }
        }

        public Message? Consume()
        {
            lock (_sync) { return _queue.Count > 0 ? _queue.Dequeue() : null; }
        }

        // 【关键】幂等消费：同一条消息处理多次也只生效一次
        public bool Process(Message msg, Func<Message, bool> handler)
        {
            // 幂等检查
            if (_consumed.ContainsKey(msg.Id))
            {
                Duplicated++;
                return true;   // 【重要】重复消息返回成功，不是失败
            }

            try
            {
                if (handler(msg))
                {
                    _consumed[msg.Id] = 1;
                    Consumed++;
                    return true;
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"      处理异常: {ex.Message}");
            }

            // 失败：重试 or 进死信
            msg.RetryCount++;
            if (msg.RetryCount >= _maxRetry)
            {
                _deadLetter.Add(msg);
                Failed++;
                Console.WriteLine($"      进入死信队列（重试 {msg.RetryCount} 次）");
            }
            else
            {
                Publish(msg);
                Console.WriteLine($"      重新入队（第 {msg.RetryCount} 次重试）");
            }

            return false;
        }

        public int QueueCount { get { lock (_sync) return _queue.Count; } }
        public int DeadLetterCount { get { lock (_sync) return _deadLetter.Count; } }
        public IReadOnlyList<Message> DeadLetters { get { lock (_sync) return _deadLetter.ToList(); } }
    }

    // ============================================================
    // 4. 削峰填谷
    // ============================================================
    // 【场景】玩家同时在线 10 万，同时发消息
    //   如果直接处理 -> 数据库瞬间被打爆
    //   用队列缓冲 -> 按数据库能承受的速度消费

    public class RateLimitedConsumer
    {
        private readonly int _capacityPerTick;

        public RateLimitedConsumer(int capacityPerTick) => _capacityPerTick = capacityPerTick;

        public int ProcessBatch(SimpleQueue queue, Func<Message, bool> handler)
        {
            int processed = 0;
            while (processed < _capacityPerTick)
            {
                var msg = queue.Consume();
                if (msg == null) break;
                queue.Process(msg, handler);
                processed++;
            }
            return processed;
        }
    }

    // ============================================================
    // 5. 最终一致性：不是「马上一致」，但「最终会一致」
    // ============================================================
    // 【场景】扣金币 + 发道具
    //   如果用队列异步发道具，会出现「金币扣了，道具还没到」
    //   这叫「最终一致性」——短暂不一致是可接受的

    public class InventoryEvent
    {
        public long PlayerId;
        public int ItemId;
        public int Count;
    }

    public class OrderService
    {
        private readonly SimpleQueue _queue;
        private int _gold = 1000;
        private readonly List<string> _inventory = new();

        public OrderService(SimpleQueue queue) => _queue = queue;

        public bool BuyItem(long playerId, int itemId, int price)
        {
            // ① 同步扣金币（必须立即准确）
            if (_gold < price)
            {
                Console.WriteLine($"    金币不足: 有 {_gold}, 需要 {price}");
                return false;
            }
            _gold -= price;
            Console.WriteLine($"    扣金币 {price}，剩余 {_gold}");

            // ② 异步发道具（可以延迟，但不能丢）
            _queue.Publish(new Message
            {
                Topic = "item.grant",
                Payload = $"{playerId}|{itemId}"
            });
            Console.WriteLine($"    已投递发道具消息（异步）");

            return true;
        }

        // 消费端：把道具放进背包
        public void OnItemGrant(Message msg)
        {
            var parts = msg.Payload.Split('|');
            _inventory.Add($"item{parts[1]}");
            Console.WriteLine($"    [消费] 发放道具 {parts[1]} 给玩家 {parts[0]}");
        }

        public int Gold => _gold;
        public int ItemCount => _inventory.Count;
    }

    // ============================================================
    // 6. 消息队列选型
    // ============================================================

    public static class MqComparison
    {
        public static readonly (string Name, string Strength, string GameUse)[] All =
        {
            ("RabbitMQ", "路由灵活、可靠", "业务消息（奖励发放、邮件）"),
            ("Kafka", "超高吞吐、可回溯", "日志采集、埋点、数据管道"),
            ("Redis Pub/Sub", "极轻量、无持久化", "在线广播（聊天、公告）"),
            ("Redis Stream", "轻量 + 持久化", "中等规模业务消息"),
            ("RocketMQ", "事务消息", "订单、支付场景"),
        };
    }

    // ============================================================
    // 演示
    // ============================================================
    public static class Demo
    {
        public static void Run()
        {
            Console.WriteLine("=== 1. 投递语义 ===");
            foreach (var s in Enum.GetValues<DeliverySemantic>())
            {
                Console.WriteLine($"    {s,-14} {SemanticsExplainer.Describe(s)}");
            }
            Console.WriteLine();
            Console.WriteLine("  【实战结论】业务消息用 AtLeastOnce + 幂等消费");
            Console.WriteLine("              不要为了 ExactlyOnce 引入复杂度");

            Console.WriteLine();
            Console.WriteLine("=== 2. 幂等消费 ===");
            var q = new SimpleQueue();
            int sideEffectCount = 0;

            var msg = new Message { Topic = "reward", Payload = "player1001:+100gold" };
            q.Publish(msg);

            Console.WriteLine("  同一条消息投递 3 次（模拟重复）:");
            for (int i = 0; i < 3; i++)
            {
                var m = new Message { Id = msg.Id, Topic = msg.Topic, Payload = msg.Payload };
                q.Process(m, _ => { sideEffectCount++; return true; });
            }
            Console.WriteLine($"    实际执行副作用: {sideEffectCount} 次（只 1 次）");
            Console.WriteLine($"    重复消息: {q.Duplicated} 条");
            Console.WriteLine("  【关键】重复消息返回成功，不是失败");

            Console.WriteLine();
            Console.WriteLine("=== 3. 失败重试与死信队列 ===");
            var q2 = new SimpleQueue(maxRetry: 3);
            q2.Publish(new Message { Topic = "broken", Payload = "always-fails" });

            Console.WriteLine("  模拟一条总是失败的消息:");
            for (int i = 0; i < 5; i++)
            {
                var m = q2.Consume();
                if (m == null) break;
                Console.WriteLine($"    第 {i + 1} 次尝试:");
                q2.Process(m, _ => throw new Exception("模拟处理失败"));
            }
            Console.WriteLine($"    死信队列: {q2.DeadLetterCount} 条");
            Console.WriteLine("  【关键】不能无限重试，要进死信队列人工处理");

            Console.WriteLine();
            Console.WriteLine("=== 4. 削峰填谷 ===");
            var q3 = new SimpleQueue();
            int processedCount = 0;

            Console.WriteLine("  瞬间灌入 100 条消息:");
            for (int i = 0; i < 100; i++)
                q3.Publish(new Message { Topic = "burst", Payload = $"msg{i}" });
            Console.WriteLine($"    队列积压: {q3.QueueCount} 条");

            var limiter = new RateLimitedConsumer(20);
            Console.WriteLine("  每批次最多处理 20 条:");
            for (int round = 0; round < 6; round++)
            {
                int n = limiter.ProcessBatch(q3, _ => { processedCount++; return true; });
                Console.WriteLine($"    第 {round + 1} 轮: 处理 {n} 条，剩余 {q3.QueueCount} 条");
                if (q3.QueueCount == 0) break;
            }
            Console.WriteLine($"  共处理 {processedCount} 条");
            Console.WriteLine("  【关键】数据库压力被摊平到多轮");

            Console.WriteLine();
            Console.WriteLine("=== 5. 最终一致性 ===");
            var q4 = new SimpleQueue();
            var order = new OrderService(q4);

            Console.WriteLine("  玩家购买道具:");
            order.BuyItem(1001, 5001, 300);
            Console.WriteLine($"    当前状态: 金币={order.Gold}, 背包道具={order.ItemCount}");
            Console.WriteLine("    【注意】金币已扣，但道具还没到（最终一致性的中间状态）");

            Console.WriteLine();
            Console.WriteLine("  消费端处理消息:");
            while (true)
            {
                var m = q4.Consume();
                if (m == null) break;
                q4.Process(m, msg => { order.OnItemGrant(msg); return true; });
            }
            Console.WriteLine($"    最终状态: 金币={order.Gold}, 背包道具={order.ItemCount}");
            Console.WriteLine("  【关键】短暂不一致可接受，但要保证「最终一致」");
            Console.WriteLine("          所以消息不能丢（AtLeastOnce）");

            Console.WriteLine();
            Console.WriteLine("=== 6. MQ 选型 ===");
            foreach (var (name, strength, use) in MqComparison.All)
            {
                Console.WriteLine($"    {name,-16} {strength,-22} {use}");
            }
        }
    }
}

问题同步做法用队列
耦合击杀 BOSS 要调 5 个服务发一条消息，谁需要谁订阅
慢等 5 件事全做完（500ms）立即返回，异步处理
峰值10 万请求同时打数据库队列缓冲，按能力消费

二、投递语义：为什么选 AtLeastOnce ★★★

三种语义的取舍
语义会不会丢会不会重复适用
AtMostOnce可能丢不会日志、埋点
AtLeastOnce不丢可能重复业务消息（主流）
ExactlyOnce不丢不重代价极高
为什么主流选 AtLeastOnce：
· 「丢数据」是不可接受的（玩家买了道具没收到）
· 「重复」是可以解决的（幂等消费）
· ExactlyOnce 需要分布式事务，复杂度和性能代价太高

结论：选 AtLeastOnce + 幂等消费，这是工业界的标准答案。

三、幂等消费：重复消息只生效一次 ★★★

本页实测的幂等实现 核心：记录已处理的消息 ID。
if (_consumed.ContainsKey(msg.Id)) return true;

本页实测输出：
同一条消息投递 3 次
实际执行副作用: 1 次（只 1 次）
重复消息: 2 条

关键细节：重复消息要返回成功
· 返回失败 → 消息队列会继续重投 → 死循环
· 返回成功 → 队列认为处理完毕 → 停止重投

消息 ID 从哪来：
· 生产端生成（推荐）
· 或消费端用业务唯一键（如订单号）

四、重试与死信队列 ★★

不能无限重试 问题：一条处理不了的消息被无限重投
· 占着队列 → 后面的消息被阻塞
· 不断重试 → 日志刷屏、CPU 空转

本页实测：
第 1 次尝试: 处理异常
重新入队（第 1 次重试）
...
第 3 次尝试: 进入死信队列

死信队列的作用：
· 把失败消息隔离出来
· 不阻塞正常消息
· 人工排查后可以重放

重试策略：
· 指数退避（1s、2s、4s、8s）
· 不要立即重试（如果是依赖服务挂了，立刻重试也没用）

五、最终一致性：短暂不一致是可接受的

本页实测的「金币扣了，道具还没到」 流程：
① 同步扣金币（必须立即准确）
② 异步发道具（可以延迟，但不能丢）

本页实测输出：
扣金币 300，剩余 700
已投递发道具消息（异步）
当前状态: 金币=700, 背包道具=0 ← 中间状态

[消费] 发放道具 5001 给玩家 1001
最终状态: 金币=700, 背包道具=1 ← 最终一致

设计要点：
· 钱相关必须同步（扣了就是扣了）
· 发东西可以异步（晚几百毫秒玩家能接受）
· 消息不能丢（所以用 AtLeastOnce）
· 超时未到账要有补偿查询机制

六、MQ 选型

MQ强项游戏场景
RabbitMQ路由灵活、可靠业务消息（奖励、邮件）
Kafka超高吞吐、可回溯日志、埋点、数据管道
Redis Pub/Sub极轻量、无持久化在线广播（聊天、公告）
Redis Stream轻量 + 持久化中等规模业务消息
RocketMQ事务消息订单、支付
Pub/Sub 不能用于业务消息 Redis Pub/Sub 的致命问题：没有持久化。
· 订阅者掉线期间的消息 → 永久丢失
· 没有 ACK 机制
· 没有重投

只适合：在线聊天广播、系统公告——丢一条无所谓。

业务消息要用：RabbitMQ / Kafka / Redis Stream。
判断标准：这条消息丢了，玩家会不会投诉？
