一、密码存储：绝对不能明文 ★★★

密码存储只有一个正确答案：慢哈希 + 随机盐。

using System;
using System.Collections.Generic;
using System.Linq;
using System.Security.Cryptography;
using System.Text;

namespace BackendSamples.Auth
{
    // ============================================================
    // 1. 密码存储：绝对不能明文
    // ============================================================
    // 【为什么不能存 MD5/SHA1】
    //   彩虹表可以秒破。加盐只能拖延几分钟。

    public static class PasswordHasher
    {
        // 【正确做法】用慢哈希（PBKDF2/BCrypt/Argon2）
        //   故意让计算变慢 -> 暴力破解成本剧增
        public static string Hash(string password, byte[] salt, int iterations = 100000)
        {
            using var pbkdf2 = new Rfc2898DeriveBytes(password, salt, iterations, HashAlgorithmName.SHA256);
            var hash = pbkdf2.GetBytes(32);
            return Convert.ToBase64String(hash);
        }

        public static byte[] GenerateSalt()
        {
            var salt = new byte[16];
            using var rng = RandomNumberGenerator.Create();
            rng.GetBytes(salt);
            return salt;
        }

        public static bool Verify(string password, string storedHash, byte[] salt, int iterations = 100000)
        {
            var computed = Hash(password, salt, iterations);
            // 【安全】用固定时间比较，防止时序攻击
            return FixedTimeEquals(computed, storedHash);
        }

        private static bool FixedTimeEquals(string a, string b)
        {
            if (a.Length != b.Length) return false;
            int diff = 0;
            for (int i = 0; i < a.Length; i++) diff |= a[i] ^ b[i];
            return diff == 0;
        }
    }

    // ============================================================
    // 2. JWT 结构：三段式
    // ============================================================
    // header.payload.signature

    public static class JwtExplainer
    {
        public static readonly (string Part, string Content, string Note)[] Parts =
        {
            ("Header", "{\"alg\":\"HS256\",\"typ\":\"JWT\"}", "算法（公开）"),
            ("Payload", "{\"sub\":\"1001\",\"exp\":1700000000}", "【不能放敏感信息】Base64 只是编码不是加密"),
            ("Signature", "HMAC(header.payload, secret)", "签名，防篡改"),
        };
    }

    // 简化版 JWT 实现（演示原理）
    public class SimpleJwt
    {
        private readonly string _secret;

        public SimpleJwt(string secret) => _secret = secret;

        public string Create(Dictionary<string, object> payload)
        {
            var header = Base64Url("{\"alg\":\"HS256\",\"typ\":\"JWT\"}");
            var body = Base64Url(Serialize(payload));
            var sig = Base64Url(Sign(header + "." + body));
            return header + "." + body + "." + sig;
        }

        public Dictionary<string, object>? Validate(string token)
        {
            var parts = token.Split('.');
            if (parts.Length != 3) return null;

            // 【关键】校验签名，防止篡改
            var expected = Base64Url(Sign(parts[0] + "." + parts[1]));
            if (expected != parts[2])
            {
                Console.WriteLine("    签名校验失败：token 被篡改");
                return null;
            }

            var payload = Deserialize(FromBase64Url(parts[1]));

            // 检查过期
            if (payload.TryGetValue("exp", out var expObj) && expObj is long exp)
            {
                if (DateTimeOffset.UtcNow.ToUnixTimeSeconds() > exp)
                {
                    Console.WriteLine("    token 已过期");
                    return null;
                }
            }

            return payload;
        }

        private string Sign(string data)
        {
            using var hmac = new HMACSHA256(Encoding.UTF8.GetBytes(_secret));
            return Convert.ToBase64String(hmac.ComputeHash(Encoding.UTF8.GetBytes(data)));
        }

        private static string Base64Url(string s) =>
            Convert.ToBase64String(Encoding.UTF8.GetBytes(s))
                .Replace('+', '-').Replace('/', '_').TrimEnd('=');

        private static string FromBase64Url(string s)
        {
            s = s.Replace('-', '+').Replace('_', '/');
            switch (s.Length % 4)
            {
                case 2: s += "=="; break;
                case 3: s += "="; break;
            }
            return Encoding.UTF8.GetString(Convert.FromBase64String(s));
        }

        // 极简 JSON（演示用，生产用 System.Text.Json）
        private static string Serialize(Dictionary<string, object> d) =>
            "{" + string.Join(",", d.Select(kv =>
                $"\"{kv.Key}\":{(kv.Value is string s ? "\"" + s + "\"" : kv.Value.ToString())}")) + "}";

        private static Dictionary<string, object> Deserialize(string json)
        {
            var result = new Dictionary<string, object>();
            json = json.Trim('{', '}');
            foreach (var pair in json.Split(','))
            {
                var idx = pair.IndexOf(':');
                if (idx < 0) continue;
                var k = pair.Substring(0, idx).Trim('"', ' ', '\n');
                var v = pair.Substring(idx + 1).Trim(' ', '\n');
                if (v.StartsWith("\"") && v.EndsWith("\""))
                    result[k] = v.Trim('"');
                else if (long.TryParse(v, out long n))
                    result[k] = n;
                else
                    result[k] = v;
            }
            return result;
        }
    }

    // ============================================================
    // 3. Token 刷新机制
    // ============================================================
    // 【问题】access token 有效期长 -> 泄露后危害大
    //         有效期短 -> 用户频繁掉线
    // 【方案】双 token

    public class TokenPair
    {
        public string AccessToken = "";
        public string RefreshToken = "";
        public long AccessExpireAt;
        public long RefreshExpireAt;
    }

    public class TokenService
    {
        private readonly SimpleJwt _jwt = new("my-secret-key");
        private readonly Dictionary<string, string> _refreshTokens = new();   // refreshToken -> userId

        // access 短（15分钟），refresh 长（7天）
        public TokenPair Issue(long userId)
        {
            long now = DateTimeOffset.UtcNow.ToUnixTimeSeconds();

            var access = _jwt.Create(new Dictionary<string, object>
            {
                ["sub"] = userId.ToString(),
                ["exp"] = now + 15 * 60,
            });

            var refresh = Guid.NewGuid().ToString("N");
            _refreshTokens[refresh] = userId.ToString();

            return new TokenPair
            {
                AccessToken = access,
                RefreshToken = refresh,
                AccessExpireAt = now + 15 * 60,
                RefreshExpireAt = now + 7 * 24 * 3600,
            };
        }

        // 用 refresh 换新的 access
        public TokenPair? Refresh(string refreshToken)
        {
            if (!_refreshTokens.TryGetValue(refreshToken, out var userId))
            {
                Console.WriteLine("    refresh token 无效");
                return null;
            }

            // 【安全】refresh token 一次性使用（旋转）
            _refreshTokens.Remove(refreshToken);

            return Issue(long.Parse(userId));
        }

        public long? ValidateAccess(string token)
        {
            var payload = _jwt.Validate(token);
            if (payload == null) return null;
            if (!payload.TryGetValue("sub", out var sub)) return null;
            return long.TryParse(sub.ToString(), out long id) ? id : null;
        }
    }

    // ============================================================
    // 4. 越权：最常见的安全漏洞
    // ============================================================
    // 水平越权：用户 A 改了 URL 里的 id，看到了用户 B 的数据
    // 垂直越权：普通用户调用了管理员接口

    public class OrderService
    {
        private readonly Dictionary<long, (long OwnerId, string Data)> _orders = new()
        {
            [1] = (1001, "订单1的内容"),
            [2] = (1002, "订单2的内容"),
        };

        // 【错误示范】只校验登录，不校验归属
        public string? GetOrder_Bad(long orderId)
        {
            return _orders.TryGetValue(orderId, out var o) ? o.Data : null;
        }

        // 【正确做法】必须校验「这个资源是不是你的」
        public string? GetOrder_Good(long orderId, long currentUserId)
        {
            if (!_orders.TryGetValue(orderId, out var o)) return null;

            if (o.OwnerId != currentUserId)
            {
                Console.WriteLine($"    越权拦截: 用户 {currentUserId} 试图访问用户 {o.OwnerId} 的订单");
                return null;
            }

            return o.Data;
        }
    }

    public static class Demo
    {
        public static void Run()
        {
            Console.WriteLine("=== 1. 密码哈希 ===");
            var salt = PasswordHasher.GenerateSalt();
            var hash = PasswordHasher.Hash("my-password-123", salt);
            Console.WriteLine($"  原始密码: my-password-123");
            Console.WriteLine($"  盐（16字节）: {Convert.ToBase64String(salt).Substring(0, 20)}...");
            Console.WriteLine($"  PBKDF2 哈希: {hash.Substring(0, 30)}...");
            Console.WriteLine($"  验证正确密码: {PasswordHasher.Verify("my-password-123", hash, salt)}");
            Console.WriteLine($"  验证错误密码: {PasswordHasher.Verify("wrong-password", hash, salt)}");
            Console.WriteLine("  【关键】慢哈希（10万次迭代）让暴力破解成本剧增");

            Console.WriteLine();
            Console.WriteLine("=== 2. JWT 结构 ===");
            foreach (var (p, c, n) in JwtExplainer.Parts)
            {
                Console.WriteLine($"  {p,-12} {c,-40} {n}");
            }
            Console.WriteLine();
            Console.WriteLine("  【重要】Payload 只是 Base64 编码，不是加密！");
            Console.WriteLine("          任何人都能解码看到内容 —— 不要放密码/身份证");

            Console.WriteLine();
            Console.WriteLine("=== 3. 双 Token 机制 ===");
            var svc = new TokenService();
            var pair = svc.Issue(1001);
            Console.WriteLine($"  登录成功:");
            Console.WriteLine($"    access token:  {pair.AccessToken.Substring(0, 40)}...");
            Console.WriteLine($"    refresh token: {pair.RefreshToken.Substring(0, 16)}...");
            Console.WriteLine($"    access 过期:   15 分钟后");
            Console.WriteLine($"    refresh 过期:  7 天后");

            Console.WriteLine();
            Console.WriteLine("  验证 access token:");
            var uid = svc.ValidateAccess(pair.AccessToken);
            Console.WriteLine($"    解析出 userId = {uid}");

            Console.WriteLine();
            Console.WriteLine("  刷新 token（一次性）:");
            var newPair = svc.Refresh(pair.RefreshToken);
            Console.WriteLine($"    新 access 签发: {newPair != null}");

            Console.WriteLine("  用旧的 refresh 再试一次:");
            var again = svc.Refresh(pair.RefreshToken);
            Console.WriteLine($"    结果: {(again == null ? "失败（已被使用）" : "成功")}");

            Console.WriteLine();
            Console.WriteLine("  篡改 token:");
            var tampered = pair.AccessToken.Substring(0, pair.AccessToken.Length - 3) + "xyz";
            svc.ValidateAccess(tampered);

            Console.WriteLine();
            Console.WriteLine("=== 4. 越权防护 ===");
            var orders = new OrderService();

            Console.WriteLine("  用户 1001 访问自己的订单 1:");
            Console.WriteLine($"    {orders.GetOrder_Good(1, 1001) ?? "拒绝"}");

            Console.WriteLine();
            Console.WriteLine("  用户 1001 试图访问别人（1002）的订单 2:");
            var bad = orders.GetOrder_Good(2, 1001);
            Console.WriteLine($"    结果: {bad ?? "拒绝"}");

            Console.WriteLine();
            Console.WriteLine("  【关键】只校验「登录状态」不够");
            Console.WriteLine("          必须校验「资源归属」—— 这就是水平越权");
        }
    }

方式安全性说明
明文灾难脱库即全部泄露
MD5 / SHA1很危险彩虹表秒破
MD5 + 盐勉强GPU 暴力破解仍很快
PBKDF2 / BCrypt / Argon2正确故意慢，抬高破解成本

二、JWT：三段式结构 ★★

Payload 不是加密，只是编码 这是最常见的误解。

JWT 的 Payload 只是 Base64Url 编码——任何人都能解码。
打开 jwt.io，粘贴 token，立刻看到全部内容。

所以绝对不能放：
· 密码、身份证号、手机号
· 内部业务数据
· 任何敏感信息

应该只放：
· sub（用户 ID）
· exp（过期时间）
· 角色/权限标识

本页输出验证：Payload {\"sub\":\"1001\",\"exp\":...} 全部是明文可读的。
签名的作用：防篡改 签名 = HMAC(header.payload, secret)

如果有人改了 payload 里的 userId：
· 签名就对不上了
· 服务端校验失败 → 拒绝

本页实测验证：
篡改 token: 签名校验失败：token 被篡改

但要小心：secret 泄露了，签名就形同虚设。
· secret 不能写在代码里
· 用足够长的随机字符串（32 字节以上）
· 定期轮换

三、双 Token 机制 ★★★

为什么需要两个 token 单 token 的困境：
· 有效期长 → 泄露后危害大（可以一直用）
· 有效期短 → 用户频繁掉线，体验差

双 token 的解法：
Token有效期存放位置作用
access15 分钟内存访问接口
refresh7 天HttpOnly Cookie换新的 access
流程：
① 登录 → 拿到两个 token
② 每次请求带 access
③ access 过期（401）→ 用 refresh 换新的
④ refresh 也过期 → 重新登录

本页实测的关键设计：refresh 是一次性的（旋转）。
用旧 refresh 再试: 失败（已被使用）
这样即使 refresh 泄露，攻击者用过一次后你也会发现。

四、越权：最常见的安全漏洞 ★★★

水平越权 vs 垂直越权 水平越权（最常见）：
用户 A 把 URL 里的 orderId 改成 B 的订单号 → 看到了 B 的订单
GET /api/orders/2（但 2 号订单属于别人）

垂直越权：
普通用户直接调用管理员接口
POST /api/admin/grant-gold

本页实测验证：
越权拦截: 用户 1001 试图访问用户 1002 的订单

根本原因：只校验了「登录了没」，没校验「这个资源是不是你的」。

防御原则：
· 每个涉及资源 ID 的接口，都要校验归属
· 或者在 SQL 里直接带条件：WHERE id = ? AND owner_id = ?
· 不要相信客户端传来的任何 ID
