12 · 长连接与短连接 ★★★

游戏后端必须用长连接——因为要服务端主动推送。但长连接带来三个必须解决的问题：粘包、心跳、断线重连。

    public static class ConnectionComparison
    {
        public static readonly (string Aspect, string Short, string Long)[] All =
        {
            ("连接建立", "每次请求都要握手", "只握手一次"),
            ("开销", "TCP+TLS 三次握手/四次挥手", "几乎没有"),
            ("服务端资源", "无状态，随便扩", "每连接占内存/文件句柄"),
            ("推送能力", "做不到（服务端无法主动发）", "天然支持"),

    // ============================================================
    // 2. 粘包与拆包：长连接的第一个坑 ★★★
    // ============================================================
    // 【为什么会粘包】
    //   TCP 是「字节流」，不是「消息流」
    //   发送方 write("AAA") + write("BBB") -> 接收方可能一次读到 "AAABBB"
    //   也可能读到 "AA" 然后 "ABBB"
    //
    // 【解决方案】在协议里加「长度前缀」

    public class LengthPrefixedCodec
    {
        private readonly List<byte> _buffer = new();

        // 编码：4字节大端长度 + 内容
        public static byte[] Encode(string message)
        {
            var payload = Encoding.UTF8.GetBytes(message);
            var result = new byte[4 + payload.Length];
            // 【关键】大端序（网络字节序）
            result[0] = (byte)(payload.Length >> 24);
            result[1] = (byte)(payload.Length >> 16);
            result[2] = (byte)(payload.Length >> 8);
            result[3] = (byte)payload.Length;
            Array.Copy(payload, 0, result, 4, payload.Length);
            return result;
        }

        // 解码：处理粘包/半包
        public List<string> Feed(byte[] chunk)
        {
            _buffer.AddRange(chunk);

            var messages = new List<string>();

            while (true)
            {
                // 长度头都不完整
                if (_buffer.Count < 4) break;

                int len = (_buffer[0] << 24) | (_buffer[1] << 16) | (_buffer[2] << 8) | _buffer[3];

                // 【关键】内容还没收全（半包）
                if (_buffer.Count < 4 + len) break;

                var payload = _buffer.GetRange(4, len).ToArray();
                messages.Add(Encoding.UTF8.GetString(payload));

                // 移除已消费的部分
                _buffer.RemoveRange(0, 4 + len);
            }

            return messages;
        }

        public int PendingBytes => _buffer.Count;
    }

    // ============================================================
    // 3. 心跳与超时检测
    // ============================================================
    // 【为什么需要心跳】
    //   TCP 连接「看起来」是连着，但对端可能已经断电/断网
    //   操作系统不会立刻发现（要等 TCP keepalive，默认2小时）

    public class Connection
    {
        public string Id = "";
        public long PlayerId;
        public DateTime LastActive;
        public int MissedHeartbeats;
        public bool Alive = true;

        // 心跳配置
        public const int HeartbeatIntervalSec = 10;   // 客户端每 10 秒发一次
        public const int TimeoutSec = 30;             // 30 秒没动静就判死

        public override string ToString() =>
            $"连接 {Id} 玩家{PlayerId} 最后活跃{(DateTime.UtcNow - LastActive).TotalSeconds:F0}秒前";
    }

    public class HeartbeatMonitor
    {
        private readonly Dictionary<string, Connection> _conns = new();
        private int _checkedCount;
        private int _kickedCount;

        public void Add(Connection c, DateTime now)
        {
            c.LastActive = now;
            _conns[c.Id] = c;
        }

        // 【关键】演示里时间是被「手动推进」的，所以必须把当前时间传进来
        //   否则 OnMessage 写的是 DateTime.UtcNow，而 CheckTimeouts 用的是模拟时间
        //   -> 两者对不上，所有连接都会被误判超时
        public void OnMessage(string connId, DateTime now)
        {
            if (_conns.TryGetValue(connId, out var c))
            {
                c.LastActive = now;
                c.MissedHeartbeats = 0;
            }
        }

        // 定时检测（每 5 秒跑一次）
        public List<Connection> CheckTimeouts(DateTime now)
        {
            var kicked = new List<Connection>();
            _checkedCount++;

            foreach (var c in _conns.Values.ToList())
            {
                if (!c.Alive) continue;

                double idle = (now - c.LastActive).TotalSeconds;

                // 【关键】超过阈值就踢
                if (idle > Connection.TimeoutSec)
                {
                    c.Alive = false;
                    _conns.Remove(c.Id);
                    kicked.Add(c);
                    _kickedCount++;
                }
            }

            return kicked;
        }

        public int AliveCount => _conns.Count;
        public int CheckedCount => _checkedCount;
        public int KickedCount => _kickedCount;
    }

    // ============================================================
    // 4. 断线重连
    // ============================================================
    // 【核心问题】玩家断线 30 秒回来后，这 30 秒发生了什么？
    //   ① 战斗还在继续 -> 需要补发状态
    //   ② 队友以为他掉线了 -> 需要通知
    //   ③ 服务端保留了会话 -> 可以直接恢复

    public class SessionRecovery
    {
        // 会话保留时间（超过就真的下线）
        public const int SessionKeepSec = 120;

        public class Session
        {
            public long PlayerId;
            public string Token = "";
            public DateTime DisconnectedAt;
            public bool InBattle;
            public List<string> MissedMessages = new();
        }

        private readonly Dictionary<long, Session> _sessions = new();

        public void OnDisconnect(long playerId, bool inBattle)
        {
            _sessions[playerId] = new Session
            {
                PlayerId = playerId,
                DisconnectedAt = DateTime.UtcNow,
                InBattle = inBattle,
            };
            Console.WriteLine($"    玩家 {playerId} 断线，会话保留 {SessionKeepSec} 秒");
            if (inBattle)
                Console.WriteLine($"    【重要】战斗中：其他玩家会看到" + "「掉线中」" + "状态");
        }

        public void BufferMessage(long playerId, string msg)
        {
            if (_sessions.TryGetValue(playerId, out var s))
                s.MissedMessages.Add(msg);
        }

        // 重连时调用
        public (bool CanRecover, Session? Session, List<string> Missed) OnReconnect(long playerId, DateTime now)
        {
            if (!_sessions.TryGetValue(playerId, out var s))
            {
                return (false, null, new List<string>());
            }

            double offline = (now - s.DisconnectedAt).TotalSeconds;

            if (offline > SessionKeepSec)
            {
                _sessions.Remove(playerId);
                Console.WriteLine($"    玩家 {playerId} 离线 {offline:F0} 秒，超过保留期，无法恢复");
                return (false, null, new List<string>());
            }

            _sessions.Remove(playerId);
            Console.WriteLine($"    玩家 {playerId} 离线 {offline:F0} 秒后重连，会话恢复");
            return (true, s, s.MissedMessages);
        }

        public int PendingSessions => _sessions.Count;
    }

对比项短连接（HTTP）长连接（TCP/WS）
连接开销每次握手只握手一次
推送能力做不到天然支持
服务端资源无状态，随便扩每连接占内存
适用REST、GM 后台实时对战、聊天、MMO

二、粘包与拆包：长连接第一个坑 ★★★

TCP 是「字节流」，不是「消息流」 这是新手最容易踩的坑。

发送方：send("AAA"); send("BBB");
接收方可能收到：
· "AAABBB"（粘包）
· "AA" 然后 "ABBB"（半包）
· "A" 然后 "AA" 然后 "BBB"

TCP 只保证「字节顺序」，不保证「消息边界」。

本页实测验证：
场景 1（粘包）: 收到 30 字节，解析出 3 条: [AAA, BBB, 测试消息]
场景 2（半包）: 第 1 次收到 6 字节 -> 0 条；第 2 次收到 24 字节 -> 解析出 3 条

解决方案：长度前缀
[4字节长度][内容][4字节长度][内容]...
长度前缀的两个细节 ① 用大端序（网络字节序）
· 不同 CPU 的字节序不同（x86 是小端）
· 协议必须统一成大端

② 要防「超长包」攻击
· 恶意客户端发「长度 = 2GB」
· 服务端尝试分配 2GB → OOM
· 必须限制单条消息最大长度（如 64KB）

这是真实的生产事故来源。

三、心跳与超时检测 ★★

为什么需要心跳 问题：TCP 连接「看起来」还连着，但对端可能已经
· 断电
· 断网
· 进程崩溃

操作系统不会立刻发现——TCP keepalive 默认2 小时才检测。

所以应用层要自己做心跳：
· 客户端每 10 秒发一次心跳
· 服务端 30 秒没收到任何消息 → 判定掉线

本页实测：
t=20s 检测: 踢掉 0 个，在线 3
t=35s 检测: 踢掉 1 个，在线 2
  踢掉: conn-2（玩家1002，闲置 35 秒）

注意：只有玩家2 被踢——玩家1、3 一直在发消息，闲置时间是 0。
（这个演示的时间是「手动推进」的，所以 OnMessage 必须接收传入的当前时间；用 DateTime.UtcNow 会让三个连接全被误踢）

参数经验值：
· 心跳间隔：5~15 秒
· 超时阈值：心跳间隔的 2~3 倍

四、断线重连：最容易被忽略的部分 ★★★

战斗中断线，不能当「重新登录」处理 问题场景：
· 玩家在战斗中掉线 30 秒
· 如果按「重新登录」处理 → 装备、血量、位置全部重置
· 队友看到的角色凭空消失又出现

正确处理：会话保留
① 断线时不销毁会话，标记为「离线」
② 与其他玩家广播「掉线中」状态（不是退出）
③ 缓存错过的消息（战斗事件）
④ 重连时恢复会话 + 补发消息 + 同步最新状态

本页实测：
玩家 3001 断线，会话保留 120 秒
期间缓存 2 条战斗消息
30 秒后重连 → 能恢复: True，补发 2 条
离线 200 秒 → 能恢复: False（超过保留期）
保留时长怎么定 取决于游戏类型：
· 回合制/卡牌：可以很长（几分钟）
· 实时对战：短（30~60 秒），否则队友体验太差
· MMO：中等（2~5 分钟）

超过保留期怎么办：
· 战斗类：判负 / 由AI托管
· MMO：角色下线，下次登录恢复到安全点

五、协议选型

协议优点缺点适用
TCP可靠、有序、简单队头阻塞大厅、聊天、回合制
UDP低延迟不可靠FPS/MOBA（配 KCP）
KCPUDP 上的可靠传输带宽消耗大实时对战
WebSocket浏览器可用有 HTTP 头开销H5、网页端
QUIC多路复用、0-RTT生态不成熟未来方向
游戏后端的常见组合 不是一个游戏只用一种协议：
· 登录/大厅 → HTTPS（一次性，不需要长连接）
· 聊天/邮件 → TCP 或 WebSocket
· 实时对战 → UDP + KCP
· 战斗结果上报 → HTTPS

按「是否需要实时推送」和「能否容忍丢包」来选。
