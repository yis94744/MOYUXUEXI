一、游戏后端的典型分层

按「有状态/无状态」和「压力特征」拆服务——这是游戏后端和 Web 后端最大的差别。

using System.Collections.Generic;
using System.Linq;

namespace BackendSamples.Overview

    // ============================================================
    // 1. 游戏后端的典型分层
    // ============================================================
    // 【核心原则】按「有状态/无状态」和「压力特征」拆服务
    //   无状态服务（登录、好友、商城）-> 随便横向扩
    //   有状态服务（战斗、房间）-> 需要特定玩家路由到特定实例

    public enum ServerType
    {
        Gateway,      // 网关：维持长连接，转发消息
        Login,        // 登录：验证、发 token
        Lobby,        // 大厅：好友、聊天、邮件
        Battle,       // 战斗：房间、帧同步/状态同步
        World,        // 世界：地图、NPC、跨服
        Match,        // 匹配：组队、段位
        Rank,         // 排行：榜单
        Gm,           // 后台：运营工具
    }

    public static class ServerRole
    {
        public static readonly (ServerType Type, string Name, bool Stateful, string Responsibility)[] All =
        {
            (ServerType.Gateway, "网关服",      false, "维持连接、协议编解码、限流、消息转发"),
            (ServerType.Login,   "登录服",      false, "账号验证、签发 token、防重放"),
            (ServerType.Lobby,   "大厅服",      false, "好友、聊天、邮件、公会（可无状态）"),
            (ServerType.Battle,  "战斗服",      true,  "房间管理、战斗逻辑、结算（有状态）"),
            (ServerType.World,   "世界服",      true,  "大地图、NPC、视野同步（有状态）"),
            (ServerType.Match,   "匹配服",      false, "匹配池、组队（需要共享状态走 Redis）"),
            (ServerType.Rank,    "排行服",      false, "榜单查询、赛季结算"),
            (ServerType.Gm,      "运营后台",    false, "发奖、封号、数据查询"),
        };
    }

        public static string HowToScale(bool stateful)

            return stateful
                ? "不能随意扩：玩家必须路由到固定实例（一致性哈希/房间分配）"

        }
    }

    // ============================================================
    // 3. 网关的核心职责（本页重点）
    // ============================================================
    // 【为什么需要网关】
    //   客户端只连一个地址，后端可以随便拆服务
    //   【关键】网关是唯一有长连接的地方

    public class GatewayService
    {
        // 连接表：连接ID -> 玩家信息
        private readonly Dictionary<string, long> _connToPlayer = new();
        private readonly Dictionary<long, string> _playerToConn = new();

        // 路由表：消息类型 -> 目标服务
        private readonly Dictionary<string, ServerType> _routes = new()
        {
            ["login.req"]   = ServerType.Login,
            ["lobby.msg"]   = ServerType.Lobby,
            ["battle.op"]   = ServerType.Battle,
            ["match.join"]  = ServerType.Match,
            ["rank.query"]  = ServerType.Rank,
        };

        public int ConnectCount => _connToPlayer.Count;

        public void OnConnect(string connId)
        {
            _connToPlayer[connId] = 0;   // 0 = 未登录
        }

        public void OnDisconnect(string connId)
        {
            if (_connToPlayer.TryGetValue(connId, out long pid) && pid > 0)
                _playerToConn.Remove(pid);
            _connToPlayer.Remove(connId);
        }

        public void BindPlayer(string connId, long playerId)
        {
            _connToPlayer[connId] = playerId;
            _playerToConn[playerId] = connId;
        }

        // 【网关核心】把消息路由到对应的后端服务
        public ServerType? Route(string connId, string msgType)
        {
            if (!_connToPlayer.ContainsKey(connId))
            {
                Console.WriteLine($"    未知连接: {connId}");
                return null;
            }

            // 【重要】除登录外，其他消息都要求已登录
            long pid = _connToPlayer[connId];
            if (pid == 0 && msgType != "login.req")
            {
                Console.WriteLine($"    未登录连接尝试发 {msgType}，拒绝");
                return null;
            }

            if (!_routes.TryGetValue(msgType, out var target))
            {
                Console.WriteLine($"    未知消息类型: {msgType}");
                return null;
            }

            return target;
        }

        // 给指定玩家推送（战斗服需要主动推消息）
        public string? GetConnOfPlayer(long playerId) =>
            _playerToConn.TryGetValue(playerId, out var c) ? c : null;

        public bool PushToPlayer(long playerId, string message)
        {
            var conn = GetConnOfPlayer(playerId);
            if (conn == null) return false;   // 不在线
            Console.WriteLine($"    推送给 {playerId}（连接 {conn}）: {message}");
            return true;
        }
    }

    // 4. 拆服务的时机：不要一开始就微服务
    // ============================================================

    public static class SplitTiming
    {
        public static readonly (string Stage, string Architecture, string Why)[] All =
        {
            ("原型/小项目", "单体（一个进程）", "开发快、部署简单"),
            ("上线初期", "单体 + 分模块", "还能扛，先不拆"),
            ("玩家增长", "网关 + 登录 + 业务", "登录压力和无状态业务分离"),
            ("需要实时对战", "独立战斗服", "战斗有状态，不能和业务混"),
            ("多区服/跨服", "世界服 + 跨服路由", "玩家分布在不同区"),
            ("大规模", "微服务 + 消息队列", "各服务独立扩缩容"),
        };
    }

    // ============================================================
    // 5. 一个真实的请求链路
    // ============================================================

    public class RequestFlow
    {
        public static void SimulateBuyItem(GatewayService gw, string connId, long playerId, int itemId)
        {
            Console.WriteLine("  客户端 -> 网关:");
            var target = gw.Route(connId, "lobby.msg");
            if (target == null) return;
            Console.WriteLine($"    路由到: {target}");

            Console.WriteLine("  大厅服 -> 校验:");
            Console.WriteLine("    ① token 有效? ✓");
            Console.WriteLine("    ② 背包有空位? ✓");
            Console.WriteLine("    ③ 金币足够? ✓");

            Console.WriteLine("  大厅服 -> 数据库（扣金币，事务内）:");
            Console.WriteLine("    UPDATE player SET gold = gold - 300 WHERE id = ? AND gold >= 300");

            Console.WriteLine("  大厅服 -> 消息队列（异步发道具）:");
            Console.WriteLine("    publish item.grant {playerId, itemId}");

            Console.WriteLine("  大厅服 -> 网关 -> 客户端:");
            gw.PushToPlayer(playerId, $"{{\"code\":0,\"itemId\":{itemId}}}");
        }
    }

服务有无状态职责
网关服无维持长连接、路由、限流
登录服无验证、发 token
大厅服无好友、聊天、邮件
战斗服有房间管理、战斗逻辑
世界服有大地图、NPC、视野
匹配服无匹配池（状态在 Redis）
排行服无榜单
运营后台无发奖、封号

二、有状态 vs 无状态：扩展方式的根本差异 ★★★

为什么这个区分如此重要
无状态有状态
怎么扩容直接加机器，负载均衡轮询必须把玩家路由到固定实例
实例挂了请求失败，重试到别处该实例上的玩家全部掉线
能否热更新可以，滚动重启不行，玩家正在战斗
所以设计原则是：
· 尽量把服务做成无状态（状态放 Redis/数据库）
· 有状态的服务要尽量小（只放不得不放的，比如战斗房间）
· 有状态服务要有重连恢复机制（页 12）

三、网关的四个职责 ★★

网关是唯一持有长连接的地方 本页实测的功能：
① 连接管理——连接ID ↔ 玩家ID 映射
② 路由——按消息类型转发到对应服务
③ 鉴权——未登录连接只能发 login.req
④ 主动推送——战斗服让网关推消息给玩家

本页输出验证：
未登录时发战斗消息 -> 拒绝
登录后 -> 路由到 Battle
战斗服推送 -> 玩家在线: True
推给不在线的玩家 -> False

价值：后端服务不需要关心连接细节——它们只需说"推给玩家 1001"。
网关容易成为瓶颈 问题：所有流量都过网关
· 单机连接数有上限（几万到几十万）
· 消息转发是纯 CPU 开销

优化：
· 网关无状态化——玩家重连可以连到任意网关
· 多网关实例 + 一致性哈希（同一玩家尽量连同一台，减少跨网关转发）
· 只做转发，不做业务（业务放后端服务）

四、什么时候拆服务

阶段架构原因
原型单体开发快
上线初期单体 + 分模块还能扛
玩家增长网关 + 登录 + 业务分离压力
实时对战独立战斗服有状态必须独立
多区服世界服 + 跨服路由玩家分布
大规模微服务 + MQ独立扩缩容
不要一开始就微服务 血的教训：
· 单体能支撑到几十万 DAU
· 拆早了：运维复杂度 > 收益
· 团队小的时候，微服务会拖死进度

正确做法：先在代码层面分模块（清晰的边界），需要时再拆进程。
判断信号：
· 某个模块的压力特征明显不同（如登录峰值）
· 某个模块经常独立上线
· 某个模块需要不同的技术栈
