一、分层的本质：依赖方向 ★★★

using System;
using System.Collections.Generic;
using System.Linq;

namespace BackendSamples.Architecture
{
    // ============================================================
    // 1. 分层的本质：依赖方向
    // ============================================================
    // 【核心】依赖只能从上往下，不能反过来

    public static class Layering
    {
        public static readonly (string Layer, string Responsibility, string Example)[] All =
        {
            ("表现层",  "接收请求、参数校验、返回响应", "Controller / API"),
            ("应用层",  "编排业务流程",                 "Service / UseCase"),
            ("领域层",  "业务规则、不依赖外部",         "Entity / Domain"),
            ("基础设施", "数据库、缓存、外部服务",      "Repository / Client"),
        };

        public static string ExplainRule() =>
            "上层依赖下层；下层【不能】依赖上层（否则改一处崩一片）";

        // 【检验方法】能不能单独编译领域层
        public static string HowToVerify() =>
            "把领域层单独拎出来编译 —— 如果必须带上 Web 框架，说明方向错了";
    }

    // ============================================================
    // 2. 依赖倒置：用接口切断依赖
    // ============================================================
    // 【问题】领域层直接用 SqlClient -> 领域层依赖数据库

    // ---- 【错误】领域层直接依赖具体实现 ----
    public class BadOrderService
    {
        // 【问题】这里直接 new 具体实现
        // 领域层就被绑死在某个数据库上了
        private readonly List<string> _fakeDb = new();

        public void CreateOrder(string order)
        {
            // 真实代码: new SqlConnection(...).Execute(...)
            _fakeDb.Add(order);
        }
    }

    // ---- 【正确】依赖接口 ----
    public interface IOrderRepository
    {
        void Save(string order);
        string? Find(string id);
    }

    public class InMemoryOrderRepo : IOrderRepository
    {
        private readonly Dictionary<string, string> _data = new();
        public void Save(string order) => _data[order] = order;
        public string? Find(string id) => _data.TryGetValue(id, out var v) ? v : null;
    }

    public class GoodOrderService
    {
        private readonly IOrderRepository _repo;

        // 【关键】依赖接口，不依赖具体实现
        public GoodOrderService(IOrderRepository repo) => _repo = repo;

        public void CreateOrder(string order) => _repo.Save(order);
        public string? GetOrder(string id) => _repo.Find(id);
    }

    // ============================================================
    // 3. 单体 vs 微服务
    // ============================================================

    public static class MonolithVsMicro
    {
        public static readonly (string Aspect, string Monolith, string Microservice)[] All =
        {
            ("开发速度",   "快（一个项目）",       "慢（多项目+通信）"),
            ("部署",       "简单（一个包）",       "复杂（编排）"),
            ("扩展",       "整体扩",               "按服务扩"),
            ("技术栈",     "统一",                 "可异构"),
            ("故障隔离",   "差（一个挂了全挂）",   "好"),
            ("调试",       "简单（一个进程）",     "难（跨服务追踪）"),
            ("团队要求",   "低",                   "高（要有运维能力）"),
        };

        public static string Recommend(int teamSize, int dau)
        {
            if (teamSize <= 5) return "单体（模块化）";
            if (teamSize <= 20 && dau < 1000000) return "单体 + 少量拆分（网关/战斗服）";
            return "微服务（但要先有 DevOps 能力）";
        }
    }

    // ============================================================
    // 4. 事件驱动：解耦的利器
    // ============================================================
    // 【问题】下单后要做 5 件事，直接调用会耦合

    public class EventBus
    {
        private readonly Dictionary<string, List<Action<object>>> _handlers = new();

        public void Subscribe(string eventName, Action<object> handler)
        {
            if (!_handlers.TryGetValue(eventName, out var list))
            {
                list = new List<Action<object>>();
                _handlers[eventName] = list;
            }
            list.Add(handler);
        }

        public void Publish(string eventName, object data)
        {
            if (!_handlers.TryGetValue(eventName, out var list)) return;
            // 【关键】快照，防止回调中修改订阅
            foreach (var h in list.ToList()) h(data);
        }

        public int HandlerCount(string eventName) =>
            _handlers.TryGetValue(eventName, out var l) ? l.Count : 0;
    }

    // ---- 【耦合】直接调用 ----
    public class CoupledOrderService
    {
        public void CreateOrder(string order)
        {
            // 【问题】新增一件事就要改这里
            SendEmail(order);
            UpdateInventory(order);
            AddPoints(order);
            NotifyWarehouse(order);
            WriteLog(order);
        }

        private void SendEmail(string o) { }
        private void UpdateInventory(string o) { }
        private void AddPoints(string o) { }
        private void NotifyWarehouse(string o) { }
        private void WriteLog(string o) { }
    }

    // ---- 【解耦】发事件 ----
    public class DecoupledOrderService
    {
        private readonly EventBus _bus;
        public DecoupledOrderService(EventBus bus) => _bus = bus;

        public void CreateOrder(string order)
        {
            // 【关键】只发事件，不关心谁处理
            _bus.Publish("order.created", order);
        }
    }

    // ============================================================
    // 5. 领域模型：贫血 vs 充血
    // ============================================================

    // ---- 贫血模型（只有数据没有行为）----
    public class AnemicPlayer
    {
        public int Gold;
        public int Level;
    }

    public class AnemicService
    {
        // 【问题】业务规则散落在各个 Service 里
        public void AddGold(AnemicPlayer p, int amount)
        {
            if (amount <= 0) throw new ArgumentException("金额非法");
            p.Gold += amount;
        }
    }

    // ---- 充血模型（数据和行为在一起）----
    public class RichPlayer
    {
        public int Gold { get; private set; }
        public int Level { get; private set; }

        public RichPlayer(int gold, int level)
        {
            Gold = gold;
            Level = level;
        }

        // 【关键】规则封装在实体里，不可能被绕过
        public void AddGold(int amount)
        {
            if (amount <= 0) throw new ArgumentException("金额必须大于 0");
            if (amount > 1000000) throw new ArgumentException("单次超过上限");
            Gold += amount;
        }

        public bool TrySpend(int amount)
        {
            if (amount <= 0) return false;
            if (Gold < amount) return false;
            Gold -= amount;
            return true;
        }
    }

    // 演示
    // ============================================================
    public static class Demo
    {
        public static void Run()
        {
            Console.WriteLine("=== 1. 分层与依赖方向 ===");
            foreach (var (layer, resp, ex) in Layering.All)
            {
                Console.WriteLine($"    {layer,-10} {resp,-32} {ex}");
            }
            Console.WriteLine();
            Console.WriteLine($"  【规则】{Layering.ExplainRule()}");
            Console.WriteLine($"  【检验】{Layering.HowToVerify()}");

            Console.WriteLine();
            Console.WriteLine("=== 2. 依赖倒置 ===");
            Console.WriteLine("  【错误】领域层直接 new 具体实现:");
            var bad = new BadOrderService();
            bad.CreateOrder("订单A");
            Console.WriteLine("    领域层被绑死在某个数据库上，换库要改代码");

            Console.WriteLine();
            Console.WriteLine("  【正确】依赖接口:");
            IOrderRepository repo = new InMemoryOrderRepo();
            var good = new GoodOrderService(repo);
            good.CreateOrder("订单B");
            Console.WriteLine($"    保存成功，查询: {good.GetOrder("订单B")}");
            Console.WriteLine("    换数据库只需换实现，Service 不用改");

            Console.WriteLine();
            Console.WriteLine("=== 3. 单体 vs 微服务 ===");
            Console.WriteLine($"    {"方面",-12} {"单体",-22} 微服务");
            foreach (var (aspect, mono, micro) in MonolithVsMicro.All)
            {
                Console.WriteLine($"    {aspect,-12} {mono,-22} {micro}");
            }
            Console.WriteLine();
            foreach (var (team, dau) in new[] { (3, 50000), (15, 500000), (50, 5000000) })
            {
                Console.WriteLine($"    {team,2} 人 / {dau,8} DAU -> {MonolithVsMicro.Recommend(team, dau)}");
            }

            Console.WriteLine();
            Console.WriteLine("=== 4. 事件驱动解耦 ===");
            var bus = new EventBus();

            // 五个独立的订阅者
            bus.Subscribe("order.created", o => Console.WriteLine($"    [邮件] 发送订单确认: {o}"));
            bus.Subscribe("order.created", o => Console.WriteLine($"    [库存] 扣减库存: {o}"));
            bus.Subscribe("order.created", o => Console.WriteLine($"    [积分] 增加积分: {o}"));
            bus.Subscribe("order.created", o => Console.WriteLine($"    [仓库] 通知发货: {o}"));
            bus.Subscribe("order.created", o => Console.WriteLine($"    [日志] 记录: {o}"));

            Console.WriteLine($"  订阅者数量: {bus.HandlerCount("order.created")}");
            Console.WriteLine();
            Console.WriteLine("  下单:");

            var orderSvc = new DecoupledOrderService(bus);
            orderSvc.CreateOrder("订单-001");

            Console.WriteLine();
            Console.WriteLine("  【价值】新增一件事");
            Console.WriteLine("    · 耦合做法: 改 CreateOrder 方法");
            Console.WriteLine("    · 事件做法: 加一个 Subscribe，不改业务代码");

            Console.WriteLine();
            Console.WriteLine("=== 5. 贫血 vs 充血模型 ===");
            Console.WriteLine("  【贫血】规则在 Service 里，容易被绕过:");
            var anemic = new AnemicPlayer { Gold = 100, Level = 1 };
            anemic.Gold = -999;   // 【危险】直接改字段，绕过所有规则
            Console.WriteLine($"    直接改字段 -> 金币变成 {anemic.Gold}（没有校验）");

            Console.WriteLine();
            Console.WriteLine("  【充血】规则在实体里，绕不过:");
            var rich = new RichPlayer(100, 1);
            try
            {
                rich.AddGold(-999);
            }
            catch (ArgumentException ex)
            {
                Console.WriteLine($"    尝试加 -999 -> {ex.Message}");
            }
            try
            {
                rich.AddGold(2000000);
            }
            catch (ArgumentException ex)
            {
                Console.WriteLine($"    尝试加 200 万 -> {ex.Message}");
            }
            rich.AddGold(500);
            Console.WriteLine($"    正常加 500 -> 金币 {rich.Gold}");

            Console.WriteLine();
            Console.WriteLine("  【结论】");
            Console.WriteLine("    · 贫血模型：简单，但规则容易漏");
            Console.WriteLine("    · 充血模型：规则集中，但要求理解领域");
            Console.WriteLine("    · 游戏后端建议：核心实体（玩家/订单）用充血");
        }
    }
}

二、依赖倒置：用接口切断依赖 ★★★

错误 vs 正确的依赖方向 【错误】领域层直接依赖数据库
class OrderService { SqlConnection conn = new(...); }
→ 领域层绑死在某个数据库上，换库要改代码，测试要连真库

【正确】依赖接口
class OrderService { IOrderRepository repo; }
· 换数据库只需换实现
· 测试时注入 Fake

本页实测：
IOrderRepository repo = new InMemoryOrderRepo();
var good = new GoodOrderService(repo);
保存成功，查询: 订单B

检验方法：把领域层单独拎出来编译——
如果必须带上 Web 框架，说明方向错了。

三、单体 vs 微服务：不要过早拆分 ★★

本页给出的决策表
团队/规模建议
≤5 人单体（模块化）
≤20 人 + <100万 DAU单体 + 少量拆分（网关/战斗服）
50 人 + 500万 DAU微服务（但先有 DevOps 能力）
本页对比的七个方面：
· 开发速度：单体快（一个项目）
· 部署：单体简单（一个包）
· 故障隔离：微服务好
· 调试：单体简单（一个进程）

结论：微服务的收益在「大规模团队 + 高并发」，代价是运维复杂度。
小团队上微服务 = 自杀。

四、事件驱动：解耦的利器 ★★★

本页实测的对比 【耦合】下单后直接调 5 个方法
SendEmail(); UpdateInventory(); AddPoints(); ...
→ 新增一件事就要改 CreateOrder

【解耦】只发事件
_bus.Publish("order.created", order);

本页实测输出：
订阅者数量: 5
[邮件] 发送订单确认: 订单-001
[库存] 扣减库存: 订单-001
[积分] 增加积分: 订单-001
[仓库] 通知发货: 订单-001
[日志] 记录: 订单-001

价值：新增一件事只需加一个 Subscribe，不改业务代码。

注意：本地事件和消息队列的区别（页 09）——
本地事件是同步的，消息队列是异步的。

五、贫血 vs 充血模型 ★★

本页实测的差别 【贫血】规则在 Service 里
class AnemicPlayer { public int Gold; }
anemic.Gold = -999; // 直接改字段，绕过所有规则
直接改字段 -> 金币变成 -999（没有校验）

【充血】规则在实体里
rich.AddGold(-999);
-> 金额必须大于 0
rich.AddGold(2000000);
-> 单次超过上限
rich.AddGold(500);
-> 金币 600

结论：
· 贫血：简单，但规则容易漏
· 充血：规则集中，但要求理解领域
· 游戏后端建议：核心实体（玩家/订单）用充血

本章动手任务

本章过关标准：
① 知道分层的本质是控制依赖方向；
② 会用依赖倒置切断对数据库的直接依赖；
③ 能按团队规模判断该不该上微服务；
④ 能用事件解耦「一件事触发多件事」；
⑤ 知道核心实体该用充血模型。

第 ① 和 第 ④ 条最关键——**前者是架构思维，后者是实战中最常用的解耦手段**。
