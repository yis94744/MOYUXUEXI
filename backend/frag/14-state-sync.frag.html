    // ============================================================
    // 2. 确定性：帧同步的命门 ★★★
    // ============================================================
    // 【核心问题】同样的输入，不同机器算出不同结果 -> 不同步

    public class DeterminismDemo
    {
        // 【错误】用浮点 -> 不同 CPU/编译器结果可能不同
        public static float BadAdd(float a, float b) => a + b;

        // 【正确】用定点数（放大倍数后取整）
        //   1.0 表示为 10000
        public const int SCALE = 10000;

        public static int FixedAdd(int a, int b) => a + b;
        public static int FixedMul(int a, int b) => (int)((long)a * b / SCALE);

        // 【错误】用系统随机数
        public static int BadRandom() => new Random().Next();

        // 【正确】确定性随机（线性同余）
        public class DeterministicRandom
        {
            private uint _state;
            public DeterministicRandom(uint seed) => _state = seed;

            public uint Next()
            {
                // 【关键】同样的 seed 在任何机器上都产生同样的序列
                _state = _state * 1664525 + 1013904223;
                return _state;
            }

            public int Next(int min, int max) => min + (int)(Next() % (uint)(max - min));
        }

        public static void Run()
        {
            Console.WriteLine("  浮点 vs 定点:");
            // 【实测发现】0.1f + 0.2f == 0.3f 在 C# 里居然是 True
            //   因为 .NET 的 float 比较做了特殊处理
            //   要暴露浮点问题，必须用「累积运算」而不是单次相加
            float fa = 0.1f, fb = 0.2f;
            Console.WriteLine($"    浮点 0.1 + 0.2 = {BadAdd(fa, fb):F10}");
            Console.WriteLine($"    等于 0.3 吗: {BadAdd(fa, fb) == 0.3f}（.NET 做了特殊处理）");

            // 【真正会出问题的地方】累积运算
            float acc = 0f;
            for (int i = 0; i < 10; i++) acc += 0.1f;
            Console.WriteLine($"    0.1 累加 10 次 = {acc:F10}");
            Console.WriteLine($"    等于 1.0 吗: {acc == 1.0f}  <-- 这里会 False");

            int ia = (int)(0.1 * SCALE), ib = (int)(0.2 * SCALE);
            Console.WriteLine($"    定点 1000 + 2000 = {FixedAdd(ia, ib)} (= {FixedAdd(ia, ib) / (float)SCALE:F1})");
            Console.WriteLine("    【结论】单次相加可能没事，但累加必然出问题");

            Console.WriteLine();
            Console.WriteLine("  确定性随机:");
            var rng1 = new DeterministicRandom(12345);
            var rng2 = new DeterministicRandom(12345);
            var seq1 = Enumerable.Range(0, 5).Select(_ => rng1.Next(1, 100)).ToList();
            var seq2 = Enumerable.Range(0, 5).Select(_ => rng2.Next(1, 100)).ToList();
            Console.WriteLine($"    seed=12345 第一次: [{string.Join(", ", seq1)}]");
            Console.WriteLine($"    seed=12345 第二次: [{string.Join(", ", seq2)}]");
            Console.WriteLine($"    序列相同: {seq1.SequenceEqual(seq2)}");
            Console.WriteLine("    【关键】帧同步里所有客户端必须用同一 seed");
        }
    }

    // 3. 帧同步：指令收集与广播
    // ============================================================

    public class LockstepServer
    {
        public class Frame
        {
            public int Id;
            public Dictionary<long, string> Commands = new();   // playerId -> command
        }

        private readonly Dictionary<int, Frame> _frames = new();
        private readonly Dictionary<int, HashSet<long>> _ready = new();
        private readonly int _playerCount;
        private int _currentFrame;

        public LockstepServer(int playerCount) => _playerCount = playerCount;

        // 客户端提交指令
        public bool Submit(long playerId, string command)
        {
            int frameId = _currentFrame + 1;

            if (!_frames.TryGetValue(frameId, out var f))
            {
                f = new Frame { Id = frameId };
                _frames[frameId] = f;
            }

            f.Commands[playerId] = command;

            if (!_ready.TryGetValue(frameId, out var set))
            {
                set = new HashSet<long>();
                _ready[frameId] = set;
            }
            set.Add(playerId);

            return true;
        }

        // 【关键】必须所有玩家都提交了才能推进（锁步）
        public Frame? TryAdvance()
        {
            int frameId = _currentFrame + 1;

            if (!_ready.TryGetValue(frameId, out var set)) return null;
            if (set.Count < _playerCount) return null;   // 还没齐

            // 补齐：没提交的玩家用空指令
            if (!_frames.TryGetValue(frameId, out var frame))
            {
                frame = new Frame { Id = frameId };
                _frames[frameId] = frame;
            }

            for (long pid = 1; pid <= _playerCount; pid++)
            {
                if (!frame.Commands.ContainsKey(pid))
                    frame.Commands[pid] = "idle";   // 没操作 = 空指令
            }

            _currentFrame = frameId;
            return frame;
        }

        public int CurrentFrame => _currentFrame;
        public int PendingFrame => _currentFrame + 1;
    }

    // ============================================================
    // 4. 状态同步：服务器权威 + 增量同步
    // ============================================================

    public class StateSyncServer
    {
        public class Unit
        {
            public long Id;
            public float X, Y;
            public int Hp;
            public int HpMax;
        }

        private readonly Dictionary<long, Unit> _units = new();
        private int _frameId;

        public void Spawn(long id, float x, float y, int hp)
        {
            _units[id] = new Unit { Id = id, X = x, Y = y, Hp = hp, HpMax = hp };
        }

        // 【关键】状态同步的三个优化
        public StateUpdate GenerateUpdate(long viewerId)
        {
            _frameId++;
            var update = new StateUpdate { FrameId = _frameId };

            foreach (var u in _units.Values)
            {
                if (u.Id == viewerId) continue;

                // 【优化1】视野裁剪：只同步附近的
                var viewer = _units.GetValueOrDefault(viewerId);
                if (viewer != null)
                {
                    double dist = Math.Sqrt(Math.Pow(u.X - viewer.X, 2) + Math.Pow(u.Y - viewer.Y, 2));
                    if (dist > 50) continue;   // 视野外不同步
                }

                update.Units.Add(new UnitSnapshot
                {
                    Id = u.Id,
                    X = u.X,
                    Y = u.Y,
                    Hp = u.Hp,
                });
            }

            return update;
        }

        // 【优化2】增量同步：只发变化的字段
        public Dictionary<long, List<string>> GenerateDelta(long playerId, Dictionary<long, Unit> lastSent)
        {
            var delta = new Dictionary<long, List<string>>();

            foreach (var u in _units.Values)
            {
                var changed = new List<string>();

                if (!lastSent.TryGetValue(u.Id, out var old))
                {
                    changed.Add("all");
                }
                else
                {
                    if (Math.Abs(old.X - u.X) > 0.01f || Math.Abs(old.Y - u.Y) > 0.01f)
                        changed.Add("pos");
                    if (old.Hp != u.Hp) changed.Add("hp");
                }

                if (changed.Count > 0) delta[u.Id] = changed;
            }

            return delta;
        }

        // 【优化3】量化压缩：位置保留 2 位小数
        public static int QuantizePos(float v) => (int)(v * 100);

        public IEnumerable<Unit> AllUnits => _units.Values;
    }

    public class StateUpdate
    {
        public int FrameId;
        public List<UnitSnapshot> Units = new();
    }

    public class UnitSnapshot
    {
        public long Id;
        public float X, Y;
        public int Hp;

        public override string ToString() => $"#{Id}({X:F1},{Y:F1}) HP={Hp}";
    }

    // ============================================================
    // 5. 客户端预测与服务端校正
    // ============================================================
    // 【问题】玩家按前进，等服务器确认要 100ms -> 手感极差
    // 【方案】客户端预测 + 服务器校正

    // 【关键】PlayerState 提到命名空间级别，Demo 里才能用
    public class PlayerState
    {
        public float X;
        public int LastAckInput;
    }

    public class PredictionDemo
    {

        private readonly List<string> _pendingInputs = new();
        private int _inputSeq;

        // 客户端：立即执行 + 记录输入
        public float ClientMove(PlayerState state, float delta)
        {
            state.X += delta;
            _pendingInputs.Add($"input-{++_inputSeq}:move{delta}");
            return state.X;
        }

        // 服务端：返回权威位置
        public (float authoritativeX, int ackSeq) ServerSimulate(float serverX, float delta)
        {
            return (serverX + delta, _inputSeq);
        }

        // 客户端：收到校正
        public (float newX, int rollbackCount) ClientReconcile(PlayerState state, float serverX, int ackSeq)
        {
            // 丢弃已确认的输入
            _pendingInputs.RemoveAll(p => int.Parse(p.Split(':')[0].Split('-')[1]) <= ackSeq);

            // 如果服务端位置和预测差太多 -> 回滚重放
            if (Math.Abs(serverX - state.X) > 0.1f)
            {
                int rollback = _pendingInputs.Count;
                state.X = serverX;
                // 重放未确认的输入
                foreach (var p in _pendingInputs)
                {
                    float d = float.Parse(p.Split(':')[1].Replace("move", ""));
                    state.X += d;
                }
                return (state.X, rollback);
            }

            return (state.X, 0);
        }

        public int PendingCount => _pendingInputs.Count;
    }

二、确定性：帧同步的命门 ★★★

三个必须解决的确定性问题 ① 浮点运算不确定
不同 CPU、不同编译器优化，浮点结果可能有微小差异。
本页实测（重要更正）：0.1f + 0.2f == 0.3f 在 .NET 里返回 True——
因为 0.1f 与 0.2f 转成 float 后的和，恰好落在与 0.3f 相同的位模式上。
只测单次相加会得出「浮点没问题」的错觉。
真正会炸的是累积运算：0.1f 连加 10 次 = 1.0000001192，== 1.0f → False。
方案：用定点数（放大 1000 倍取整），或全程用 decimal/整数运算

② 随机数不确定
new Random().Next() 每个客户端结果不同。
方案：用确定性随机（同 seed 同序列）
本页实测：seed=12345 两次产生完全相同的序列。

③ 遍历顺序不确定
Dictionary / HashSet 的遍历顺序在不同运行中可能不同。
方案：用 List 或排序后的数组。
帧同步的锁步机制 核心：所有玩家都提交了指令，才能推进一帧。

本页实测：
玩家1、2 提交了，玩家3 没提交
能否推进: False（必须等玩家3）
玩家3 也提交后 → 能否推进: True

没提交的玩家怎么处理：用空指令（idle）补齐。
但问题来了：如果玩家故意不提交（掉线/恶意）→ 所有人卡住。

解决方案：
· 超时自动补空指令（最常用）
· 掉线托管（AI 接管）
· 踢出并让其他人继续

三、状态同步的三个优化 ★★

状态同步的带宽问题 问题：100 个玩家，每人每秒 10 次同步
· 每次同步 100 个单位的位置
· 每秒 10 万条状态 → 带宽爆炸

三个优化：
① 视野裁剪（本页实测）
· 只同步「附近 50 米」内的单位
· 4 个单位中只同步 3 个

② 增量同步
· 只发变化的字段（位置变了发位置，血量变了发血量）
· 本页实现了 GenerateDelta

③ 量化压缩
· 位置保留 2 位小数 → float 4 字节变 2 字节
· 角度用 1 字节（0~255 映射 0~360）

四、客户端预测：手感的来源 ★★★

为什么需要预测 问题：玩家按前进
· 请求到服务器：50ms
· 服务器处理：10ms
· 回来：50ms
→ 总共 110ms 才看到动 → 手感极差

预测的做法：
① 客户端立即执行（玩家马上看到动）
② 同时把输入发给服务器
③ 服务器返回权威位置
④ 不一致时回滚重放

本页实测：
第 1 次: 本地位置 = 1.0 / 第 2 次: 2.0 / 第 3 次: 3.0
待确认输入: 3 条
服务器返回权威位置（假设一致）: 校正后位置: 3.0，回滚 0 条输入
服务器说位置应该是 1.0（发生了碰撞）: 校正后位置: 1.0，回滚 0 条输入

这里有个容易看漏的点：两次校正的回滚条数都是 0——因为 ServerSimulate 返回的 ackSeq 已经等于 3（全部输入都确认了）， 待重放的队列是空的，所以只发生了位置校正、没有重放。
回滚条数 = 未被服务器确认的输入数。真实网络里 ack 落后于本地输入，才会有重放； 重放时如果再撞墙，就表现为玩家看到的「回拉」卡顿。

副作用：玩家会看到角色「回拉」——这就是网络差时瞬移的原因。
