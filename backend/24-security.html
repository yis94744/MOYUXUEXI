    // ============================================================
    // 2. SQL 注入：演示被注入的过程 ★★★
    // ============================================================

    public class SqlInjectionDemo
    {
        // 【模拟】内存数据库
        private readonly Dictionary<string, string> _users = new()
        {
            ["admin"] = "hash_admin_pwd",
            ["player1"] = "hash_player_pwd",
        };

        private readonly string _secret = "SECRET_DATA";

        // ---- 【错误】字符串拼接 ----
        public string Login_Bad(string username, string password)
        {
            // 真实代码： "SELECT * FROM users WHERE name='" + username + "' AND pwd='" + password + "'"
            string sql = $"SELECT * FROM users WHERE name='{username}' AND pwd='{password}'";

            // 【危险】模拟拼接后的 SQL 被解析
            // 如果 username = "admin' --"，SQL 变成：
            //   SELECT * FROM users WHERE name='admin' --' AND pwd='...'
            //   -- 注释掉了后面的密码检查！

            if (username.Contains("'") && username.Contains("--"))
            {
                return $"【被注入】SQL: {sql}\n    结果: 登录成功（绕过了密码校验）\n    泄露数据: {_secret}";
            }

            var exists = _users.ContainsKey(username) && _users[username] == password;
            return exists ? "登录成功" : "用户名或密码错误";
        }

        // ---- 【正确】参数化查询 ----
        public string Login_Good(string username, string password)
        {
            // 真实代码： command.Parameters.AddWithValue("@name", username)
            // 参数被当作「数据」而不是「SQL 片段」

            // 即使传 "admin' --"，也只会被当成用户名字面量
            if (username.Contains("'") || username.Contains("--"))
            {
                return $"【安全】参数化查询: 无匹配（'admin\\' --' 被当作字面量）";
            }

            var exists = _users.ContainsKey(username) && _users[username] == password;
            return exists ? "登录成功" : "用户名或密码错误";
        }

        public static void Run()
        {
            var db = new SqlInjectionDemo();

            Console.WriteLine("  正常登录:");
            Console.WriteLine($"    admin + 正确密码: {db.Login_Good("admin", "hash_admin_pwd")}");
            Console.WriteLine($"    admin + 错误密码: {db.Login_Good("admin", "wrong")}");

            Console.WriteLine();
            Console.WriteLine("  【注入攻击】用户名传 \"admin' --\"");
            Console.WriteLine("    【错误做法】字符串拼接:");
            Console.WriteLine("      " + db.Login_Bad("admin' --", "anything").Replace("\n", "\n      "));

            Console.WriteLine();
            Console.WriteLine("    【正确做法】参数化查询:");
            Console.WriteLine("      " + db.Login_Good("admin' --", "anything"));
        }
    }

    // ============================================================
    // 3. XSS：转义是必须的
    // ============================================================

    public class XssDemo
    {
        // 【错误】直接拼接用户输入到 HTML
        public static string Render_Bad(string userInput)
        {
            return $"<div>{userInput}</div>";
        }

        // 【正确】HTML 转义
        public static string Render_Good(string userInput)
        {
            var sb = new StringBuilder();
            foreach (char c in userInput)
            {
                sb.Append(c switch
                {
                    '<' => "&lt;",
                    '>' => "&gt;",
                    '&' => "&amp;",
                    '"' => "&quot;",
                    '\'' => "&#39;",
                    _ => c.ToString()
                });
            }
            return $"<div>{sb}</div>";
        }

        public static void Run()
        {
            string attack = "<script>alert('xss')</script>";

            Console.WriteLine($"  攻击载荷: {attack}");
            Console.WriteLine();
            Console.WriteLine($"  【错误】直接渲染:");
            Console.WriteLine($"    {Render_Bad(attack)}");
            Console.WriteLine("    -> 浏览器会执行 script，窃取 cookie");

            Console.WriteLine();
            Console.WriteLine($"  【正确】转义后:");
            Console.WriteLine($"    {Render_Good(attack)}");
            Console.WriteLine("    -> 显示成文本，不执行");

            Console.WriteLine();
            Console.WriteLine("  【关键】");
            Console.WriteLine("    · 输出到 HTML -> HTML 转义");
            Console.WriteLine("    · 输出到 JS  -> JS 转义");
            Console.WriteLine("    · 输出到 URL -> URL 编码");
            Console.WriteLine("    【转义要匹配输出上下文】");
        }
    }

    // 4. 敏感信息脱敏
    // ============================================================

    public static class Masking
    {
        public static string MaskPhone(string phone)
        {
            if (string.IsNullOrEmpty(phone) || phone.Length < 7) return "***";
            return phone.Substring(0, 3) + "****" + phone.Substring(phone.Length - 4);
        }

        public static string MaskEmail(string email)
        {
            var at = email.IndexOf('@');
            if (at <= 1) return "***";
            return email[0] + "***" + email.Substring(at);
        }

        public static string MaskIdCard(string id)
        {
            if (string.IsNullOrEmpty(id) || id.Length < 8) return "***";
            return id.Substring(0, 4) + "**********" + id.Substring(id.Length - 4);
        }

        public static void Run()
        {
            Console.WriteLine($"  手机号: {MaskPhone("13812345678")}");
            Console.WriteLine($"  邮箱:   {MaskEmail("test@example.com")}");
            Console.WriteLine($"  身份证: {MaskIdCard("110101199001011234")}");

            Console.WriteLine();
            Console.WriteLine("  【什么时候脱敏】");
            Console.WriteLine("    · 客服后台展示（客服不该看到完整手机号）");
            Console.WriteLine("    · 日志（避免日志泄露）");
            Console.WriteLine("    · 导出报表");
            Console.WriteLine();
            Console.WriteLine("  【什么时候不脱敏】");
            Console.WriteLine("    · 玩家看自己的信息");
            Console.WriteLine("    · 内部审计（需要完整数据，但要有权限）");
        }
    }

    // ============================================================
    // 5. CSRF：跨站请求伪造
    // ============================================================

    public class CsrfDemo
    {
        // 【问题】用户登录了银行网站，然后访问了恶意网站
        //   恶意网站用 <img src="bank.com/transfer?to=hacker&amount=1000">
        //   浏览器自动带上银行的 cookie -> 转账成功

        public static (bool Allowed, string Reason) CheckRequest(
            string method, string? csrfToken, string? origin)
        {
            // 【防护1】GET 请求不应有副作用
            if (method == "GET")
            {
                // 如果 GET 能改数据，就是设计错误
                return (false, "GET 请求不应修改数据");
            }

            // 【防护2】必须带 CSRF Token
            if (string.IsNullOrEmpty(csrfToken))
                return (false, "缺少 CSRF Token");

            // 【防护3】校验 Origin / Referer
            if (!string.IsNullOrEmpty(origin) && !origin.StartsWith("https://mygame.com"))
                return (false, "来源不合法: " + origin);

            return (true, "通过");
        }

        public static void Run()
        {
            Console.WriteLine("  模拟三种请求:");

            Console.WriteLine($"    GET /transfer -> {CheckRequest("GET", null, null).Reason}");
            Console.WriteLine($"    POST 无 token -> {CheckRequest("POST", null, null).Reason}");
            Console.WriteLine($"    POST 有 token 但来源可疑 -> {CheckRequest("POST", "abc", "https://evil.com").Reason}");
            Console.WriteLine($"    POST 有 token + 正确来源 -> {CheckRequest("POST", "abc", "https://mygame.com").Reason}");

            Console.WriteLine();
            Console.WriteLine("  【关键防护】");
            Console.WriteLine("    · GET 只读，不修改数据");
            Console.WriteLine("    · 状态修改用 POST/PUT/DELETE");
            Console.WriteLine("    · 必须带 CSRF Token");
            Console.WriteLine("    · Cookie 设 SameSite=Strict");
        }
    }

    二、SQL 注入：亲眼看到被注入 ★★★

    本页实测的注入过程 正常登录：
    admin + 正确密码 -> 登录成功

    注入攻击：用户名传 admin 加两个单引号加 --
    【错误】字符串拼接：
    SELECT * FROM users WHERE name=...
    后面的密码检查被注释掉了
    【被注入】SQL: SELECT * FROM users WHERE name='admin' --' AND pwd='anything'
    结果: 登录成功（绕过了密码校验）
    泄露数据: SECRET_DATA

    【正确】参数化查询：
    【安全】参数化查询: 无匹配（'admin\' --' 被当作字面量）
    整段被当作字面量，不是 SQL 片段

    规则：永远不要拼接 SQL，永远用参数化。

    三、XSS：转义要匹配上下文 ★★

    本页实测的转义效果 攻击载荷：script 标签

    【错误】直接渲染：
    浏览器执行 script，窃取 cookie

    【正确】HTML 转义：
    把尖括号、引号转成实体
    显示成文本，不执行

    关键：转义要匹配输出上下文
    · 输出到 HTML → HTML 转义
    · 输出到 JS → JS 转义
    · 输出到 URL → URL 编码
    用错转义方式等于没转义。

    四、脱敏：什么时候要，什么时候不要

    本页实测的脱敏 手机号: 138****5678
    邮箱: t***@example.com
    身份证: 1101**********1234

    必须脱敏的场景：
    · 客服后台——客服不该看到完整手机号
    · 日志——避免日志泄露
    · 导出报表

    不该脱敏的场景：
    · 玩家看自己的信息
    · 内部审计（但要有权限）

    五、CSRF 与加密选择

    CSRF：借用户的身份发请求 攻击原理：
    ① 用户登录了游戏官网
    ② 访问了恶意网站
    ③ 恶意网站构造一个请求到游戏接口
    ④ 浏览器自动带上 cookie → 发奖给了黑客

    本页实测的三道防线：
    GET /transfer -> GET 请求不应修改数据
    POST 无 token -> 缺少 CSRF Token
    POST 来源可疑 -> 来源不合法
    POST 全部正确 -> 通过

    关键：GET 只读，修改必须用 POST。
    密码只能用慢哈希 ★★★
    场景用什么为什么
    密码慢哈希 PBKDF2单向，不可逆
    身份证号对称加密 AES需要还原
    传输TLS/HTTPS防窃听
    签名防篡改HMAC只要校验
    最重要的一条：
    密码只能用「慢哈希」，绝对不能用可逆加密。
    因为：你能解密，就说明数据库泄露后攻击者也能解密。
