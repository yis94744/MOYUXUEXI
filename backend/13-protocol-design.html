    // ============================================================
    // 2. 手写二进制协议（理解 Protobuf 原理）
    // ============================================================
    // Protobuf 的核心：字段编号 + 类型 + 值
    // 【关键】用编号而不是字段名 -> 省空间

    public class BinaryWriter
    {
        private readonly List<byte> _buf = new();

        public void WriteVarint(ulong value)
        {
            // 【Varint】小数字占 1 字节，大数字才占更多
            while (value >= 0x80)
            {
                _buf.Add((byte)(value | 0x80));
                value >>= 7;
            }
            _buf.Add((byte)value);
        }

        public void WriteInt32(int value) => WriteVarint((ulong)(uint)value);

        public void WriteString(string s)
        {
            var bytes = Encoding.UTF8.GetBytes(s);
            WriteVarint((ulong)bytes.Length);   // 长度前缀
            _buf.AddRange(bytes);
        }

        // 写字段：fieldNumber + wireType
        public void WriteField(int fieldNumber, int wireType) =>
            WriteVarint((ulong)((fieldNumber << 3) | wireType));

        public void WriteIntField(int fieldNumber, int value)
        {
            WriteField(fieldNumber, 0);   // wireType 0 = varint
            WriteInt32(value);
        }

        public void WriteStringField(int fieldNumber, string value)
        {
            WriteField(fieldNumber, 2);   // wireType 2 = length-delimited
            WriteString(value);
        }

        public byte[] ToArray() => _buf.ToArray();
        public int Length => _buf.Count;
    }

    public class BinaryReader
    {
        private readonly byte[] _data;
        private int _pos;

        public BinaryReader(byte[] data) => _data = data;

        public ulong ReadVarint()
        {
            ulong result = 0;
            int shift = 0;
            while (_pos < _data.Length)
            {
                byte b = _data[_pos++];
                result |= (ulong)(b & 0x7F) << shift;
                if ((b & 0x80) == 0) break;
                shift += 7;
            }
            return result;
        }

        public int ReadInt32() => (int)(uint)ReadVarint();

        public string ReadString()
        {
            int len = (int)ReadVarint();
            var s = Encoding.UTF8.GetString(_data, _pos, len);
            _pos += len;
            return s;
        }

        // 读字段头
        public (int FieldNumber, int WireType) ReadField()
        {
            ulong tag = ReadVarint();
            return ((int)(tag >> 3), (int)(tag & 0x7));
        }

        public bool HasMore => _pos < _data.Length;
        public int Position => _pos;
    }

    // ============================================================
    // 3. 消息定义与编解码
    // ============================================================

    public class PlayerLoginReq
    {
        public long PlayerId;
        public string Token = "";

        // 【关键】字段编号一旦分配，【永远不要改】
        //   改编号 = 老客户端解析错乱
        private const int F_PLAYER_ID = 1;
        private const int F_TOKEN = 2;

        public byte[] Serialize()
        {
            var w = new BinaryWriter();
            w.WriteIntField(F_PLAYER_ID, (int)PlayerId);
            w.WriteStringField(F_TOKEN, Token);
            return w.ToArray();
        }

        public static PlayerLoginReq Deserialize(byte[] data)
        {
            var r = new BinaryReader(data);
            var req = new PlayerLoginReq();

            while (r.HasMore)
            {
                var (field, wireType) = r.ReadField();

                // 【关键】按字段编号分派，不认识的跳过（向前兼容）
                switch (field)
                {
                    case F_PLAYER_ID:
                        req.PlayerId = r.ReadInt32();
                        break;
                    case F_TOKEN:
                        req.Token = r.ReadString();
                        break;
                    default:
                        // 【重要】未知字段要能跳过，否则新服务端读老客户端会崩
                        SkipField(r, wireType);
                        break;
                }
            }

            return req;
        }

        private static void SkipField(BinaryReader r, int wireType)
        {
            if (wireType == 0) r.ReadVarint();
            else if (wireType == 2) r.ReadString();   // 简化：当字符串跳过
        }

        public override string ToString() => $"登录请求(player={PlayerId}, token={Token.Substring(0, Math.Min(8, Token.Length))}...)";
    }

    // 4. 协议版本兼容 ★★★
    // ============================================================
    // 【最难的问题】老客户端 + 新服务端 / 新客户端 + 老服务端

    public static class CompatibilityRules
    {
        public static readonly (string Rule, string Explanation)[] All =
        {
            ("字段编号永不复用", "删掉的字段编号也不能给新字段用（老客户端会误解）"),
            ("只增不改", "改字段类型会导致解析失败（int32 改成 string 必崩）"),
            ("新增字段要给默认值", "老客户端不发这个字段，服务端要用默认值处理"),
            ("required 不要用", "改成 optional，否则老消息解析失败"),
            ("保留已删除编号", "reserved 1, 2, 5; 防止被误用"),
        };

        public static void Print()
        {
            foreach (var (rule, exp) in All)
                Console.WriteLine($"    {rule,-20} {exp}");
        }
    }

    public class PlayerLoginReqV2
    {
        public long PlayerId;
        public string Token = "";
        public string DeviceId = "";     // 【新增】v2 才有
        public int ClientVersion;         // 【新增】

        private const int F_PLAYER_ID = 1;
        private const int F_TOKEN = 2;
        private const int F_DEVICE_ID = 3;
        private const int F_CLIENT_VER = 4;

        public byte[] Serialize()
        {
            var w = new BinaryWriter();
            w.WriteIntField(F_PLAYER_ID, (int)PlayerId);
            w.WriteStringField(F_TOKEN, Token);
            w.WriteStringField(F_DEVICE_ID, DeviceId);
            w.WriteIntField(F_CLIENT_VER, ClientVersion);
            return w.ToArray();
        }
    }

    // 老版本解析（不认识字段 3、4）
    public static class OldClientParser
    {
        public static (long playerId, string token, int skipped) Parse(byte[] data)
        {
            var r = new BinaryReader(data);
            long pid = 0;
            string token = "";
            int skipped = 0;

            while (r.HasMore)
            {
                var (field, wireType) = r.ReadField();
                switch (field)
                {
                    case 1: pid = r.ReadInt32(); break;
                    case 2: token = r.ReadString(); break;
                    default:
                        // 【关键】跳过不认识的字段，不崩溃
                        if (wireType == 0) r.ReadVarint();
                        else if (wireType == 2) r.ReadString();
                        skipped++;
                        break;
                }
            }

            return (pid, token, skipped);
        }
    }

    二、字段编号：省空间的核心 ★★★

    为什么用编号而不是字段名 JSON 方式：
    {"playerId":1001,"token":"abc123def456"}
    → 40 字节（字段名占了一大半）

    二进制 + 编号：
    08 E9 07 12 0C 61 62 63 31 32 33 64 65 66 34 35 36
    → 17 字节
    · 08 = 字段1 + 类型0（varint）
    · E9 07 = 1001（varint 编码）
    · 12 = 字段2 + 类型2（字符串）
    · 0C = 长度 12
    · 后面是 "abc123def456"

    省 57%——而且数据越大，比例越高。
    Varint：小数字占更少字节 原理：
    · 每个字节用低 7 位存数据
    · 最高位是「还有后续字节」标志

    效果：
    · 1 → 1 字节
    · 1001 → 2 字节
    · 100000 → 3 字节
    · 2000000000 → 5 字节

    为什么有效：游戏里绝大多数数字都很小（等级、数量、ID）。
    注意：负数用 varint 会变成 10 字节（因为补码最高位是1）——负数要用 zigzag 编码。

    三、向前兼容：新老版本共存 ★★★

    五条铁律
    规则违反的后果
    字段编号永不复用老客户端发旧字段 → 被当新字段解析
    只增不改类型int32 改成 string → 必崩
    新增字段给默认值老客户端不发 → 服务端拿到 0 或 null
    不要用 required老消息没有这个字段 → 解析失败
    删除的编号要 reserved防止被误用
    本页实测验证了「跳过未知字段」：
    v2 客户端发送 23 字节（含 deviceId、clientVersion）
    v1 服务端解析: playerId=2002, token=xyz789
    跳过了 2 个不认识的字段（没崩溃）

    这就是向前兼容的核心：老代码遇到新字段要跳过，而不是报错。

    四、压缩：什么时候值得

    实测阈值
    消息大小建议
    < 128 字节不压缩
    128~1KB可选
    1KB~10KB建议压缩
    > 10KB必须压缩
    为什么小消息不压：
    · 压缩头本身就有几字节
    · 压缩/解压的 CPU 开销 > 省下的带宽

    算法选择：
    · LZ4——极快，压缩比 2~3x，游戏首选
    · zlib——压缩比高但慢，用于资源包
    · Brotli——压缩比最高，用于静态资源
