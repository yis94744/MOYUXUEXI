    一、日志分级：什么该记

    using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;

namespace BackendSamples.Logging
{
    // ============================================================
    // 1. 日志分级：什么该记，什么不该记
    // ============================================================

    public enum LogLevel
    {
        Trace = 0,    // 极详细（每帧、每次循环）
        Debug = 1,    // 调试信息
        Info = 2,     // 关键流程（登录、下单）
        Warn = 3,     // 可疑但不致命
        Error = 4,    // 错误
        Fatal = 5,    // 致命（进程要挂）
    }

    public static class LevelGuide
    {
        public static readonly (LogLevel Level, string When, string Example)[] All =
        {
            (LogLevel.Trace, "开发期追踪细节", "收到请求 header: ..."),
            (LogLevel.Debug, "开发期调试", "计算伤害: base=100, crit=1.5"),
            (LogLevel.Info, "生产环境保留", "玩家 1001 登录成功"),
            (LogLevel.Warn, "可疑，需关注", "缓存未命中率 85%"),
            (LogLevel.Error, "业务失败", "支付回调验签失败"),
            (LogLevel.Fatal, "进程级故障", "数据库连接池耗尽"),
        };
    }

    // ============================================================
    // 2. 结构化日志：不要拼字符串
    // ============================================================
    // 【错误】log.Info($"玩家 {id} 买了 {item}")
    //   问题：无法按 id 查询、无法聚合统计
    // 【正确】log.Info("item_purchased", new { playerId = id, itemId = item })
    //   好处：可以按字段查询和聚合

    public class StructuredLog
    {
        public DateTime Time;
        public LogLevel Level;
        public string Event;
        public Dictionary<string, object> Fields = new();
        public string? TraceId;

        public string ToJson()
        {
            var fields = string.Join(",", Fields.Select(kv =>
                $"\"{kv.Key}\":{(kv.Value is string s ? "\"" + s + "\"" : kv.Value)}"));
            return $"{{\"time\":\"{Time:O}\",\"level\":\"{Level}\",\"event\":\"{Event}\"" +
                   (TraceId != null ? $",\"traceId\":\"{TraceId}\"" : "") +
                   (fields.Length > 0 ? "," + fields : "") + "}";
        }

        public override string ToString() => ToJson();
    }

    public class Logger
    {
        private readonly List<StructuredLog> _logs = new();
        private readonly LogLevel _minLevel;
        private readonly string _traceId;

        public Logger(LogLevel minLevel = LogLevel.Info, string traceId = "")
        {
            _minLevel = minLevel;
            _traceId = traceId;
        }

        public void Log(LogLevel level, string evt, Dictionary<string, object>? fields = null)
        {
            if (level < _minLevel) return;   // 【关键】低于阈值的直接丢弃（省 IO）

            _logs.Add(new StructuredLog
            {
                Time = DateTime.UtcNow,
                Level = level,
                Event = evt,
                Fields = fields ?? new(),
                TraceId = string.IsNullOrEmpty(_traceId) ? null : _traceId,
            });
        }

        public void Info(string evt, Dictionary<string, object>? f = null) => Log(LogLevel.Info, evt, f);
        public void Warn(string evt, Dictionary<string, object>? f = null) => Log(LogLevel.Warn, evt, f);
        public void Error(string evt, Dictionary<string, object>? f = null) => Log(LogLevel.Error, evt, f);

        public IReadOnlyList<StructuredLog> Logs => _logs;
        public int Count => _logs.Count;

        // 按事件统计（这就是结构化日志的价值）
        public Dictionary<string, int> GroupByEvent() =>
            _logs.GroupBy(l => l.Event).ToDictionary(g => g.Key, g => g.Count());
    }

    // ============================================================
    // 3. 链路追踪：一次请求的完整路径
    // ============================================================
    // 【问题】一个请求经过 5 个服务，出问题时不知道慢在哪
    // 【方案】每个请求分配 traceId，全链路传递

    public class Span
    {
        public string Name = "";
        public long StartMs;
        public long DurationMs;
        public string TraceId = "";
        public string? ParentSpanId;
        public string SpanId = Guid.NewGuid().ToString("N").Substring(0, 8);

        public override string ToString() =>
            $"[{TraceId.Substring(0, 8)}/{SpanId}] {Name,-22} {DurationMs,5}ms";
    }

    public class Tracer
    {
        private readonly List<Span> _spans = new();
        private readonly Stack<Span> _stack = new();
        public string TraceId = Guid.NewGuid().ToString("N");

        public void Begin(string name)
        {
            var s = new Span
            {
                Name = name,
                TraceId = TraceId,
                StartMs = DateTimeOffset.UtcNow.ToUnixTimeMilliseconds(),
                ParentSpanId = _stack.Count > 0 ? _stack.Peek().SpanId : null,
            };
            _stack.Push(s);
        }

        public void End()
        {
            if (_stack.Count == 0) return;
            var s = _stack.Pop();
            s.DurationMs = DateTimeOffset.UtcNow.ToUnixTimeMilliseconds() - s.StartMs;
            _spans.Add(s);
        }

        public IReadOnlyList<Span> Spans => _spans.OrderBy(s => s.StartMs).ToList();

        public long TotalMs => _spans.Where(s => s.ParentSpanId == null).Sum(s => s.DurationMs);
    }

    // ============================================================
    // 4. 关键指标（Metrics）
    // ============================================================
    // 【三类指标】
    //   Counter  计数器（只增不减）：请求总数、错误数
    //   Gauge    瞬时值（可增可减）：在线人数、队列长度
    //   Histogram 分布（P50/P95/P99）：响应耗时

    public class Metrics
    {
        private readonly Dictionary<string, long> _counters = new();
        private readonly Dictionary<string, double> _gauges = new();
        private readonly Dictionary<string, List<double>> _histograms = new();

        public void Inc(string name, long delta = 1)
        {
            _counters.TryGetValue(name, out long v);
            _counters[name] = v + delta;
        }

        public void SetGauge(string name, double value) => _gauges[name] = value;

        public void Observe(string name, double value)
        {
            if (!_histograms.TryGetValue(name, out var list))
            {
                list = new List<double>();
                _histograms[name] = list;
            }
            list.Add(value);
        }

        public long GetCounter(string name) => _counters.TryGetValue(name, out var v) ? v : 0;
        public double GetGauge(string name) => _gauges.TryGetValue(name, out var v) ? v : 0;

        // 【关键】P99 比分均值更有价值
        public (double P50, double P95, double P99, double Max) Percentiles(string name)
        {
            if (!_histograms.TryGetValue(name, out var list) || list.Count == 0)
                return (0, 0, 0, 0);

            var sorted = list.OrderBy(x => x).ToList();
            return (
                sorted[(int)(sorted.Count * 0.50)],
                sorted[(int)(sorted.Count * 0.95)],
                sorted[Math.Min((int)(sorted.Count * 0.99), sorted.Count - 1)],
                sorted[^1]
            );
        }

        public IEnumerable<string> AllNames =>
            _counters.Keys.Concat(_gauges.Keys).Concat(_histograms.Keys).Distinct();
    }

    // ============================================================
    // 5. 告警：什么值得报警
    // ============================================================

    public static class AlertRules
    {
        public static readonly (string Name, string Condition, string Why)[] All =
        {
            ("错误率", "5xx 占比 > 1%（持续5分钟）", "真实故障"),
            ("P99 延迟", "> 1s（持续5分钟）", "用户可感知的卡顿"),
            ("连接池", "使用率 > 90%", "即将耗尽"),
            ("队列积压", "> 10000 条", "消费跟不上"),
            ("CPU", "> 85%（持续10分钟）", "容量不足"),
            ("磁盘", "> 90%", "写入会失败"),
        };
    }

    // ============================================================
    // 演示
    // ============================================================
    public static class Demo
    {
        public static void Run()
        {
            Console.WriteLine("=== 1. 日志分级 ===");
            foreach (var (lv, when, ex) in LevelGuide.All)
            {
                Console.WriteLine($"    {lv,-7} {when,-18} {ex}");
            }
            Console.WriteLine();
            Console.WriteLine("  【重要】低级别日志不要在生产开启");
            Console.WriteLine("          字符串拼接本身就有开销");

            Console.WriteLine();
            Console.WriteLine("=== 2. 结构化日志 ===");
            var logger = new Logger(LogLevel.Info, "abc123def456");

            logger.Info("player_login", new() { ["playerId"] = 1001, ["ip"] = "1.2.3.4" });
            logger.Info("player_login", new() { ["playerId"] = 1002, ["ip"] = "5.6.7.8" });
            logger.Warn("cache_miss_rate_high", new() { ["rate"] = 0.85 });
            logger.Error("payment_verify_failed", new() { ["orderId"] = "ORD-001" });

            Console.WriteLine("  日志输出（JSON 格式）:");
            foreach (var l in logger.Logs)
            {
                Console.WriteLine($"    {l.ToJson()}");
            }

            Console.WriteLine();
            Console.WriteLine("  按事件聚合（这是结构化日志的价值）:");
            foreach (var kv in logger.GroupByEvent())
            {
                Console.WriteLine($"    {kv.Key}: {kv.Value} 次");
            }

            Console.WriteLine();
            Console.WriteLine("  对比字符串拼接日志:");
            Console.WriteLine($"    log.Info(\"玩家 \" + id + \" 登录\")  -> 无法按 id 查询");
            Console.WriteLine($"    结构化日志  -> 可按 playerId 字段过滤、聚合、告警");

            Console.WriteLine();
            Console.WriteLine("=== 3. 链路追踪 ===");
            var tracer = new Tracer();
            tracer.Begin("HTTP /api/buy");
            System.Threading.Thread.Sleep(3);

            tracer.Begin("auth");
            System.Threading.Thread.Sleep(1);
            tracer.End();

            tracer.Begin("load_player");
            System.Threading.Thread.Sleep(5);
            tracer.End();

            tracer.Begin("deduct_gold");
            System.Threading.Thread.Sleep(2);
            tracer.End();

            tracer.Begin("publish_event");
            System.Threading.Thread.Sleep(1);
            tracer.End();

            tracer.End();

            Console.WriteLine($"  TraceId: {tracer.TraceId.Substring(0, 8)}");
            foreach (var s in tracer.Spans)
            {
                var indent = s.ParentSpanId == null ? "" : "  └─ ";
                Console.WriteLine($"    {indent}{s}");
            }
            Console.WriteLine($"  总耗时: {tracer.TotalMs} ms");
            Console.WriteLine();
            Console.WriteLine("  【价值】一眼看出哪一步慢");
            Console.WriteLine("          本例最慢是 load_player（5ms）");

            Console.WriteLine();
            Console.WriteLine("=== 4. 指标 ===");
            var metrics = new Metrics();

            // 模拟 100 次请求
            var rnd = new Random(42);
            for (int i = 0; i < 100; i++)
            {
                metrics.Inc("http_requests_total");
                double elapsed = rnd.Next(1, 100) + (i % 10 == 0 ? rnd.Next(500, 1500) : 0);
                metrics.Observe("http_request_duration_ms", elapsed);
                if (elapsed > 1000) metrics.Inc("http_errors_total");
            }
            metrics.SetGauge("online_players", 1523);
            metrics.SetGauge("queue_length", 342);

            Console.WriteLine($"  计数器 http_requests_total = {metrics.GetCounter("http_requests_total")}");
            Console.WriteLine($"  计数器 http_errors_total   = {metrics.GetCounter("http_errors_total")}");
            Console.WriteLine($"  瞬时值 online_players      = {metrics.GetGauge("online_players"):F0}");
            Console.WriteLine($"  瞬时值 queue_length        = {metrics.GetGauge("queue_length"):F0}");

            var (p50, p95, p99, max) = metrics.Percentiles("http_request_duration_ms");
            Console.WriteLine();
            Console.WriteLine("  耗时分布:");
            Console.WriteLine($"    P50 = {p50:F0} ms   （一半请求快于此）");
            Console.WriteLine($"    P95 = {p95:F0} ms   （95% 请求快于此）");
            Console.WriteLine($"    P99 = {p99:F0} ms   （99% 请求快于此）");
            Console.WriteLine($"    Max = {max:F0} ms");
            Console.WriteLine();
            Console.WriteLine("  【关键】看 P99 而不是平均值");
            Console.WriteLine("          平均值会被少数快请求拉低，掩盖长尾问题");

            Console.WriteLine();
            Console.WriteLine("=== 5. 告警规则 ===");
            foreach (var (name, cond, why) in AlertRules.All)
            {
                Console.WriteLine($"    {name,-10} {cond,-28} {why}");
            }
            Console.WriteLine();
            Console.WriteLine("  【告警设计原则】");
            Console.WriteLine("    · 只报「需要人处理」的（否则会被忽略）");
            Console.WriteLine("    · 加持续时间（避免瞬时抖动误报）");
            Console.WriteLine("    · 每条告警都要有处理手册");
        }
    }
}

    级别什么时候用生产环境
    Trace开发期追踪细节关闭
    Debug开发期调试关闭
    Info关键流程（登录、下单）保留
    Warn可疑但非致命保留
    Error业务失败保留
    Fatal进程级故障保留 + 告警

    二、结构化日志：不要拼字符串 ★★

    字符串拼接日志的问题 错误做法：
    log.Info($"玩家 {id} 买了 {itemId}，花费 {price}")
    → 输出：玩家 1001 买了 5001，花费 300

    问题：
    · 想查「所有买了 5001 的玩家」→ 只能文本搜索，慢且不准
    · 想统计「平均花费」→ 要正则解析
    · 字段顺序变了 → 解析全崩

    正确做法：结构化
    log.Info("item_purchased", new { playerId, itemId, price })
    → 输出 JSON：{"event":"item_purchased","playerId":1001,...}

    本页实测：能按 event 聚合成
    player_login: 2 次
    cache_miss_rate_high: 1 次
    这是字符串日志做不到的。

    三、链路追踪：一次请求的完整路径 ★★

    traceId 的价值 问题：一个请求经过 5 个服务，出问题时
    · 不知道慢在哪一步
    · 不知道哪个服务报的错
    · 日志散落各处，拼不起来

    方案：每个请求分配 traceId，全链路传递。

    本页实测输出：
    TraceId: aa575c4d
    [aa575c4d/ca9080cd] HTTP /api/buy    76ms
      └─ [aa575c4d/2cf0aa88] auth        15ms
      └─ [aa575c4d/5d9c1f8f] load_player 16ms ← 与 deduct_gold 几乎并列
      └─ [aa575c4d/6c45c18a] deduct_gold 15ms
      └─ [aa575c4d/5f326e0d] publish_event 15ms

    一眼看出：load_player 与 deduct_gold 几乎并列最慢（16ms / 15ms）。
    没有追踪的话，只能加日志重新部署再观察。

    四、指标：看 P99 而不是平均值 ★★★

    平均值会骗人 问题：100 个请求，99 个耗时 10ms，1 个耗时 2000ms
    · 平均值 = (99×10 + 2000) / 100 ≈ 30ms → 看起来很好
    · 但那个 2000ms 的请求：玩家已经骂娘了

    本页实测输出：
    P50 = 50 ms   一半请求快于此
    P95 = 1161 ms 95% 请求快于此
    P99 = 1336 ms 99% 请求快于此
    Max = 1336 ms

    看 P99 能发现长尾问题，平均值会掩盖它。

    三类指标：
    · Counter（只增）：请求总数、错误数
    · Gauge（可增减）：在线人数、队列长度
    · Histogram（分布）：响应耗时

    五、告警：只报需要人处理的

    告警疲劳是真实问题 常见错误：
    · CPU 超过 50% 就告警 → 一天几百条
    · 每个错误都告警 → 没人看
    · 没有持续时间判断 → 瞬时抖动也报

    正确的告警设计：
    · 只报「需要人处理」的
    · 加持续时间（持续 5 分钟才报）
    · 每条告警配处理手册（收到后第一步做什么）

    推荐的六条基础告警：
    · 错误率 > 1%（持续5分钟）
    · P99 延迟 > 1s
    · 连接池使用率 > 90%
    · 队列积压 > 10000
    · CPU > 85%（持续10分钟）
    · 磁盘 > 90%
