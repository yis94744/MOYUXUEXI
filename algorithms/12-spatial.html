    一、基准：朴素两两检测

    -- ============================================================
--  空间划分与碰撞优化：让 O(n^2) 变成 O(n) ★★
-- ============================================================
--  场景：1000 个敌人要互相检测碰撞/范围
--    朴素做法：两两检测 = 1000 x 999 / 2 ≈ 50 万次
--    空间划分：每个物体只和【附近的】比较 -> 几千次
--
--  这一份实现并实测四种空间划分方案。

print("=== 1. 朴素两两检测：基准 ===")

local function bruteForceCheck(entities, radius)
  local pairs_checked = 0
  local collisions = 0

  for i = 1, #entities do
    for j = i + 1, #entities do
      pairs_checked = pairs_checked + 1

      local a, b = entities[i], entities[j]
      local dx, dy = a.x - b.x, a.y - b.y
      local distSq = dx * dx + dy * dy

      if distSq < radius * radius then
        collisions = collisions + 1
      end
    end
  end

  return pairs_checked, collisions
end

for i = 1, N do
  entities[i] = {
    id = i,
    x = math.random() * 1000,           -- 1000x1000 的地图
    y = math.random() * 1000,
  }
end

print("=== 2. 均匀网格（Uniform Grid）===")
-- 最简单的空间划分：把地图切成格子，物体只和同格+邻格的比较
-- 【优点】实现极简、查询 O(1)、内存可控
-- 【缺点】物体分布不均时退化（全挤在一格里）

local Grid = {}
Grid.__index = Grid

function Grid.new(width, height, cellSize)
  local cols = math.ceil(width / cellSize)
  local rows = math.ceil(height / cellSize)

  local self = setmetatable({
    cellSize = cellSize,
    cols = cols,
    rows = rows,
    cells = {},
    entityCount = 0,
  }, Grid)

  -- 初始化格子
  for cy = 0, rows - 1 do
    self.cells[cy] = {}
    for cx = 0, cols - 1 do
      self.cells[cy][cx] = {}
    end
  end

  return self
end

function Grid:getCell(x, y)
  local cx = math.floor(x / self.cellSize)
  local cy = math.floor(y / self.cellSize)
  -- 边界钳制
  cx = math.max(0, math.min(self.cols - 1, cx))
  cy = math.max(0, math.min(self.rows - 1, cy))
  return cx, cy
end

function Grid:insert(entity)
  local cx, cy = self:getCell(entity.x, entity.y)
  local cell = self.cells[cy][cx]
  cell[#cell + 1] = entity
  entity._cellX = cx
  entity._cellY = cy
  self.entityCount = self.entityCount + 1
  return cx, cy
end

-- 查询某个格子及周围 8 格内的所有对象
function Grid:queryNeighbors(x, y)
  local results = {}
  local cx, cy = self:getCell(x, y)

  for dy = -1, 1 do
    for dx = -1, 1 do
      local nx, ny = cx + dx, cy + dy
      if nx >= 0 and nx < self.cols and ny >= 0 and ny < self.rows then
        local cell = self.cells[ny][nx]
        for i = 1, #cell do
          results[#results + 1] = cell[i]
        end
      end
    end
  end

  return results
end

-- 用网格做碰撞检测
local function gridCheck(entities, radius, cellSize)
  local grid = Grid.new(1000, 1000, cellSize)

  for _, e in ipairs(entities) do
    grid:insert(e)
  end

  local pairs_checked = 0
  local collisions = 0

  for _, a in ipairs(entities) do
    local neighbors = grid:queryNeighbors(a.x, a.y)

    for _, b in ipairs(neighbors) do
      if a.id < b.id then                -- 避免重复检测同一对
        pairs_checked = pairs_checked + 1

        local dx, dy = a.x - b.x, a.y - b.y
        if dx * dx + dy * dy < radius * radius then
          collisions = collisions + 1
        end
      end
    end
  end

  return pairs_checked, collisions, grid
end

print("=== 4. 四叉树（Quadtree）===")
-- 解决网格的「分布不均」问题：物体密集的地方自动细分
-- 【优点】自适应、适合分布不均的场景
-- 【缺点】构建有开销、动态物体频繁移动时要重建

local QuadTree = {}
QuadTree.__index = QuadTree

function QuadTree.new(x, y, w, h, capacity, maxDepth)
  return setmetatable({
    x = x, y = y, w = w, h = h,
    capacity = capacity or 4,      -- 每格最多存几个
    maxDepth = maxDepth or 6,      -- 最大深度（防止无限细分）
    depth = 0,
    entities = {},
    children = nil,                -- nil 表示还没细分
  }, QuadTree)
end

function QuadTree:contains(e)
  return e.x >= self.x and e.x < self.x + self.w
     and e.y >= self.y and e.y < self.y + self.h
end

function QuadTree:subdivide()
  local hw, hh = self.w / 2, self.h / 2
  local d = self.depth + 1

  self.children = {
    QuadTree.new(self.x,      self.y,      hw, hh, self.capacity, self.maxDepth),
    QuadTree.new(self.x + hw, self.y,      hw, hh, self.capacity, self.maxDepth),
    QuadTree.new(self.x,      self.y + hh, hw, hh, self.capacity, self.maxDepth),
    QuadTree.new(self.x + hw, self.y + hh, hw, hh, self.capacity, self.maxDepth),
  }
  for _, c in ipairs(self.children) do c.depth = d end
end

function QuadTree:insert(e)
  if not self:contains(e) then return false end

  -- 有空间且未细分 -> 直接存
  if not self.children and #self.entities < self.capacity then
    self.entities[#self.entities + 1] = e
    return true
  end

  -- 达到容量且未到最大深度 -> 细分
  if not self.children and self.depth < self.maxDepth then
    self:subdivide()
    -- 把已有对象下推到子节点
    local toRedistribute = self.entities
    self.entities = {}
    for _, old in ipairs(toRedistribute) do
      for _, child in ipairs(self.children) do
        if child:insert(old) then break end
      end
    end
  end

  -- 插入子节点
  if self.children then
    for _, child in ipairs(self.children) do
      if child:insert(e) then return true end
    end
  end

  -- 已经到了最大深度，只能存在这里
  self.entities[#self.entities + 1] = e
  return true
end

-- 查询矩形范围内的对象
function QuadTree:query(range, results)
  results = results or {}

  if not self:intersects(range) then return results end

  for _, e in ipairs(self.entities) do
    if e.x >= range.x and e.x < range.x + range.w
   and e.y >= range.y and e.y < range.y + range.h then
      results[#results + 1] = e
    end
  end

  if self.children then
    for _, child in ipairs(self.children) do
      child:query(range, results)
    end
  end

  return results
end

function QuadTree:intersects(range)
  return not (range.x + range.w < self.x or range.x > self.x + self.w
           or range.y + range.h < self.y or range.y > self.y + self.h)
end

-- 统计节点数
function QuadTree:countNodes()
  local n = 1
  if self.children then
    for _, c in ipairs(self.children) do n = n + c:countNodes() end
  end
  return n
end

print("=== 5. 空间哈希（Spatial Hash）===")
-- 网格的改进版：用哈希表存格子，不需要预先知道地图边界
-- 【优点】无限大世界、稀疏数据、动态增删方便
-- 【缺点】哈希冲突、需要处理负坐标

local SpatialHash = {}
SpatialHash.__index = SpatialHash

function SpatialHash.new(cellSize)
  return setmetatable({
    cellSize = cellSize,
    buckets = {},
    total = 0,
  }, SpatialHash)
end

function SpatialHash:key(cx, cy)
  -- 【关键】把二维坐标编码成一维 key
  -- 用字符串会分配内存，用数字运算更快
  return cx * 73856093 + cy * 19349663
end

function SpatialHash:insert(e)
  local cx = math.floor(e.x / self.cellSize)
  local cy = math.floor(e.y / self.cellSize)
  local k = self:key(cx, cy)

  if not self.buckets[k] then self.buckets[k] = {} end
  self.buckets[k][#self.buckets[k] + 1] = e
  self.total = self.total + 1
end

function SpatialHash:queryNeighbors(x, y)
  local results = {}
  local cx = math.floor(x / self.cellSize)
  local cy = math.floor(y / self.cellSize)

  for dy = -1, 1 do
    for dx = -1, 1 do
      local k = self:key(cx + dx, cy + dy)
      local bucket = self.buckets[k]
      if bucket then
        for i = 1, #bucket do
          results[#results + 1] = bucket[i]
        end
      end
    end
  end

  return results
end

        shCollisions = shCollisions + 1
      end
    end
  end
end

    二、均匀网格与那个致命陷阱 ★★★

    实测发现的静默漏检（这是本页最重要的内容） 在测试"格子大小对结果的影响"时，出现了一个异常现象：

    格子大小检测对数检出碰撞数正确性
    53333漏检
    10112111漏检
    20424167正确
    401701167正确
    20032974167正确
    正确答案是 167 个碰撞（暴力法的结果）。
    但格子大小 5 时只检出 33 个，格子 10 时只检出 111 个——都漏检了！

    为什么：查询半径是 20，而格子只有 5。
    只查「3×3 邻格」最多只能覆盖 15 个单位的范围，距离在 15~20 之间的碰撞全被漏掉了。

    最危险的地方：格子 5 的"检测对数只有 33"，看起来比其他方案快得多——如果你只看性能数字，会以为这是最优配置，实际上结果是错的。
    铁律：格子大小必须 >= 查询半径
    格子大小结果
    格子 < 半径漏检（结果是错的！）
    格子 = 半径刚好够，但边界情况容易出问题
    格子 = 2 倍半径推荐值
    格子 >> 半径每格对象过多，退化成暴力法
    这个坑的可怕之处在于：它不会报错、不会崩溃，只是"某些碰撞检测不到"。
    表现为——子弹偶尔穿人、范围技能偶尔漏怪、玩家偶尔穿过墙体。
    这类 bug 极难复现，因为取决于物体的精确位置是否恰好落在某个边界上。

    三、四叉树：处理分布不均

    网格的弱点与四叉树的解法 网格的问题：如果所有物体都挤在地图的一个角落，那一格里会有几百个对象——退化成暴力法。

    四叉树的解法：物体密集的地方自动细分，稀疏的地方保持大格子。

    但四叉树有自己的代价：
    均匀网格四叉树
    构建复杂度O(n)O(n log n)
    查询复杂度O(1) 定位O(log n) 下降
    动态更新O(1) 改格子索引O(log n) 删除+重插
    内存固定动态（节点多时更大）
    适用动态物体静态物体
    关键结论：频繁移动的物体，网格方案的更新成本远低于四叉树。
    所以四叉树更适合地形、建筑、植被这类静态或缓慢移动的对象。

    四、空间哈希：无限地图的选择

    空间哈希解决了什么 均匀网格需要预先知道地图边界（要分配 cols × rows 的数组）。
    但开放世界、程序化生成的地图可能非常大甚至无限。

    空间哈希用哈希表代替二维数组：
    • 格子的 key = hash(cellX, cellY)
    • 只创建有物体的格子，稀疏数据下内存占用极小
    • 支持负坐标（无限地图）

    注意：哈希会有冲突——不同格子可能映射到同一个 key。实践中这个概率极低，且冲突只影响性能（多查几个对象），不影响正确性。

    五、四种方案完整对比

    方案构建查询动态更新内存适用场景
    暴力两两无O(n²)无无n < 100
    均匀网格O(n)O(1)O(1)固定分布均匀、地图固定
    四叉树O(n log n)O(log n)O(log n)动态分布不均、静态为主
    空间哈希O(n)O(1)O(1)稀疏无限地图、动态增删
    实战选型建议 ① 首选均匀网格或空间哈希——实现简单、性能好，90% 的场景够用。
    ② 四叉树只适合静态物体——地形、建筑、植被。动态物体用它会得不偿失。
    ③ 格子大小取 2 倍查询半径——并且一定要用 Profiler 实测你项目的最优值。
    ④ 记住 Broad Phase 与 Narrow Phase 的分工——空间划分只是粗筛，筛出候选后仍要做精确检测。
    ⑤ Unity 的 Physics.OverlapSphere 已经优化过——但有自己的开销和 GC 问题（见 C# 站页 26）。当你需要自定义单位范围查询、或在服务端做逻辑时，才需要自己实现。
