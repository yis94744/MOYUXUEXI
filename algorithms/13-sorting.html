local function bench(fn, ...)
  local t = os.clock()
  local r = fn(...)
  return (os.clock() - t), r
end

local function isSorted(t)
  for i = 2, #t do
    if t[i - 1] > t[i] then return false end
  end
  return true
end

-- 冒泡排序 O(n^2)
local function bubbleSort(arr)
  local a = {}
  for i = 1, #arr do a[i] = arr[i] end
  local n = #a
  for i = 1, n - 1 do
    local swapped = false
    for j = 1, n - i do
      if a[j] > a[j + 1] then
        a[j], a[j + 1] = a[j + 1], a[j]
        swapped = true
      end
    end
    if not swapped then break end     -- 优化：已有序就提前结束
  end
  return a
end

-- 插入排序 O(n^2) 但小数组很快
local function insertionSort(arr)
  local a = {}
  for i = 1, #arr do a[i] = arr[i] end
  for i = 2, #a do
    local key = a[i]
    local j = i - 1
    while j >= 1 and a[j] > key do
      a[j + 1] = a[j]
      j = j - 1
    end
    a[j + 1] = key
  end
  return a
end

-- 希尔排序（插入排序的改进）
local function shellSort(arr)
  local a = {}
  for i = 1, #arr do a[i] = arr[i] end
  local n = #a
  local gap = math.floor(n / 2)
  while gap > 0 do
    for i = gap + 1, n do
      local temp = a[i]
      local j = i
      while j > gap and a[j - gap] > temp do
        a[j] = a[j - gap]
        j = j - gap
      end
      a[j] = temp
    end
    gap = math.floor(gap / 2)
  end
  return a
end

-- 归并排序 O(n log n)，稳定
local function mergeSort(arr)
  local a = {}
  for i = 1, #arr do a[i] = arr[i] end

  local function merge(left, right)
    local result = {}
    local i, j = 1, 1
    while i <= #left and j <= #right do
      -- 【重要】用 <= 保证稳定性
      if left[i] <= right[j] then
        result[#result + 1] = left[i]
        i = i + 1
      else
        result[#result + 1] = right[j]
        j = j + 1
      end
    end
    while i <= #left do result[#result + 1] = left[i]; i = i + 1 end
    while j <= #right do result[#result + 1] = right[j]; j = j + 1 end
    return result
  end

  local function sort(t)
    if #t <= 1 then return t end
    local mid = math.floor(#t / 2)
    local left, right = {}, {}
    for i = 1, mid do left[i] = t[i] end
    for i = mid + 1, #t do right[i - mid] = t[i] end
    return merge(sort(left), sort(right))
  end

  return sort(a)
end

-- 快速排序 O(n log n) 平均，原地
local function quickSort(arr)
  local a = {}
  for i = 1, #arr do a[i] = arr[i] end

  local function qsort(lo, hi)
    if lo >= hi then return end

    -- 【优化】三数取中选基准，避免有序数组退化
    local mid = math.floor((lo + hi) / 2)
    if a[mid] < a[lo] then a[mid], a[lo] = a[lo], a[mid] end
    if a[hi] < a[lo] then a[hi], a[lo] = a[lo], a[hi] end
    if a[hi] < a[mid] then a[hi], a[mid] = a[mid], a[hi] end
    a[mid], a[hi] = a[hi], a[mid]     -- 基准放到末尾

    local pivot = a[hi]
    local i = lo - 1
    for j = lo, hi - 1 do
      if a[j] <= pivot then
        i = i + 1
        a[i], a[j] = a[j], a[i]
      end
    end
    a[i + 1], a[hi] = a[hi], a[i + 1]

    qsort(lo, i)
    qsort(i + 2, hi)
  end

  qsort(1, #a)
  return a
end

-- 堆排序 O(n log n)，原地，最坏也 O(n log n)
local function heapSort(arr)
  local a = {}
  for i = 1, #arr do a[i] = arr[i] end
  local n = #a

  local function siftDown(i, size)
    while true do
      local largest = i
      local l, r = i * 2, i * 2 + 1
      if l <= size and a[l] > a[largest] then largest = l end
      if r <= size and a[r] > a[largest] then largest = r end
      if largest == i then break end
      a[i], a[largest] = a[largest], a[i]
      i = largest
    end
  end

  for i = math.floor(n / 2), 1, -1 do siftDown(i, n) end
  for i = n, 2, -1 do
    a[1], a[i] = a[i], a[1]
    siftDown(1, i - 1)
  end
  return a
end

-- 用最小堆找最大的 K 个（复习页 07）
local function topK(data, k)
  local heap = {}     -- 最小堆，只存 K 个

  local function siftUp(i)
    while i > 1 do
      local parent = math.floor(i / 2)
      if heap[parent] <= heap[i] then break end
      heap[parent], heap[i] = heap[i], heap[parent]
      i = parent
    end
  end

  local function siftDown(i)
    local n = #heap
    while true do
      local smallest = i
      local l, r = i * 2, i * 2 + 1
      if l <= n and heap[l] < heap[smallest] then smallest = l end
      if r <= n and heap[r] < heap[smallest] then smallest = r end
      if smallest == i then break end
      heap[i], heap[smallest] = heap[smallest], heap[i]
      i = smallest
    end
  end

  for _, v in ipairs(data) do
    if #heap < k then
      heap[#heap + 1] = v
      siftUp(#heap)
    elseif v > heap[1] then
      heap[1] = v
      siftDown(1)
    end
  end

  -- 【坑】最小堆"依次弹出"得到的是【升序】（堆顶始终是最小的）
  --   想要降序，必须弹出后再反转，或者用头插
  local result = {}
  while #heap > 0 do
    result[#result + 1] = heap[1]          -- 先弹出最小的
    heap[1] = heap[#heap]
    heap[#heap] = nil
    siftDown(1)
  end
  -- 反转 -> 降序
  local n = #result
  for i = 1, math.floor(n / 2) do
    result[i], result[n - i + 1] = result[n - i + 1], result[i]
  end
  return result
end

-- 快速选择 O(n) 平均
local function quickSelect(arr, k, compare)
  compare = compare or function(a, b) return a < b end
  local a = {}
  for i = 1, #arr do a[i] = arr[i] end

  local function select(lo, hi, target)
    if lo == hi then return a[lo] end

    -- 分区
    local pivot = a[hi]
    local i = lo - 1
    for j = lo, hi - 1 do
      if compare(a[j], pivot) then
        i = i + 1
        a[i], a[j] = a[j], a[i]
      end
    end
    a[i + 1], a[hi] = a[hi], a[i + 1]
    local pivotIndex = i + 1

    if target == pivotIndex then
      return a[pivotIndex]
    elseif target < pivotIndex then
      return select(lo, pivotIndex - 1, target)
    else
      return select(pivotIndex + 1, hi, target)
    end
  end

  return select(1, #a, k)
end

    二、三个"不用排序"的技巧 ★★★

    技巧一：Top-K 用堆（实测 5 毫秒 vs 85~96 毫秒） 场景：排行榜只要前 100 名。
    错误做法：全部排序再取前 100。
    正确做法：维护一个大小为 K 的最小堆。
    方法复杂度实测（10 万数据，K=10）
    完整排序O(n log n)约 85~96 毫秒
    堆的 Top-KO(n log K)约 5 毫秒
    快 17~21 倍（4 次运行分别为 17.4 / 18.0 / 19.2 / 21.5 倍）。
    K 越小，优势越大。
    注意：本页的 Top-K 用 Lua 手写堆，而"完整排序"是 table.sort 的 C 实现；如果两者都用 Lua 实现，倍数会小一些（页 07 的对比就是 8~9 倍）。这也是"同一结论在不同实现下倍数不同"的典型例子。
    技巧二：找第 K 名用快速选择（O(n)） 场景：只要知道"第 100 名是多少分"，不需要知道完整排名。
    快速选择（QuickSelect）的思路：
    · 类似快速排序的分区
    · 但只递归包含目标的那一半，另一半直接丢弃
    · 所以平均复杂度是 O(n)，不是 O(n log n)
    本页实测：10 万数据找第 100 大——快速选择明显快于完整排序，结果一致。
    技巧三：小数组用插入排序（实测反超快排） 实测数据（每个规模重复 500 次取总时间）：
    数据量插入排序快速排序结果
    8约 1 ms约 1 ms基本持平（差异在计时精度以下）
    161~2 ms2~3 ms插入排序略快（1~3 倍，波动大）
    323~5 ms4~6 ms两者接近
    6412~14 ms8~12 ms快排稳定反超
    分界点大约在 32——32 之前插入排序不慢，64 之后快排稳定领先。
    原因：小数组时递归与分区开销占比大，而插入排序常数极小、对近乎有序的数据很高效。
    注意：8 和 16 这两档单次耗时只有 1~2 微秒，已经逼近 os.clock 的计时精度，所以倍数波动很大——这也是"分界点"只能给区间、不能给精确值的原因。
    实战应用：标准库的快排会在数组小于阈值时切换到插入排序——这叫混合排序（IntroSort）。

    三、稳定性：什么时候重要

    排行榜的典型需求 需求："先按分数降序，分数相同的先到先得"。
    用不稳定排序的问题：分数相同时，顺序可能是乱的——玩家会看到"我明明先到，却排在后面"。
    两种解法：
    ① 用稳定排序（归并排序）
    ② 在比较函数里写全条件：
    if a.score ~= b.score then return a.score > b.score end
    return a.joinOrder < b.joinOrder
    本页实测：方案 ② 输出 A(100), C(100), E(100), B(90), D(90)——分数相同的保持了加入顺序。
    实战中方案 ② 更常用——因为它不依赖排序算法本身的稳定性。

    ⑤ 能实现多级排序（分数 + 时间）。

    第 ③④ 条是这一页最有价值的内容——它们能帮你避免"为了取前 10 个排了 10 万个"这种浪费。
