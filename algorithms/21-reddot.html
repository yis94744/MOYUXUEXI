    -- ============================================================
--  红点系统：从朴素实现到百万级节点优化 ★★★
-- ============================================================
--  红点是手游里最容易被低估的系统：
--    看起来只是「显示一个小圆点」，
--    但在大型项目里，红点节点的数量可以达到【几十万个】，
--    而且每次条件变化都可能触发全树重算。
--
--  这一份代码用 Lua 演示全部核心原理（C# 实现思路完全一致）。

print("=== 1. 朴素实现：每次查询都递归重算 ===")
-- 问题：每次 UI 刷新都从根节点往下递归，节点多时会卡

local SimpleRedDot = {}

function SimpleRedDot.createSimple()
  local tree = {
    name = "root",
    selfVisible = false,      -- 节点自身是否满足显示条件
    children = {},
  }

  -- 添加节点
  local function addNode(parent, node)
    node.children = node.children or {}
    parent.children[#parent.children + 1] = node
    return node
  end

  -- 【朴素查询】每次都递归全树
  local function isVisible(node)
    if node.selfVisible then return true end
    for _, child in ipairs(node.children) do
      if isVisible(child) then return true end
    end
    return false
  end

  -- 构建一棵测试树：3 层，每层 10 个子节点
  local function buildTree(parent, depth)
    if depth == 0 then return 0 end
    local count = 0
    for i = 1, 10 do
      local child = addNode(parent, { name = parent.name .. "." .. i, selfVisible = false })
      count = count + 1 + buildTree(child, depth - 1)
    end
    return count
  end

  local total = buildTree(tree, 3)

  -- 模拟：只有最底层一个节点是亮的
  local function setLeafVisible(node, depth)
    if depth == 0 then node.selfVisible = true return true end
    for _, child in ipairs(node.children) do
      if setLeafVisible(child, depth - 1) then return true end
    end
    return false
  end
  setLeafVisible(tree, 3)

  -- 测性能：反复查询根节点
  local N = 20000
  local t1 = os.clock()
  local result
  for i = 1, N do
    result = isVisible(tree)
  end
  local simpleTime = os.clock() - t1

  print(string.format("  节点总数: %d", total))
  print(string.format("  查询 %d 次耗时: %.4f 秒", N, simpleTime))
  print(string.format("  单次查询: %.1f 微秒", simpleTime / N * 1000000))
  print("  【问题】每次查询都遍历全树，节点越多越慢")

  return tree, total, simpleTime
end

local simpleTree, simpleTotal, simpleTime = SimpleRedDot.createSimple()

print("=== 2. 优化一：脏标记 + 缓存 ===")
-- 核心思路：
--   节点状态变化时，只向上标记父节点「需要重算」，
--   查询时如果缓存有效就直接返回缓存值。
--   这是【空间换时间】的经典应用。

local DirtyRedDot = {}

function DirtyRedDot.create()
  local tree = {
    name = "root",
    selfVisible = false,
    children = {},
    cachedVisible = false,     -- 缓存的可见性
    dirty = true,              -- 是否需要重算
    parent = nil,
  }

  local function addNode(parent, node)
    node.children = node.children or {}
    node.parent = parent
    node.cachedVisible = false
    node.dirty = true
    parent.children[#parent.children + 1] = node
    return node
  end

  -- 【关键】向上标记脏，而不是向下重算
  local function markDirtyUpward(node)
    local cur = node
    while cur do
      if cur.dirty then break end        -- 已经是脏的，祖先肯定也脏，可以提前退出
      cur.dirty = true
      cur = cur.parent
    end
  end

  -- 设置节点自身状态（触发向上标脏）
  local function setSelfVisible(node, visible)
    if node.selfVisible == visible then return end
    node.selfVisible = visible
    markDirtyUpward(node)
  end

  -- 【惰性求值】只在需要时重算，且只算脏的那些
  local recalcCount = 0
  local function isVisible(node)
    if not node.dirty then
      return node.cachedVisible          -- 缓存有效，O(1) 返回
    end

    recalcCount = recalcCount + 1
    local result = node.selfVisible
    if not result then
      for _, child in ipairs(node.children) do
        if isVisible(child) then
          result = true
          break
        end
      end
    end

    node.cachedVisible = result
    node.dirty = false
    return result
  end

  -- 构建同样的树
  local function buildTree(parent, depth)
    if depth == 0 then return 0 end
    local count = 0
    for i = 1, 10 do
      local child = addNode(parent, { name = parent.name .. "." .. i })
      count = count + 1 + buildTree(child, depth - 1)
    end
    return count
  end

  local total = buildTree(tree, 3)

  local function getLeaf(node, depth)
    if depth == 0 then return node end
    return getLeaf(node.children[1], depth - 1)
  end

  setSelfVisible(getLeaf(tree, 3), true)

  local N = 20000
  local t1 = os.clock()
  local result
  for i = 1, N do
    result = isVisible(tree)
  end
  local dirtyTime = os.clock() - t1

  print(string.format("  节点总数: %d", total))
  print(string.format("  查询 %d 次耗时: %.4f 秒", N, dirtyTime))
  print(string.format("  单次查询: %.1f 微秒", dirtyTime / N * 1000000))
  print(string.format("  实际重算次数: %d（其余命中缓存）", recalcCount))
  print(string.format("  相比朴素实现快 %.0f 倍", simpleTime / dirtyTime))

  return { tree = tree, setSelfVisible = setSelfVisible, isVisible = isVisible, getLeaf = getLeaf }
end

local dirty = DirtyRedDot.create()

print("=== 3. 优化二：批量更新的合并 ===")
-- 如果一帧内改了 1000 个节点，逐个标脏会重复走同一条路径。
-- 用「脏集合 + 统一提交」把重复工作合并。

local BatchRedDot = {}

function BatchRedDot.create()
  local pendingSet = {}       -- 待处理的节点集合
  local pendingCount = 0

  local function markDirty(node)
    if node.dirty then return end      -- 已标过，跳过（这就是合并的效果）
    node.dirty = true
    pendingSet[node] = true
    pendingCount = pendingCount + 1
  end

  local function flush()
    -- 统一向上传播（去重后的节点）
    for node in pairs(pendingSet) do
      local cur = node.parent
      while cur and not cur.dirty do
        cur.dirty = true
        cur = cur.parent
      end
    end
    pendingSet = {}
    pendingCount = 0
  end

  return { markDirty = markDirty, flush = flush, getPendingCount = function() return pendingCount end }
end

local batch = BatchRedDot.create()

-- 构建一个测试场景：1000 个节点同时变化
local nodes = {}
for i = 1, 1000 do
  nodes[i] = { dirty = false, parent = nil }
end
-- 让它们共享部分父节点（模拟真实树结构）
local sharedParent = { dirty = false, parent = nil }
for i = 1, 500 do nodes[i].parent = sharedParent end

local t2 = os.clock()
for i = 1, 1000 do
  batch.markDirty(nodes[i])
end
local markTime = os.clock() - t2

print(string.format("  标记 1000 个节点耗时: %.6f 秒", markTime))
print(string.format("  实际入队数量: %d", batch.getPendingCount()))
batch.flush()
print("  提交后：共享父节点只被传播一次（去重生效）")

print("=== 5. 优化四：非整型红点的增量计数 ===")
-- 当红点还显示数量（如「3 个未领奖励」）时，
-- 聚合不再用「或」而是用「求和」，可以维护增量计数。

local CounterRedDot = {}

function CounterRedDot.create()
  local tree = { count = 0, selfCount = 0, children = {}, parent = nil }

  local function addChild(parent, node)
    node.children = node.children or {}
    node.parent = parent
    node.count = node.count or 0          -- 聚合值（含自身与全部子孙）
    node.selfCount = node.selfCount or 0  -- 仅自身
    parent.children[#parent.children + 1] = node
    return node
  end

  -- 增量更新：只加差值，不重算全树
  local function addCount(node, delta)
    if delta == 0 then return end
    node.selfCount = node.selfCount + delta

    -- 【关键】向上累加，O(树高) 而不是 O(节点数)
    local cur = node
    while cur do
      cur.count = cur.count + delta
      cur = cur.parent
    end
  end

  return { tree = tree, addChild = addChild, addCount = addCount }
end

local counter = CounterRedDot.create()
local c1 = counter.addChild(counter.tree, {})
local c2 = counter.addChild(c1, {})
local c3 = counter.addChild(c2, {})

counter.addCount(c3, 3)
counter.addCount(c2, 2)

print(string.format("  叶子节点 count = %d", c3.count))
print(string.format("  中间节点 count = %d", c2.count))
print(string.format("  根节点 count = %d", counter.tree.count))
print("  【关键】更新是 O(树高) 而不是 O(节点数)——与节点总数无关")
print("  适用：计数类红点（可累加）；不适用：布尔类红点（需用脏标记方案）")

    朴素实现的问题：O(N) 查询 每次查询都从根节点递归整棵树。

    实测数据（1110 个节点，查询 2 万次）：
    • 朴素递归：0.0060 秒（单次 0.3 微秒）
    • 脏标记：0.0010 秒（单次 0.1 微秒）
    • 快 6 倍，且实际重算只有 4 次

    注意：1110 个节点时差距还不算大。但当节点达到 10 万级时，朴素实现会让 UI 卡死——因为每次打开界面都要遍历全部节点。

    二、核心优化：脏标记 + 缓存 ★★

    脏标记的两个关键点 ① 向上标脏，而不是向下重算
    节点状态变化时，只沿着父链向上标记"需要重算"，而不是立刻重算整棵子树。
    代价：O(树高)，通常只有 3~6 步。

    ② 惰性求值
    真正重算发生在查询时，而且只算被标脏的节点。
    这样"一帧内改了 1000 个节点"的场景，也只会在下一次查询时重算一次。

    ③ 提前退出
    标脏时如果发现祖先已经是脏的，就可以停止向上——因为再往上肯定也脏了。
    这个细节能省下大量重复工作。

    三、进阶优化：批量合并

    为什么需要批量合并 考虑这个场景：一帧内玩家获得了 100 个道具，每个都触发红点变化。
    如果逐个标脏，同一个父节点会被重复标记 100 次——虽然第二次之后就提前退出了，但仍有开销。

    做法：先把所有变化的节点放进一个集合，帧末统一提交。
    集合天然去重，共享的父节点只会被传播一次。

    四、计数类红点：增量更新 ★★

    布尔红点和计数红点，聚合方式完全不同 布尔红点（"有没有"）：聚合用 或 运算——任一子节点亮，父节点就亮。
    计数红点（"有几个"）：聚合用 求和 运算——父节点 = 自身 + 所有子节点。

    关键差别：计数类红点可以用增量更新！
    当某个叶子从 3 变成 5（增加 2），只需要沿着父链每层加 2，复杂度 O(树高)。
    这与节点总数完全无关——10 万节点和 100 节点一样快。

    但布尔红点不能这样做：一个子节点从亮变灭，父节点是否还亮取决于其他子节点。所以必须用脏标记方案。
    实战中的组合方案 真实项目通常是四种方案组合使用：
    ① 树结构 + 脏标记——解决通用场景（布尔红点）
    ② 批量合并——解决一帧内大量变化
    ③ UI 引用计数——只重算被界面引用的分支
    ④ 计数类走增量——避免整树重算

    面试加分点：能说出"不同红点类型用不同聚合策略"，而不是只会一个递归全树。

    五、UI 引用计数：把 10 万降到 200

    这是大型项目的关键优化 红点树可能很大，但玩家同时能看到的节点很少——只有当前展开的界面分支。

    做法：
    ① 每个节点维护一个引用计数，表示"有多少个 UI 元素正在监听它"；
    ② 只有引用计数 > 0 的节点才参与重算；
    ③ 未展开的分支直接返回缓存值，不递归。

    效果：假设 10 万节点里只有 200 个被界面引用，重算规模从 10 万降到 200——提升 500 倍。

    实现要点：UI 元素在 OnEnable 时 retain、OnDisable 时 release，与事件订阅的模式完全一致（见页 05）。

    七、完整架构设计

    生产环境的红点系统通常分成四层 ① 数据层（RedDotNode）
    树的节点结构：name、selfValue、children、parent、cachedValue、dirty。

    ② 逻辑层（RedDotManager）
    负责：注册节点、设置条件、标脏、重算、通知。这是核心。

    ③ 绑定层（RedDotBinder）
    UI 组件挂上它，声明"我监听哪个红点路径"。它负责 retain/release 和刷新显示。

    ④ 配置层
    红点树的结构和触发条件应该配置化——因为策划要能改。
    常见做法：节点树用 JSON/Excel 配置，触发条件绑定到事件名。
    三个上线后才会暴露的坑 ① 事件没解绑导致内存泄漏
    UI 关闭了但红点监听还在，节点引用着已销毁的 UI 对象——这和页 05 讲的事件泄漏是同一个问题。

    ② 初始化时全树标脏
    登录时一次性设置几百个条件，每个都触发向上标脏。应该先静默设置，最后统一提交流程一次。

    ③ 服务器数据和客户端不同步
    红点状态本质上来自服务器。断线重连、跨天刷新、活动开启——都要有完整的重算机制，不能只靠增量。
