print("=== 模式 1：对象池（Object Pool）===")
-- 【解决的问题】频繁创建/销毁对象的开销与 GC 压力
-- 【适用】子弹、特效、伤害数字、敌人
-- 【不适用】长期存在的对象（管理器）、创建成本极低的简单对象

local ObjectPool = {}
ObjectPool.__index = ObjectPool

function ObjectPool.new(factory, reset, prewarm)
  local self = setmetatable({
    _factory = factory,
    _reset = reset or function() end,
    _free = {},
    _createdCount = 0,
    _reusedCount = 0,
  }, ObjectPool)

  -- 预热：提前创建，避免运行时卡顿
  for i = 1, (prewarm or 0) do
    local obj = self._factory()
    self._free[#self._free + 1] = obj
    self._createdCount = self._createdCount + 1
  end

  return self
end

function ObjectPool:get()
  local obj = table.remove(self._free)
  if obj then
    self._reusedCount = self._reusedCount + 1
    return obj, true                      -- true 表示复用的
  end
  self._createdCount = self._createdCount + 1
  return self._factory(), false           -- false 表示新建的
end

function ObjectPool:release(obj)
  self._reset(obj)
  self._free[#self._free + 1] = obj
end

function ObjectPool:stats()
  return {
    created = self._createdCount,
    reused = self._reusedCount,
    free = #self._free,
  }
end

-- 实测：不复用 vs 复用
local N = 200000

local t1 = os.clock()
for i = 1, N do
  local bullet = { x = 0, y = 0, vx = 1, vy = 1, life = 3, id = i }
  bullet.life = bullet.life - 0.016
end
local noPoolTime = os.clock() - t1

local pool = ObjectPool.new(
  function() return { x = 0, y = 0, vx = 0, vy = 0, life = 3 } end,
  function(o) o.life = 3 end,
  50
)

local t2 = os.clock()
for i = 1, N do
  local bullet = pool:get()
  bullet.life = bullet.life - 0.016
  pool:release(bullet)
end
local poolTime = os.clock() - t2

local st = pool:stats()
print(string.format("  不复用（每次建新表）: %.4f 秒", noPoolTime))
print(string.format("  对象池（复用）:       %.4f 秒", poolTime))
print(string.format("  统计: 共创建 %d 个对象，复用了 %d 次", st.created, st.reused))
print()
print("  【实测结论 —— 和直觉相反】")
print(string.format("  在这个测试里，对象池并没有更快（%.2fx）！", noPoolTime / poolTime))
print()
print("  原因：这个测试的「对象」是一个只有 5 个字段的小表，")
print("        创建它的成本本身就极低（Lua 分配小块内存很快）。")
print("        而对象池的 get/release 有函数调用和表操作开销，")
print("        两者大致相抵。")
print()
print("  【那为什么还要用对象池？】真正的原因不是「建对象慢」，而是：")

print("=== 模式 2：缓存（Memoization）===")
-- 【解决的问题】重复计算相同的值
-- 【适用】纯函数、配置查找、坐标转换
-- 【注意】缓存本身占内存，要有失效/淘汰策略

local function memoize(fn)
  local cache = {}
  local hits, misses = 0, 0

  local wrapper = function(k)
    local v = cache[k]
    if v ~= nil then
      hits = hits + 1
      return v
    end
    misses = misses + 1
    v = fn(k)
    cache[k] = v
    return v
  end

  return wrapper, function() return hits, misses end
end

-- 一个"昂贵"的计算
local function expensiveCalc(n)
  local sum = 0
  for i = 1, 1000 do sum = sum + i * n end
  return sum
end

local memoFn, getStats = memoize(expensiveCalc)

local t3 = os.clock()
for i = 1, 10000 do expensiveCalc(i % 100) end
local noMemo = os.clock() - t3

local t4 = os.clock()
for i = 1, 10000 do memoFn(i % 100) end
local withMemo = os.clock() - t4

local hits, misses = getStats()
print(string.format("  未缓存: %.4f 秒", noMemo))
print(string.format("  已缓存: %.4f 秒", withMemo))
if withMemo > 0 then
  print(string.format("  快 %.1f 倍", noMemo / withMemo))
end
print(string.format("  命中 %d 次，未命中 %d 次（命中率 %.1f%%）", hits, misses, hits / (hits + misses) * 100))
print("  【注意】key 只用了 100 个不同值，所以缓存效果极好")
print("          如果 key 每次都不同（比如用对象做 key），缓存反而拖慢")

print()
print("=== 模式 3：批处理（Batching）===")
-- 【解决的问题】每次操作都有固定开销时，合并操作能摊薄开销
-- 【适用】Draw Call、网络包、数据库写入、文件 IO
-- 【特征】固定开销 N 次 -> 1 次

local BATCH_SIZE = 50

-- 模拟：每次操作有 1 单位的固定开销
local function simulateBatch(totalItems, batchSize)
  local operations = 0
  local i = 1
  while i <= totalItems do
    -- 一次批处理操作
    operations = operations + 1
    -- 处理这一批中的每个元素（每个元素也有微小开销）
    local endIdx = math.min(i + batchSize - 1, totalItems)
    i = endIdx + 1
  end
  return operations
end

print(string.format("  处理 10000 个元素："))
for _, bs in ipairs({ 1, 10, 50, 100 }) do
  local ops = simulateBatch(10000, bs)
  print(string.format("    批大小 %-4d -> %d 次操作（固定开销减少 %.0f 倍）",
    bs, ops, 10000 / ops))
end
print("  【实战例子】")
print("    · Unity 的 Static Batching：把静态物体合并成一个 Draw Call")
print("    · 网络：把 10 个属性变化打包成一个包发送")
print("    · 存档：把多次修改攒到关服/切场景时一次性写入")

print("=== 模式 4：分帧（Time Slicing）===")
-- 【解决的问题】单帧做了太多事，导致卡顿
-- 【做法】把工作量拆到多帧，每帧做一部分
-- 【适用】寻路、AI 决策、资源加载、排序、路径计算

local TimeSlicer = {}
TimeSlicer.__index = TimeSlicer

function TimeSlicer.new(workFn, itemsPerFrame)
  return setmetatable({
    _workFn = workFn,
    _itemsPerFrame = itemsPerFrame or 10,
    _queue = {},
    _completed = 0,
  }, TimeSlicer)
end

function TimeSlicer:push(item)
  self._queue[#self._queue + 1] = item
end

function TimeSlicer:update()
  local processed = 0
  while #self._queue > 0 and processed < self._itemsPerFrame do
    local item = table.remove(self._queue, 1)
    self._workFn(item)
    processed = processed + 1
    self._completed = self._completed + 1
  end
  return processed
end

function TimeSlicer:pendingCount() return #self._queue end

-- 演示：100 个任务，每帧处理 10 个
local slicer = TimeSlicer.new(function(item) end, 10)
for i = 1, 100 do slicer:push(i) end

print("  100 个任务，每帧处理 10 个：")
local frame = 0
while slicer:pendingCount() > 0 do
  frame = frame + 1
  local processed = slicer:update()
  if frame <= 3 or frame >= 9 then
    print(string.format("    帧 %-2d: 处理 %d 个，剩余 %d 个", frame, processed, slicer:pendingCount()))
  elseif frame == 4 then
    print("    ...")
  end
end
print(string.format("  总共用了 %d 帧完成", frame))
print()
print("  【实战经验】")
print("    · 每帧预算通常 1~2 毫秒，用 Profiler 测出实际耗时再定")
print("    · 必须显示进度条，否则玩家会以为卡了")
print("    · 优先级加队列：重要的任务先处理")

print("=== 模式 5：空间换时间 ===")
-- 【核心】预计算并存储结果，用内存换取 CPU
-- 【适用】查表代替计算、预烘焙光照、预计算的寻路路径

print("  例子：三角函数查表")
local SAMPLE = 360
local sinTable = {}
for i = 0, SAMPLE - 1 do
  sinTable[i] = math.sin(i * math.pi / 180)
end

local function fastSin(deg)
  local idx = math.floor(deg) % SAMPLE
  if idx < 0 then idx = idx + SAMPLE end
  return sinTable[idx]
end

local M = 1000000
local t5 = os.clock()
local s1 = 0
for i = 1, M do s1 = s1 + math.sin(i % 360 * math.pi / 180) end
local mathTime = os.clock() - t5

local t6 = os.clock()
local s2 = 0
for i = 1, M do s2 = s2 + fastSin(i % 360) end
local tableTime = os.clock() - t6

print(string.format("  直接算 math.sin: %.4f 秒", mathTime))
print(string.format("  查表:            %.4f 秒", tableTime))
print()
print("  【实测结论 —— 查表反而更慢】")
print("  在现代 CPU 上，math.sin 已经被优化得很好（甚至用 SIMD 指令），")
print("  而查表需要：取模运算 -> 浮点转整数 -> 表查找 -> 边界检查，")
print("  这些操作加起来超过了一次硬件 sin 计算。")
print()
print("  【那什么时候查表有价值？】")
print("    · 计算【非常复杂】时（不是 sin 这种单指令操作）")
print("      例：复杂的伤害公式、多层嵌套的属性计算")
print("    · 需要【确定性】结果时（跨平台浮点一致性，用于帧同步）")
print("    · 在【没有硬件浮点单元】的嵌入式设备上（老式功能机）")
print()
print("  【教训】「空间换时间」是个诱人的说法，但必须先测量。")
print("          现代 CPU 的算力经常比你的内存访问更快。")

print("=== 模式 6：惰性计算（Lazy Evaluation）===")
-- 【核心】不到真正需要时不算
-- 【适用】UI 未展开的分支、不常用的数据、初始化成本高的对象

local LazyValue = {}
LazyValue.__index = LazyValue

function LazyValue.new(computeFn)
  return setmetatable({
    _computeFn = computeFn,
    _value = nil,
    _computed = false,
    _computeCount = 0,
  }, LazyValue)
end

function LazyValue:get()
  if not self._computed then
    self._value = self._computeFn()
    self._computed = true
    self._computeCount = self._computeCount + 1
  end
  return self._value
end

function LazyValue:isComputed() return self._computed end
function LazyValue:computeCount() return self._computeCount end

local lazy = LazyValue.new(function()
  -- 模拟昂贵计算
  local sum = 0
  for i = 1, 100000 do sum = sum + i end
  return sum
end)

print("  创建后是否已计算: " .. tostring(lazy:isComputed()))
print("  第一次取值: " .. lazy:get())
print("  计算次数: " .. lazy:computeCount())
print("  第二次取值: " .. lazy:get())
print("  计算次数: 「 .. lazy:computeCount() .. 」（没有重复计算）")

print("  【红点系统的应用】红点树不用全部重算，只在 UI 查询时才计算被引用的分支")
print("  【Unity 的应用】Addressables 的按需加载、ScriptableObject 的延迟初始化")

print("=== 模式 7：减少间接层（Flatten）===")
-- 【核心】每次函数调用、每次表查找、每次指针跳转都有成本
-- 【做法】把热路径上的间接调用展开

print("  例子：热循环里的属性访问")
local entity = { stats = { hp = 100, maxHp = 100 }, transform = { x = 0, y = 0 } }

local ITER = 2000000

-- 写法 A：每次通过嵌套表访问
local t7 = os.clock()
local a = 0
for i = 1, ITER do
  a = a + entity.stats.hp
end
local nestedTime = os.clock() - t7

-- 写法 B：缓存到局部变量
local stats = entity.stats
local t8 = os.clock()
local b = 0
for i = 1, ITER do
  b = b + stats.hp
end
local flatTime = os.clock() - t8

-- 写法 C：缓存到纯局部变量（最快）
local hp = entity.stats.hp
local t9 = os.clock()
local c = 0
for i = 1, ITER do
  c = c + hp
end
local localTime = os.clock() - t9

print(string.format("  嵌套访问 entity.stats.hp:  %.4f 秒", nestedTime))
print(string.format("  缓存中间表 stats.hp:       %.4f 秒", flatTime))
print(string.format("  缓存到局部变量 hp:          %.4f 秒", localTime))
print(string.format("  最快比最慢快 %.2f 倍", nestedTime / localTime))
print("  【结论】热循环里，把频繁访问的东西缓存到局部变量最划算")

print()

print("=== 模式 9：位运算代替集合 ===")
-- 【核心】用一位表示一个布尔标志，比用表快得多、省得多

local Flags = {
  NONE     = 0,
  FIRE     = 1,     -- 2^0
  ICE      = 2,     -- 2^1
  POISON   = 4,     -- 2^2
  STUN     = 8,     -- 2^3
  SHIELD   = 16,    -- 2^4
}

local M2 = 2000000

-- 用表存状态
local stateTable = {}
local t12 = os.clock()
for i = 1, M2 do
  stateTable.fire = true
  stateTable.stun = false
  local has = stateTable.fire and not stateTable.stun
end
local tableStateTime = os.clock() - t12

-- 用位存状态
local stateBits = 0
local t13 = os.clock()
for i = 1, M2 do
  stateBits = stateBits | Flags.FIRE       -- 设置
  stateBits = stateBits & ~Flags.STUN      -- 清除
  local has = (stateBits & Flags.FIRE) ~= 0  -- 检查
end
local bitStateTime = os.clock() - t13

print(string.format("  用多个布尔字段: %.4f 秒", tableStateTime))
print(string.format("  用位掩码:       %.4f 秒", bitStateTime))
print()
print("  【实测结论 —— 位运算在 Lua 里没有优势】")
print("  在 Lua 5.4 中，位运算通过元方法/内部函数实现，")
print("  而表字段访问是高度优化的哈希查找——两者性能接近。")
print()
print("  【但位运算仍有不可替代的价值】这些优势不是速度：")
print("    1. 【省内存】一个 int 存 32 个布尔标志，网络同步时省带宽")
print("    2. 【原子性】一个整数可以一次性比较/传递，不担心部分更新")
print("    3. 【序列化方便】存成整数比存 32 个 bool 紧凑得多")
print("    4. 【C# 里更快】在 Unity 的 C# 中，位运算是原生 CPU 指令，")
print("                    比 Dictionary<string,bool> 或 bool[] 明显更快")
print()
print("  【结论】在 Lua 里用位运算是为了【省内存和带宽】，不是为了速度。")
print("          在 C# 里它既有内存优势也有速度优势。")
print()
print("  【其他用途】")
print("    · 游戏对象的图层（LayerMask）")
print("    · 技能的免疫标记（免疫哪些控制类型）")
print("    · 存档的标记位（哪些成就已完成）")

local arr = {}

print("  实测：开方 vs 平方距离")
local M3 = 3000000
local x1, y1, x2, y2 = 1.0, 2.0, 4.0, 6.0

local t16 = os.clock()
local sum1 = 0
for i = 1, M3 do
  local d = math.sqrt((x1 - x2) ^ 2 + (y1 - y2) ^ 2)
  if d < 10 then sum1 = sum1 + 1 end
end
local sqrtTime = os.clock() - t16

local t17 = os.clock()
local sum2 = 0
local dx, dy = x1 - x2, y1 - y2
local dxSq, dySq = dx * dx, dy * dy
local threshold = 100          -- 10 的平方
for i = 1, M3 do
  if dxSq + dySq < threshold then sum2 = sum2 + 1 end

一、三个需要亲自验证的"常识" ★★★

反例一：对象池不一定更快 实测结果：200000 次创建 vs 200000 次池复用——对象池并没有稳定胜出，比值在 0.93x ~ 1.26x 之间随机波动（本机连跑 3 次的比值分别是 1.18x、1.26x、1.06x）。

结论要说得准确一点：不是"对象池一定更慢"，而是"在这个微基准里，对象池的收益落在噪声范围内"——它既没有稳定变快，也没有稳定变慢。
所以"用对象池一定更快"这个说法不成立，但"用对象池一定更慢"同样不成立。

为什么：测试里的"对象"是一个只有 5 个字段的小表，创建它的成本本身就极低。而对象池的 get/release 有函数调用和表操作开销，两者大致相抵。

那为什么还要用对象池？真正的原因不是"建对象慢"，而是：
① GC 压力——200000 个短命对象会让 GC 频繁触发，而 GC 停顿体现在帧率抖动上，微基准测不出来；
② 初始化成本——如果对象初始化要读配置、算属性、加载资源，复用就明显划算；
③ 内存碎片——长期运行下频繁分配会让堆碎片化。

正确的判断方式：不要凭微基准决定，用 Profiler 看 GC Alloc 和帧率曲线。
反例二：三角函数查表反而更慢 实测结果：直接算 math.sin 100 万次用 0.052~0.058 秒；查表用 0.066~0.075 秒——查表稳定慢 15%~30%（约 0.85x）。

为什么：现代 CPU 的 sin 已经被优化得很好（甚至用 SIMD 指令），而查表需要：取模运算 → 浮点转整数 → 表查找 → 边界检查，这些加起来超过了一次硬件 sin 计算。

那什么时候查表有价值？
① 计算非常复杂时（不是 sin 这种单指令操作）——复杂伤害公式、多层嵌套属性计算；
② 需要确定性结果时（跨平台浮点一致性，用于帧同步）；
③ 在没有硬件浮点单元的设备上（老式功能机）。

教训："空间换时间"是个诱人的说法，但现代 CPU 的算力经常比你的内存访问更快。
反例三：位运算在 Lua 里没有速度优势 实测结果：布尔字段 0.055~0.060 秒，位掩码 0.070~0.071 秒——位运算稳定更慢约 20%（0.78~0.80x）。

为什么：Lua 5.4 的位运算通过内部函数实现，而表字段访问是高度优化的哈希查找，两者性能接近。

但位运算仍有不可替代的价值（这些优势都不是速度）：
① 省内存——一个 int 存 32 个布尔标志；
② 省带宽——网络同步时一个整数比 32 个 bool 紧凑得多；
③ 原子性——一个整数可以一次性比较/传递，不担心部分更新；
④ C# 里确实更快——在 Unity 的 C# 中，位运算是原生 CPU 指令，比 bool[] 或 Dictionary 明显更快。

结论：在 Lua 里用位运算是为了省内存和带宽，不是为了速度。在 C# 里它两者兼得。

注意"缓存"的适用条件 实测中缓存快了 70~83 倍——但这是因为只有 100 个不同的 key 被查询了 10000 次（命中率 99%）。

如果 key 每次都不同（比如用对象实例做 key），那么：
• 缓存永远不命中
• 却要为每次查询付出写入缓存的开销
• 而且缓存会无限增长直到内存爆掉

所以缓存必须考虑：命中率、内存上限、淘汰策略（见下一页 LRU）。

四、优化的正确流程 ★★★

这是全页最重要的内容 优化必须遵循的顺序：
① 测量——用 Profiler 找出真正的瓶颈（不要猜）
② 分析——确认它占了多少时间（优化只占 1% 的东西没有意义）
③ 优化——只改瓶颈部分
④ 复测——验证真的变快了，且没有引入 bug

常见的错误：
• 优化了 1% 的代码，却增加了 200 行复杂度
• 凭"感觉"优化，结果改错了地方（本页的三个反例就是例证）
• 为了性能牺牲可读性，但性能没提升——因为本来就不是瓶颈

80/20 法则：20% 的代码占 80% 的耗时。找到那 20%，比盲目优化全部代码有效得多。
