18 · 决策树与效用 AI ★

行为树解决的是"怎么执行"，决策树解决的是"做什么"。这两者经常被混淆，但它们是互补的——而效用系统（Utility AI）又是决策树的平滑版本。

这一页在路线中的位置 上一页（17）行为树讲"行为怎么组织与执行"。这一页讲"在多个行为之间怎么选"。
关键衔接：决策树本质是树结构（06）的应用；效用系统大量使用数学曲线（参见 C# 站的页 09 插值）。
下一页（19）GOAP——它比决策树更灵活，能自动组合出未预设的行为序列。
验证方式：本页代码在 Lua 5.4.7 实测运行，输出结果是真实的。

一、从硬编码到数据化

--  决策树与效用 AI：让 AI 做出「聪明」的选择 ★
-- ============================================================
--  行为树解决的是「怎么执行」，决策树解决的是「做什么」。
--  两者是互补的：
--    行为树：我要巡逻 -> 执行巡逻动作（含子步骤）
--    决策树：面对 3 个敌人，我该打还是跑？（单次决策）

print("=== 1. 最简单的决策树：嵌套 if ===")
-- 决策树本质就是一棵 if-else 树，但把它「数据化」后可以配置

local function simpleDecision(hp, enemyCount, hasPotion)
  if hp < 30 then
    if hasPotion then
      return "喝药水"
    else
      return "逃跑"
    end
  else
    if enemyCount >= 3 then
      return "使用群攻技能"
    elseif enemyCount >= 1 then
      return "普通攻击"
    else
      return "巡逻"
    end
  end
end

print("    · 条件复杂后嵌套层级失控")
print("    · 无法复用、无法测试")

  { hp = 80, enemyCount = 5, hasPotion = false, desc = "满血多敌" },

for _, c in ipairs(cases) do

print("=== 2. 数据化决策树：节点可配置 ===")
-- 把决策逻辑抽成节点，支持从配置加载

local DecisionNode = {}
DecisionNode.__index = DecisionNode

-- 创建条件节点
function DecisionNode.newCondition(condFn, trueNode, falseNode, label)
  return setmetatable({
    type = "condition",
    condFn = condFn,
    trueNode = trueNode,
    falseNode = falseNode,
    label = label or "condition",
  }, DecisionNode)
end

-- 创建动作节点（叶子）
function DecisionNode.newAction(actionName)
  return setmetatable({
    type = "action",
    actionName = actionName,
  }, DecisionNode)
end

-- 执行决策
function DecisionNode:evaluate(ctx, path)
  path = path or {}
  path[#path + 1] = self.label or self.actionName or "?"

  if self.type == "action" then
    return self.actionName, path
  end

  local condResult = self.condFn(ctx)
  local nextNode = condResult and self.trueNode or self.falseNode

  if not nextNode then
    return "无匹配分支", path
  end

  return nextNode:evaluate(ctx, path)
end

-- 用数据构建同一棵树
local tree = DecisionNode.newCondition(
  function(ctx) return ctx.hp < 30 end,
  -- hp < 30 分支
  DecisionNode.newCondition(
    function(ctx) return ctx.hasPotion end,
    DecisionNode.newAction("喝药水"),
    DecisionNode.newAction("逃跑"),
    "hasPotion?"
  ),
  -- hp >= 30 分支
  DecisionNode.newCondition(
    function(ctx) return ctx.enemyCount >= 3 end,
    DecisionNode.newAction("使用群攻技能"),
    DecisionNode.newCondition(
      function(ctx) return ctx.enemyCount >= 1 end,
      DecisionNode.newAction("普通攻击"),
      DecisionNode.newAction("巡逻"),
      "enemyCount>=1?"
    ),
    "enemyCount>=3?"
  ),
  "hp<30?"
)

print("  数据化后的优势：")
print("    · 结构可以序列化成 JSON/Excel，策划可配置")
print("    · 每次决策返回完整路径，便于调试")
print()
print("  决策路径可视化（这是调试 AI 最有用的信息）：")
for _, c in ipairs(cases) do
  local action, path = tree:evaluate(c)
  print(string.format("    [%s]", c.desc))
  print(string.format("      路径: %s", table.concat(path, " -> ")))
  print(string.format("      决策: %s", action))
end

print("=== 3. 效用系统（Utility AI）：用评分代替硬判断 ===")
-- 决策树的问题是「硬边界」：
--   hp = 31 和 hp = 29 会导致完全不同的行为，过渡很突兀。
-- 效用系统给每个行为打分，选最高分的，过渡平滑。

print("  【核心思想】每个行为有一个「有多想做」的分数")
print("  分数通常由多条曲线组合而成。")

-- 常用的效用曲线
local Curve = {}

-- 线性：值越低越想逃跑
function Curve.linear(x, minX, maxX)
  if maxX == minX then return 0 end
  return math.max(0, math.min(1, (x - minX) / (maxX - minX)))
end

-- 反线性：值越低分数越高
function Curve.inverseLinear(x, minX, maxX)
  return 1 - Curve.linear(x, minX, maxX)
end

-- 二次曲线：低值区域更敏感
function Curve.quadratic(x, minX, maxX)
  local t = Curve.linear(x, minX, maxX)
  return t * t
end

-- 平滑曲线（smoothstep）：两端平缓，中间陡峭
function Curve.smoothstep(x, minX, maxX)
  local t = Curve.linear(x, minX, maxX)
  return t * t * (3 - 2 * t)
end

-- 阈值曲线：超过阈值立即满分
function Curve.threshold(x, threshold, maxX)
  if x >= threshold then return 1 end
  return Curve.linear(x, 0, threshold)
end

print()
print("  常用效用曲线：")
print("    linear(线性)        - 均匀变化")
print("    inverseLinear(反线性) - 越低越想要")
print("    quadratic(二次)     - 低值区更敏感")
print("    smoothstep(平滑)    - 两端平缓中间陡")
print("    threshold(阈值)     - 超过阈值立即满分")

print("=== 4. 完整的效用 AI 实现 ===")

local UtilityAI = {}
UtilityAI.__index = UtilityAI

function UtilityAI.new()
  return setmetatable({ behaviors = {} }, UtilityAI)
end

-- 注册行为：名字 + 评分函数
function UtilityAI:addBehavior(name, scoreFn)
  self.behaviors[#self.behaviors + 1] = {
    name = name,
    scoreFn = scoreFn,
  }
  return self
end

-- 选最高分
function UtilityAI:select(ctx)
  local best, bestScore = nil, -1
  local allScores = {}

  for _, b in ipairs(self.behaviors) do
    local score = b.scoreFn(ctx)
    allScores[#allScores + 1] = { name = b.name, score = score }

    if score > bestScore then
      bestScore = score
      best = b.name
    end
  end

  return best, bestScore, allScores
end

-- 定义角色 AI
local ai = UtilityAI.new()

-- 喝药水：血量越低越想喝，且必须有药
ai:addBehavior("喝药水", function(ctx)
  if not ctx.hasPotion then return 0 end
  return Curve.inverseLinear(ctx.hp, 0, 60) * 0.9
end)

-- 逃跑：血量低 + 敌人多
ai:addBehavior("逃跑", function(ctx)
  local hpFactor = Curve.inverseLinear(ctx.hp, 0, 50)
  local enemyFactor = Curve.linear(ctx.enemyCount, 0, 5)
  return hpFactor * 0.7 + enemyFactor * 0.3
end)

-- 群攻：敌人越多越想用（但同时要自己有血才敢用）
ai:addBehavior("使用群攻技能", function(ctx)
  local enemyFactor = Curve.threshold(ctx.enemyCount, 3, 6)
  local hpFactor = Curve.linear(ctx.hp, 20, 60)      -- 残血时不敢放
  return enemyFactor * hpFactor * 0.9
end)

-- 普通攻击：有敌人且血量还行
ai:addBehavior("普通攻击", function(ctx)
  local enemyFactor = Curve.linear(ctx.enemyCount, 0, 3)
  local hpFactor = Curve.linear(ctx.hp, 20, 80)
  return enemyFactor * hpFactor * 0.6
end)

-- 巡逻：没敌人时
ai:addBehavior("巡逻", function(ctx)
  local noEnemy = Curve.inverseLinear(ctx.enemyCount, 0, 2)
  return noEnemy * 0.5
end)

end

数据化的最大价值：策划能改 硬编码的 if-else 只有一个问题，但它是致命的——策划改不了。

每一次"把血量阈值从 30 改成 35"，都要走需求 → 开发 → 提测 → 发版的完整流程。
而数据化之后，策划在 Excel 里改个数字、重新导出配置就完成了。

数据结构设计：
{ type = "condition", key = "hp", op = "<", value = 30, trueNode = {...}, falseNode = {...} }
这样一份 JSON 就能描述整棵决策树。

二、效用系统：用评分代替硬判断 ★★

决策树的根本问题：硬边界 看本页的实测对比（4 个敌人、无药水，血量从 100 降到 0）：
血量决策树效用系统
100 ~ 40使用群攻技能使用群攻技能
30使用群攻技能（突变点）逃跑（平滑过渡）
20 及以下逃跑逃跑
决策树在血量 30 处突然改变行为——31 打、29 跑。
而效用系统的分数是连续变化的，行为切换发生在"两个分数交叉"的时刻，看起来更自然。
五种常用效用曲线
曲线形状用途
linear均匀上升通用的"越多越好"
inverseLinear均匀下降"越低越想要"（逃跑、喝药）
quadratic低值区更敏感强调极端情况
smoothstep两端平缓、中间陡最自然的手感
threshold超过阈值立即满分"至少要有 3 个敌人"
组合方式：多个维度加权求和。例如"逃跑"的分数 = 血量权重 x 血量因子 + 敌人数权重 x 敌人因子。

三、模糊逻辑：处理"有点危险"

传统逻辑 vs 模糊逻辑 传统逻辑：hp < 30 是"危险"，否则是"安全"——非黑即白。
模糊逻辑：血量 40 时，危险度 0.4、安全度 0.6——同时是两者。

实测数据（用 smoothstep 模拟隶属度）：
• hp = 25 → 危险度 1.00，安全度 0.00
• hp = 40 → 危险度 0.40，安全度 0.60
• hp = 60 → 危险度 0.00，安全度 1.00

用途：需要"渐变式"行为过渡的 AI——潜行游戏里敌人的警觉度、恐怖游戏的紧张感、赛车 AI 的激进程度。

四、三者对比与选型

方案决策方式优点缺点适用
决策树条件判断（硬边界）清晰、可配置、易调试边界突变、规则多时树很深技能选择、UI 状态、明确规则
效用系统评分取最高平滑、可量化、易扩展参数难调、难调试NPC 日常、策略 AI
模糊逻辑隶属度 + 规则最自然、能表达模糊概念最复杂、最难调需要细腻情绪表现的 AI
实战中的组合方式 最常见的做法是混合使用：
• 决策树判断"要不要做"——比如"现在该不该进入战斗状态"
• 效用系统决定"做哪个"——进入战斗后，用评分选具体技能
• 行为树执行"怎么做"——选中的技能用行为树的子节点来执行

三层各司其职，比用单一方案硬套要清晰得多。
