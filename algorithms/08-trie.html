print("=== 1. 朴素敏感词过滤：暴力匹配的代价 ===")

local function naiveFilter(text, words)
  local result = text
  for _, word in ipairs(words) do
    -- 逐个词做全文查找并替换
    local start = 1
    while true do
      local s, e = string.find(result, word, start, true)  -- true = 纯文本查找
      if not s then break end
      result = result:sub(1, s - 1) .. string.rep("*", #word) .. result:sub(e + 1)
      start = s + #word
    end
  end
  return result
end

print("=== 2. Trie：一次扫描完成所有匹配 ===")
-- 核心思想：把所有敏感词构建成一棵树，用文本去树上走

local Trie = {}
Trie.__index = Trie

function Trie.new()
  return setmetatable({
    root = {},          -- 根节点（空表）
    wordCount = 0,
  }, Trie)
end

-- 插入一个词
function Trie:insert(word)
  local node = self.root
  for i = 1, #word do
    local ch = word:sub(i, i)
    if not node[ch] then
      node[ch] = {}     -- 创建子节点
    end
    node = node[ch]
  end
  node.isEnd = true     -- 标记词尾
  node.word = word      -- 记录完整词（方便后续处理）
  self.wordCount = self.wordCount + 1
end

-- 批量插入
function Trie:insertAll(words)
  for _, w in ipairs(words) do
    self:insert(w)
  end
end

-- 【核心】从指定位置开始匹配，返回最长匹配长度（0 表示没匹配）
function Trie:matchAt(text, startPos)
  local node = self.root
  local longest = 0
  local i = startPos

  while i <= #text do
    local ch = text:sub(i, i)
    node = node[ch]
    if not node then break end        -- 走不下去了

    if node.isEnd then
      longest = i - startPos + 1      -- 记录最长匹配
    end
    i = i + 1
  end

  return longest
end

-- 过滤（用 * 替换）
function Trie:filter(text)
  local result = {}
  local i = 1

  while i <= #text do
    local len = self:matchAt(text, i)
    if len > 0 then
      -- 命中：替换为等长星号，跳过整个词
      result[#result + 1] = string.rep("*", len)
      i = i + len
    else
      result[#result + 1] = text:sub(i, i)
      i = i + 1
    end
  end

  return table.concat(result)
end

-- 只检测是否包含（不替换，更快）
function Trie:contains(text)
  local i = 1
  while i <= #text do
    if self:matchAt(text, i) > 0 then return true end
    i = i + 1
  end
  return false
end

-- 返回所有命中的词
function Trie:findAll(text)
  local found = {}
  local i = 1
  while i <= #text do
    local node = self.root
    local j = i
    while j <= #text do
      local ch = text:sub(j, j)
      node = node[ch]
      if not node then break end
      if node.isEnd then
        found[#found + 1] = { word = node.word, pos = i }
      end
      j = j + 1
    end
    i = i + 1
  end
  return found
end

print("=== 4. AC 自动机：Trie + 失败指针 ===")
-- Trie 有个问题：每个位置都要从头尝试匹配。
-- AC 自动机给每个节点加「失败指针」，
-- 匹配失败时跳到最长公共后缀继续，实现真正的 O(n)。

local ACMatcher = {}
ACMatcher.__index = ACMatcher

function ACMatcher.new()
  return setmetatable({
    root = {},
    nodes = {},      -- 所有节点列表（用于构建失败指针）
  }, ACMatcher)
end

function ACMatcher:insertAll(words)
  for _, w in ipairs(words) do
    self:insert(w)
  end
end

function ACMatcher:insert(word)
  local node = self.root
  for i = 1, #word do
    local ch = word:sub(i, i)
    if not node[ch] then
      node[ch] = { ch = ch, depth = (node.depth or 0) + 1, fail = nil }
      self.nodes[#self.nodes + 1] = node[ch]
    end
    node = node[ch]
  end
  node.isEnd = true
  node.word = word
end

-- 构建失败指针（BFS）
function ACMatcher:build()
  local queue = {}
  local head = 1

  -- 第一层节点的 fail 指向根
  for ch, child in pairs(self.root) do
    if type(child) == "table" and child.ch then
      child.fail = self.root
      queue[#queue + 1] = child
    end
  end

  -- BFS 逐层构建
  while head <= #queue do
    local current = queue[head]
    head = head + 1

    for ch, child in pairs(current) do
      if type(child) == "table" and child.ch and child.ch == ch then
        -- 找 child 的 fail：从 current.fail 开始找相同字符的子节点
        local f = current.fail
        while f and not f[ch] do
          f = f.fail
        end
        child.fail = (f and f[ch]) or self.root

        -- 合并终止标记（重要：让后缀匹配也能被识别）
        if child.fail.isEnd then
          child.isEnd = true
          child.word = child.word or child.fail.word
        end

        queue[#queue + 1] = child
      end
    end
  end
end

-- AC 匹配：一次遍历找出所有命中
function ACMatcher:search(text)
  local result = {}
  local node = self.root

  for i = 1, #text do
    local ch = text:sub(i, i)

    -- 沿着失败指针回溯，直到找到匹配或回到根
    while node ~= self.root and not node[ch] do
      node = node.fail
    end

    node = node[ch] or self.root

    -- 检查当前节点及其失败链上是否有词结尾
    local temp = node
    while temp ~= self.root do
      if temp.isEnd then
        result[#result + 1] = {
          word = temp.word,
          pos = i - #temp.word + 1,
        }
      end
      temp = temp.fail
    end
  end

  return result
end

    实测发现了一个重要的语义差异 暴力法和 Trie 的过滤结果不一致——但这不是 bug，是语义差异：

    词库里有「垃圾」和「垃圾游戏」两个词：
    • 暴力法：先替换「垃圾」→ 变成「****游戏」，之后「垃圾游戏」再也匹配不到
    • Trie：最长匹配优先 → 直接命中「垃圾游戏」整体替换

    实际输出对比：
    • 暴力法：这******游戏真难玩（只替换了「垃圾」）
    • Trie：这************真难玩（替换了整个「垃圾游戏」）

    结论：Trie 的"最长匹配"才是正确行为。否则「垃圾游戏」会被替换成「****游戏」，剩余部分仍然可见。
    这是 Trie 相比暴力法的正确性优势，不只是性能优势。

    二、AC 自动机：Trie 的进化版

    Trie 的瓶颈在哪里 Trie 虽然比暴力法快得多，但仍有一个浪费：每个位置都要从根节点重新开始尝试匹配。

    AC 自动机（Aho-Corasick）给每个节点加了一个失败指针（fail pointer）：
    当匹配失败时，跳到"最长公共后缀"对应的节点继续，而不是回到根。

    复杂度对比（文本长度 n，最长的词长度 m）：
    • 暴力法：O(n x 词数)
    • Trie：O(n x m) 最坏情况
    • AC 自动机：O(n + 命中数)——严格线性，一次扫描搞定

    三、UTF-8 中文处理：一个必踩的坑

    string.sub 按字节切，会让中文乱码 Lua 的 string.sub 是按字节操作的，而一个中文字符占 3 个字节（UTF-8）。

    "垃圾":sub(1, 3) —— 看起来是取第一个字符，实际上只取了"垃"的 3 个字节中的一部分，结果是乱码。

    正确做法：Lua 5.3+ 提供了 utf8 库：
    • utf8.len(s) —— 字符数（不是字节数）
    • utf8.codes(s) —— 按字符遍历
    • utf8.offset(s, n) —— 取第 n 个字符的字节位置

    做敏感词过滤必须用 utf8 库，否则中文词永远匹配不对。
