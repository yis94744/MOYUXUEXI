print("=== 1. 方案一：枚举 + switch（最简单，也最快）===")

local State = {
  IDLE = "IDLE",
  PATROL = "PATROL",
  CHASE = "CHASE",
  ATTACK = "ATTACK",
}

local EnumFSM = {}
EnumFSM.__index = EnumFSM

function EnumFSM.new()
  return setmetatable({
    state = State.IDLE,
    stateTime = 0,
    log = {},
  }, EnumFSM)
end

function EnumFSM:update(dt, ctx)
  self.stateTime = self.stateTime + dt

  -- 【核心】一个大 switch（Lua 里用 if 链）
  if self.state == State.IDLE then
    if ctx.distance < 10 then
      self:transition(State.PATROL)
    end

  elseif self.state == State.PATROL then
    if ctx.distance < 5 then
      self:transition(State.CHASE)
    elseif ctx.distance > 15 then
      self:transition(State.IDLE)
    end

  elseif self.state == State.CHASE then
    if ctx.distance < 2 then
      self:transition(State.ATTACK)
    elseif ctx.distance > 10 then
      self:transition(State.PATROL)
    end

  elseif self.state == State.ATTACK then
    if ctx.distance > 3 then
      self:transition(State.CHASE)
    end
  end
end

function EnumFSM:transition(newState)
  if self.state == newState then return end
  self.log[#self.log + 1] = self.state .. " -> " .. newState
  self.state = newState
  self.stateTime = 0
end

print("  结构：if-elseif 链 + 字符串状态")
print("  优点：最快、最简单、无额外对象")
print("  缺点：状态多了难维护、无法复用")

print()
print("=== 2. 方案二：表驱动（状态 -> 处理函数）===")
-- 把每个状态的处理逻辑放进表里，避免长 if 链

local TableFSM = {}
TableFSM.__index = TableFSM

function TableFSM.new()
  local self = setmetatable({
    state = State.IDLE,
    stateTime = 0,
    log = {},
  }, TableFSM)

  -- 【关键】状态处理表
  self.handlers = {
    [State.IDLE] = function(s, dt, ctx)
      if ctx.distance < 10 then s:transition(State.PATROL) end
    end,
    [State.PATROL] = function(s, dt, ctx)
      if ctx.distance < 5 then s:transition(State.CHASE)
      elseif ctx.distance > 15 then s:transition(State.IDLE) end
    end,
    [State.CHASE] = function(s, dt, ctx)
      if ctx.distance < 2 then s:transition(State.ATTACK)
      elseif ctx.distance > 10 then s:transition(State.PATROL) end
    end,
    [State.ATTACK] = function(s, dt, ctx)
      if ctx.distance > 3 then s:transition(State.CHASE) end
    end,
  }
  return self
end

function TableFSM:update(dt, ctx)
  self.stateTime = self.stateTime + dt
  local handler = self.handlers[self.state]
  if handler then handler(self, dt, ctx) end
end

function TableFSM:transition(newState)
  if self.state == newState then return end
  self.log[#self.log + 1] = self.state .. " -> " .. newState
  self.state = newState
  self.stateTime = 0
end

-- 对比：给状态加 enter/exit 很方便（表驱动的一大优势）
print("  结构：状态字符串 -> 处理函数 的映射表")
print("  优点：无长 if 链、易于按状态扩展 enter/exit")
print("  缺点：一次表查找开销；状态处理函数散落在表里")

print("=== 3. 方案三：类式状态机（可复用、可继承）===")
-- 每个状态是一个对象，有 enter/update/exit

local StateBase = {}
StateBase.__index = StateBase

function StateBase.new(name)
  return setmetatable({ name = name }, StateBase)
end
function StateBase:enter(owner, ctx) end
function StateBase:update(owner, ctx, dt) end
function StateBase:exit(owner, ctx) end

local ClassFSM = {}
ClassFSM.__index = ClassFSM

function ClassFSM.new()
  return setmetatable({
    current = nil,
    states = {},
    log = {},
  }, ClassFSM)
end

function ClassFSM:addState(state)
  self.states[state.name] = state
  return self
end

function ClassFSM:changeState(name, ctx)
  local newState = self.states[name]
  if not newState then
    error("未注册的状态: " .. tostring(name))
  end

  if self.current then
    self.current:exit(self, ctx)
    self.log[#self.log + 1] = self.current.name .. " -> " .. name
  end

  self.current = newState
  self.current:enter(self, ctx)
end

function ClassFSM:update(ctx, dt)
  if self.current then
    self.current:update(self, ctx, dt)
  end
end

-- 定义具体状态
local IdleState = setmetatable({}, { __index = StateBase })
IdleState.__index = IdleState
function IdleState.new()
  local s = StateBase.new("IDLE")
  return setmetatable(s, IdleState)
end
function IdleState:update(owner, ctx, dt)
  if ctx.distance < 10 then owner:changeState("PATROL", ctx) end
end

local PatrolState = setmetatable({}, { __index = StateBase })
PatrolState.__index = PatrolState
function PatrolState.new()
  local s = StateBase.new("PATROL")
  return setmetatable(s, PatrolState)
end
function PatrolState:update(owner, ctx, dt)
  if ctx.distance < 5 then owner:changeState("CHASE", ctx)
  elseif ctx.distance > 15 then owner:changeState("IDLE", ctx) end
end

local ChaseState = setmetatable({}, { __index = StateBase })
ChaseState.__index = ChaseState
function ChaseState.new()
  local s = StateBase.new("CHASE")
  return setmetatable(s, ChaseState)
end
function ChaseState:update(owner, ctx, dt)
  if ctx.distance < 2 then owner:changeState("ATTACK", ctx)
  elseif ctx.distance > 10 then owner:changeState("PATROL", ctx) end
end

local AttackState = setmetatable({}, { __index = StateBase })
AttackState.__index = AttackState
function AttackState.new()
  local s = StateBase.new("ATTACK")
  return setmetatable(s, AttackState)
end
function AttackState:update(owner, ctx, dt)
  if ctx.distance > 3 then owner:changeState("CHASE", ctx) end
end

local classFSM = ClassFSM.new()
classFSM:addState(IdleState.new())
classFSM:addState(PatrolState.new())
classFSM:addState(ChaseState.new())
classFSM:addState(AttackState.new())
classFSM:changeState("IDLE", {})

print("  结构：每个状态是独立对象，有 enter/update/exit")
print("  优点：可复用、可继承、状态逻辑内聚")
print("  缺点：对象开销大、一次虚调用")

print()

    实测结论推翻了一个常见说法 很多资料说"类式状态机因为有虚函数调用，所以比 if-else 慢"。
    但实测结果不是这样（50 万次 update）：

    • 枚举 + if 链：0.0490 秒（基准）
    • 表驱动：0.0450 秒（0.92x）
    • 类式状态机：0.0420 秒（0.86x）

    三种方案在同一量级，类式反而略快。

    为什么：状态数少时，"if-else 链"需要逐条比较字符串；而类式状态机是一次表查找加一次方法调用——在状态数较多时反而更高效。

    所以：性能不是选型依据，别为了"性能"牺牲可维护性。

    二、怎么选？看这四个维度

    方案结构适用不适合
    枚举 + if一串 if-elseif状态 < 5、逻辑简单状态会增长、需要复用
    表驱动状态 → 处理函数 映射需要 enter/exit 钩子状态逻辑复杂、需要继承
    类式状态机每个状态是对象状态逻辑复杂、需要复用状态极简单时过度设计
    层次状态机状态树 + 事件冒泡状态有嵌套结构状态间关系扁平
    栈式状态机状态栈暂停、弹窗、中断恢复频繁随机跳转
    实战建议 从枚举开始，需要时再重构。
    • 一个只有 3 个状态的怪物巡逻逻辑 → 用 if-else 就够了，别过度设计；
    • 一个十几状态的玩家角色控制器 → 用类式状态机，否则维护成本会失控；
    • UI 界面栈 → 用栈式状态机，天然契合"打开/关闭"的语义。

    重构的信号：当你的 if-else 链超过 50 行、或者同一个状态转换写了两遍时，就该重构了。

    三、层次状态机（HSM）★

    解决"状态爆炸"的问题 问题场景：角色有"站立/移动"两个大状态，"移动"里又分"走/跑/冲刺"。
    朴素做法：状态数 = 2 x 3 = 6，每个都要写与其他 5 个的转换条件 → 连线爆炸。

    层次状态机：把状态组织成树，共性的转换逻辑上提到父状态。
    层级状态负责的转换
    父Grounded（地面）处理"按跳跃键 → 转到空中"
    子Idle / Walk / Run处理互相之间的转换
    父Airborne（空中）处理"落地 → 转到地面"
    子Jump / Fall处理互相之间的转换
    关键规则：事件冒泡
    子状态处理不了的事件会向上传给父状态。比如在 Walk 状态下按跳跃：Walk 不处理 → 冒泡给 Grounded → 切换到 Airborne。

    收益：跳跃逻辑只写一遍，三个子状态都能用。

    四、栈式状态机

    天然契合"中断/恢复"语义 场景：游戏中打开菜单，游戏逻辑应该暂停；关闭菜单后恢复到打开之前的状态。

    用普通状态机的问题：你需要记住"打开菜单前是什么状态"，还要处理嵌套弹窗。
    用栈式状态机：
    push(MenuState) → 自动暂停 PlayingState
    pop() → 自动恢复 PlayingState
    再开一层弹窗就是再 push 一次，天然支持任意嵌套。

    典型用途：UI 界面栈、暂停系统、对话系统、技能打断、教程引导。

    五、七个实战要点

    这些是踩过坑才知道的 ① 状态用字符串还是数字
    字符串可读性好、调试方便；数字性能好、能对比。大型项目的常见做法：用字符串常量定义，编译期或初始化时转成数字 ID。

    ② transition 必须去重
    if (currentState == newState) return; —— 否则会重复触发 enter/exit，导致特效重复播放、音效重叠。

    ③ 必须记录 stateTime
    几乎所有逻辑都需要"进入这个状态多久了"（攻击前摇、无敌时间、冷却）。在 enter 时清零，每帧累加。

    ④ 状态退出要清理
    离开 ATTACK 状态时，要取消连招计时、隐藏武器拖尾、重置动画参数。不清理会有"残留状态"的诡异 bug。

    ⑤ 避免在 update 里直接改状态
    同一帧里状态被改了三次？很难调试。更好的做法：先记录"想转到哪个状态"，update 结束后统一处理转换。

    ⑥ 加一个状态变更日志
    打印每次转换（IDLE -> CHASE -> ATTACK）。调试 AI 时，这个日志能让你五分钟定位问题，而不是盲猜一小时。

    ⑦ UI 也是状态机
    按钮的 Normal/Hover/Pressed/Disabled、界面的 Loading/Ready/Error——本质都是状态机。用状态机思想写 UI，比一堆 bool 标志位清晰得多。

    六、状态机 vs 行为树：什么时候该换

    信号说明建议
    状态数 > 10 且互相都能转转换矩阵接近全连接换行为树
    转换条件写了重复逻辑同样一段判断出现在多处先试层次状态机
    状态内有"步骤序列"进入某状态后要做 3 件事该状态内部用行为树
    策划想改 AI 但改不动逻辑硬编码在代码里换行为树（可配置）
    状态数 < 8 且逻辑清晰一眼能看懂转换关系保持状态机，别过度设计
    混合使用是最实际的方案 大状态用状态机，状态内部用行为树——这是很多商业项目的实际做法。

    例如：
    • PatrolState 内部的行为树：走到 A 点 → 等待 2 秒 → 走到 B 点 → 循环
    • CombatState 内部的行为树：选择技能 → 释放 → 冷却 → 再选择
    • FleeState 内部的行为树：找最远的出口 → 逃跑 → 检测是否安全

    这样既有状态机的清晰结构，又有行为树的可组合性。
